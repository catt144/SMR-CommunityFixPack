-- TEMPORARY LoadFirst (option B) order mutator, game 1.1.1.405907.
-- Rewrites the SAVED enable order to ORDER below through the game's own
-- TurnModOff/TurnModOn helpers (ModManager.lua:35-41), requests the account
-- save through THIS KIT's persistent slot (a changed write calls
-- SaveAccountStorage(1000), Mod.lua:1487-1503), and quits at the menu.
-- The pack's LoadFirst is vetoed for THIS launch only, so the pack does not
-- promote itself in the same launch and the next boot loads exactly ORDER.
-- Edit ORDER before each arm. Restores nothing by itself: the owner's original
-- order is restored by arming this same payload with that order.
LoadFirstSet = {}

-- the veto must exist before the pack's 00_Core adopts the table; the kit
-- loads before the pack on this rig's original order, and when it does not,
-- the pack's promotion in this launch is simply overwritten below
SMRFixPack_Disabled = rawget(_G, "SMRFixPack_Disabled") or {}
SMRFixPack_Disabled.LoadFirst = true

ORDER = {
    "SMR_CommunityFixPackTestKit",
    "SMR_TrainHubDev_20260918",
    "SMR_CommunityFixPack",
    "SMR_CommunityOptInPack",
    "SMR_RailShaftDev_20260923",
}

local function say(s)
    ModLog("[LOADFIRST-SET] " .. tostring(s):gsub("%%", "%%%%"))
    local flush = rawget(_G, "FlushLogFile")
    if flush then pcall(flush) end
end

local function ids(list)
    local out = {}
    for i, v in ipairs(list or {}) do out[i] = tostring(type(v) == "table" and v.id or v) end
    return table.concat(out, ",")
end

function LoadFirstSet.Apply()
    local before = GetModsEnabledByUser()
    say("BEFORE saved=" .. ids(before))
    local def = rawget(_G, "Mods") and Mods["SMR_CommunityFixPack"]
    local opts = type(def) == "table" and rawget(def, "options")
    local pack = rawget(_G, "SMRFixPack")
    local lf = type(pack) == "table" and type(pack.fixes) == "table" and pack.fixes.LoadFirst
    say("BEFORE option LoadFirst=" .. tostring(type(opts) == "table" and rawget(opts, "LoadFirst"))
        .. " status=" .. tostring(lf and lf.status) .. " detail=" .. tostring(lf and lf.detail))
    for _, id in ipairs(before) do TurnModOff(id) end
    for _, id in ipairs(ORDER) do TurnModOn(id) end
    local after = GetModsEnabledByUser()
    say("AFTER saved=" .. ids(after))
    local err = WriteModPersistentData("loadfirst-set " .. tostring(os.time()) .. " " .. ids(after))
    say("SAVE_REQUEST err=" .. tostring(err))
    return ids(after) == table.concat(ORDER, ",")
end

function LoadFirstSet.Boot()
    CreateRealTimeThread(function()
        Sleep(1)
        local deadline = RealTime() + 300000
        while not GetPreGameMainMenu() do
            if RealTime() > deadline then say("TIMEOUT menu"); quit(); return end
            Sleep(500)
        end
        Sleep(3000)
        local ok, res = pcall(LoadFirstSet.Apply)
        say("APPLY ok=" .. tostring(ok) .. " matched=" .. tostring(res))
        -- the shipped writer saves 1000 ms after the request; quit also flushes a pending save
        Sleep(5000)
        say("DONE")
        quit()
    end)
end

LoadFirstSet.Boot()
