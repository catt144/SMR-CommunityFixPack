local function serving_centers(farm)
	local out, seen = {}, {}
	local ok, list = pcall(function() return farm.command_centers end)
	if not ok or type(list) ~= "table" then return out, false end
	local function add(node, depth)
		if not node or depth > 4 then return end
		local h = rawget(node, "handle")
		if h and seen[h] then return end
		if h then seen[h] = true end
		out[#out + 1] = node
		local ok_e, ext = pcall(function() return node.linked_extenders end)
		if ok_e and type(ext) == "table" then
			for _, e in ipairs(ext) do
				local ok_w, working = pcall(function() return e.working end)
				if ok_w and working then add(e, depth + 1) end
			end
		end
	end
	for _, c in ipairs(list) do add(c, 1) end
	return out, true
end

function C47.VegPeek(when)
	local map = rawget(_G, "CurrentMap")
	if not (map and map.MapGet) then
		say(fmt("VEG %s :: CurrentMap:MapGet unavailable — NOT SAMPLED", tostring(when)))
		return
	end
	local ok, list = pcall(function() return map:MapGet("map", "VegetationTaskRequester") end)
	if not ok or type(list) ~= "table" then
		say(fmt("VEG %s :: MapGet failed ok=%s — NOT SAMPLED. ⛔ A zero here would be UNSAMPLED, not 'none exist'.", tostring(when), tostring(ok)))
		return
	end
	say(fmt("VEG %s :: %d live VegetationTaskRequester(s) on the map", tostring(when), #list))
	for i = 1, 3 do
		local o = list[i]
		if o then
			local ok_d, dist = pcall(function() return o.supply_dist_modifier end)
			local ok_p, prio = pcall(function() return o.priority end)
			local ok_a, amt  = pcall(function() return o.request and o.request:GetActualAmount() end)
			say(fmt("VEG %s :: sample %s | supply_dist_modifier=%s (ok=%s) | priority=%s (ok=%s) | request actual amount=%s | preset=%s",
				tostring(when), idof(o), tostring(dist), tostring(ok_d),
				tostring(prio), tostring(ok_p), tostring(ok_a and amt),
				tostring(select(2, pcall(function() return o.preset and o.preset.id end)))))
		end
	end
end

function C47.Census()
	if not C47.Guard("Census") then return false end
	C47.Label("ORGANIC", "census of farms and Seeds depots",
		"reads only; the interventions below are separately labelled FORCED")
	C47.Now("census")
	C47.farms, C47.depots = {}, {}
	local map = rawget(_G, "CurrentMap")
	local farms = C47.LabelAll("OpenFarmBase")
	for i, b in ipairs(farms) do
		local centers = select(1, serving_centers(b))
		C47.farms[#C47.farms + 1] = { obj = b, role = "FARM", handle = rawget(b, "handle"), win = nil }
		local ok_ms, maxs = pcall(function() return b.consumption_max_storage end)
		say(fmt("CENSUS FARM [%d] %s :: stored=%s of %s | consumption_amount=%s | tick=%s ms | working=%s | serving centres+extenders=%d",
			i, idof(b), tostring(rawget(b, "consumption_stored_resources")), tostring(maxs),
			tostring(rawget(b, "consumption_amount")),
			tostring(select(2, pcall(function() return b.building_update_time end))),
			tostring(rawget(b, "working")), #centers))
	end
	C47.Applicable("C48B-population(OpenFarmBase)", #C47.farms, #C47.farms,
		"city labels walked this load")
	if #C47.farms == 0 then
		say("CENSUS ⛔ no OpenFarmBase — wrong save staged. Nothing measured.")
		return false
	end
	local seen = {}
	for _, cls in ipairs{ "StorageDepot", "UniversalStorageDepot", "MechanizedDepot", "StorageSeeds" } do
		local ok, list = pcall(function() return map and map:MapGet("map", cls) end)
		if ok and type(list) == "table" then
			for _, d in ipairs(list) do
				local h = rawget(d, "handle")
				local ok_a, acc = pcall(function() return d:DoesAcceptResource("Seeds") end)
				if h and not seen[h] and (not ok_a or acc) then
					seen[h] = true
					C47.depots[#C47.depots + 1] = d
				end
			end
		end
	end
	C47.Applicable("C48B-population(Seeds-accepting depots)", #C47.depots, #C47.depots,
		"CurrentMap:MapGet over four depot classes, filtered on DoesAcceptResource('Seeds')")
	C47.VegPeek("census")
	return true
end

---- depot totals, computed in-code so the discriminator sits in one line ------

local function depot_reading()
	local vals, total, ok_n = {}, 0, 0
	for _, d in ipairs(C47.depots or empty_table) do
		local ok, stored = pcall(function()
			if d.GetStoredAmount then return d:GetStoredAmount() end
			return (d.stockpiled_amount or empty_table).Seeds
		end)
		local v = (ok and type(stored) == "number") and stored or nil
		vals[rawget(d, "handle") or #vals + 1] = { obj = d, v = v }
		if v then total = total + v; ok_n = ok_n + 1 end
	end
	return { vals = vals, total = total, ok_n = ok_n }
end

local function depot_delta(name, open, close)
	local unmoved, fell, rose, big_fall, big_rise = 0, 0, 0, nil, nil
	for h, rec in pairs(open.vals) do
		local o, c = rec.v, close.vals[h] and close.vals[h].v
		if o and c then
			local d = c - o
			if d == 0 then unmoved = unmoved + 1
			elseif d < 0 then fell = fell + 1; if not big_fall or d < big_fall.d then big_fall = { d = d, obj = rec.obj } end
			else rose = rose + 1; if not big_rise or d > big_rise.d then big_rise = { d = d, obj = rec.obj } end end
		end
	end
	say(fmt("DEPOTS %s :: total %d → %d (Δ %+d over %d/%d readable depots) | unmoved=%d fell=%d rose=%d | biggest fall %s %s | biggest rise %s %s",
		tostring(name), open.total, close.total, close.total - open.total,
		close.ok_n, #(C47.depots or empty_table), unmoved, fell, rose,
		big_fall and idof(big_fall.obj) or "-", tostring(big_fall and big_fall.d),
		big_rise and idof(big_rise.obj) or "-", tostring(big_rise and big_rise.d)))
end

---- a measurement window ------------------------------------------------------

local function fresh_win()
	return { min = nil, max = nil, last_buf = nil, n = 0, stalled = 0,
		deliveries = 0, delivered = 0, sizes = {}, drains = 0, drain_total = 0 }
end

local function sample_once()
	C47.samples = C47.samples + 1
	for _, rec in ipairs(C47.farms or empty_table) do
		local w = rec.win
		if w then
			local cur = rawget(rec.obj, "consumption_stored_resources")
			if type(cur) == "number" then
				w.n = w.n + 1
				if not w.min or cur < w.min then w.min = cur end
				if not w.max or cur > w.max then w.max = cur end
				if w.last_buf and cur > w.last_buf then
					local size = cur - w.last_buf
					w.deliveries = w.deliveries + 1
					w.delivered = w.delivered + size
					w.sizes[size] = (w.sizes[size] or 0) + 1
				elseif w.last_buf and cur < w.last_buf then
					w.drains = w.drains + 1
					w.drain_total = w.drain_total + (w.last_buf - cur)
				end
				w.last_buf = cur
			end
			if not rawget(rec.obj, "working") then w.stalled = w.stalled + 1 end
		end
	end
end

function C47.RunWindow(name)
	if not C47.Guard("Window " .. tostring(name)) then return false end
	local open = depot_reading()
	C47.VegPeek("window '" .. tostring(name) .. "' open")
	for _, rec in ipairs(C47.farms or empty_table) do rec.win = fresh_win() end
	local start_gt, start_rt = GameTime(), RealTime()
	say(fmt("WINDOW '%s' OPEN :: GameTime=%d | target +%d game ms | speed=%sx | sampling every %d real ms",
		tostring(name), start_gt, C47.window_gt,
		tostring(GetTimeFactor() / const.DefaultTimeFactor), C47.sample_ms))
	local target = start_gt + C47.window_gt
	local last_gt, still, complete = start_gt, 0, true
	while true do
		sample_once()
		local gt, rt = GameTime(), RealTime()
		if gt == last_gt then
			still = still + 1
			if still == C47.freeze_samples then
				say(fmt("WINDOW '%s' ⛔ CLOCK STOPPED for %d consecutive samples — window is PARTIAL.", tostring(name), still))
				C47.FreezeDiagnostic("clock stopped in window " .. tostring(name))
				C47.frozen_at = gt
				complete = false
				break
			end
		else still = 0; last_gt = gt end
		if gt - target >= 0 then break end
		if rt - (start_rt + C47.max_real_ms) >= 0 then
			say(fmt("WINDOW '%s' ⛔ REAL-TIME CAP — window is PARTIAL.", tostring(name)))
			complete = false
			break
		end
		Sleep(C47.sample_ms)
	end
	local close = depot_reading()
	local end_gt, end_rt = GameTime(), RealTime()
	local w = { name = name, start_gt = start_gt, end_gt = end_gt,
		real_ms = end_rt - start_rt, complete = complete,
		depot_open = open, depot_close = close, farms = {} }
	for _, rec in ipairs(C47.farms or empty_table) do
		w.farms[#w.farms + 1] = { id = idof(rec.obj), win = rec.win }
		rec.win = nil
	end
	C47.windows = C47.windows or {}
	C47.windows[#C47.windows + 1] = w
	say(fmt("WINDOW '%s' CLOSED :: game %d ms (%s h) | real %d ms | complete=%s",
		tostring(name), end_gt - start_gt,
		tostring((end_gt - start_gt) / C47.MS_PER_HOUR), w.real_ms, tostring(complete)))
	depot_delta("window '" .. tostring(name) .. "'", open, close)
	C47.VegPeek("window '" .. tostring(name) .. "' close")
	return complete
end

---- ⛔⛔ THE INTERVENTION -----------------------------------------------------

function C47.ApplyBrake()
	if not C47.Guard("ApplyBrake") then return false end
	C47.Label("FORCED", "supply_dist_modifier 100 → 150 on VegetationTaskRequester, plus a request rebuild per live requester",
		"the ONLY mutations this leg makes; the class value is restored before quit and nothing is ever saved to disk")
	local classes = rawget(_G, "g_Classes")
	local cls = classes and classes.VegetationTaskRequester
	if not cls then
		say("BRAKE ⛔ g_Classes.VegetationTaskRequester did not resolve — INTERVENTION NOT APPLIED, window 2 is VOID as an intervention reading.")
		return false
	end
	local before = cls.supply_dist_modifier
	say(fmt("BRAKE class value before=%s (expected 100, inherited from TaskRequester, _TaskRequest.lua:96)", tostring(before)))
	cls.supply_dist_modifier = 150
	say(fmt("BRAKE class value set to %s — the SurfaceDeposit number and rationale (SurfaceDeposit.lua:84)", tostring(cls.supply_dist_modifier)))
	C47.brake = { before = before, total = 0, rebuilt = 0, failed = 0, skipped = 0,
		identity_unchanged = 0, amount_mismatch = 0, amount_delta = 0 }

	local map = rawget(_G, "CurrentMap")
	local ok, list = pcall(function() return map and map:MapGet("map", "VegetationTaskRequester") end)
	if not ok or type(list) ~= "table" then
		say("BRAKE ⛔ requester walk failed — class value is set but live requests keep their old weighting. INTERVENTION PARTIAL AT BEST.")
		return false
	end
	C47.brake.total = #list
	say(fmt("BRAKE rebuilding %d live requester(s): per requester, centres:RemoveBuilding → fresh CreateResourceRequests (bakes the new modifier via Request_New) → centres:AddBuilding — the TaskRequester:SetPriority re-registration template (_TaskRequest.lua:157-181)", #list))
	for i, o in ipairs(list) do
		local ok_one, why = pcall(function()
			if not IsValid(o) then C47.brake.skipped = C47.brake.skipped + 1 return end
			local old_req = o.request
			local old_amt = old_req and old_req:GetActualAmount() or nil
			local centers = o.command_centers or empty_table
			for _, c in ipairs(centers) do
				pcall(function() c:RemoveBuilding(o) end)
			end
			o.task_requests = {}
			o.request = false
			o:CreateResourceRequests()
			local new_req = o.request
			for _, c in ipairs(centers) do
				pcall(function() c:AddBuilding(o) end)
			end
			if new_req and new_req ~= old_req then
				C47.brake.rebuilt = C47.brake.rebuilt + 1
				local new_amt = new_req:GetActualAmount()
				if old_amt and new_amt and old_amt ~= new_amt then
					C47.brake.amount_mismatch = C47.brake.amount_mismatch + 1
					C47.brake.amount_delta = C47.brake.amount_delta + (new_amt - old_amt)
				end
			else
				if new_req == old_req then C47.brake.identity_unchanged = C47.brake.identity_unchanged + 1 end
				C47.brake.failed = C47.brake.failed + 1
			end
		end)
		if not ok_one then C47.brake.failed = C47.brake.failed + 1 end
		if i % 250 == 0 then Sleep(1) end
	end
	local b = C47.brake
	local took = b.total > 0 and (b.rebuilt * 100 / b.total) >= 90
	say(fmt("BRAKE REBUILD :: total=%d rebuilt=%d failed=%d skipped=%d | request identity unchanged=%d | old-vs-new amount mismatches=%d (net Δ %+d seeds — soil drift between creation times, COUNTED not absorbed) | ⭐ INTERVENTION TOOK=%s (bar: ≥90%% rebuilt)",
		b.total, b.rebuilt, b.failed, b.skipped, b.identity_unchanged,
		b.amount_mismatch, b.amount_delta, tostring(took)))
	C47.VegPeek("after brake")
	say("BRAKE ⚠️ RemoveBuilding idles any drone that was heading to a rebuilt requester (DroneControl.lua:720-728) — a one-time transient, which is what the settle gap before window 2 exists to absorb.")
	return took
end

function C47.RestoreBrake()
	local classes = rawget(_G, "g_Classes")
	local cls = classes and classes.VegetationTaskRequester
	if cls and C47.brake then
		cls.supply_dist_modifier = C47.brake.before
		say(fmt("BRAKE class value RESTORED to %s — live requests created under 150 keep it until this process exits; nothing persists (this leg never saves)", tostring(cls.supply_dist_modifier)))
	end
end

---- report --------------------------------------------------------------------

function C47.Report()
	say(fmt("REPORT SPAN :: samples=%d at %d ms | frozen at=%s | windows=%d",
		C47.samples, C47.sample_ms, tostring(C47.frozen_at or "not frozen"),
		#(C47.windows or empty_table)))
	for _, w in ipairs(C47.windows or empty_table) do
		for _, f in ipairs(w.farms) do
			local win = f.win
			local sizes, parts, exact280, bulk = {}, {}, 0, 0
			for s, c in pairs(win.sizes) do sizes[#sizes + 1] = { s = s, c = c } end
			table.sort(sizes, function(a, b) return a.s < b.s end)
			for _, e in ipairs(sizes) do
				parts[#parts + 1] = fmt("%dx%d", e.s, e.c)
				if e.s == 280 then exact280 = exact280 + e.c end
				if e.s >= 2000 then bulk = bulk + e.c end
			end
			local hours = (w.end_gt - w.start_gt) / C47.MS_PER_HOUR
			say(fmt("REPORT WINDOW '%s' %s :: complete=%s | game %d ms (%s h) | buffer min=%s max=%s | stalled %d of %d | deliveries=%d totalling %d (mean %s) | ⭐ exactly-280=%d (%s%% of deliveries) | ⭐ bulk ≥2000=%d | drains=%d totalling %d (per game hour %s)",
				tostring(w.name), f.id, tostring(w.complete),
				w.end_gt - w.start_gt, tostring(hours),
				tostring(win.min), tostring(win.max), win.stalled, win.n,
				win.deliveries, win.delivered,
				win.deliveries > 0 and tostring(win.delivered / win.deliveries) or "n/a",
				exact280, win.deliveries > 0 and tostring(exact280 * 100 / win.deliveries) or "n/a",
				bulk, win.drains, win.drain_total,
				hours > 0 and tostring(win.drain_total / hours) or "n/a"))
			say(fmt("REPORT WINDOW '%s' %s :: ⭐ DELIVERY SIZE DISTRIBUTION (size x count): %s — the 280-wall collapsing and/or ≥2000 arrivals appearing in the braked window is the discriminator; ⚠️ a single 2000-2800 value can also be a Broadleaf offer (2000 × soil), so the wall share and the depot totals carry the verdict together",
				tostring(w.name), f.id, #parts > 0 and table.concat(parts, ", ") or "(none sampled)"))
		end
		depot_delta("window '" .. tostring(w.name) .. "' (re-emitted)", w.depot_open, w.depot_close)
	end
	say("REPORT ⛔ NOT CLAIMED: a drone's DECISION is still unread (the pairing is C-side). What this leg reads is whether the developers' own brake number CHANGES the routing signature — delivery sizes and depot totals — under an intervention, on this one save.")
	say("REPORT ⚠️ CONFOUNDS: cheats enabled, oversized terraforming speed-run colony, drone carry dial at +2 (a trip is 3000, not the vanilla 1000), one save, one map, windows ordered in time (baseline first).")
	return true
end

---- the leg body --------------------------------------------------------------

function C47.Boot()
	CreateRealTimeThread(function()
		Sleep(1)
		say("=== c48-brake C48B === payload loaded. Staged save: " .. STAGED_SAVE)
		CreateRealTimeThread(function()
			local deadline = RealTime() + C47.watchdog
			while C47.stage ~= "done" do
				if RealTime() - deadline >= 0 then
					say("TIMEOUT " .. tostring(C47.stage))
					pcall(C47.Report)
					pcall(C47.RestoreBrake)
					local f = rawget(_G, "FlushLogFile"); if f then f() end
					Sleep(1000) quit() return
				end
				Sleep(1000)
			end
		end)
		if rawget(_G, "ShowStartGamePopup") then ShowStartGamePopup = function() end end
		C47.stage = "waiting-menu"
		if not C47.WaitFor(5 * 60 * 1000, function() return not not GetPreGameMainMenu() end) then
			say("TIMEOUT waiting-menu") C47.stage = "done" Sleep(2000) quit() return
		end
		Sleep(2000)
		C47.ReadConditions("main menu, before the load")
		C47.stage = "loading"
		local ok, res = C47.TryYield("load " .. STAGED_SAVE, function()
			return C47.Load(STAGED_SAVE, C47.speed)
		end)
		if not (ok and res) then
			say("⛔ LOAD FAILED or the speed did not take — nothing below this line exists.")
			C47.ErrorWatchNote() C47.stage = "done" Sleep(2000) quit() return
		end
		C47.ReadConditions("after the load")
		C47.stage = "census"
		local ok_c, res_c = C47.TryYield("leg Census", C47.Census)
		if not (ok_c and res_c) then
			C47.ErrorWatchNote() C47.stage = "done" Sleep(2000) quit() return
		end
		C47.stage = "baseline"
		C47.TryYield("window baseline", function() return C47.RunWindow("baseline") end)
		C47.stage = "intervening"
		local ok_b, took = C47.TryYield("leg ApplyBrake", C47.ApplyBrake)
		if not (ok_b and took) then
			say("⛔ THE INTERVENTION DID NOT FULLY TAKE — window 2 still runs and is reported, but it is PARTIAL/VOID as an intervention reading; the verdict lines must say so.")
		end
		C47.stage = "settling"
		local settle_target = GameTime() + C47.settle_gt
		while GameTime() < settle_target do Sleep(200) end
		say(fmt("SETTLE done :: %d game ms absorbed (drones idled by the rebuild have re-tasked)", C47.settle_gt))
		C47.stage = "braked"
		C47.TryYield("window braked", function() return C47.RunWindow("braked") end)
		C47.stage = "reporting"
		C47.Try("C47.Report", C47.Report)
		C47.Try("C47.RestoreBrake", C47.RestoreBrake)
		C47.ReadConditions("at close")
		C47.Now("close")
		C47.ErrorWatchNote()
		say("=== c48-brake C48B === DONE. ⛔ Nothing was saved: this run never calls SaveGame, so the staged copy on disk is byte-identical to the one staged before the launch.")
		C47.stage = "done"
		Sleep(3000)
		quit()
	end)
end

say("[C48B] 98_C48Brake loaded — self-driving in 1 tick.")
C47.Boot()
