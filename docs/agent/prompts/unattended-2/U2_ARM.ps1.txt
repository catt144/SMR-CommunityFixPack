# unattended-2 ARM / DISARM script.  Parked source; copy to a scratch .ps1 and run.
#
# Rule C11: arming is a script FILE, never an inline one-liner - an inline edit's
# quoting was mangled once and the game launched unarmed.
# C11 corollary (unattended-1 I2): never pipe this script's output through
# Select-Object - a piped consumer terminates the upstream pipeline and can kill
# the script before its write executes. Nothing here is piped.
#
# G5 ARM GATE (binding): after writing, EVERY piece is read BACK OFF DISK and the
# script exits non-zero if any is missing. A launch never happens on the word of
# the writer. Cost of the whole failure class it removes: 8 min -> 0.2 s.
#
# S3 (corun-batch-1 ledger): a parked .ps1 needs a BOM, because PS 5.1 reads a
#    no-BOM .ps1 as ANSI and any non-ASCII character breaks the parse. This body
#    is kept ASCII-only as well, belt and braces.
#
# ENCODING: PowerShell 5.1's Set-Content -Encoding utf8 writes a BOM and once
#    silently prefixed one onto metadata.lua. Use WriteAllLines with an explicit
#    no-BOM encoder for metadata.lua, as below.
#
#   Usage:  .\U2_ARM.ps1 arm            # 97_U2Common + 98_U2Run, REAL run
#           .\U2_ARM.ps1 arm-rehearsal  # same, but flips REHEARSAL to true
#           .\U2_ARM.ps1 disarm         # deletes both, restores metadata, sweeps
#
# arm-rehearsal exists so the parked source stays authoritative at REHEARSAL =
# false (the setting the chain's acceptance needs) while a deliberate
# instrument-proving run can still be armed by a SCRIPT rather than a hand edit
# (C11). The flip is read back off disk and printed, and the run's own MODE
# banner repeats it in the log, so a rehearsal can never be mistaken for a
# measurement after the fact.

param([Parameter(Mandatory = $true)][ValidateSet("arm", "arm-rehearsal", "disarm")][string]$mode)

$ErrorActionPreference = "Stop"
$park = "c:\Dev\SMR-BugFixPack\docs\agent\prompts\unattended-2"
$kit  = "c:\Dev\SMR-BugFixPack-TestKit"
$meta = Join-Path $kit "metadata.lua"
$noBom = New-Object System.Text.UTF8Encoding($false)

$common = @{ src = "97_U2Common.lua.txt"; dst = "Code\97_U2Common.lua" }
$run    = @{ src = "98_U2Run.lua.txt";    dst = "Code\98_U2Run.lua" }
$pairs  = @($common, $run)

$arming = ($mode -ne "disarm")
$rehearsal = ($mode -eq "arm-rehearsal")

if ($arming) {
  foreach ($p in $pairs) {
    Copy-Item -LiteralPath (Join-Path $park $p.src) -Destination (Join-Path $kit $p.dst) -Force
    Write-Output ("WROTE " + $p.dst)
  }
  if ($rehearsal) {
    $rf = Join-Path $kit "Code\98_U2Run.lua"
    $rt = [System.IO.File]::ReadAllText($rf)
    $rt2 = $rt.Replace("local REHEARSAL = false", "local REHEARSAL = true")
    if ($rt2 -eq $rt) { Write-Output "ARM FAIL: could not find 'local REHEARSAL = false' to flip"; exit 1 }
    [System.IO.File]::WriteAllText($rf, $rt2, $noBom)
    Write-Output "WROTE 98_U2Run.lua with REHEARSAL flipped to true"
  }
  $lines = Get-Content -LiteralPath $meta -Encoding UTF8
  $out = New-Object System.Collections.Generic.List[string]
  $inserted = $false
  foreach ($l in $lines) {
    $out.Add($l)
    if (-not $inserted -and $l -match '99_FixtureCarry\.lua') {
      foreach ($p in $pairs) {
        $leaf = Split-Path $p.dst -Leaf
        $out.Add(("`t`t`"Code/" + $leaf + "`",  -- TEMPORARY unattended-2"))
      }
      $inserted = $true
    }
  }
  if (-not $inserted) { Write-Output "ARM FAIL: anchor 99_FixtureCarry.lua not found"; exit 1 }
  [System.IO.File]::WriteAllLines($meta, $out, $noBom)
  Write-Output "WROTE metadata.lua (no BOM)"
} else {
  foreach ($p in $pairs) {
    $f = Join-Path $kit $p.dst
    if (Test-Path -LiteralPath $f) { Remove-Item -LiteralPath $f -Force; Write-Output ("DELETED " + $p.dst) }
  }
  $lines = Get-Content -LiteralPath $meta -Encoding UTF8
  $kept = $lines | Where-Object { $_ -notmatch 'Code/9[78]_U2' }
  [System.IO.File]::WriteAllLines($meta, $kept, $noBom)
  Write-Output ("metadata.lua: removed " + ($lines.Count - $kept.Count) + " line(s), no BOM")
}

# ---- gate (both directions) --------------------------------------------------
$fail = $false
$m = Get-Content -LiteralPath $meta -Raw -Encoding UTF8
foreach ($p in $pairs) {
  $f = Join-Path $kit $p.dst
  $present = Test-Path -LiteralPath $f
  $listed  = $m -match [regex]::Escape('"Code/' + (Split-Path $p.dst -Leaf) + '"')
  if ($arming) {
    if (-not $present) { Write-Output ("ARM GATE FAIL: missing " + $p.dst); $fail = $true }
    elseif ((Get-Content -LiteralPath $f -Raw -Encoding UTF8) -notmatch 'TEMPORARY') {
      Write-Output ("ARM GATE FAIL: no TEMPORARY marker in " + $p.dst); $fail = $true
    } else { Write-Output ("ARM GATE OK   " + $p.dst) }
    if (-not $listed) { Write-Output ("ARM GATE FAIL: metadata does not list " + $p.dst); $fail = $true }
    else { Write-Output ("ARM GATE OK   metadata lists " + $p.dst) }
  } else {
    if ($present) { Write-Output ("DISARM FAIL: still present " + $p.dst); $fail = $true }
    if ($listed)  { Write-Output ("DISARM FAIL: metadata still lists " + $p.dst); $fail = $true }
  }
}

# The run payload MUST drive itself - this is unattended and nobody will type a
# console line. The shared harness must NOT, or two flows would race.
if ($arming) {
  $runText = Get-Content -LiteralPath (Join-Path $kit "Code\98_U2Run.lua") -Raw -Encoding UTF8
  if ($runText -match '(?m)^\s*U2\.Boot\(\)') { Write-Output "ARM GATE OK   98_U2Run drives itself (unattended by design)" }
  else { Write-Output "ARM GATE FAIL: 98_U2Run does not call U2.Boot() - the run would launch and do nothing"; $fail = $true }
  # Read the mode back OFF DISK. A run armed as a rehearsal must say so here and
  # a real run must say so here; neither may be inferred from the command typed.
  if ($runText -match '(?m)^\s*local REHEARSAL = true') {
    Write-Output "ARM GATE OK   MODE ON DISK = HARNESS REHEARSAL (every verdict in that run is VOID by construction)"
    if (-not $rehearsal) { Write-Output "ARM GATE FAIL: 'arm' was requested but the armed file says REHEARSAL = true"; $fail = $true }
  } elseif ($runText -match '(?m)^\s*local REHEARSAL = false') {
    Write-Output "ARM GATE OK   MODE ON DISK = REAL VERIFICATION RUN (stops at the gate if the pack is not loaded)"
    if ($rehearsal) { Write-Output "ARM GATE FAIL: 'arm-rehearsal' was requested but the armed file says REHEARSAL = false"; $fail = $true }
  } else { Write-Output "ARM GATE FAIL: cannot read the REHEARSAL mode off disk"; $fail = $true }
  $comText = Get-Content -LiteralPath (Join-Path $kit "Code\97_U2Common.lua") -Raw -Encoding UTF8
  if ($comText -match '(?m)^\s*U2\.Boot\(\)') { Write-Output "ARM GATE FAIL: 97_U2Common calls U2.Boot() at file scope - two flows would race"; $fail = $true }
  else { Write-Output "ARM GATE OK   97_U2Common starts nothing on its own" }

  # G1 RESOLUTION CROSS-CHECK (unattended-1 harness-defect class 1). A parse
  # sweep is a SYNTAX verdict, not a resolution one: U1.ErrorWatchNote was
  # called by all 7 payloads and never existed. Diff the U2.* names USED by the
  # payload against the names DEFINED by the harness.
  $defined = @{}
  foreach ($mm in [regex]::Matches($comText, '(?m)^\s*(?:function\s+U2\.(\w+)|U2\.(\w+)\s*=)')) {
    $n = $mm.Groups[1].Value; if (-not $n) { $n = $mm.Groups[2].Value }
    if ($n) { $defined[$n] = $true }
  }
  $missing = @()
  foreach ($mm in [regex]::Matches($runText, 'U2\.(\w+)\s*\(')) {
    $n = $mm.Groups[1].Value
    if (-not $defined.ContainsKey($n) -and $n -ne "Payload") { $missing += $n }
  }
  $missing = $missing | Sort-Object -Unique
  if ($missing.Count -gt 0) { Write-Output ("ARM GATE FAIL: 98_U2Run calls undefined U2 names: " + ($missing -join ", ")); $fail = $true }
  else { Write-Output ("ARM GATE OK   resolution cross-check: every U2.* the payload calls is defined (" + $defined.Count + " definitions)") }
}

# stale-probe sweep, BOTH repos (WORKFLOW probe hygiene rule 4)
$hits = @()
foreach ($dir in @("c:\Dev\SMR-BugFixPack\Code", "c:\Dev\SMR-BugFixPack-TestKit\Code")) {
  $hits += (Select-String -Path (Join-Path $dir "*.lua") -Pattern "TEMPORARY" -List | ForEach-Object { $_.Path })
}
if ($arming) { Write-Output ("PROBE SWEEP: armed: " + (($hits | ForEach-Object { Split-Path $_ -Leaf }) -join ", ") + " - declared by unattended-2") }
elseif ($hits.Count -eq 0) { Write-Output "PROBE SWEEP: clean" }
else { Write-Output ("PROBE SWEEP: ARMED AFTER DISARM - " + ($hits -join ", ")); $fail = $true }

if ($fail) { Write-Output ($mode.ToUpper() + " GATE: RED"); exit 1 }
Write-Output ($mode.ToUpper() + " GATE: GREEN")
exit 0
