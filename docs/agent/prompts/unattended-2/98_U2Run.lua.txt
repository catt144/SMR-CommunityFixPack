-- Declared by: unattended-2 chain prompt 1 (01_OPUS_RUN.md), Job 2. Armed
-- together with 97_U2Common.lua; THIS file is what starts the run.
--
-- Probe hygiene rule 5: present in TestKit Code/ only while the run is actually
-- happening; it and its metadata.lua line die in the commit that records the
-- answers. This .lua.txt is the source of truth and is itself deleted at chain
-- close (it survives in git).
--
-- ⛔ WHAT IT MAY DO. It loads a STAGED COPY, reads, calls the pack's own
--    sanitizer entry points, saves twice under names of its own, and runs the
--    probe suite. It NEVER touches PT35FIXTURE.savegame.sav itself — the chain's
--    protected standing fixture — and it never passes save_as_last. Nothing is
--    forced in the world: no cheat, no meteor, no research grant, no popup. The
--    only world change is GAME SPEED, which C4 makes mandatory for a live read.
--
-- ⛔ FORCED vs ORGANIC (chain rule 11). EVERYTHING here is FORCED/STAGED. The
--    staged copy is forced, the sanitizer calls are forced, the saves are
--    forced. No organic claim is available from this run and no `tested` grant
--    is available from it either — the unattended ceiling is MECHANISM.
--
-- WHAT EACH LOAD IS FOR:
--   load 1  U2STAGE  — a fresh copy of the fixture, which this build's F48 pass
--                      has never seen. The PostLoadGame handler fires during
--                      LoadGame, so the pass's own per-track before/after lines
--                      are already in the log when step 5 starts reading. That
--                      is F48 acceptance (a)-first-half and (c).
--   load 2  U2RT1    — the round trip. The flag is now in the save, so the
--                      automatic pass must stand down (READ, not inferred), the
--                      track signatures must be identical to load 1's, and a
--                      DIRECT call must repair zero — acceptance (a)-second-half
--                      and (b).
--   load 3  U2RT2    — R4 round trip number two for PT-35 leg A, then the C43
--                      suite run last, so a probe cannot contaminate a reading
--                      taken before it.

U2.tag      = "U2RUN"
U2.watchdog = 25 * 60 * 1000

-- ⛔⛔ THE ONE SWITCH IN THIS FILE, AND IT IS DECLARED IN THE BANNER.
--
--   false = the REAL run. If the Community Fix Pack is not loaded, the run
--           STOPS at the gate rather than banking readings about code that
--           never executed. This is the setting the chain's acceptance needs.
--   true  = a HARNESS REHEARSAL. The run proceeds with the pack absent, every
--           verdict-bearing line is stamped VOID, and its log may be quoted for
--           the flow working and for NOTHING ELSE.
--
-- Run 1 (2026-08-11 01:17) had no gate at all and was voided by a disabled
-- pack six steps in. Run 2 was the rehearsal that proved the corrected save
-- names and the three-load flow. The re-run the owner unblocks sets this false.
local REHEARSAL = false

-- ⛔ FULL FILENAMES. The engine writes `savename` VERBATIM and appends nothing:
--    run 1 asked for "U2RT1", got a file called `U2RT1`, and could not load it
--    back. U2.Save now guards this as well; both belts are deliberate.
local STAGE = "U2STAGE.savegame.sav"
local RT1   = "U2RT1.savegame.sav"
local RT2   = "U2RT2.savegame.sav"

local say = U2.Say
local fmt = U2.Fmt

-- banked across loads
local S1, S1b, S2, S3          -- track signature sets
local N1, N1b, N2, N2b, N3, N3b -- sanitizer reading sets

function U2.Payload()
	----------------------------------------------------------------- load 1 --
	if not U2.Step("2-load-staged-copy", function()
		return U2.Load(STAGE, 3)
	end, "untimed") then return end

	if not U2.Step("3-conditions-and-gate", function()
		say(fmt("MODE %s", REHEARSAL
			and "HARNESS REHEARSAL — this run proves the load/save/reload flow and NOTHING about F48, C43, F100 or PT-35. Every verdict below is void by construction."
			or "REAL VERIFICATION RUN — the gate below stops the run if the pack is not loaded."))
		-- ⚠️ REGISTERED ids, not file names (corun-batch-1 asked for
		-- "Opt_CohortHousing" and got NOT REGISTERED beside "CohortHousing").
		U2.ReadConditions("unattended-2 verification run, load 1 of 3, staged copy of PT35FIXTURE",
			{ "SaveSanitizer", "NoHomeless" })
		say("F100 ⛔ the acceptance reading for F100 is NOT taken here — the deactivation line prints at mod load, long before this payload runs, and an absence verdict may only come from the FINAL archived log (EF-047). The line above records what the module's status is NOW; the boot lines are read agent-side after this process exits.")
		return U2.RequirePackLoaded("load 1, before any verdict-bearing read", REHEARSAL)
	end) then return end

	U2.Step("4-fixture-confirm", function()
		U2.ReadTurbineFixture("load 1, staged copy")
		U2.ReadUpgradeFixture("load 1, staged copy")
	end)

	U2.Step("5-f48-after-the-automatic-pass", function()
		U2.ReadF48Flag("load 1, after PostLoadGame")
		S1 = U2.SnapAllTracks("load1-after-auto-pass")
		say("F48 the automatic pass has already run (PostLoadGame fires inside LoadGame). Its own per-track 'connections A -> B' lines and its 'repaired N of M track(s) walked' line are ABOVE this one — that pair is acceptance (c) and the first half of (a).")
		U2.RunF48("load1-immediate-rerun (idempotence, PRE-RELOAD)")
		S1b = U2.SnapAllTracks("load1-after-direct-rerun")
		U2.DiffNumbers(S1, S1b, "F48 idempotence within load 1 (PRE-RELOAD ONLY — the round trip is load 2)")
	end)

	U2.Step("6-pt35-leg-a-load-1", function()
		N1 = U2.ReadSanitizerNumbers("load1-before-heal-calls")
		U2.RunHealFamily("load 1")
		N1b = U2.ReadSanitizerNumbers("load1-after-heal-calls")
		U2.DiffNumbers(N1, N1b, "PT-35 leg A do-no-harm, load 1, both passes called (PRE-RELOAD ONLY)")
	end)

	U2.Step("7-save-round-trip-1", function()
		return U2.Save(RT1)
	end, "untimed")

	----------------------------------------------------------------- load 2 --
	if not U2.Step("8-load-round-trip-1", function()
		return U2.Load(RT1, 3)
	end, "untimed") then return end

	U2.Step("9-f48-across-the-round-trip", function()
		U2.ReadConditions("load 2 of 3, the R4 round trip", { "SaveSanitizer" })
		U2.ReadF48Flag("load 2, after PostLoadGame — a TRUE here is the automatic pass standing down, read rather than inferred from an absence of lines")
		S2 = U2.SnapAllTracks("load2-after-reload")
		U2.DiffNumbers(S1b, S2, "F48 acceptance (a) SECOND HALF — does the repaired state SURVIVE save+reload")
	end)

	U2.Step("10-f48-do-no-harm-on-a-clean-save", function()
		U2.RunF48("load2-do-no-harm (acceptance (b): an already-repaired save must report ZERO)")
		local after = U2.SnapAllTracks("load2-after-direct-call")
		U2.DiffNumbers(S2, after, "F48 acceptance (b) — a direct call on an already-clean save must move nothing")
	end)

	U2.Step("11-pt35-leg-a-round-trip-1", function()
		N2 = U2.ReadSanitizerNumbers("load2-before-heal-calls")
		U2.DiffNumbers(N1b, N2, "PT-35 leg A ACROSS SAVE+RELOAD #1 (R4) — the entry's own claim is that these do not move")
		U2.RunHealFamily("load 2")
		N2b = U2.ReadSanitizerNumbers("load2-after-heal-calls")
		U2.DiffNumbers(N2, N2b, "PT-35 leg A do-no-harm, load 2, both passes called")
		U2.ReadTurbineFixture("load 2, after the round trip")
		U2.ReadUpgradeFixture("load 2, after the round trip")
	end)

	U2.Step("12-save-round-trip-2", function()
		return U2.Save(RT2)
	end, "untimed")

	----------------------------------------------------------------- load 3 --
	if not U2.Step("13-load-round-trip-2", function()
		return U2.Load(RT2, 3)
	end, "untimed") then return end

	U2.Step("14-pt35-leg-a-round-trip-2", function()
		U2.ReadConditions("load 3 of 3, the second R4 round trip", { "SaveSanitizer" })
		U2.ReadF48Flag("load 3, after PostLoadGame")
		S3 = U2.SnapAllTracks("load3-after-second-reload")
		U2.DiffNumbers(S2, S3, "F48 — the repaired state across a SECOND save+reload")
		N3 = U2.ReadSanitizerNumbers("load3-before-heal-calls")
		U2.DiffNumbers(N2b, N3, "PT-35 leg A ACROSS SAVE+RELOAD #2 (R4)")
		U2.RunHealFamily("load 3")
		N3b = U2.ReadSanitizerNumbers("load3-after-heal-calls")
		U2.DiffNumbers(N3, N3b, "PT-35 leg A do-no-harm, load 3, both passes called")
		U2.DiffNumbers(N1, N3b, "PT-35 leg A START -> FINISH (three loads, two round trips, six pass calls)")
		U2.ReadTurbineFixture("load 3, final")
		U2.ReadUpgradeFixture("load 3, final")
	end)

	----------------------------------------------------------------- C43 -----
	-- LAST, deliberately: RunAll drives 87 probes with synthetic input, and a
	-- reading taken after it would be a reading taken after the probes.
	U2.Step("15-c43-full-suite", function()
		local t = rawget(_G, "SMRTest")
		if type(t) ~= "table" or type(t.RunAll) ~= "function" then
			say("C43 NOT-APPLICABLE (SMRTest.RunAll missing — the Test Kit is not loaded)")
			return
		end
		say("C43 ⛔ the acceptance reading is ZERO engine error lines naming Test Kit code in the FINAL archived log, both refusal-carrying probes reporting SKIP, and every other probe's verdict unchanged against the retail baseline (docs/archive/logs/Mars.exe-20260803-22.23.59-6a22b86d.log, 78 PASS / 0 FAIL / 9 SKIP / 0 ERROR over 87 probes). Absence is NOT read from here (EF-047).")
		U2.Try("SMRTest.RunAll()", function()
			local counts = t.RunAll()
			return type(counts) == "table"
				and string.format("PASS=%d FAIL=%d SKIP=%d ERROR=%d",
					counts.PASS or -1, counts.FAIL or -1, counts.SKIP or -1, counts.ERROR or -1)
				or tostring(counts)
		end)
	end)

	U2.ErrorWatchNote()
end

U2.Boot()
