---- output ------------------------------------------------------------------
--
-- ModLog is the only print path proven to reach the log file on disk
-- (Mod.lua:109-132). Its emitter formats its single argument a second time
-- (lib.lua:164-174), so a literal '%' raises there — escape it. Flush per line
-- so a killed run still shows how far it got.
local function say(msg)
	msg = "[" .. tostring(U2.tag) .. "] " .. tostring(msg)
	local modlog = rawget(_G, "ModLog")
	if modlog then
		modlog((msg:gsub("%%", "%%%%")))
	else
		local dbg = rawget(_G, "DebugPrint")
		if dbg then dbg(msg .. "\n") end
	end
	print(msg)
	local flush = rawget(_G, "FlushLogFile")
	if flush then pcall(flush) end
end
U2.Say = say

local function fmt(...)
	local ok, s = pcall(string.format, ...)
	return ok and s or "<fmt error>"
end
U2.Fmt = fmt

local function id_of(obj)
	if not obj then return "nil" end
	if type(obj) ~= "table" then return tostring(obj) end
	return tostring(obj.class or "?") .. "#" .. tostring(rawget(obj, "handle"))
end
U2.IdOf = id_of

-- G3. The ONLY sanctioned call wrapper. Prints ok and the error text, always.
-- ⛔ pcall cannot survive a yield — use U2.TryYield for anything that Sleeps.
function U2.Try(what, fn, ...)
	local ok, a, b = pcall(fn, ...)
	say(fmt("TRY %s ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

function U2.TryYield(what, fn, ...)
	local ok, a, b = sprocall(fn, ...)
	say(fmt("TRY %s (yielding) ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

-- G2. The condition-sampled rule, mechanised.
function U2.Applicable(label, count, population, witness)
	local applicable = (population or 0) > 0
	say(fmt("READ %s = %s | population=%s | APPLICABLE=%s | witness=%s",
		tostring(label), tostring(count), tostring(population),
		tostring(applicable), tostring(witness or "-- NONE NAMED --")))
	return applicable
end

---- flow helpers ------------------------------------------------------------

local function shutdown(reason)
	U2.stage = "done"
	say("SHUTDOWN " .. tostring(reason))
	local flush = rawget(_G, "FlushLogFile")
	if flush then flush() end
	Sleep(U2.quit_delay)
	quit()
end
U2.Shutdown = shutdown

-- untimed = true suppresses the END timing line entirely (EF-045 correction C8):
-- a step that spans a load or a save blocks render mode and under-reports itself
-- by up to 11.5x. Loads take their number from the engine's own line.
local function step(name, fn, untimed)
	U2.stage = name
	local t = RealTime()
	say("STEP " .. name .. " BEGIN")
	local ok, res = sprocall(fn)
	if not ok then
		say("STEP " .. name .. " ERROR")
		shutdown("ERROR " .. name)
		return false
	end
	if untimed then
		say("STEP " .. name .. " END (no timing — this step blocks render mode)")
	else
		say(fmt("STEP %s END %dms", name, RealTime() - t))
	end
	if res == false then
		shutdown("STOP " .. name .. " — the step asked to stop; its reason is the line above")
		return false
	end
	return true
end
U2.Step = step

local function wait_for(timeout, ready)
	local deadline = RealTime() + timeout
	while not ready() do
		if RealTime() - deadline >= 0 then return false end
		Sleep(500)
	end
	return true
end
U2.WaitFor = wait_for

local function game_is_live()
	local city   = rawget(_G, "MainCity")
	local colony = rawget(_G, "UIColony")

end

---- load / save -------------------------------------------------------------

function U2.Load(savefile, speed)
	local err = LoadGame(savefile, {})            -- ⛔ never pass save_as_last
	if err then
		say("LOADERR " .. tostring(savefile) .. " " .. tostring(err))
		return false
	end
	U2.loads = U2.loads + 1
	say(fmt("LOAD OK #%d file=%s — LoadGame returned no error", U2.loads, tostring(savefile)))
	if not wait_for(5 * 60 * 1000, game_is_live) then
		say("LOADERR " .. tostring(savefile) .. " — city never came live")
		return false
	end
	Sleep(U2.settle_time)                          -- 15 s is the MEASURED datum
	speed = speed or 3
	local before = GetTimeFactor() / const.DefaultTimeFactor
	local ok = pcall(function() UIColony:SetGameSpeed(speed) end)
	Sleep(500)
	local after = GetTimeFactor() / const.DefaultTimeFactor
	say(fmt("READ speed: SetGameSpeed(%d) call=%s | before=%s after=%s",
		speed, tostring(ok), tostring(before), tostring(after)))
	return true
end

-- ⛔ save_as_last is NEVER passed — it would repoint the owner's Continue button.
-- ⛔ G4: the witness is the on-disk file (read agent-side, game closed) plus the
--    LoadGame that follows in this very run. ListForTag is printed and is NOT
--    the witness (EF-049).
function U2.Save(name)
	local landed = false
	U2.TryYield("SaveGame " .. tostring(name), function()
		local err, nm = SaveGame(name, { savename = name, silent = true, no_screenshot = true })
		say(fmt("SAVE SaveGame returned err=%s name=%s", tostring(err), tostring(nm)))
		landed = not err
		return err
	end)
	Sleep(1000)
	local sg = rawget(_G, "Savegame")
	if sg then
		U2.Try("Savegame.ListForTag (EF-049 open question, NOT the witness)", function()
			local list = sg.ListForTag("savegame")
			return type(list) == "table" and #list or ("non-table:" .. type(list))
		end)
	end
	say("SAVE WITNESS for " .. tostring(name)
		.. " = the on-disk file read agent-side after this process exits, plus the LoadGame of it below. NOT ListForTag (EF-049).")
	return landed
end

--
-- WORKFLOW EXTERNAL VALIDITY rule: the pack count is READ, never assumed.
function U2.ReadConditions(note, named)
	local pack = rawget(_G, "SMRFixPack")
	local active, total, inactive = 0, 0, {}
	if pack and pack.order then
		for _, id in ipairs(pack.order) do
			total = total + 1
			local f = pack.fixes and pack.fixes[id]
			local st = f and f.status or "missing"
			if st == "active" then active = active + 1 else inactive[#inactive + 1] = id .. "=" .. st end
		end
	end
	say(fmt("CONDITIONS pack=%d/%d active | map=%s | speed=%s | loads_this_process=%d | note=%s",
		active, total, tostring(rawget(_G, "CurrentMapName")),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), U2.loads, tostring(note or "-")))
	if #inactive > 0 then
		say("CONDITIONS not-active: " .. table.concat(inactive, ", "))
	end
	for _, id in ipairs(named or empty_table) do
		local f = pack and pack.fixes and pack.fixes[id]
		say(fmt("CONDITIONS by-name %s = %s | detail=%s", tostring(id),
			tostring(f and f.status or "NOT REGISTERED"), tostring(f and f.detail or "-")))
	end
	return active, total
end

-- The F03 half's fixture. ⛔ The label name is NOT guessed: the finder walks
-- every label of every city and reports whatever carries a non-empty
-- `upgrades_built` (Building:HasUpgrade reads that table, Building.lua). A
-- guessed class name that resolves to nothing prints a plausible zero forever
-- (corun-batch-2 ledger, three reader defects in one function).
function U2.ReadUpgradeFixture(when)
	local seen, upgraded, examples = {}, 0, {}
	for _, city in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, list in pairs(city.labels or empty_table) do
			if type(list) == "table" then
				for _, o in ipairs(list) do
					if type(o) == "table" and not seen[o] then
						seen[o] = true
						local built = rawget(o, "upgrades_built")
						if type(built) == "table" and next(built) then
							upgraded = upgraded + 1
							if #examples < 4 then
								local ids = {}
								for id, v in sorted_pairs(built) do
									if v then ids[#ids + 1] = tostring(id) end
								end
								examples[#examples + 1] = fmt("%s{%s}", id_of(o), table.concat(ids, ","))
							end
						end
					end
				end
			end
		end
	end
	say(fmt("PT35 upgraded buildings=%d | examples: %s | when=%s",
		upgraded, (#examples > 0 and table.concat(examples, " ") or "<none>"), tostring(when)))
	U2.Applicable("PT35-F03-half-fixture(an APPLIED building upgrade exists)", upgraded, upgraded,
		"every label of every city walked this load, upgrades_built read per object — the fixture's Remote Medic is the intended subject")
	return upgraded
end

---- THE SANITIZER DO-NO-HARM READING SET (PT-35 leg A) -----------------------
--
-- ⛔ Every number here is one the sanitizer's passes could plausibly move. The
--    set is taken as an ORDERED list so two takings can be compared entry by
--    entry and the comparison prints WHICH entry moved, not just how many.
--
-- Reading 3 deliberately reads the Large turbine's electricity_production RAW
-- (both the own-field and the resolved value) rather than a computed figure: a
-- nil must be VISIBLE as nil, because a reader that silently zeroes cannot tell
-- "unchanged" from "never read anything" (G3).
local UPGRADE_MOD_ID = "^(%d+)_upgrade%d+_mod_%d+$"

local function count_label_mods(container)
	local by_label = type(container) == "table" and container.label_modifiers
	if type(by_label) ~= "table" then return 0, 0 end
	local total, upgrade_shaped = 0, 0
	for _, modifiers in pairs(by_label) do
		for id in pairs(modifiers or empty_table) do
			total = total + 1
			if type(id) == "string" and string.match(id, UPGRADE_MOD_ID) then
				upgrade_shaped = upgrade_shaped + 1
			end
		end
	end
	return total, upgrade_shaped
end

function U2.ReadSanitizerNumbers(when)
	local out = {}
	local function put(name, value) out[#out + 1] = { name = name, value = tostring(value) } end

	local colony = rawget(_G, "UIColony")

	-- 1-2: the three turbine labels the F35 pass is driven by
	local tmods, tpercent = 0, 0
	for _, label in ipairs({ "WindTurbine", "WindTurbine_Large", "WindTurbine_Diffuser" }) do
		for id, mod in pairs(((colony or empty_table).label_modifiers or empty_table)[label] or empty_table) do
			if type(mod) == "table" then
				tmods = tmods + 1
				tpercent = tpercent + (mod.percent or 0)
				say(fmt("PT35   turbine label %s carries id=%s prop=%s percent=%s amount=%s | when=%s",
					label, tostring(id), tostring(mod.prop), tostring(mod.percent),
					tostring(mod.amount), tostring(when)))
			end
		end
	end
	put("turbine_label_modifiers", tmods)
	put("turbine_percent_sum", tpercent)

	-- 3: does it reach the building? read RAW.
	local large = label_all("WindTurbine_Large")
	local first = large[1]
	local own = first and rawget(first, "electricity_production")
	local resolved
	if first then local ok, v = pcall(function() return first.electricity_production end) resolved = ok and v or ("raised:" .. tostring(v)) end
	put("large_turbine_electricity_production_own", tostring(own))
	put("large_turbine_electricity_production_resolved", tostring(resolved))

	-- 4-6: the containers the F03 pass sweeps
	local ctotal, cupg = count_label_mods(colony)
	put("colony_label_modifiers", ctotal)
	put("colony_upgrade_shaped_ids", cupg)
	local citytotal, cityupg, domes, dometotal, domeupg = 0, 0, 0, 0, 0
	for _, city in ipairs(rawget(_G, "Cities") or empty_table) do
		local a, b = count_label_mods(city)
		citytotal, cityupg = citytotal + a, cityupg + b
		for _, dome in ipairs((city.labels or empty_table).Dome or empty_table) do
			domes = domes + 1
			local c, d = count_label_mods(dome)
			dometotal, domeupg = dometotal + c, domeupg + d
		end
	end
	put("city_label_modifiers", citytotal)
	put("city_upgrade_shaped_ids", cityupg)
	put("dome_count", domes)
	put("dome_label_modifiers", dometotal)
	put("dome_upgrade_shaped_ids", domeupg)

	-- 7: turbine population, so a changed reading cannot be blamed on a changed world
	put("turbines_large", #large)
	put("turbines_plain", #label_all("WindTurbine"))
	put("turbines_diffuser", #label_all("WindTurbine_Diffuser"))

	local parts = {}
	for _, r in ipairs(out) do parts[#parts + 1] = r.name .. "=" .. r.value end
	say(fmt("PT35NUM %s :: %s", tostring(when), table.concat(parts, " ")))

	-- the do-no-harm population, stated with what sampled it
	U2.Applicable("PT35-F03-do-no-harm-population(live upgrade-shaped modifier ids)",
		cupg + cityupg + domeupg, cupg + cityupg + domeupg,
		"colony + every city + every dome label_modifiers walked this load")
	return out
end

-- Compares two takings entry by entry and NAMES every entry that moved.
function U2.DiffNumbers(a, b, what)
	if type(a) ~= "table" or type(b) ~= "table" then
		say("PT35DIFF NOT-APPLICABLE (a reading set is missing) :: " .. tostring(what))
		return nil
	end
	local changed, n = {}, math.max(#a, #b)
	for i = 1, n do
		local ra, rb = a[i], b[i]
		local na = ra and ra.name or ("<missing#" .. i .. ">")
		if not ra or not rb or ra.name ~= rb.name or ra.value ~= rb.value then
			changed[#changed + 1] = fmt("%s: %s -> %s", na,
				ra and ra.value or "<absent>", rb and rb.value or "<absent>")
		end
	end
	say(fmt("PT35DIFF %s :: %d of %d readings changed%s", tostring(what), #changed, n,
		#changed > 0 and (" — " .. table.concat(changed, " | ")) or ""))
	return #changed
end

---- F48 — the track snapshot ------------------------------------------------
--
-- ⭐ SnapTrack is corun-batch-1's, verbatim in substance: node_idx is included
--    because the failure path forces start_el.node_idx = 1 (Tracks.lua:579)
--    while the success-path renumbering (:632-635) never runs — a DUPLICATE
--    node_idx is the residue signature. connections_total is the number PT-37
--    watched move 559 -> 558.
function U2.SnapTrack(track, when)
	if not track then say("SNAP NOT-APPLICABLE (no track)") return nil end
	local list = track.elements or empty_table
	local conn, dup, seen = 0, {}, {}
	for _, el in ipairs(list) do
		local n = el.node_idx
		if seen[n] then dup[#dup + 1] = tostring(n) end
		seen[n] = true
		conn = conn + #(el.connections or empty_table)
	end
	local sig = fmt("start_el=%s end_el=%s n=%d connections_total=%d duplicate_node_idx=[%s]",
		id_of(track.start_el), id_of(track.end_el), #list, conn, table.concat(dup, ","))
	say(fmt("SNAP %s %s :: %s", tostring(when), id_of(track), sig))
	return sig
end

-- Every track, in a stable order, as one comparable signature list. The order is
-- the TrackBase label's own walk order, and the handle is printed on every line
-- so a re-ordered label cannot masquerade as a changed track.
function U2.SnapAllTracks(when)
	local out, broken, sites = {}, 0, 0
	for _, tr in ipairs(label_all("TrackBase")) do
		for _, el in ipairs(tr.elements or empty_table) do
			if el.broken then broken = broken + 1 end
		end
		sites = sites + #(tr.repair_cgs or empty_table)
		out[#out + 1] = { name = id_of(tr), value = U2.SnapTrack(tr, when) }
	end
	say(fmt("SNAPALL %s :: %d track(s), %d broken element(s), %d repair site(s)",
		tostring(when), #out, broken, sites))
	U2.Applicable("F48-track-population(tracks with elements to re-order)", #out, #out,
		"TrackBase label walked this load")
	return out
end

---- watchdog + flow ---------------------------------------------------------

function U2.Flow()
	CreateRealTimeThread(function()
		local deadline = RealTime() + U2.watchdog
		while U2.stage ~= "done" do
			if RealTime() - deadline >= 0 then
				say("WATCHDOG TIMEOUT at stage " .. tostring(U2.stage))
				local flush = rawget(_G, "FlushLogFile")
				if flush then flush() end
				Sleep(1000)
				quit()
				return
			end
			Sleep(1000)
		end
	end)

	say(fmt("=== unattended-2 %s === watchdog %dms", tostring(U2.tag), U2.watchdog))

	if not step("1-waiting-menu", function()
		if not wait_for(U2.menu_timeout, function() return not not GetPreGameMainMenu() end) then
			return false
		end
		Sleep(2000)
	end) then return end

	local ok = sprocall(U2.Payload)
	if not ok then
		say("PAYLOAD ERROR at stage " .. tostring(U2.stage))
		shutdown("ERROR payload")
		return
	end
	shutdown("done")
end

function U2.Boot()
	CreateRealTimeThread(function()
		-- CreateRealTimeThread runs its body synchronously up to the first yield,
		-- so this Sleep is what puts everything below after ModsLoadCode().
		Sleep(1)
		U2.Flow()
	end)
end
