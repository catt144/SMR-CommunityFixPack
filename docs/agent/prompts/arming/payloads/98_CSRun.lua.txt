-- SMR Community Fix Pack — Test Kit :: COMBINED SITTING payload. TEMPORARY
--
-- Declared by: docs/agent/prompts/COMBINED_SITTING.md. One cell per launch; the
-- CELL line below is rewritten by CS_ARM.ps1 and READ BACK OFF DISK by its gate,
-- so a cell can never be misattributed after the fact.
--
-- ⛔ THE CELL LINE AND THE SAVEFILE LINE ARE SCRIPT-EDITED. Their parked values
--    are the dry run's, deliberately: if the ARM script ever fails to rewrite
--    them, the run does the harmless prep cell rather than an attended one.

local CELL = "dry"
local SAVEFILE = "CSHOST.savegame.sav"

--------------------------------------------------------------------------------
-- The cell table. `fix`/`opt`/`res` are what the CONFIG GATE demands, and the
-- gate reads the GLOBALS in this process — not the junctions, because two of
-- the three mods are switched by the owner in the MOD MANAGER this sitting and
-- their junctions stay in place throughout (D13's state 3: the def loads, the
-- code does not).
--
-- `attended = true` means: the payload hands the keyboard over and NOTHING in
-- the harness will close the game afterwards.

local CELLS = {
	-- prep, nobody at the keyboard: does the owner's campaign actually carry
	-- subsurface Exotic Minerals deposits on its already-generated asteroid
	-- maps, does the LoadGame re-sign sweep fire on them, and does the camera
	-- drive work? Answering this in prep is what stops moment C being the
	-- corun-batch-1 M1 failure (3 minutes budgeted, ~25 spent, subject gone).
	dry      = { fix = true,  opt = true,  res = false, attended = false, load = true,  f102 = true,  residue = true  },

	-- ⭐ moment C — F102's minute. Packs ON, rescue OUT.
	c1       = { fix = true,  opt = true,  res = false, attended = true,  load = true,  f102 = true,  residue = false },

	-- ⭐ moment A — the PT-20 redo, done RIGHT. State 3: both packs
	--    Mod-Manager-disabled AND a full process restart. Rescue OUT, because a
	--    loaded Save Rescue would strip this save's residue in the first two
	--    seconds and PT-20 would no longer be sampling the thing it exists to
	--    sample (see the brief §3, the one deviation this prep took).
	-- ⭐ f102 = true here, ADDED 2026-08-14 mid-sitting after the owner asked
	--    "is this the new art?" at moment C. It is the same deposit on the same
	--    asteroid with the packs GONE, so this cell measures something F102's
	--    safety case has only ever ARGUED from source: that the class-default
	--    retarget REVERTS on uninstall and the vanilla entity is still there.
	--    Expected: entity reads back 'SignPreciousMineralsDeposit', and NO
	--    're-signed' line, because the fix's OnMsg.LoadGame handler is gone with
	--    it. It also gives the owner the matching half of a before/after they can
	--    photograph. Read-only; it touches nothing PT-20 measures.
	a1       = { fix = false, opt = false, res = false, attended = true,  load = true,  f102 = true,  residue = true  },

	-- ⭐ moment B — the D13 attended after-sweep. Same state-3 window, rescue IN.
	b1       = { fix = false, opt = false, res = true,  attended = true,  load = true,  f102 = false, residue = true  },

	-- ⭐ moment B(4) — the stand-down, after the owner re-enables both packs.
	b2       = { fix = true,  opt = true,  res = true,  attended = true,  load = true,  f102 = false, residue = true  },

	-- close-out control, nobody at the keyboard: the rig as the owner left it.
	zrestore = { fix = true,  opt = true,  res = false, attended = false, load = false, f102 = false, residue = false },
}

--------------------------------------------------------------------------------

local cfg = CELLS[CELL]

-- ⛔ FILE SCOPE, NOT INSIDE THE PAYLOAD, and the prep dry run is why.
--    CS.Flow() decides whether to arm the watchdog, and Flow runs BEFORE
--    Payload — so setting this in Payload left the unattended prep cell with no
--    watchdog at all while every gate line read GREEN. Same class as the r1-void
--    ARM defect: a guard that is *read* earlier than it is *written*.
CS.attended = cfg and cfg.attended or false

function CS.Payload()
	CS.tag = CELL

	if not cfg then
		CS.Say("⛔ UNKNOWN CELL '" .. tostring(CELL) .. "' — this payload knows nothing about it. Nothing measured.")
		return
	end

	CS.attended = cfg.attended
	CS.Say(CS.Fmt("CELL %s :: wants fix=%s opt-in=%s rescue=%s | attended=%s | savefile=%s",
		CELL, tostring(cfg.fix), tostring(cfg.opt), tostring(cfg.res),
		tostring(cfg.attended), tostring(cfg.load and SAVEFILE or "-- none --")))

	-- ⛔ THE GATE THAT STOPS. Before any load, before any reading.
	if not CS.Step("2-config-gate", function()
		return CS.RequireConfig(CELL, cfg.fix, cfg.opt, cfg.res)
	end) then return end

	if cfg.load then
		if not CS.Step("3-load", function()
			-- speed 1: EF-056 — this is a copy of a REAL campaign and its
			-- autosave timer is still that campaign's. Every autosave was
			-- byte-copied by name before this sitting's first launch, but the
			-- cheapest protection is still not letting game time run.
			return CS.Load(SAVEFILE, 1)
		end, true) then return end
	end

	if cfg.f102 then
		CS.Step("4-f102", function()
			CS.ListFixes(CELL)
			local found, first = CS.CensusDeposits(CELL)
			if first then
				CS.ShowDeposit(first)
			else
				-- The drop rule, stated in advance rather than improvised: if the
				-- loaded maps carry no such deposit, the owner is NOT sent hunting.
				-- The recorded respawn call from F102's own negative-repro leg is
				-- the fallback, and it is the entry's verbatim line.
				CS.Say("F102 ⛔ NO DEPOSIT ON ANY LOADED MAP. Not sending anyone hunting: the fallback is F102's own recorded call, run from the console — UIColony:SpawnAsteroid(Presets.DiscoveryAsteroidPreset.Default[\"Asteroid_D\"], true) — which is what the 2026-08-12 negative repro used. It generates a fresh asteroid, which tests the class-default retarget but NOT the LoadGame sweep over already-placed deposits.")
				CS.Try("SpawnAsteroid fallback availability", function()
					local p = rawget(_G, "Presets")
					local d = p and p.DiscoveryAsteroidPreset and p.DiscoveryAsteroidPreset.Default
					return tostring(type(d) == "table" and d["Asteroid_D"] ~= nil)
				end)
			end
			CS.Say(CS.Fmt("F102 VERDICT-INPUT %s :: deposits found = %d. ⛔ The verdict words are the OWNER'S EYES: 'signs render / selectable: yes-no'. This log can prove the entity string and the sweep count; it cannot prove anything rendered.", CELL, #found))
		end)
	end

	if cfg.residue then
		CS.Step("5-residue", function()
			local rec = CS.ReadResidue(CELL .. "/after-load")
			if CELL == "b1" or CELL == "b2" then
				CS.AssertCleared(rec, CELL .. "/after-load")
			end
		end)
	end

	CS.ErrorWatchNote()

	if not cfg.attended then
		CS.Say("CELL " .. CELL .. " :: unattended — closing cleanly so the log tail flushes (EF-047).")
		return
	end

	--------------------------------------------------------------------------
	-- The attended half. Everything below is what the owner is asked to do, in
	-- the log, so the transcript is not the only record of what was asked
	-- (corun-pt15 ledger 3: three of four owner quotes existed only in chat).

	local script = {}
	if CELL == "c1" then
		script = {
			"MOMENT C — F102's minute (checklist item 11, local half).",
			-- ⛔ ONE sign, not three. Measured in prep on this exact fixture and
			--    again at this cell's own census line above: the campaign carries
			--    a single subsurface Exotic Minerals deposit, on slot 3. The
			--    'three signs' of checklist item 11 came from the 2026-08-12
			--    console-spawned D-type, which is a different map that no save on
			--    disk carries. Do not send the owner hunting for two that are not
			--    there. (This line said 'three' on the first attended run,
			--    2026-08-14, and the count printed right above it said 1.)
			"1. LOOK at the ONE deposit sign the camera is parked on and has already selected: does the new art render, and does its panel come up when you click it?",
			"2. Say the verdict words: 'sign renders: yes-no' and 'selectable: yes-no'.",
			"3. Then: main menu -> MOD MANAGER -> turn OFF 'Community Fix Pack' AND 'Community Fix Pack: Opt-In Modules'. LEAVE the Test Kit ON.",
			"4. QUIT THE GAME COMPLETELY. ⛔ The disable does NOT take effect until a full process restart (D13, measured 2026-08-10) — a return to the main menu leaves the mixed state (2) that made the old PT-20 comparison unreadable.",
		}
	elseif CELL == "a1" then
		script = {
			"MOMENT A — the PT-20 redo, done RIGHT (state 3: both packs off AND restarted).",
			"1. Check the gate line above says pack=0/0 and opt-in=0/0. If it does not, the game will already have closed itself.",
			"1b. FREE RIDE-ALONG (added at the owner's question, moment C): the camera is parked on the SAME deposit as moment C, with the packs GONE. This is the uninstall half of F102 — the entity line above says whether the retarget reverted. Photograph it if you want the before/after pair, then fly back to the colony.",
			"2. Play ~10 minutes of ORDINARY play. Nothing special, nothing forced. The original bugs coming back is expected and fine.",
			"3. Then SAVE from the in-game menu under the name: PT20REDO",
			"4. Then LOAD PT20REDO back from the load list. The save must behave normally.",
			"5. QUIT THE GAME COMPLETELY. Do NOT touch the Mod Manager — the next launch needs this exact state.",
		}
	elseif CELL == "b1" then
		script = {
			"MOMENT B — the D13 attended after-sweep. ⛔ THE DIALOGS ARE THE ONLY THING YOUR EYES CAN SAMPLE THAT NO LOG CAN: WaitMessage writes no line in either direction, so a clean run here is what finally grants D13 'tested'.",
			"1. A 'Save Rescue' dialog should already be on your screen, or about to be. READ IT. Does its text match the frozen spec (removed / repaired / kept on purpose / 'stores nothing in your save')? Screenshot it if you can.",
			"2. Say: 'report dialog RAISED: yes-no', and read out the numbers it shows.",
			"3. Then SAVE from the in-game menu under the name: D13SWEEP",
			"4. Then LOAD D13SWEEP back. ⛔ EXPECTED: NOTHING HAPPENS — no dialog at all. Say 'second load silent: yes-no'.",
			"5. Then: main menu -> MOD MANAGER -> turn BOTH packs back ON. QUIT COMPLETELY.",
			"⛔ STOP THE LEG and call the agent if a KEEP name is missing or a REMOVE name survived — the CLEARED lines above say which.",
		}
	elseif CELL == "b2" then
		script = {
			"MOMENT B(4) — the stand-down, with both packs back on.",
			"1. EXPECTED: ONE 'Save Rescue' dialog saying there is nothing to do because the packs are still installed. Say 'stand-down dialog ONCE: yes-no'.",
			"2. The log line above should read removed 0.",
			"3. QUIT THE GAME COMPLETELY. The agent then pulls the rescue junction and puts the rig back to standing config.",
		}
	end
	CS.Handover(script)

	-- Only the D13 cells need the residue re-read on the owner's own loads; for
	-- PT-20 a gate read per load is the whole point (every reading carries its
	-- pack=0/0 line beside it).
	CS.InstrumentOwnerLoads(CELL == "b1" or CELL == "b2")
end

CS.Boot()
