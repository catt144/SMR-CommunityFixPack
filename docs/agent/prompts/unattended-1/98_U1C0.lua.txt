-- SMR Community Fix Pack — Test Kit :: unattended-1 CYCLE 0.  TEMPORARY
--
-- Declared by: unattended-1 chain prompt 1, for prompt 2. Armed together with
-- 97_U1Common.lua and nothing else; both die in the commit that records this
-- cycle's answers (probe hygiene rule 5).
--
-- CYCLE 0 — THE PROOF CYCLE. Every read, no mutation, one load.
--   P   the SAVE primitive proof (README chain rule 10) — the gate for legs
--       A and D. Runs FIRST so a failure costs one cycle, not six.
--   X0  run-conditions header + pack count as READ
--   X1  leg A fixture confirm  (Large Wind Turbine + dome + upgraded MedCenter)
--   X2  leg D applicability    (all six load-heal families, cold-load reading)
--   X3  legs B/C fixture confirm (tracks, break candidates, drones)
--   X4  leg F ride-along       (C42 passage read)
--   X5  leg E fixture confirm  (dust-storm/devil preconditions, NO trigger)
--
-- ⛔ The proof writes ONE throwaway savegame, U1C0PROOF.savegame.sav, and loads
--    it back. Prompt 2 deletes that file agent-side with the game closed, in
--    the same step that deletes the staged copy. There is no in-game delete:
--    io, os and AsyncFileDelete are all keys of ModEnvBlacklist
--    (CommonLua/Classes/Mod.lua:1329/:1424/:1425), so the mod environment has
--    no file-delete primitive at all.

U1 = rawget(_G, "U1") or {}
U1.tag       = "U1C0"
U1.watchdog  = 6 * 60 * 1000      -- 3x the ~1.5 min prediction, 6 min floor
U1.save_file = "U1STAGE.savegame.sav"
U1.proof_save = "U1C0PROOF.savegame.sav"

function U1.Payload()
	local say, fmt, step = U1.Say, U1.Fmt, U1.Step

	if not step("2-load-staged", function() return U1.Load(U1.save_file, 3) end, true) then return end

	if not step("X0-conditions", function() U1.ReadConditions("cycle 0, cold load of the staged copy") end) then return end

	----------------------------------------------------------------------
	-- P — THE SAVE PRIMITIVE PROOF.  Everything legs A and D do rests here.
	----------------------------------------------------------------------
	if not step("P-save-primitive", function()
		say("SAVEPROOF step 1 — environment reachability")
		say(fmt("SAVEPROOF type(SaveGame)=%s type(Savegame)=%s type(Savegame.ListForTag)=%s",
			type(rawget(_G, "SaveGame")), type(rawget(_G, "Savegame")),
			type(rawget(_G, "Savegame") or empty_table) == "table"
				and type((rawget(_G, "Savegame") or empty_table).ListForTag) or "n/a"))

		-- ⛔ untimed: this step contains both a save and a load, and EF-045 C8
		-- forbids printing a RealTime delta across either.
		say("SAVEPROOF step 2 — write a throwaway under a name we chose")
		if not U1.Save(U1.proof_save, "U1C0PROOF") then
			say("SAVEPROOF FAIL — the write did not land. Legs A and D become routed gaps.")
			U1.save_proven = false
			return
		end

		say("SAVEPROOF step 3 — load it back")
		if not U1.Load(U1.proof_save, 3) then
			say("SAVEPROOF FAIL — the file exists but will not load back. Legs A and D become routed gaps.")
			U1.save_proven = false
			return
		end

		say("SAVEPROOF step 4 — confirm the reloaded world is live and the pack is the same")
		U1.ReadConditions("after reloading the throwaway")
		say("SAVEPROOF PASS — SaveGame + ListForTag + LoadGame round trip complete. Deletion is agent-side, game closed.")
	end, true) then return end

	if not U1.save_proven then
		say("⛔ SAVE PRIMITIVE NOT PROVEN — prompt 2 must route legs A and D as gaps and skip cycles 1, 2 and 6.")
	end

	----------------------------------------------------------------------
	-- Fixture confirms. Every one of these is a LABEL READ and changes
	-- nothing. A confirm that fails turns its leg into a SKIP line, never a
	-- re-choice of save (README stop condition 3).
	----------------------------------------------------------------------
	if not step("X1-legA-fixture", function()
		U1.ReadSanitizerNumbers()
		-- The MedCenter half of the fixture, stated as its own reading: the
		-- F03 leak target is an upgrade modifier on a parent_dome, so what
		-- matters is whether ANY building in a dome carries applied upgrades.
		local centers = U1.LabelAll("MedicalCenter")
		local upgraded = 0
		for _, b in ipairs(centers) do
			local n = 0
			for _ in pairs(b.upgrade_modifiers or empty_table) do n = n + 1 end
			if n > 0 then upgraded = upgraded + 1 end
		end
		say(fmt("PT35 fixture MedCenter: %d Medical Center(s), %d carrying applied upgrade modifiers | LEG A FIXTURE OK=%s",
			#centers, upgraded, tostring(#centers > 0)))
	end) then return end

	if not step("X2-legD-applicability", function()
		U1.ReadAllHeals("cycle 0 baseline (cold load of the staged copy)")
		say("NOTE cycle 0 reads the heal families ONCE. Any heal line printed by the pack ABOVE this marker fired on this cold load; that is the load-1 set leg D compares against.")
	end) then return end

	if not step("X3-legBC-fixture", function()
		local tracks, candidates = U1.ReadTracks()
		U1.ReadDrones()
		U1.ReadF99Residue("cycle 0 baseline — nothing has been broken yet")
		local el, tr = U1.PickBreakCandidate()
		say(fmt("LEGS B/C FIXTURE: %d track(s), %d break-candidate(s), first candidate=%s on %s | OK=%s",
			tracks, candidates, U1.IdOf(el), U1.IdOf(tr), tostring(el ~= nil)))
	end) then return end

	if not step("X4-legF-c42", function()
		U1.ReadC42()
	end) then return end

	if not step("X5-legE-fixture", function()
		local presets = (rawget(_G, "Presets") or empty_table).MapSettings or empty_table
		local storms  = presets.DustStorm
		local devils  = presets.DustDevils
		local map     = rawget(_G, "MainMap")
		local md      = map and map.mapdata
		say(fmt("LEG E fixture: Presets.MapSettings.DustStorm=%s DustDevils=%s | map DustStorm setting=%s DustDevils setting=%s",
			type(storms), type(devils),
			tostring(md and md.MapSettings_DustStorm), tostring(md and md.MapSettings_DustDevils)))
		local has = rawget(_G, "HasDustStorm")
		say(fmt("LEG E fixture: HasDustStorm=%s now=%s | CheatDustStorm=%s GenerateDustDevilIn=%s",
			type(has), tostring(type(has) == "function" and has(map) or "?"),
			type(rawget(_G, "CheatDustStorm")), type(rawget(_G, "GenerateDustDevilIn"))))
		say("LEG E fixture read only — nothing was triggered on this cycle.")
	end) then return end

	U1.ErrorWatchNote()
	say("=== cycle 0 complete ===")
end

U1.Boot()
