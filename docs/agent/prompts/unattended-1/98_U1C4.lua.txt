U1 = rawget(_G, "U1") or {}
U1.tag         = "U1C4"
-- Worst case = 30 s overhead + rounds x (10 s witness + repair_wait) = 30 s +
-- 4 x 4:10 = 17.2 min. Watchdog is above that AND 3x the ~6 min prediction.
U1.watchdog    = 18 * 60 * 1000
U1.save_file   = "U1STAGE.savegame.sav"
U1.rounds      = 4                -- break/repair attempts, one element each
U1.repair_wait = 4 * 60 * 1000    -- per-round patience for the drones
U1.break_wait  = 10 * 1000        -- patience for the DAMAGE to become visible

local function set_speed(n)
	local ok = pcall(function() UIColony:SetGameSpeed(n) end)
	Sleep(300)
	return ok, GetTimeFactor() / const.DefaultTimeFactor
end

-- Picks a candidate this cycle has not already used. Returns el, track.
local function pick_fresh(used)
	for _, tr in ipairs(U1.LabelAll("TrackBase")) do
		for _, el in ipairs(tr.elements or empty_table) do
			if el ~= tr.start_el and el ~= tr.end_el and not el.station and not el.broken
					and not used[U1.IdOf(el)] then
				return el, tr
			end
		end
	end
	return nil
end

-- The liveness witness. Returns a description of what was actually seen, or nil.
local function break_witness(el, track)
	local seen = nil
	U1.WaitFor(U1.break_wait, function()
		local broken = el.broken and true or false
		local sites  = #(track.elements_under_construction or empty_table)
		local cgs    = #(track.repair_cgs or empty_table)
		if broken or sites > 0 or cgs > 0 then
			seen = U1.Fmt("broken=%s sites=%d repair_cgs=%d", tostring(broken), sites, cgs)
			return true
		end
		return false
	end)
	return seen
end

function U1.Payload()
	local say, fmt, step = U1.Say, U1.Fmt, U1.Step
	local attempted, completed, timed_out, unwitnessed = 0, 0, 0, 0
	local used = {}

	if not step("2-load-staged", function() return U1.Load(U1.save_file, 3) end, true) then return end

	if not step("C0-baseline", function()
		U1.ReadConditions("leg C, cold load — speed 3 for the break, 20 for the organic wait")
		U1.ReadTracks()
		U1.ReadF99Residue("C0 baseline")
		local n, hubs = U1.ReadDrones()
		if n == 0 or hubs == 0 then
			say("LEG C FIXTURE FAILURE — no drones and/or no hub. An ORGANIC repair cannot happen, so every round below would be unsampled. Prompt 2 records SKIP <no drones/hub reachable> and routes the gap; it does NOT substitute a cheat completion.")
			return false
		end
	end) then return end

	for round = 1, U1.rounds do
		local el, track
		local witnessed = false
		local label = "C" .. tostring(round)

		if not step(label .. "-pick", function()
			el, track = pick_fresh(used)
			if not el then
				say(fmt("%s no UNUSED break candidate left — stopping the loop after %d attempt(s)", label, attempted))
				return
			end
			used[U1.IdOf(el)] = true
			say(fmt("%s TARGET element=%s track=%s elements=%d repair_cgs=%d under_construction=%d",
				label, U1.IdOf(el), U1.IdOf(track), #(track.elements or empty_table),
				#(track.repair_cgs or empty_table),
				#(track.elements_under_construction or empty_table)))
		end) then return end

		if not el then break end

		if not step(label .. "-break", function()
			local bt = rawget(_G, "BreakTracks")
			if type(bt) ~= "function" then
				say("LEG C ABORT — BreakTracks unreachable. Record verbatim and route.")
				return false
			end
			-- Speed 3 for the break so the damage is observable before drones can
			-- erase it. This is the whole repair for the voided run.
			local ok_s, sp = set_speed(3)
			attempted = attempted + 1
			say(fmt("%s FORCED: BreakTracks({element}) at speed %s (SetGameSpeed ok=%s) — ⛔ ORGANIC WINDOW OPENS AFTER THIS LINE; any [LUA ERROR] until the CLOSE marker happened with NO cheat on the stack",
				label, tostring(sp), tostring(ok_s)))
			local ok, err = pcall(bt, { el })
			say(fmt("%s break call ok=%s err=%s", label, tostring(ok), tostring(err)))
		end) then return end

		if not step(label .. "-witness", function()
			local seen = break_witness(el, track)
			if not seen then
				unwitnessed = unwitnessed + 1
				say(fmt("%s ⛔ BREAK NOT OBSERVED within %dms — no broken flag, no repair site, no repair group. This round is UNSAMPLED and is NOT scored a completion. (This is the check the voided first run did not have.)",
					label, U1.break_wait))
				return
			end
			witnessed = true
			say(fmt("%s BREAK WITNESSED: %s — there is now real damage for drones to repair", label, seen))
		end) then return end

		-- ⛔ Only a WITNESSED round gets an organic wait, and only a witnessed
		-- round can be scored a completion. This one line is the whole repair.
		if not step(label .. "-organic-repair", function()
			if not witnessed then
				say(fmt("%s skipping the organic wait — nothing was broken to repair", label))
				U1.ReadF99Residue(label .. " after the unsampled round")
				return
			end
			local ok_s, sp = set_speed(20)
			say(fmt("%s organic wait at speed %s (SetGameSpeed ok=%s) — drones only, polled, never forced", label, tostring(sp), tostring(ok_s)))
			local ok = U1.WaitFor(U1.repair_wait, function()
				return #(track.elements_under_construction or empty_table) == 0 and not el.broken
			end)
			if ok then
				completed = completed + 1
				say(fmt("%s ORGANIC REPAIR COMPLETED by drones — no cheat was called | broken=%s sites=%d repair_cgs=%d",
					label, tostring(el.broken and true or false),
					#(track.elements_under_construction or empty_table),
					#(track.repair_cgs or empty_table)))
			else
				timed_out = timed_out + 1
				say(fmt("%s ORGANIC REPAIR DID NOT COMPLETE within %dms — this round is UNSAMPLED, not a zero. under_construction=%d broken=%s",
					label, U1.repair_wait,
					#(track.elements_under_construction or empty_table), tostring(el.broken and true or false)))
			end
			say(fmt("%s ⛔ ORGANIC WINDOW CLOSES", label))
			set_speed(3)
			U1.ReadF99Residue(label .. " after the organic round")
		end) then return end
	end

	if not step("C-verdict", function()
		U1.ReadTracks()
		local a, b = U1.ReadF99Residue("C final, pre-reload")
		say(fmt("LEG C SAMPLED CONDITION: %d break(s) forced, %d WITNESSED-then-repaired ORGANICALLY to completion, %d timed out (unsampled), %d never showed damage at all (unsampled). Final residue a=%d b=%d.",
			attempted, completed, timed_out, unwitnessed, a, b))
		say("LEG C ⛔ THE ONLY SENTENCES PROMPT 2 MAY WRITE FROM THIS:")
		say("  if TrackElement.lua:805 appears inside an ORGANIC WINDOW above -> F99 has a no-cheat occurrence; it moves off `cand`; severity is still the owner's.")
		say("  if it does not, and the WITNESSED-completed count is N > 0 -> zero occurrences across N organic completions, on this save, in this configuration. A rate bound with its condition stated. NOT a refutation, NOT evidence of unreachability.")
		say("  if the WITNESSED-completed count is 0 -> the leg sampled NOTHING about reachability and must say so; the run is a fixture result, not an F99 result.")
	end) then return end

	U1.ErrorWatchNote()
	say("=== cycle 4 (leg C) complete ===")
end

U1.Boot()
