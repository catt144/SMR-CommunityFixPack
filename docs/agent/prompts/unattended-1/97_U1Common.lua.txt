---- output ------------------------------------------------------------------

-- ModLog is the only print path proven to reach the log file on disk
-- (Mod.lua:109-132). Its emitter formats its single argument a second time
-- (lib.lua:164-174), so a literal '%' raises there — escape it. Flush per line
-- so a killed run still shows how far it got.
local t0 = false
local function say(msg)
	if not t0 then t0 = RealTime() end
	msg = "[" .. tostring(U1.tag) .. "] " .. tostring(msg)
	local modlog = rawget(_G, "ModLog")
	if modlog then
		modlog((msg:gsub("%%", "%%%%")))
	else
		local dbg = rawget(_G, "DebugPrint")
		if dbg then dbg(msg .. "\n") end
	end
	print(msg)
	local flush = rawget(_G, "FlushLogFile")
	if flush then pcall(flush) end
end
U1.Say = say

local function fmt(...)
	local ok, s = pcall(string.format, ...)
	return ok and s or "<fmt error>"
end
U1.Fmt = fmt

local function id_of(obj)
	if not obj then return "nil" end
	if type(obj) ~= "table" then return tostring(obj) end
	return tostring(obj.class or "?") .. "#" .. tostring(rawget(obj, "handle"))
end
U1.IdOf = id_of

---- flow helpers ------------------------------------------------------------

local function shutdown(reason)
	U1.stage = "done"
	say("SHUTDOWN " .. tostring(reason))
	local flush = rawget(_G, "FlushLogFile")
	if flush then flush() end
	Sleep(U1.quit_delay)
	quit()
end
U1.Shutdown = shutdown

-- Steps Sleep, so sprocall is mandatory (pcall cannot survive a yield).
--
-- untimed = true suppresses the END timing line entirely (EF-045, correction
-- C8: do not even PRINT a RealTime() delta for such a step, because a future
-- reader will quote it). ⛔ It is set for SAVE steps as well as LOAD steps, and
-- that is not caution: SaveGame's first act is WaitRenderMode("ui")
-- (CommonLua/Savegame.lua:1079) — the same no-frames-render block that made
-- LoadGame under-report itself 11.5x — so a saved step's RealTime delta is
-- wrong for exactly EF-045's reason. A save has no engine-printed duration
-- line either, so its honest report is no number at all.
local function step(name, fn, untimed)
	U1.stage = name
	local t = RealTime()
	say("STEP " .. name .. " BEGIN")
	local ok, res = sprocall(fn)
	if not ok then
		say("STEP " .. name .. " ERROR")
		shutdown("ERROR " .. name)
		return false
	end
	if untimed then
		say("STEP " .. name .. " END (no timing — this step blocks render mode; a load takes its number from the engine's own 'Game loaded on map ... in Nms' line, a save has none)")
	else
		say(fmt("STEP %s END %dms", name, RealTime() - t))
	end
	if res == false then
		shutdown("STOP " .. name .. " — the step asked to stop; its reason is the last line before this one")
		return false
	end
	return true
end
U1.Step = step

-- Never raises: error() does not unwind mod code in this engine, so a raising
-- wait would spin until the watchdog killed the process (95_AutoRun's lesson).
local function wait_for(timeout, ready)
	local deadline = RealTime() + timeout
	while not ready() do
		if RealTime() - deadline >= 0 then return false end
		Sleep(500)
	end
	return true
end
U1.WaitFor = wait_for

-- Returns true on success. Sets a speed and reads it back every time: a loaded
-- save arrives PAUSED (C4) and every game-time thread below it is dead until
-- this runs.
function U1.Load(savefile, speed)
	local err = LoadGame(savefile, {})            -- ⛔ never pass save_as_last
	if err then
		say("LOADERR " .. tostring(savefile) .. " " .. tostring(err))
		return false
	end
	U1.loads = U1.loads + 1
	say(fmt("LOAD OK #%d file=%s — LoadGame returned no error", U1.loads, tostring(savefile)))
	if not wait_for(5 * 60 * 1000, game_is_live) then
		say("LOADERR " .. tostring(savefile) .. " — city never came live")
		return false
	end
	Sleep(U1.settle_time)                          -- 15 s is the MEASURED-sufficient datum
	speed = speed or 3
	local before = GetTimeFactor() / const.DefaultTimeFactor
	local ok = pcall(function() UIColony:SetGameSpeed(speed) end)
	Sleep(500)
	local after = GetTimeFactor() / const.DefaultTimeFactor
	say(fmt("READ speed: SetGameSpeed(%d) call=%s | before=%s after=%s",
		speed, tostring(ok), tostring(before), tostring(after)))
	return true
end

---- the SAVE primitive ------------------------------------------------------
--
-- SaveGame(display_name, params) — CommonLua/Savegame.lua:1071. Src-verified
-- 2026-08-04 (chain prompt 1), never executed by us before this chain:
--   * plain global, and NOT a key of ModEnvBlacklist
--     (CommonLua/Classes/Mod.lua:1267-1428, re-derived this session);
--   * it YIELDS — WaitRenderMode("ui") at :1079, DoSaveGame's WaitChangeMapDone
--     / WaitSaveGameDone at :1038-1039, and Savegame._WrappedSave (= _Wrap of
--     _InternalSave, :874) which does WaitThread → CreateRealTimeThread →
--     WaitThread (:337-344). It must therefore be called from a REAL-TIME
--     thread, exactly as LoadGame had to be (corun-rig S2);
--   * params.savename routes to Savegame.WithName (:1051 → :895), bypassing
--     _UniqueName, so the filename on disk is EXACTLY what we pass. Both
--     WithName and WithTag set params.backup = false (:908/:890) — no .bak;
--   * params.silent skips LoadingScreenOpen/Close (:1075-1077, :1088-1090) —
--     required by chain rule 13, an unattended run may not raise a screen it
--     cannot dismiss;
--   * params.no_screenshot skips the scene render-mode round trip
--     (:1010-1023);
--   * ⛔ params.save_as_last is NEVER passed: it writes LocalStorage.last_save
--     (:1057-1059), i.e. it would repoint the owner's Continue button at our
--     throwaway file. This is a do-no-harm hazard, not a preference.
--
-- Returns true only if the file is afterwards visible to
-- Savegame.ListForTag("savegame") (= _Wrap(_InternalListForTag), :941/:469;
-- the tag is parsed back OUT of the filename by ^.*%.([%w_]+)%.sav$ at :477,
-- which is why every throwaway name must end ".savegame.sav").

local function list_saves()
	local sg = rawget(_G, "Savegame")
	if type(sg) ~= "table" or type(sg.ListForTag) ~= "function" then
		return nil, "Savegame.ListForTag unavailable"
	end
	local err, list = sg.ListForTag("savegame")
	if err then return nil, tostring(err) end
	return list or {}
end
U1.ListSaves = list_saves

local function has_save(list, savename)
	for _, e in ipairs(list or empty_table) do
		if e.savename == savename then return true end
	end
	return false
end
U1.HasSave = has_save

-- savename MUST end in ".savegame.sav".
function U1.Save(savename, display)
	if type(rawget(_G, "SaveGame")) ~= "function" then
		say("SAVEFAIL SaveGame is not a function in this environment — the primitive is env-blacklisted after all")
		return false
	end
	local before, lerr = list_saves()
	if not before then
		say("SAVEFAIL cannot list saves: " .. tostring(lerr))
		return false
	end
	if has_save(before, savename) then
		say(fmt("SAVEFAIL %s already exists before we wrote it — refusing to overwrite anything", savename))
		return false
	end
	say(fmt("SAVE writing %s (%d savegame-tagged files before)", savename, #before))
	local err, name = SaveGame(display or savename, {
		savename     = savename,
		silent       = true,
		no_screenshot = true,
	})
	say(fmt("SAVE returned err=%s name=%s", tostring(err), tostring(name)))
	if err then return false end
	local after = list_saves()
	local seen = has_save(after, savename)
	say(fmt("SAVE listed-after=%s (%d savegame-tagged files after)", tostring(seen), #(after or empty_table)))
	if not seen then return false end
	U1.save_proven = true
	return true
end

---- watchdog + flow ---------------------------------------------------------
--
-- The payload file sets U1.watchdog and U1.Payload, then calls U1.Flow().

function U1.Flow()
	CreateRealTimeThread(function()
		local deadline = RealTime() + U1.watchdog
		while U1.stage ~= "done" do
			if RealTime() - deadline >= 0 then
				say("WATCHDOG TIMEOUT at stage " .. tostring(U1.stage))
				local flush = rawget(_G, "FlushLogFile")
				if flush then flush() end
				Sleep(1000)
				quit()
				return
			end
			Sleep(1000)
		end
	end)

	say(fmt("=== unattended-1 %s === watchdog %dms", tostring(U1.tag), U1.watchdog))

	if not step("1-waiting-menu", function()
		if not wait_for(U1.menu_timeout, function() return not not GetPreGameMainMenu() end) then
			return false
		end
		Sleep(2000)
	end) then return end

	local ok = sprocall(U1.Payload)
	if not ok then
		say("PAYLOAD ERROR at stage " .. tostring(U1.stage))
		shutdown("ERROR payload")
		return
	end
	shutdown("done")
end

---- shared readers ----------------------------------------------------------
--
-- Every one of these PRINTS and RETURNS. The printed line is the record; the
-- return value is what a payload compares across loads. Each reader states its
-- own APPLICABLE verdict, because a count of 0 taken where the condition does
-- not exist is an UNSAMPLED condition, not a negative result.

-- Run-conditions header (WORKFLOW EXTERNAL VALIDITY rule). Pack count is READ,
-- never assumed — the owner's campaign runs every opt-in ON, which is not
-- default config.
function U1.ReadConditions(note)
	local pack = rawget(_G, "SMRFixPack")
	local active, total, inactive = 0, 0, {}
	if pack and pack.order then
		for _, id in ipairs(pack.order) do
			total = total + 1
			local f = pack.fixes and pack.fixes[id]
			local st = f and f.status or "missing"
			if st == "active" then active = active + 1 else inactive[#inactive + 1] = id .. "=" .. st end
		end
	end
	say(fmt("CONDITIONS pack=%d/%d active | map=%s | speed=%s | loads_this_process=%d | note=%s",
		active, total, tostring(rawget(_G, "CurrentMapName")),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), U1.loads, tostring(note or "-")))
	if #inactive > 0 then
		say("CONDITIONS not-active: " .. table.concat(inactive, ", "))
	end
	return active, total
end

-- H1 — Fix_AstrogeologistExtractors (F92). The 2026-08-02 defect was an
-- identity test that missed on every load and re-applied: 1 modifier became 2
-- after one save+reload, i.e. +20% instead of +10%. So the number that matters
-- is the COUNT of colony label modifiers per label, not the presence of one.
function U1.ReadHealAstro()
	local colony = rawget(_G, "UIColony")
	local mods = colony and colony.label_modifiers
	local get_profile = rawget(_G, "GetCommanderProfile")
	local profile = type(get_profile) == "function" and get_profile() or nil
	local pid = type(profile) == "table" and profile.id or "none"
	if type(mods) ~= "table" then
		say("HEAL H1 astro: NOT-APPLICABLE (no UIColony.label_modifiers)")
		return nil
	end
	local total, detail = 0, {}
	for label, by_id in pairs(mods) do
		local all = 0
		for _ in pairs(by_id or empty_table) do all = all + 1 end
		if all > 0 then
			total = total + all
			detail[#detail + 1] = fmt("%s=%d", tostring(label), all)
		end
	end
	table.sort(detail)
	say(fmt("HEAL H1 astro: commander_profile=%s | colony label_modifier counts: %s | TOTAL=%d",
		tostring(pid), table.concat(detail, " "), total))
	return total, table.concat(detail, " ")
end

		local by_label = dome.label_modifiers

	end
	say(fmt("HEAL H2 saint: %d dome(s), %d carrying label modifiers, %d modifier(s) total | APPLICABLE=%s",
		#domes, with_mods, total, tostring(#domes > 0)))
	return total
end

-- Compares two ReadAllHeals() results and prints one verdict line per family.
function U1.CompareHeals(a, b, label)
	local keys = { "astro", "saint", "bio", "suns", "meteor", "rains" }
	local changed = 0
	for _, k in ipairs(keys) do
		local same = a[k] == b[k]
		if not same then changed = changed + 1 end
		say(fmt("HEALDIFF %s %s: %s  [%s] -> [%s]", tostring(label), k,
			same and "SAME" or "CHANGED", tostring(a[k]), tostring(b[k])))
	end
	say(fmt("HEALDIFF %s VERDICT: %d of 6 families changed (prediction: 0)", tostring(label), changed))
	return changed
end

-- Legs B and C — the F99 residue read, in the exact shape the checklist rider

function U1.ReadF99Residue(when)

	for _, c in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, t in ipairs((c.labels or empty_table).TrackBase or empty_table) do

	return a, b
end

-- Leg F — C42, ride-along. A stale entry is a unit in a passage element's
-- `units` list whose own `holder` is something else.
function U1.ReadC42()
	local stale, elements, units, passages = 0, 0, 0, 0
	for _, c in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, p in ipairs((c.labels or empty_table).Passage or empty_table) do
			passages = passages + 1
			for _, el in ipairs(p.elements or empty_table) do
				elements = elements + 1
				for _, u in ipairs(el.units or empty_table) do
					units = units + 1
					if u.holder ~= el then stale = stale + 1 end
				end
			end
		end
	end
	say(fmt("C42STALE %d  | %d passage(s), %d element(s), %d unit entr(ies) | APPLICABLE=%s",
		stale, passages, elements, units, tostring(passages > 0)))
	return stale, passages, elements, units
end

-- Track fixture read (legs B and C need to know what is there before they act).
function U1.ReadTracks()
	local tracks = U1.LabelAll("TrackBase")
	local els, broken, sites, candidates = 0, 0, 0, 0
	for _, tr in ipairs(tracks) do
		local list = tr.elements or empty_table
		els = els + #list
		sites = sites + #(tr.elements_under_construction or empty_table)
		for _, el in ipairs(list) do
			if el.broken then broken = broken + 1 end
			-- BreakTracks' own filter, Meteors.lua:601: neither endpoint, no station
			if el ~= tr.start_el and el ~= tr.end_el and not el.station and not el.broken then
				candidates = candidates + 1
			end
		end
	end
	say(fmt("TRACKS %d track(s), %d element(s), %d already broken, %d repair site(s), %d break-candidate(s)",
		#tracks, els, broken, sites, candidates))
	return #tracks, candidates
end

-- Picks one element matching BreakTracks' own filter (Meteors.lua:601).
function U1.PickBreakCandidate()
	for _, tr in ipairs(U1.LabelAll("TrackBase")) do
		for _, el in ipairs(tr.elements or empty_table) do
			if el ~= tr.start_el and el ~= tr.end_el and not el.station and not el.broken then
				return el, tr
			end
		end
	end
	return nil
end

-- Drone/repair fixture read — leg C's organic path needs drones and resources
-- reachable from the track, and a leg that cannot say so has not sampled it.
function U1.ReadDrones()
	local drones = U1.LabelAll("Drone")
	local hubs   = U1.LabelAll("DroneHub")
	local rcs    = U1.LabelAll("RocketLanding")
	local idle = 0
	for _, d in ipairs(drones) do
		if d.command == "Idle" then idle = idle + 1 end
	end
	say(fmt("DRONES %d drone(s) (%d idle), %d hub(s), %d rocket landing pad(s)",
		#drones, idle, #hubs, #rcs))
	return #drones, #hubs
end

---- error-watch marker -------------------------------------------------------

-- ⚠️ ADDED 2026-08-04 by chain prompt 2's pre-launch probe-hygiene sweep, in the
-- open (chain rule 5). All seven payload files call U1.ErrorWatchNote() as their
-- last act and the function did not exist: a nil call raises, sprocall catches it
-- in U1.Flow, and EVERY cycle would have shut down as "PAYLOAD ERROR at stage
-- <last step>" with its closing banner missing. The parse sweep could not see
-- this — 8/8 GREEN is a syntax verdict, not a resolution one.

-- straight to the log file and the mod environment has no file read (io and os
-- are both keys of ModEnvBlacklist, Mod.lua:1424/:1425), so a count printed from
-- in here would be invented. It prints the boundary a reader greps between
-- instead, and names the attribution duty (WORKFLOW "never silently discount a
-- line").
function U1.ErrorWatchNote()

Attribute each one against the FORCED / ORGANIC window markers printed above; report every unexplained line verbatim with its position, never discount one. stage_at_close=%s loads_this_process=%d",
		tostring(U1.tag), tostring(U1.tag), tostring(U1.stage), U1.loads))
end
