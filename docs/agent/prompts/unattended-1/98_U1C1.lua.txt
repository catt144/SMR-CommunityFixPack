--                                        bonus(es) this save started without"

--    90_SaveSanitizer (F35/F03) prints NOTHING from PostLoadGame unless a pass
--    throws — it is leg A's subject, driven directly, not watched here.

--    WHAT NUMBERS GET COMPARED. Not the lines — the counts behind them, read
--    identically at every load by U1.ReadAllHeals(). Prediction: every family
--    identical across loads 1, 2 and 3, and no heal line after load 1.
--
--    THE HONEST LIMIT, STATED BEFORE ANY NUMBER IS QUOTED. A family whose
--    condition does not exist on this save is UNSAMPLED, not refuted. Each
--    reader prints its own APPLICABLE verdict for that reason. Cycle 6 (leg D2)
--    is what turns unsampled families into sampled ones, by forcing the defect
--    state; this cycle measures the save as it actually is.
--
-- ⚠️ Requires the SAVE primitive. If cycle 0's proof did not pass, this cycle
--    is not run at all — leg D becomes a routed gap (README stop condition 1).

U1 = rawget(_G, "U1") or {}
U1.tag        = "U1C1"
U1.watchdog   = 6 * 60 * 1000     -- 3x the ~1.7 min prediction, 6 min floor
U1.save_file  = "U1STAGE.savegame.sav"
U1.round_save = "U1C1HEAL.savegame.sav"

function U1.Payload()
	local say, fmt, step = U1.Say, U1.Fmt, U1.Step
	local r1, r2, r3

	if not step("2-load-1-staged", function() return U1.Load(U1.save_file, 3) end, true) then return end

	if not step("D1-read-load-1", function()
		U1.ReadConditions("leg D1, load 1 — cold load of the staged copy")
		r1 = U1.ReadAllHeals("load 1 (cold, staged copy)")
	end) then return end

	if not step("D1-save", function()
		if not U1.Save(U1.round_save, "U1C1HEAL") then
			say("LEG D1 ABORT — the save primitive failed here even though cycle 0 proved it. Record this verbatim: it is an unforeseen-issue finding, not a leg result.")
			return false
		end
	end, true) then return end

	if not step("3-load-2-roundtrip", function() return U1.Load(U1.round_save, 3) end, true) then return end

	if not step("D1-read-load-2", function()
		U1.ReadConditions("leg D1, load 2 — first reload of the round-trip save")
		r2 = U1.ReadAllHeals("load 2 (first reload)")
		U1.CompareHeals(r1, r2, "load1->load2")
	end) then return end

	if not step("4-load-3-roundtrip", function() return U1.Load(U1.round_save, 3) end, true) then return end

	if not step("D1-read-load-3", function()
		U1.ReadConditions("leg D1, load 3 — second reload of the same file")
		r3 = U1.ReadAllHeals("load 3 (second reload)")
		U1.CompareHeals(r2, r3, "load2->load3")
		U1.CompareHeals(r1, r3, "load1->load3")
	end) then return end

	if not step("D1-verdict", function()
		say("LEG D1 READING COMPLETE. The verdict is assembled by prompt 2 from THREE things, and all three must be quoted together:")
		say("  1. the three HEALDIFF VERDICT lines above (prediction: 0 families changed on each);")
		say("  2. every pack heal line in this log, WITH the load it followed — a heal line after load 2 or load 3 is the defect;")
		say("  3. each family's APPLICABLE verdict — a family that read NOT-APPLICABLE was not sampled and must be reported as unsampled, never as clean.")
	end) then return end

	U1.ErrorWatchNote()
	say("=== cycle 1 (leg D1) complete ===")
end

U1.Boot()
