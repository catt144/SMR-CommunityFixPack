-- LEG D2 — THE LOAD-HEAL ROUND-TRIP SWEEP, forced-defect half.
--
-- WHY THIS HALF EXISTS AND CYCLE 1 IS NOT ENOUGH. Cycle 1 reads the save as it
-- actually is. If a heal family's condition is not present on the staged copy,
-- its "no change across three loads" is an UNSAMPLED condition, not a clean
-- result — the project's own rule after the 2026-08-03 refuted/unsampled
-- correction. This cycle CREATES the condition for the two families whose
-- defect state is exactly and reversibly definable, then measures whether the
-- heal fires exactly once and lands on exactly the right number.
--
-- ⛔ FORCED vs ORGANIC (README rule 11). FORCED: the removal of already-healed
--    state, so that the load-time heal has something to do. ORGANIC: nothing —
--    this is a MECHANISM leg and its ceiling is MECHANISM (WORKFLOW triage mode
--    1). It says how the heal behaves when it fires; it says nothing about how
--    often a real save arrives needing it.
--
-- ⭐ WHY THIS CYCLE CARRIES THE LEG. Evidence, not caution: the three archived
--    co-run #1 loads of a COPY OF THIS VERY SAVE
--    (docs/archive/corun1*.log, 2026-08-04, pack 81/81) fired **zero** load-time
--    heal lines between them. So leg D1's natural-state reading is expected to
--    be "nothing fired, nothing repeated" — true, but weak, because nothing was
--    going to fire. THIS cycle is where leg D actually samples anything, and two
--    of its three families are forceable on ANY save whatsoever, so the leg
--    produces a real measurement no matter what the staged copy contains.
--
-- THE THREE FAMILIES, AND WHY ONLY THESE THREE.
--   H5 Fix_MeteorFrequency (F88). The defect state is one GameVar:
--      SMRFixPack_MeteorLatch = false. The heal (OnMsg.PostLoadGame,
--      Fix_MeteorFrequency.lua:164-184) restarts the global "Meteors" game-time
--      thread and writes the pack version back into the latch. F88's own defect
--      was an UNLATCHED restart on every load, so "fires once, then never
--      again across reloads" IS the repair under test. Fully determined, one
--      assignment, reversible by the heal itself.
--   H1 Fix_AstrogeologistExtractors (F92). The defect state is the ABSENCE of
--      the colony label modifiers the profile owes. The labels are not guessed:
--      the module's own MISSING list is
--      { AutomaticMetalsExtractor / production_per_day1 / 10%,
--        MicroGAutoWaterExtractor / water_production / 10% }
--      (Fix_AstrogeologistExtractors.lua:67-70) under commander profile id
--      "astrogeologist" (:62). Stripping exactly those and reloading is the
--      2026-08-02 keyboard finding re-run on the rig: the identity-keyed
--      version re-applied on every load and 1 modifier became 2 after one
--      save+reload. The pass number to check is therefore EXACTLY 1 per label
--      after the heal, and still exactly 1 after another round trip.
--   H6 Fix_RainsDeadlock (C34), added 2026-08-04 after the owner said they did
--      not know what this save carries — which is the right answer, and the fix
--      for it is to stop needing to know. `RainsDisasterThreads` is a shipped
--      GameVar (`Lua/TerraformingDisasters.lua:323`), so it exists and persists
--      on EVERY save; the C34 pass logs `RainsDisasterThreads was %s —
--      recreated as an empty table` and rebuilds it whenever it is not a table
--      (`Fix_RainsDeadlock.lua:126-131`). Setting it to `false` is therefore the
--      module's own documented defect state, reachable with no fixture at all.
--      ⚠️ It is forced LAST, immediately before the save, because between the
--      assignment and the reload the shipped code would index a boolean
--      (`TerraformingDisasters.lua:190/:249/:280/:328`) if a rain event fired in
--      that window. The window is seconds and the game is not running a rain,
--      but if a `[LUA ERROR]` naming `TerraformingDisasters.lua` appears there,
--      it is a CONSEQUENCE OF OUR FORCING — report it as that, not as a defect.
--   H2 Saint, H3 biorobots and H4 suns are left alone deliberately: forcing them
--      means editing colonist traits, dome membership or built objects, which is
--      a bigger mutation than the measurement is worth and is not reversible by
--      the heal alone. They stay reported as cycle 1 found them — sampled or
--      unsampled, said plainly.
--
-- ⚠️ Requires the SAVE primitive. If cycle 0's proof did not pass, this cycle
--    is not run at all and leg D2 is a routed gap.
--
-- MUTATING LEG — runs after every read cycle. Four loads, two throwaway saves.

U1 = rawget(_G, "U1") or {}
U1.tag       = "U1C6"
U1.watchdog  = 8 * 60 * 1000      -- 3x the ~2.5 min prediction (four loads, two saves)
U1.save_file = "U1STAGE.savegame.sav"
U1.save_a    = "U1C6FORCED.savegame.sav"
U1.save_b    = "U1C6HEALED.savegame.sav"

local ASTRO_LABELS = {
	{ Label = "AutomaticMetalsExtractor", Prop = "production_per_day1" },
	{ Label = "MicroGAutoWaterExtractor", Prop = "water_production" },
}

-- Reads the two labels the astro heal owns, precisely. Returns "n1/n2".
local function read_astro_exact(when)
	local say, fmt = U1.Say, U1.Fmt
	local colony = rawget(_G, "UIColony")
	local mods = colony and colony.label_modifiers
	local parts = {}
	for _, want in ipairs(ASTRO_LABELS) do
		local n, detail = 0, {}
		for id, m in pairs((type(mods) == "table" and mods[want.Label]) or empty_table) do
			if type(m) == "table" and m.prop == want.Prop then
				n = n + 1
				detail[#detail + 1] = fmt("%s:%s", tostring(id), tostring(m.percent))
			end
		end
		table.sort(detail)
		parts[#parts + 1] = tostring(n)
		say(fmt("D2 ASTRO %s %s(%s): %d modifier(s) [%s]", tostring(when),
			want.Label, want.Prop, n, table.concat(detail, ",")))
	end
	return table.concat(parts, "/")
end

-- Reads the C34 structure precisely: its Lua type and how many entries it holds.
-- "false/0" is the forced defect state; "table/N" is healthy.
local function read_rains_exact(when)
	local say, fmt = U1.Say, U1.Fmt
	local threads = rawget(_G, "RainsDisasterThreads")
	local n = 0
	if type(threads) == "table" then for _ in pairs(threads) do n = n + 1 end end
	local s = fmt("%s/%d", type(threads), n)
	say(fmt("D2 RAINS %s: RainsDisasterThreads type=%s entries=%d | g_RainDisaster=%s",
		tostring(when), type(threads), n, tostring(rawget(_G, "g_RainDisaster"))))
	return s
end

function U1.Payload()
	local say, fmt, step = U1.Say, U1.Fmt, U1.Step
	local astro_applicable = false
	local a0, a1, a2, latch0, latch1, latch2, r0, r1, r2

	if not U1.save_proven then
		say("⛔ LEG D2 NOT RUN — the SAVE primitive was not proven in cycle 0. Routed gap.")
		return
	end

	if not step("2-load-staged", function() return U1.Load(U1.save_file, 3) end, true) then return end

	if not step("D2-0-baseline", function()
		U1.ReadConditions("leg D2, cold load — before any forcing")
		U1.ReadAllHeals("D2 baseline")
		a0 = read_astro_exact("baseline")
		latch0 = U1.ReadHealMeteorLatch()
		r0 = read_rains_exact("baseline")
		local get_profile = rawget(_G, "GetCommanderProfile")
		local profile = type(get_profile) == "function" and get_profile() or nil
		local pid = type(profile) == "table" and profile.id or "none"
		astro_applicable = (pid == "astrogeologist")
		say(fmt("D2 APPLICABILITY: commander profile=%s | H1 astro forceable=%s | H5 meteor latch forceable=true | H6 rains structure forceable=true",
			tostring(pid), tostring(astro_applicable)))
		if not astro_applicable then
			say("D2 NOTE — H1 is NOT forceable on this save because the colony does not run the Astrogeologist profile. That family stays UNSAMPLED and must be reported as unsampled, not as clean. H5 and H6 are forced below regardless; both are applicable on every save, which is why this leg produces a real measurement whatever the save contains.")
		end
	end) then return end

	----------------------------------------------------------------------
	-- Force the defect state
	----------------------------------------------------------------------
	if not step("D2-1-force", function()
		say("D2 FORCED (1/3): SMRFixPack_MeteorLatch = false  — recreating F88's pre-heal state")
		SMRFixPack_MeteorLatch = false
		say(fmt("D2 latch now=%s", tostring(rawget(_G, "SMRFixPack_MeteorLatch"))))

		if astro_applicable then
			local colony = rawget(_G, "UIColony")
			local removed = 0
			for _, want in ipairs(ASTRO_LABELS) do
				local by_id = (type(colony.label_modifiers) == "table" and colony.label_modifiers[want.Label]) or nil
				local kill = {}
				for id, m in pairs(by_id or empty_table) do
					if type(m) == "table" and m.prop == want.Prop then kill[#kill + 1] = id end
				end
				for _, id in ipairs(kill) do
					colony:SetLabelModifier(want.Label, id, nil)
					removed = removed + 1
				end
			end
			say(fmt("D2 FORCED (2/3): stripped %d Astrogeologist extractor modifier(s) — recreating F92's pre-heal state", removed))
			read_astro_exact("after forcing")
		else
			say("D2 FORCED (2/3): SKIPPED — H1 not applicable on this save")
		end

		-- ⛔ LAST, and immediately before the save. Between this assignment and
		-- the reload the shipped code would index a boolean
		-- (TerraformingDisasters.lua:190/:249/:280/:328) if a rain event fired.
		-- The window is seconds; a [LUA ERROR] naming TerraformingDisasters.lua
		-- inside it is a CONSEQUENCE OF THIS FORCING and is reported as that.
		say("D2 FORCED (3/3): RainsDisasterThreads = false — recreating C34's structure defect. ⛔ FORCING WINDOW OPENS")
		RainsDisasterThreads = false
		read_rains_exact("after forcing")
	end) then return end

	if not step("D2-2-save-forced", function()
		if not U1.Save(U1.save_a, "U1C6FORCED") then
			say("LEG D2 ABORT at the first save — record verbatim as an unforeseen-issue finding.")
			return false
		end
	end, true) then return end

	----------------------------------------------------------------------
	-- Load 2: the heal must FIRE, exactly once, landing on exactly 1 per label
	----------------------------------------------------------------------
	if not step("3-load-healing", function() return U1.Load(U1.save_a, 3) end, true) then return end

	if not step("D2-3-read-healed", function()
		U1.ReadConditions("leg D2, load 2 — the load that should heal")
		say("D2 ⛔ FORCING WINDOW CLOSED at the load above. THE HEAL LINES FOR THIS LOAD ARE THE PACK LINES IMMEDIATELY ABOVE THIS MARKER. Expect exactly one meteor one-shot-heal line, exactly one C34 'RainsDisasterThreads was false — recreated as an empty table' line, and (if applicable) one Astrogeologist applied-N line with N = 2.")
		a1 = read_astro_exact("after the healing load")
		latch1 = U1.ReadHealMeteorLatch()
		r1 = read_rains_exact("after the healing load")
		say(fmt("D2 PREDICTION CHECK load 2: astro counts %s -> %s (prediction: back to the baseline %s, and NEVER higher) | latch %s -> %s (prediction: back to the pack version, not false) | rains %s -> %s (prediction: back to a table)",
			tostring(a0), tostring(a1), tostring(a0), tostring(latch0), tostring(latch1),
			tostring(r0), tostring(r1)))
	end) then return end

	if not step("D2-4-save-healed", function()
		if not U1.Save(U1.save_b, "U1C6HEALED") then
			say("LEG D2 ABORT at the second save — record verbatim.")
			return false
		end
	end, true) then return end

	----------------------------------------------------------------------
	-- Load 3: the heal must NOT fire again, and nothing may grow
	----------------------------------------------------------------------
	if not step("4-load-idempotence", function() return U1.Load(U1.save_b, 3) end, true) then return end

	if not step("D2-5-read-idempotence", function()
		U1.ReadConditions("leg D2, load 3 — the idempotence load")
		say("D2 ⛔ ANY pack heal line between the previous LOAD OK marker and this line is the DEFECT: the state was already correct in the file that was just loaded.")
		a2 = read_astro_exact("after the idempotence load")
		latch2 = U1.ReadHealMeteorLatch()
		r2 = read_rains_exact("after the idempotence load")
		say(fmt("D2 PREDICTION CHECK load 3: astro %s -> %s (prediction: identical) | latch %s -> %s (prediction: identical) | rains %s -> %s (prediction: identical)",
			tostring(a1), tostring(a2), tostring(latch1), tostring(latch2),
			tostring(r1), tostring(r2)))
	end) then return end

	if not step("D2-verdict", function()
		say(fmt("LEG D2 SUMMARY  astro: baseline=%s healed=%s idempotence=%s | latch: %s / %s / %s | rains: %s / %s / %s | SAMPLED: H5 yes, H6 yes, H1 %s",
			tostring(a0), tostring(a1), tostring(a2),
			tostring(latch0), tostring(latch1), tostring(latch2),
			tostring(r0), tostring(r1), tostring(r2),
			astro_applicable and "yes" or "NO — unsampled, say so"))
		say("LEG D2 PASS requires ALL of: (1) a heal line for each forced family after load 2 and NONE after load 3; (2) the load-2 numbers equal to the baseline, never above it — a number ABOVE baseline is F92's compounding defect reproduced; (3) load-3 numbers identical to load 2.")
		say("LEG D2 ⛔ CEILING: MECHANISM. It says how these heals behave when they fire. It says nothing about how often a real save needs them, and nothing at all about H2/H3/H4, which were not forced.")
	end) then return end

	U1.ErrorWatchNote()
	say("=== cycle 6 (leg D2) complete ===")
end

U1.Boot()
