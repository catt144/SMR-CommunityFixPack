-- SMR Community Fix Pack — Test Kit :: unattended-1 CYCLE 2 — LEG A.  TEMPORARY
--
-- Declared by: unattended-1 chain prompt 1, for prompt 2. Armed with
-- 97_U1Common.lua; both die in the commit that records this cycle's answers.
--
-- LEG A — PT-35 "the save sanitizer does no harm", case A only.
--
-- WHAT PT-35 CASE A ACTUALLY ASKS (PLAYTEST_CHECKLIST.md:782-798), translated
-- from a keyboard test into reads, which is the whole reason it routed
-- UNATTENDED: "nothing on screen may change" becomes "the read-back numbers do
-- not change", which is the entry's own claim.
--   1. read the numbers;
--   2. SMRFixPack.Sanitizer.RepairTurbineBuff() must return 0;
--   3. SMRFixPack.Sanitizer.RepairLeakedUpgradeModifiers() must return 0;
--   4. the numbers must be unchanged;
--   5. save, reload, run both again — still 0, numbers still unchanged.
--      A bonus that GREW on the second run is the FAIL.
--
-- WHY BOTH RETURNING 0 IS THE PREDICTION AND NOT AN ASSUMPTION. Both passes are
-- written to be conservative and idempotent by construction:
--   * repair_turbine_buff (90_SaveSanitizer.lua:52-101) early-returns 0 unless
--     FrictionlessComposites is researched, and treats ANY existing non-zero
--     percent modifier for that prop on that label as "already buffed", so it
--     cannot double-apply;
--   * repair_leaked_upgrade_modifiers (:151-176) only removes ids matching
--     "^(%d+)_upgrade%d+_mod_%d+$" whose handle no longer resolves to a valid
--     object, and both passes have already run once on this load via
--     OnMsg.PostLoadGame (:205) before we call them by hand.
-- The second point is what makes 0 the prediction rather than a hope: the
-- automatic pass has already drained anything drainable, so the manual call is
-- being asked to repeat work that is finished. A non-zero return is therefore
-- interesting BOTH ways — it would mean the automatic pass did not run, or that
-- the passes are not idempotent.
--
-- ⚠️ Requires the SAVE primitive (step 5). If cycle 0's proof did not pass,
--    prompt 2 runs this cycle only up to step 4 and records the save/reload
--    half as a routed gap — the first four steps are still a real result.

U1 = rawget(_G, "U1") or {}
U1.tag       = "U1C2"
U1.watchdog  = 6 * 60 * 1000      -- 3x the ~1.5 min prediction, 6 min floor
U1.save_file = "U1STAGE.savegame.sav"
U1.leg_save  = "U1C2PT35.savegame.sav"

local function run_both(when)
	local say, fmt = U1.Say, U1.Fmt
	local san = (rawget(_G, "SMRFixPack") or empty_table).Sanitizer
	if type(san) ~= "table" then
		say("PT35 ABORT — SMRFixPack.Sanitizer is not present; the pack is not loaded as expected")
		return nil
	end
	local ok1, n1 = pcall(san.RepairTurbineBuff)
	local ok2, n2 = pcall(san.RepairLeakedUpgradeModifiers)
	say(fmt("PT35 CALL %s: RepairTurbineBuff ok=%s returned=%s | RepairLeakedUpgradeModifiers ok=%s returned=%s | PREDICTION both 0",
		tostring(when), tostring(ok1), tostring(n1), tostring(ok2), tostring(n2)))
	return (ok1 and n1 or -1), (ok2 and n2 or -1)
end

function U1.Payload()
	local say, fmt, step = U1.Say, U1.Fmt, U1.Step
	local before1, after1, before2, after2

	if not step("2-load-1-staged", function() return U1.Load(U1.save_file, 3) end, true) then return end

	if not step("A1-read-before", function()
		U1.ReadConditions("leg A, load 1 — cold load of the staged copy, BEFORE any manual sanitizer call")
		before1 = U1.ReadSanitizerNumbers()
	end) then return end

	if not step("A2-call-both", function()
		local t, u = run_both("load 1")
		U1.legA_turbine_1, U1.legA_upgrade_1 = t, u
	end) then return end

	if not step("A3-read-after", function()
		after1 = U1.ReadSanitizerNumbers()
		U1.CompareSanitizer(before1, after1, "load1 before->after")
	end) then return end

	if not U1.save_proven then
		say("⛔ LEG A stops here — the SAVE primitive was not proven in cycle 0. The save/reload half (PT-35 step 3) is a ROUTED GAP; steps 1-2 above stand on their own.")
		U1.ErrorWatchNote()
		return
	end

	if not step("A4-save", function()
		if not U1.Save(U1.leg_save, "U1C2PT35") then
			say("LEG A ABORT at the save — record verbatim; cycle 0 had proven this primitive, so a failure here is an unforeseen-issue finding.")
			return false
		end
	end, true) then return end

	if not step("3-load-2-reload", function() return U1.Load(U1.leg_save, 3) end, true) then return end

	if not step("A5-read-after-reload", function()
		U1.ReadConditions("leg A, load 2 — reload of the save written after the manual calls")
		before2 = U1.ReadSanitizerNumbers()
		U1.CompareSanitizer(after1, before2, "load1-after->load2-before")
	end) then return end

	if not step("A6-call-both-again", function()
		local t, u = run_both("load 2")
		U1.legA_turbine_2, U1.legA_upgrade_2 = t, u
	end) then return end

	if not step("A7-read-final", function()
		after2 = U1.ReadSanitizerNumbers()
		U1.CompareSanitizer(before2, after2, "load2 before->after")
		U1.CompareSanitizer(before1, after2, "start->finish")
		say(fmt("PT35 SUMMARY returns: load1 turbine=%s upgrade=%s | load2 turbine=%s upgrade=%s | PASS requires all four = 0 AND every PT35DIFF VERDICT = 0 changed",
			tostring(U1.legA_turbine_1), tostring(U1.legA_upgrade_1),
			tostring(U1.legA_turbine_2), tostring(U1.legA_upgrade_2)))
	end) then return end

	U1.ErrorWatchNote()
	say("=== cycle 2 (leg A) complete ===")
end

U1.Boot()
