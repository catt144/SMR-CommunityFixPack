# c48-brake ARM / DISARM script.  Parked source; copy to a scratch .ps1 and run.
#
# Inherits every lesson of C47_ARM.ps1 (c47-farm, recoverable at git fd1102a):
# rule C11 (arming is a FILE), the G5 read-back gate, the S3 BOM note
# (ASCII-only body), the no-BOM metadata write, and the family-wide strip
# pattern Code/9[78]_C4[78] - which matches 98_C48Brake as well as every
# c47-farm payload, so a disarm here also sweeps any stray from that chain.
#
#   Usage:  .\C48B_ARM.ps1 arm-brake    # 97_C47Common + 98_C48Brake: the intervention leg
#           .\C48B_ARM.ps1 disarm      # removes everything this script can add
#
# ------------------------------------------------------------------------------
# THE GATE THIS MODE NEEDS THAT NO EARLIER MODE HAD, and why it is explicit.
#
#   Every earlier unattended gate in this family BANNED mutation outright. This
#   leg exists to make exactly ONE declared mutation - the SurfaceDeposit brake
#   number applied to VegetationTaskRequester, plus the request rebuild that
#   makes it reach the C-side pairing - so the gate flips from "no mutating
#   calls" to "exactly the declared mutating calls, and the declaration text
#   must be present". A payload that mutates without declaring, or declares
#   without the FORCED label, fails. SaveGame and the farm/crop mutators stay
#   banned: the staged copy must be byte-identical after the run.
# ------------------------------------------------------------------------------

param([Parameter(Mandatory = $true)][ValidateSet("arm-brake", "disarm")][string]$mode)

$ErrorActionPreference = "Stop"
$park = "c:\Dev\SMR-BugFixPack\docs\agent\prompts\c48-brake"
$kit  = "c:\Dev\SMR-BugFixPack-TestKit"
$meta = Join-Path $kit "metadata.lua"
$noBom = New-Object System.Text.UTF8Encoding($false)

$common  = @{ src = "97_C47Common.lua.txt"; dst = "Code\97_C47Common.lua" }
$payload = @{ src = "98_C48Brake.lua.txt"; dst = "Code\98_C48Brake.lua" }
$pairs   = @()
if ($mode -eq "arm-brake") { $pairs = @($common, $payload) }

$autorunLine = 'Code/96_AutoRunFlag.lua'

# --- clean slate every time ---------------------------------------------------
foreach ($p in @($common, $payload)) {
  $f = Join-Path $kit $p.dst
  if (Test-Path -LiteralPath $f) { Remove-Item -LiteralPath $f -Force; Write-Output ("DELETED " + $p.dst) }
}
$lines = Get-Content -LiteralPath $meta -Encoding UTF8
$kept = @()
foreach ($l in $lines) {
  if ($l -match 'Code/9[78]_C4[78]') { continue }
  if ($l -match [regex]::Escape($autorunLine)) { continue }
  $kept += $l
}
[System.IO.File]::WriteAllLines($meta, $kept, $noBom)
Write-Output ("metadata.lua: stripped " + ($lines.Count - $kept.Count) + " arming line(s)")

# --- arm ----------------------------------------------------------------------
if ($mode -ne "disarm") {
  foreach ($p in $pairs) {
    Copy-Item -LiteralPath (Join-Path $park $p.src) -Destination (Join-Path $kit $p.dst) -Force
    Write-Output ("WROTE " + $p.dst)
  }
  $lines = Get-Content -LiteralPath $meta -Encoding UTF8
  $out = New-Object System.Collections.Generic.List[string]
  $inserted = $false
  foreach ($l in $lines) {
    $out.Add($l)
    if (-not $inserted -and $l -match '99_FixtureCarry\.lua') {
      foreach ($p in $pairs) {
        $leaf = Split-Path $p.dst -Leaf
        $out.Add(("`t`t`"Code/" + $leaf + "`",  -- TEMPORARY c48-brake"))
      }
      $inserted = $true
    }
  }
  if (-not $inserted) { Write-Output "ARM FAIL: anchor 99_FixtureCarry.lua not found"; exit 1 }
  [System.IO.File]::WriteAllLines($meta, $out, $noBom)
  Write-Output "WROTE metadata.lua (no BOM)"
}

# ---- gate --------------------------------------------------------------------
$fail = $false
$m = Get-Content -LiteralPath $meta -Raw -Encoding UTF8

# The permanent wave-11 probe must survive every mode of this script.
if ($m -match [regex]::Escape('"Code/61_Probes_Wave11.lua"')) {
  Write-Output "GATE OK       metadata still lists the permanent Code/61_Probes_Wave11.lua"
} else {
  Write-Output "GATE FAIL: metadata no longer lists Code/61_Probes_Wave11.lua"; $fail = $true
}
if (Test-Path -LiteralPath (Join-Path $kit "Code\61_Probes_Wave11.lua")) {
  Write-Output "GATE OK       Code/61_Probes_Wave11.lua is on disk"
} else {
  Write-Output "GATE FAIL: Code/61_Probes_Wave11.lua is missing from disk"; $fail = $true
}

if ($mode -eq "arm-brake") {
  if ($m -match [regex]::Escape('"' + $autorunLine + '"')) {
    Write-Output "ARM GATE FAIL: 96_AutoRunFlag is still listed - it would start a NEW COLONY and race this payload's LoadGame"; $fail = $true
  } else { Write-Output "ARM GATE OK   96_AutoRunFlag is NOT listed" }
  foreach ($p in $pairs) {
    $f = Join-Path $kit $p.dst
    if (-not (Test-Path -LiteralPath $f)) { Write-Output ("ARM GATE FAIL: missing " + $p.dst); $fail = $true; continue }
    $t = Get-Content -LiteralPath $f -Raw -Encoding UTF8
    if ($t -notmatch 'TEMPORARY') { Write-Output ("ARM GATE FAIL: no TEMPORARY marker in " + $p.dst); $fail = $true }
    else { Write-Output ("ARM GATE OK   " + $p.dst) }
    if ($m -match [regex]::Escape('"Code/' + (Split-Path $p.dst -Leaf) + '"')) { Write-Output ("ARM GATE OK   metadata lists " + $p.dst) }
    else { Write-Output ("ARM GATE FAIL: metadata does not list " + $p.dst); $fail = $true }
  }

  $brText = Get-Content -LiteralPath (Join-Path $kit $payload.dst) -Raw -Encoding UTF8
  $brCode = ($brText -split "`r?`n" | ForEach-Object { ($_ -replace '--.*$', '') } | Where-Object { $_.Trim() -ne '' }) -join "`n"
  $brCode = $brCode -replace 'OnMsg\.[A-Za-z_]\w*', 'OnMsgHandler'

  # self-drive (unattended)
  if ($brCode -match '(?m)^\s*C47\.Boot\s*\(') { Write-Output "ARM GATE OK   payload self-drives (C47.Boot() at file scope)" }
  else { Write-Output "ARM GATE FAIL: payload does not call C47.Boot() at file scope"; $fail = $true }
  if ($brCode -match 'quit\s*\(') { Write-Output "ARM GATE OK   payload quits the process itself" }
  else { Write-Output "ARM GATE FAIL: payload never calls quit()"; $fail = $true }

  # STILL banned, mutation or not: nothing that writes a save or drives a farm.
  $w = @()
  foreach ($banned in @('GetCurrentHarvestType', 'SaveGame', 'CheatFill', 'SetHarvestType',
                        'PlaceVegetation', 'Consume_Internal', 'DoneObject', 'SetAmount', 'AddAmount')) {
    if ($brCode -match ([regex]::Escape($banned) + '\s*\(')) { $w += $banned }
  }
  if ($w.Count -gt 0) { Write-Output ("ARM GATE FAIL: the payload CALLS " + ($w -join ", ") + " - outside the declared mutation set"); $fail = $true }
  else { Write-Output "ARM GATE OK   payload calls nothing outside the declared mutation set (no SaveGame / farm / crop / amount mutator)" }

  # THE DECLARED-MUTATION GATE: the intervention must be present, labelled and
  # restorable - a brake payload without these is either inert or smuggling.
  foreach ($needed in @('supply_dist_modifier', 'RemoveBuilding', 'AddBuilding', 'CreateResourceRequests', 'RestoreBrake')) {
    if ($brCode -match [regex]::Escape($needed)) { Write-Output ("ARM GATE OK   declared mutation present: " + $needed) }
    else { Write-Output ("ARM GATE FAIL: declared mutation missing: " + $needed); $fail = $true }
  }
  if ($brText -match '"FORCED"') { Write-Output "ARM GATE OK   the intervention is labelled FORCED" }
  else { Write-Output "ARM GATE FAIL: no FORCED label on the intervention"; $fail = $true }
  if ($brCode -match '150') { Write-Output "ARM GATE OK   the brake value is the SurfaceDeposit 150" }
  else { Write-Output "ARM GATE FAIL: the payload never states the 150"; $fail = $true }

  # both windows must exist, or there is no comparison
  foreach ($win in @('"baseline"', '"braked"')) {
    if ($brText -match [regex]::Escape($win)) { Write-Output ("ARM GATE OK   window present: " + $win) }
    else { Write-Output ("ARM GATE FAIL: window missing: " + $win); $fail = $true }
  }

  # G1 resolution cross-check
  $comText = Get-Content -LiteralPath (Join-Path $kit $common.dst) -Raw -Encoding UTF8
  $both = $comText + "`n" + $brText
  $defined = @{}
  foreach ($mm in [regex]::Matches($both, '(?m)^\s*function\s+C47\.([A-Za-z_]\w*)')) { $defined[$mm.Groups[1].Value] = $true }
  foreach ($mm in [regex]::Matches($both, '(?m)^\s*C47\.([A-Za-z_]\w*)\s*=')) { $defined[$mm.Groups[1].Value] = $true }
  $used = @{}
  foreach ($mm in [regex]::Matches($brCode, 'C47\.([A-Za-z_]\w*)')) { $used[$mm.Groups[1].Value] = $true }
  $un = @(); foreach ($k in $used.Keys) { if (-not $defined.ContainsKey($k)) { $un += $k } }
  if ($un.Count -gt 0) { Write-Output ("ARM GATE FAIL: unresolved C47.* names: " + ($un -join ", ")); $fail = $true }
  else { Write-Output ("ARM GATE OK   all " + $used.Count + " C47.* names resolve") }
}

if ($mode -eq "disarm") {
  foreach ($p in @($common, $payload)) {
    if (Test-Path -LiteralPath (Join-Path $kit $p.dst)) { Write-Output ("DISARM FAIL: still present " + $p.dst); $fail = $true }
  }
  if ($m -match 'Code/9[678]_(C4[78]|AutoRunFlag)') { Write-Output "DISARM FAIL: metadata still lists an arming line"; $fail = $true }
  if (-not $fail) { Write-Output "DISARM OK - kit is clean, and the permanent wave-11 probe is intact" }
}

if ($fail) { Write-Output "GATE RESULT: FAIL - DO NOT LAUNCH"; exit 1 }
Write-Output "GATE RESULT: PASS"
exit 0
