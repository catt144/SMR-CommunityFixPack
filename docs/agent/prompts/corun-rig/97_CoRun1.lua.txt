-- ModLog is the only print path proven to reach the log file on disk
-- (Mod.lua:109-132). Its emitter formats its single argument a second time
-- (lib.lua:164-174), so a literal '%' raises there — escape it. Flush per line
-- so a killed run still shows how far it got (2026-07-25 wedge lesson).
local t0 = false
local function say(msg)
	if not t0 then t0 = RealTime() end
	msg = "[CORUN1] +" .. tostring(RealTime() - t0) .. "ms " .. tostring(msg)
	local modlog = rawget(_G, "ModLog")
	if modlog then
		modlog((msg:gsub("%%", "%%%%")))
	else
		local dbg = rawget(_G, "DebugPrint")
		if dbg then dbg(msg .. "\n") end
	end
	print(msg)
	local flush = rawget(_G, "FlushLogFile")
	if flush then pcall(flush) end
end
CoRun1.Say = say

local function fmt(...)
	local ok, s = pcall(string.format, ...)
	return ok and s or "<fmt error>"
end

local function id_of(obj)
	if not obj then return "nil" end
	if type(obj) ~= "table" then return tostring(obj) end
	local h = rawget(obj, "handle")
	local cls = obj.class or "?"
	return tostring(cls) .. "#" .. tostring(h)
end

local function find_in_holder(unit)
	local h = unit and unit.holder
	if not h or type(h) ~= "table" then return "no-holder" end
	local units = h.units
	if not units then return "units=nil" end
	return tostring(table.find(units, unit)) .. "/" .. tostring(#units)
end

---- flow helpers ------------------------------------------------------------

local function shutdown(reason)
	CoRun1.stage = "done"
	say("SHUTDOWN " .. tostring(reason))
	local flush = rawget(_G, "FlushLogFile")
	if flush then flush() end
	Sleep(CoRun1.quit_delay)
	quit()
end

-- Steps Sleep, so sprocall is mandatory (pcall cannot survive a yield).
local function step(name, fn)
	CoRun1.stage = name
	local t = RealTime()
	say("STEP " .. name .. " BEGIN")
	local ok, res = sprocall(fn)
	if not ok then
		say("STEP " .. name .. " ERROR")
		shutdown("ERROR " .. name)
		return false
	end
	say(fmt("STEP %s END %dms", name, RealTime() - t))
	if res == false then
		shutdown("TIMEOUT " .. name)
		return false
	end
	return true
end

-- Never raises: error() does not unwind mod code in this engine, so a raising
-- wait would spin until the watchdog killed the process (95_AutoRun's lesson).
local function wait_for(timeout, ready)
	local deadline = RealTime() + timeout
	while not ready() do
		if RealTime() - deadline >= 0 then return false end
		Sleep(500)
	end
	return true
end

local function game_is_live()
	local city   = rawget(_G, "MainCity")
	local colony = rawget(_G, "UIColony")
	return not not (city and city.labels and colony and colony.labels)
end

-- Owner gate. The console runs in the mod sandbox but reads and writes of
-- NON-blacklisted names go straight to the real _G (console.lua:48,53), so
-- `CoRun1_Go = 1` typed as a console statement is visible here. That is
-- SOURCE-verified, never executed by us — so every gate has a TIMEOUT and the
-- run proceeds without the owner rather than wedging on an unproven primitive.
-- The log records which of the two happened, which is what verifies it.
local function wait_gate(n, timeout, what)
	say(fmt("GATE %d WAITING (%s) — owner types in the console:  CoRun1_Go = %d", n, what, n))
	local got = wait_for(timeout, function()
		local v = rawget(_G, "CoRun1_Go")
		return type(v) == "number" and v >= n
	end)
	say(fmt("GATE %d %s", n, got and "RELEASED by owner" or "TIMED OUT — proceeding on the clock"))
	return got
end

local function label_all(name)
	local out = {}
	local cities = rawget(_G, "Cities") or empty_table
	for _, city in ipairs(cities) do
		for _, o in ipairs((city.labels or empty_table)[name] or empty_table) do
			out[#out + 1] = o
		end
	end
	return out
end

		tostring(CoRun1.wrap_done), tostring(CoRun1.wrap_enter),
		tostring(CoRun1.wrap_exit_holder), tostring(CoRun1.wrap_comfort),

---- L1: F11 pre-wrapper watch (Tier A) --------------------------------------

local function leg1_f11_watch()
	local trains = label_all("Train")
	local target, best = false, -1
	for _, t in ipairs(trains) do
		local n = #(t.units or empty_table)
		if n > best then best, target = n, t end
	end
	if not target then
		say("L1 SKIP — no train found")
		return
	end
	say(fmt("L1 TARGET train=%s units=%d command=%s station=%s map=%s",
		id_of(target), #(target.units or empty_table), tostring(target.command),
		id_of(target.current_station), tostring(target:GetMap() == MainMap and "MainMap" or "other")))

	local vas = rawget(_G, "ViewAndSelectObject")
	if vas then pcall(vas, target) end
	say("L1 camera moved to the target train (ViewAndSelectObject)")

	CoRun1.f11_watch = true
	local start = {
		exit  = CoRun1.count.exit_holder_train,
		comf  = CoRun1.count.comfort_travel,
		san   = CoRun1.count.sanity_forest,
	}

	-- Objective counters run for the whole window; the poll below is the
	-- liveness witness beside them (a wedge is a train whose command stays
	-- UnloadTrain while #units never falls).
	local watched = {}
	for _, t in ipairs(trains) do
		watched[t] = { cmd = t.command, units = #(t.units or empty_table), unload_since = false }
	end

	local deadline = RealTime() + 180 * 1000
	local wedges = 0
	while RealTime() < deadline do
		if rawget(_G, "CoRun1_Go") and rawget(_G, "CoRun1_Go") >= 1 then break end
		for t, st in pairs(watched) do
			if IsValid(t) then
				local cmd, n = t.command, #(t.units or empty_table)
				if cmd ~= st.cmd or n ~= st.units then
					say(fmt("L1 TRAIN %s %s(%d) -> %s(%d) station=%s",
						id_of(t), tostring(st.cmd), st.units, tostring(cmd), n,
						id_of(t.current_station)))
					st.cmd, st.units = cmd, n
					st.unload_since = (cmd == "UnloadTrain") and GameTime() or false
				elseif cmd == "UnloadTrain" and st.unload_since
					and GameTime() - st.unload_since > 20 * const.HourDuration then
					wedges = wedges + 1
					say(fmt("L1 ⛔ WEDGE CANDIDATE train=%s has been in UnloadTrain for %d game-hours with %d units",
						id_of(t), (GameTime() - st.unload_since) / const.HourDuration, n))
					st.unload_since = GameTime()
				end
			end
		end
		Sleep(1000)
	end
	wait_gate(1, 60 * 1000, "L1 F11 watch — say when you have seen one unload")
	CoRun1.f11_watch = false

	say(fmt("L1 COUNTERS over the window: train-holder removals=%d, 'travel time' comfort calls=%d, 'seen forest' sanity calls=%d, wedge candidates=%d",
		CoRun1.count.exit_holder_train - start.exit,
		CoRun1.count.comfort_travel - start.comf,
		CoRun1.count.sanity_forest - start.san,
		wedges))
end

---- L2: C41 amplification (Tier A) ------------------------------------------

local function leg2_c41()
	-- Half A: the mouse-range poll. C41's ONE measured lead is that
	-- terminal.GetMousePos() can return coordinates outside desktop.box
	-- (F76 pass #8: mouse (6148,2350) vs desk (0,0)-(3840,2160)). This needs
	-- no picker at all — poll the same call and count excursions.
	local dbox = terminal.desktop.box
	say(fmt("L2A mouse-range poll BEGIN desk=%s — owner: move the cursor over both monitors", tostring(dbox)))
	local n, out = 0, 0
	local minx, maxx, miny, maxy = 1000000, -1000000, 1000000, -1000000
	local deadline = RealTime() + 60 * 1000
	while RealTime() < deadline do
		if rawget(_G, "CoRun1_Go") and rawget(_G, "CoRun1_Go") >= 2 then break end
		local p = terminal.GetMousePos()
		local x, y = p:xy()
		n = n + 1
		if x < minx then minx = x end
		if x > maxx then maxx = x end
		if y < miny then miny = y end
		if y > maxy then maxy = y end
		if not dbox:Point2DInside(p) then
			out = out + 1
			if out <= 5 or out % 25 == 0 then
				say(fmt("L2A ⭐ OUT-OF-RANGE mouse=%s desk=%s (hit %d of %d reads)", tostring(p), tostring(dbox), out, n))
			end
		end
		Sleep(200)
	end
	say(fmt("L2A RESULT %d/%d reads outside desktop.box | x range %d..%d | y range %d..%d | desk=%s",
		out, n, minx, maxx, miny, maxy, tostring(dbox)))

	-- Half B: the amplification loop proper. Open the shipped picker N times
	-- and record, per cycle, whether the dialog exists afterwards and where it
	-- landed. FORCED: the open — OpenResourceSelector is called directly
	-- instead of being reached through RCTransport:InteractWithObject's
	-- storage branch (RCTransport.lua:419-424). The context is that call
	-- site's own context, verbatim. ORGANIC: the anchor — ResourceItems:Init
	-- reads the live mouse either way (ResourceItems.lua:11), which is the only
	-- quantity C41's lead is about.
	local rc = false
	for _, v in ipairs(label_all("RCTransport")) do
		if IsValid(v) then rc = v break end
	end
	local depot = false
	for _, d in ipairs(label_all("UniversalStorageDepot")) do
		local ok, stored = pcall(function() return d:GetStoredAmount() end)
		if IsValid(d) and ok and (stored or 0) > 0 and #(d.resource or empty_table) > 0 then
			depot = d
			break
		end
	end
	if not (rc and depot) then
		say(fmt("L2B SKIP — need an RCTransport and a stocked UniversalStorageDepot (rc=%s depot=%s)", id_of(rc), id_of(depot)))
		return
	end
	local _, stored = pcall(function() return depot:GetStoredAmount() end)
	say(fmt("L2B amplification BEGIN rc=%s depot=%s stored=%s resources=%d cycles=20 — owner: watch for the picker FAILING to appear",
		id_of(rc), id_of(depot), tostring(stored), #(depot.resource or empty_table)))

	CoRun1.picker_log = true
	local appeared, missing = 0, 0
	for i = 1, 20 do
		local mouse = terminal.GetMousePos()
		CoRun1.count.picker_items = -1
		local ok, err = pcall(function()
			OpenResourceSelector(rc, { target = depot, meta_key = const.vkControl,
				close_on_rmb = true, interaction_mode = "load" })
		end)
		Sleep(400)
		local dlg = GetDialog("ResourceItems")
		if dlg then
			appeared = appeared + 1
			say(fmt("L2B cycle %d/20 dialog=ALIVE box=%s items=%d mouse_at_open=%s",
				i, tostring(dlg.box), CoRun1.count.picker_items, tostring(mouse)))
		else
			missing = missing + 1
			say(fmt("L2B cycle %d/20 dialog=MISSING ok=%s err=%s items=%d mouse_at_open=%s",
				i, tostring(ok), tostring(err), CoRun1.count.picker_items, tostring(mouse)))
		end
		pcall(CloseResourceSelector)
		Sleep(400)
	end
	CoRun1.picker_log = false
	say(fmt("L2B RESULT %d/20 opened, %d/20 missing | layouts logged=%d, of which anchor outside desktop.box=%d",
		appeared, missing, CoRun1.count.picker_layouts, CoRun1.count.picker_anchor_out))
end

---- L3: F11 cross-map Done-timing (ride-along) ------------------------------

local function leg3_done_timing()
	-- Co-run #0's pair was same-map, so OnTransferToMapDone never fired and
	-- routes (a)/(b) were untouched. This one SELECTS for cross-map and
	-- asserts it before driving anything.
	local rocket = false
	for _, r in ipairs(label_all("AllRockets")) do
		if IsValid(r) and r:GetMap() == MainMap and r.landed and IsKindOf(r, "Holder") then
			rocket = r
			break
		end
	end
	local col = false
	for _, t in ipairs(label_all("Train")) do
		if IsValid(t) and rocket and t:GetMap() ~= rocket:GetMap() then
			for _, u in ipairs(t.units or empty_table) do
				if IsValid(u) and IsKindOf(u, "Colonist") then col = u break end
			end
		end
		if col then break end
	end
	if not (rocket and col) then
		say(fmt("L3 SKIP — cross-map pair not available (rocket=%s, off-map rider=%s). This is a §6 gap, not a failure.",
			id_of(rocket), id_of(col)))
		return
	end
	local cross = rocket:GetMap() ~= col:GetMap()
	say(fmt("L3 PAIR rider=%s holder=%s rocket=%s cross_map=%s", id_of(col), id_of(col.holder), id_of(rocket), tostring(cross)))
	if not cross then
		say("L3 ABORT — cross_map is false; this leg refuses to repeat co-run #0's same-map result")
		return
	end

	say(fmt("L3 BEFORE find=%s #units=%s holder=%s col_map=%s rocket_map=%s",
		find_in_holder(col), tostring(col.holder and #(col.holder.units or empty_table)),
		id_of(col.holder), tostring(col:GetMap()), tostring(rocket:GetMap())))

	-- The instrument is ORDER, not a stack trace: `debug` is on
	-- ModEnvBlacklist so a caller cannot be printed from mod code. Every wrap
	-- stamps a shared sequence number, so a Holder.OnExitHolder that lands
	-- BETWEEN OnTransferToMapDone ENTER and EXIT is route (a), and one that
	-- lands inside EnterTransporter with no Done around it is route (b).
	CoRun1.trace_unit = col
	CoRun1.trace = true
	CoRun1.count.done_fired = 0
	say("L3 FORCED: col:SetCommand(\"EnterTransporter\", rocket) — verbatim the body of CargoTransporter:ExpeditionLoadCrew (CargoTransporter.lua:300-302), driven on a chosen target")
	col:SetCommand("EnterTransporter", rocket)
	Sleep(15 * 1000)
	CoRun1.trace = false

	say(fmt("L3 AFTER find=%s holder=%s holder_is_rocket=%s Done_fired=%d",
		find_in_holder(col), id_of(col.holder), tostring(col.holder == rocket), CoRun1.count.done_fired))
end

---- L4: F99 hex tie-break (ride-along, mutates track — runs LAST) -----------

local function leg4_hex_tiebreak()
	-- BreakTracks (Meteors.lua:599-607) is the shipped producer: for each
	-- element that is neither start_el nor end_el and has no station, it calls
	-- track:BreakTrackElement(element, cg). Driving that call site directly on
	-- a chosen element is the same technique F11's sitting used — it removes
	-- the lottery, not the mechanism, and it costs no disaster.
	local target_track, target_el = false, false
	for _, tr in ipairs(label_all("TrackBase")) do
		for _, el in ipairs(tr.elements or empty_table) do
			if IsValid(el) and not el.broken and el ~= tr.start_el and el ~= tr.end_el and not el.station then
				target_track, target_el = tr, el
				break
			end
		end
		if target_el then break end
	end
	if not target_el then
		say("L4 SKIP — no breakable mid-track element found (needs: not start_el, not end_el, no station)")
		return
	end

	local map = ResolveMap(target_el)
	local q, r = target_el.q, target_el.r
	local before = HexGetTrackGridElement(map.object_hex_grid, q, r)
	say(fmt("L4 CONTROL (before the break) hex=(%d,%d) HexGetTrackGridElement -> %s | is the element itself=%s",
		q, r, id_of(before), tostring(before == target_el)))

	say(fmt("L4 FORCED: track:BreakTrackElement(element) — the shipped call site BreakTracks uses (Meteors.lua:604), on a chosen element. track=%s element=%s",
		id_of(target_track), id_of(target_el)))
	local ok, err = pcall(function() return target_track:BreakTrackElement(target_el, nil) end)
	if not ok then
		say("L4 BREAK FAILED: " .. tostring(err))
		return
	end
	Sleep(1000)

	local site = target_el.broken
	say(fmt("L4 STATE after break: element=%s visible=%s | site=%s site.broken==element=%s | site is a TrackGridElement=%s",
		id_of(target_el), tostring(target_el:GetVisible()), id_of(site),
		tostring(site and site.broken == target_el),
		tostring(site and IsKindOf(site, "TrackGridElement"))))
	say(fmt("L4 both candidates occupy the same hex: element q,r=(%s,%s) site q,r=(%s,%s)",
		tostring(target_el.q), tostring(target_el.r),
		tostring(site and site.q), tostring(site and site.r)))

	local after = HexGetTrackGridElement(map.object_hex_grid, q, r)
	local verdict
	if after == target_el then verdict = "THE HIDDEN ELEMENT"
	elseif after == site then verdict = "THE REPAIR SITE"
	elseif not after then verdict = "NEITHER (nil)"
	else verdict = "A THIRD OBJECT" end
	say(fmt("L4 ⭐ TIE-BREAK READ: HexGetTrackGridElement(hex_grid, %d, %d) -> %s  ==> %s", q, r, id_of(after), verdict))
	say(fmt("L4 F99 consequence: process_alien_elements' neighbour walk (TrackElement.lua:705-711) sees %s on this hex", verdict))

	-- Second read one frame later, in case placement is deferred.
	Sleep(2000)
	local after2 = HexGetTrackGridElement(map.object_hex_grid, q, r)
	say(fmt("L4 CONFIRM (2 s later): -> %s | same as first read=%s", id_of(after2), tostring(after2 == after)))
end

---- the leg body ------------------------------------------------------------

function CoRun1.Flow()
	CreateRealTimeThread(function()
		local deadline = RealTime() + CoRun1.watchdog
		while CoRun1.stage ~= "done" do
			if RealTime() - deadline >= 0 then
				say("WATCHDOG TIMEOUT at stage " .. tostring(CoRun1.stage))
				local flush = rawget(_G, "FlushLogFile")
				if flush then flush() end
				Sleep(1000)
				quit()
				return
			end
			Sleep(1000)
		end
	end)

	say("=== CO-RUN #1 — the payload run (03_OPUS_PAYLOAD_RUN.md) ===")
	say(fmt("save file: %s | watchdog %dms", CoRun1.save_file, CoRun1.watchdog))

	if not step("1-waiting-menu", function()
		if not wait_for(CoRun1.menu_timeout, function() return not not GetPreGameMainMenu() end) then
			return false
		end
		Sleep(2000)
	end) then return end

	-- ⛔ Timed by the engine's own "Game loaded on map ... in Nms" line, NOT by
	-- the RealTime() delta this step prints (correction C1 / EF-045).
	if not step("2-loading-save", function()
		local err = LoadGame(CoRun1.save_file, {})
		if err then
			say("LOADERR " .. tostring(err))
			return false
		end
		say("LOAD OK — LoadGame returned no error")
	end) then return end

	if not step("3-waiting-city", function() return wait_for(5 * 60 * 1000, game_is_live) end) then return end
	if not step("3b-settling", function() Sleep(CoRun1.settle_time) end) then return end

	if not step("4-census", leg0_census) then return end

	-- C4: a freshly loaded save arrives PAUSED. Everything game-time below is
	-- dead until this runs, so it runs before any leg and reads back.
	if not step("4b-speed", function()
		local before = GetTimeFactor() / const.DefaultTimeFactor
		local ok = pcall(function() UIColony:SetGameSpeed(3) end)
		Sleep(500)
		local after = GetTimeFactor() / const.DefaultTimeFactor
		say(fmt("READ S5 game speed: SetGameSpeed(3) call=%s | before=%s after=%s", tostring(ok), tostring(before), tostring(after)))
	end) then return end

	say("=== OWNER SITS DOWN NOW — legs 1 and 2 need your eyes; see CORUN1_BRIEF.md ===")

	if not step("L1-f11-watch", leg1_f11_watch) then return end
	if not step("L2-c41-amplify", leg2_c41) then return end

	say("=== OWNER IS FREE — the remaining legs need no eyes ===")
	wait_gate(3, 30 * 1000, "you may leave; ride-alongs start now")

	if not step("L3-done-timing", leg3_done_timing) then return end
	if not step("L4-hex-tiebreak", leg4_hex_tiebreak) then return end

	say("=== CO-RUN #1 legs complete ===")
	shutdown("done")
end

	end
