-- Probe hygiene rule 5: present in Code/ only while its run is happening.
--
-- WHY THERE IS A RUN 3. Run 2 pre-registered a bottom-right CORNER slam, saw
-- only the x clamp fire, and recorded the corner half as REFUTED on the stated
-- grounds that "the second monitor is not tall enough to produce a y anchor
-- past 2160". ⛔ That reason was false. The owner's display readouts:
--
--     display 1  Odyssey G7      3840 x 2160  at (0,0)      <- game is fullscreen here
--     display 2  BenQ RD280UG    3840 x 2560  at (3840,0)
--     virtual desktop            0..7680 x 0..2560
--
-- Display 2 is 400 px TALLER than the game's box, so a cursor in its bottom
-- strip gives y > 2160 and fires the y clamp too. Run 2's parked cursor sat at
-- y≈1259 and never sampled it. "Refuted" was really "unsampled", and this run
-- exists to make the difference stop mattering: the owner's stated reason for
-- wanting it is that a later agent must not be able to explain some future
-- error away as "your mouse was at the bottom of the BenQ".
--
-- ⛔ THE PREDICTION IS COMPUTED AND LOGGED BEFORE ANY PICKER OPENS, from the
-- measured dialog size, so this run cannot be read as fitting a story to
-- whatever arrived. ResourceItems:UpdateLayout (ResourceItems.lua:45-71),
-- Margins = box(0,0,0,0) (inherited, ItemsMenu.lua:5), safe area measured
-- 0 0 3840 2160:
--     x = anchor.x - width/2                 then, if x+width > 3840:  x = 3840 - width
--     y = anchor.y - height                  then, if y+height > 2160: y = 2160 - height
-- With run 2's measured mw=1616 mh=429 that is box (2224,1731)-(3840,2160):
-- the bottom-right corner exactly, both clamps firing.
--
-- The flip branch (:53-56) must NOT fire: it needs y < 0, and y here is
-- anchor.y - height ≈ 2350-429 ≈ 1921. Logged so the distinction is on record.

CoRun1 = rawget(_G, "CoRun1") or {}

CoRun1.save_file    = "CORUN1.savegame.sav"
CoRun1.menu_timeout = 5 * 60 * 1000
CoRun1.settle_time  = 15 * 1000
CoRun1.watchdog     = 15 * 60 * 1000
CoRun1.quit_delay   = 3 * 1000
CoRun1.stage        = "idle"
CoRun1.picker_log   = false
CoRun1.count = { layouts = 0, items = -1 }

local t0 = false
local function say(msg)
	if not t0 then t0 = RealTime() end
	msg = "[CORUN1C] +" .. tostring(RealTime() - t0) .. "ms " .. tostring(msg)
	local modlog = rawget(_G, "ModLog")
	if modlog then
		modlog((msg:gsub("%%", "%%%%")))
	else
		local dbg = rawget(_G, "DebugPrint")
		if dbg then dbg(msg .. "\n") end
	end
	print(msg)
	local flush = rawget(_G, "FlushLogFile")
	if flush then pcall(flush) end
end

local function fmt(...)
	local ok, s = pcall(string.format, ...)
	return ok and s or "<fmt error>"
end

local function id_of(obj)
	if not obj then return "nil" end
	if type(obj) ~= "table" then return tostring(obj) end
	return tostring(obj.class or "?") .. "#" .. tostring(rawget(obj, "handle"))
end

---- instrument, at mod-file scope -------------------------------------------

do
	local RI = rawget(_G, "ResourceItems")
	if RI and type(RI.UpdateLayout) == "function" then
		local orig = RI.UpdateLayout
		function RI:UpdateLayout()
			local r = orig(self)
			if CoRun1.picker_log then
				CoRun1.count.layouts = CoRun1.count.layouts + 1
				local dbox = terminal.desktop.box
				local a = self.align_pos
				say(fmt("L2D LAYOUT #%d anchor=%s anchor_x_out=%s anchor_y_out=%s box=%s mw=%s mh=%s scale=%s desk=%s",
					CoRun1.count.layouts, tostring(a),
					tostring(a and a:x() > dbox:maxx()), tostring(a and a:y() > dbox:maxy()),
					tostring(self.box), tostring(self.measure_width),
					tostring(self.measure_height), tostring(self.scale), tostring(dbox)))
			end
			return r
		end
		CoRun1.wrap_picker = true
	end
	if RI and type(RI.GetItems) == "function" then
		local orig = RI.GetItems
		function RI:GetItems(category_id)
			local b = orig(self, category_id)
			if CoRun1.picker_log then CoRun1.count.items = #(b or empty_table) end
			return b
		end
	end
end

---- flow helpers ------------------------------------------------------------

local function shutdown(reason)
	CoRun1.stage = "done"
	say("SHUTDOWN " .. tostring(reason))
	local flush = rawget(_G, "FlushLogFile")
	if flush then flush() end
	Sleep(CoRun1.quit_delay)
	quit()
end

local function step(name, fn)
	CoRun1.stage = name
	local t = RealTime()
	say("STEP " .. name .. " BEGIN")
	local ok, res = sprocall(fn)
	if not ok then say("STEP " .. name .. " ERROR") shutdown("ERROR " .. name) return false end
	say(fmt("STEP %s END %dms", name, RealTime() - t))
	if res == false then shutdown("TIMEOUT " .. name) return false end
	return true
end

local function wait_for(timeout, ready)
	local deadline = RealTime() + timeout
	while not ready() do
		if RealTime() - deadline >= 0 then return false end
		Sleep(500)
	end
	return true
end

local function label_all(name)
	local out = {}
	for _, city in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, o in ipairs((city.labels or empty_table)[name] or empty_table) do
			out[#out + 1] = o
		end
	end
	return out
end

---- L2D: both clamps --------------------------------------------------------

local function leg2d_corner()
	local dbox = terminal.desktop.box
	local sx1, sy1, sx2, sy2 = UIL.GetSafeArea()
	say(fmt("L2D env: desk=%s safe=%s %s %s %s screen=%s scale=%s",
		tostring(dbox), tostring(sx1), tostring(sy1), tostring(sx2), tostring(sy2),
		tostring(UIL.GetScreenSize()), tostring(terminal.desktop.scale)))

	local rc = false
	for _, v in ipairs(label_all("RCTransport")) do
		if IsValid(v) then rc = v break end
	end
	local depot = false
	for _, d in ipairs(label_all("UniversalStorageDepot")) do
		local ok, stored = pcall(function() return d:GetStoredAmount() end)
		if IsValid(d) and ok and (stored or 0) > 0 and #(d.resource or empty_table) > 0 then
			depot = d break
		end
	end
	if not (rc and depot) then
		say(fmt("L2D SKIP — need an RCTransport and a stocked depot (rc=%s depot=%s)", id_of(rc), id_of(depot)))
		return
	end

	say("L2D ⛔ PREDICTION, computed before any picker opens. With the run-2 dialog size (mw=1616 mh=429) and an anchor whose x>3840 AND y>2160, BOTH clamps must fire and the box must be (2224,1731)-(3840,2160) — the bottom-right corner. The flip branch must NOT fire (it needs y<0; y here starts at anchor.y-height ~= 1921). If instead only the x clamp fires again, the y clamp is unreachable from this geometry after all and run 2's REFUTED verdict was right for the wrong reason.")

	say(fmt("L2D ⭐ PUT THE CURSOR IN THE BOTTOM STRIP OF THE BENQ — below y=%d on the taller monitor — and leave it. Waiting for a mouse read with x>%d AND y>%d, 3 times running.",
		dbox:maxy(), dbox:maxx(), dbox:maxy()))
	local streak, armed, last = 0, false, false
	local deadline = RealTime() + 150 * 1000
	while RealTime() < deadline do
		local p = terminal.GetMousePos()
		last = p
		if p:x() > dbox:maxx() and p:y() > dbox:maxy() then
			streak = streak + 1
			if streak >= 3 then
				say(fmt("L2D ARMED — cursor at %s (x past %d AND y past %d)", tostring(p), dbox:maxx(), dbox:maxy()))
				armed = true
				break
			end
		else
			streak = 0
		end
		Sleep(300)
	end
	if not armed then
		say(fmt("L2D NOT ARMED in 150 s — last read %s. The corner case stays UNSAMPLED; ⛔ do not record this as a negative.", tostring(last)))
		return
	end

	CoRun1.picker_log = true
	local opened, missing, both, xonly, corner = 0, 0, 0, 0, 0
	for i = 1, 12 do
		local mouse = terminal.GetMousePos()
		CoRun1.count.items = -1
		local ok, err = pcall(function()
			OpenResourceSelector(rc, { target = depot, meta_key = const.vkControl,
				close_on_rmb = true, interaction_mode = "load" })
		end)
		Sleep(400)
		local dlg = GetDialog("ResourceItems")
		if dlg then
			opened = opened + 1
			local b, a = dlg.box, dlg.align_pos
			local ax_out = a and a:x() > dbox:maxx()
			local ay_out = a and a:y() > dbox:maxy()
			if ax_out and ay_out then both = both + 1 elseif ax_out then xonly = xonly + 1 end
			local at_corner = b and b:maxx() >= sx2 - 2 and b:maxy() >= sy2 - 2
			if at_corner then corner = corner + 1 end
			say(fmt("L2D cycle %d/12 dialog=ALIVE anchor=%s (x_out=%s y_out=%s) box=%s CORNER=%s items=%d mouse=%s",
				i, tostring(a), tostring(ax_out), tostring(ay_out), tostring(b),
				tostring(at_corner), CoRun1.count.items, tostring(mouse)))
		else
			missing = missing + 1
			say(fmt("L2D cycle %d/12 dialog=MISSING ok=%s err=%s items=%d mouse=%s",
				i, tostring(ok), tostring(err), CoRun1.count.items, tostring(mouse)))
		end
		pcall(CloseResourceSelector)
		Sleep(400)
	end
	CoRun1.picker_log = false
	say(fmt("L2D ⭐ RESULT %d/12 opened, %d/12 missing | anchors with BOTH x and y out=%d | anchors with x only=%d | boxes in the BOTTOM-RIGHT CORNER=%d | layouts=%d",
		opened, missing, both, xonly, corner, CoRun1.count.layouts))
end

---- the leg body ------------------------------------------------------------

function CoRun1.Flow()
	CreateRealTimeThread(function()
		local deadline = RealTime() + CoRun1.watchdog
		while CoRun1.stage ~= "done" do
			if RealTime() - deadline >= 0 then
				say("WATCHDOG TIMEOUT at " .. tostring(CoRun1.stage))
				local flush = rawget(_G, "FlushLogFile")
				if flush then flush() end
				Sleep(1000)
				quit()
				return
			end
			Sleep(1000)
		end
	end)

	say("=== CO-RUN #1 RUN 3 — the corner case run 2 recorded as refuted and had only left unsampled ===")

	if not step("1-waiting-menu", function()
		if not wait_for(CoRun1.menu_timeout, function() return not not GetPreGameMainMenu() end) then return false end
		Sleep(2000)
	end) then return end

	if not step("2-loading-save", function()
		local err = LoadGame(CoRun1.save_file, {})
		if err then say("LOADERR " .. tostring(err)) return false end
		say("LOAD OK")
	end) then return end

	if not step("3-waiting-city", function()
		return wait_for(5 * 60 * 1000, function()
			local c, u = rawget(_G, "MainCity"), rawget(_G, "UIColony")
			return not not (c and c.labels and u and u.labels)
		end)
	end) then return end
	if not step("3b-settling", function() Sleep(CoRun1.settle_time) end) then return end

	if not step("4-speed", function()
		local before = GetTimeFactor() / const.DefaultTimeFactor
		local ok = pcall(function() UIColony:SetGameSpeed(3) end)
		Sleep(500)
		say(fmt("READ game speed: call=%s before=%s after=%s", tostring(ok), tostring(before),
			tostring(GetTimeFactor() / const.DefaultTimeFactor)))
	end) then return end

	if not step("L2D-corner", leg2d_corner) then return end
	say("=== CO-RUN #1 RUN 3 complete ===")
	shutdown("done")
end

CreateRealTimeThread(function()
	Sleep(1)
	if rawget(_G, "SMRTest_EnablePathLeg") then
		say("standing down: 98_EnablePathLeg.lua drives this session")
		return
	end
	CoRun1.Flow()
end)
