-- SMR Community Fix Pack — Test Kit :: c47-farm SAMPLER PAYLOAD.  TEMPORARY
--
-- Declared by: chain c47-farm prompt 01 (01_BUILD_AND_MEASURE.md), Jobs 2 and 4.
-- Runs UNATTENDED: it self-drives from file scope, carries its own watchdog and
-- quits the process itself. Nobody is at the keyboard.
--
-- ⛔ NOT A SUITE PROBE, BY CONSTRUCTION. It is a long-running instrument, it is
-- never added to the shipped Test Kit code list permanently, and it is disarmed
-- at close-out. The permanent half of this leg is Code/61_Probes_Wave11.lua.
--
-- WHAT IT DOES, in order:
--   boot -> wait for the pre-game main menu
--        -> read run conditions for all three mods + the banner-path gate
--        -> load the STAGED COPY of the owner's named save C47FARM (never the
--           original) and SET A SPEED (this leg needs game time to pass)
--        -> census every OpenFarmBase and every ForestationPlantBase by handle
--        -> sample buffers on a fixed real-time cadence for N sols of GAME time
--        -> emit C47.Report() and quit
--
-- ⭐⭐ THE METHOD, AND WHY IT PATCHES NOTHING. The out-of-resource event this
--    leg counts is broadcast as an engine message, so the instrument SUBSCRIBES
--    instead of wrapping:
--        BaseBuilding:UpdateNotWorkingBuildingsNotification()  BaseBuilding.lua:136-138
--          -> UpdateObjectInNotification(self, self:ShouldShowNotWorkingNotification(), "NotWorkingBuildings", self:GetMap())
--             -> AddObjectToNotification    Notifications.lua:231-249  -> Msg("AddNotificationObject", ...)     :247
--             -> RemoveObjectFromNotification Notifications.lua:275-290 -> Msg("RemoveNotificationObject", ...)  :283
--    Declaring OnMsg handlers costs the game nothing and cannot alter a vanilla
--    path. The add/remove PAIR is the flap count, which is the number the
--    owner's complaint is actually about.
--    ⚠️ The entry cites NotificationUI.lua:209/:220 for these; the shipped tree
--       has the two OnMsg handlers at :211 and :223, and the Msg() emissions are
--       in Notifications.lua, not NotificationUI.lua. Corrected here at source.
--
-- ⛔⛔ THE ONE READ THIS FILE MAY NEVER MAKE, and it would be the obvious one:
--    `GetCurrentHarvestType()` **MUTATES** the building —
--    `self.current_harvest_idx = self.current_harvest_idx % 3 + 1` on entry
--    (ThreeHarvestTypesBuilding.lua:41-42). Calling it to "read the current
--    crop" would advance the rotation of every farm on every sample and change
--    the very thing being measured. The census therefore reads
--    `GetHarvestTypeInfo(1..3)` (:20-23, pure) and the `consumption_amount`
--    FIELD, never the rotating getter.
--
-- ⚠️ SECOND HAZARD, same shape: `HasSufficientSoilQuality` and the reason
--    getters are pure (checked at source), so `GetNotWorkingReason()` is safe to
--    poll. It is polled; nothing else on the building is called.
--
-- ⭐ THE VOICE ARITHMETIC IS RE-DERIVED HERE, AND IT CORRECTS THE ENTRY.
--    C47.md §(c) says the 60 s cooldown is "per farm, not per notification", so
--    that N flapping farms could make N voice lines per 60 s. **Read at source,
--    that is wrong.** `QueueVoice` (CommonLua/Voice.lua:343-360) does:
--        local now = RealTime()
--        local cooldown = VoicesCooldown[voiced_text_id] or now
--        if cooldown > now then return end
--        VoicesCooldown[voiced_text_id] = now + voice_cooldown
--    The cooldown table is keyed on the **voiced TEXT id**, not on the object,
--    and the clock is **RealTime()**, not game time. `VoicePerObject` only
--    decides WHICH handler fires (NotificationUI.lua:211 rather than :199) and
--    what gets passed as `id`; it does not key the cooldown. On top of that the
--    preset leaves `VoiceUnique` at its class default "text"
--    (NotificationPreset.lua:34), so a second add is also dropped whenever a
--    matching text is already sitting in `VoiceQueue`.
--    ⇒ **"A building has stopped working" can be queued at most once per 60
--      REAL seconds for the whole colony**, from any not-working building in the
--      game — not once per farm. This file counts BOTH models and prints them
--      side by side, so the entry can be corrected against a number instead of
--      an assertion, and it also prints the live `VoicesCooldown` key set as the
--      witness that the key really is the text.
--
-- ⛔ WHAT NO LINE BELOW MAY BE READ AS CLAIMING: that a banner was drawn on a
--    screen or that a voice line was audible. Counting the message measures
--    that the game raised the event and that the queueing rule would admit it.
--    A screen and a speaker need the owner's eyes and ears — the difference
--    between `tested-unattended` and `tested-attended` (owner, 2026-08-15).

local say = C47.Say
local fmt = C47.Fmt
local idof = C47.IdOf

C47.tag  = "C47S"
C47.mode = "REAL"

local STAGED_SAVE = "C47STAGE.savegame.sav"
local NOTIF_ID    = "NotWorkingBuildings"

-- The run shape. 5x is const.fastGameSpeed (Lua/_GameConst.lua:27) and is the
-- rig's usual high-speed condition; the owner's observation was made at high
-- game speed, which is exactly when the game-time drain and the REAL-time voice
-- cooldown diverge.
C47.speed        = 5
C47.target_sols  = 5
C47.sample_ms    = 500          -- real ms between buffer samples
C47.window_ms    = 60 * 1000    -- real ms between interim summary lines
C47.max_real_ms  = 16 * 60 * 1000   -- hard real-time cap on the sampling window
C47.watchdog     = 24 * 60 * 1000   -- hard stop for the whole process

-- File-scope state, declared here so the ARM script's G1 resolution
-- cross-check can see every name.
C47.tracked        = nil    -- array of per-object records
C47.by_handle      = nil    -- handle -> record
C47.adds_total     = 0      -- every add on NOTIF_ID, whatever the object
C47.removes_total  = 0
C47.adds_untracked = 0
C47.voice_last     = nil    -- RealTime() of the last voice this leg judged eligible
C47.voice_eligible = 0      -- SHIPPED model: one text-keyed 60 s real-time gate
C47.voice_perobj   = 0      -- the model C47.md ASSUMED: 60 s per object
C47.msgs_armed     = false
C47.start_gt       = nil
C47.start_rt       = nil
C47.end_gt         = nil
C47.end_rt         = nil
C47.samples        = 0
C47.census_done    = false

local VOICE_COOLDOWN_MS = 60 * 1000   -- read back live in the census; this is the fallback

---- the two subscriptions ---------------------------------------------------
--
-- Registered at FILE SCOPE, before anything is loaded, so no add or remove can
-- happen behind the instrument's back. Both bodies are wrapped: an error raised
-- inside an OnMsg handler does NOT unwind in this engine (EF-008), it reports
-- and carries on, and a handler that throws on every notification in a five-sol
-- run would bury the log.

local function note_add(obj)
	C47.adds_total = C47.adds_total + 1
	local rec = C47.by_handle and C47.by_handle[rawget(obj, "handle")]
	if not rec then C47.adds_untracked = C47.adds_untracked + 1 end

	-- ⭐ THE SHIPPED VOICE MODEL, and it is evaluated for EVERY add on this id,
	--    tracked object or not: the cooldown table is keyed on the voiced TEXT
	--    (Voice.lua:355-358), so a Water Extractor stopping consumes exactly the
	--    same 60 real seconds a farm's add would have. Scoring it only over the
	--    farms would overstate the farms' share of the voice budget.
	--    ⚠️ This is an UPPER BOUND on what the game would actually queue: the
	--    real path additionally requires notification:IsVisible() (a same-map
	--    test, NotificationPreset.lua:134-138), CurrentMap:GameTime() past
	--    const.Voice.NoVoiceBefore, and — because the preset leaves VoiceUnique
	--    at its "text" default — that no matching text is already sitting in
	--    VoiceQueue. Each of those can only SUBTRACT.
	local now = RealTime()
	if not C47.voice_last or (now - C47.voice_last) >= VOICE_COOLDOWN_MS then
		C47.voice_last = now
		C47.voice_eligible = C47.voice_eligible + 1
		if rec then rec.voice_shipped = rec.voice_shipped + 1 end
	end

	if not rec then return end
	rec.adds = rec.adds + 1
	rec.add_gt = GameTime()

	-- the model C47.md §(c) ASSUMED: a 60 s window PER OBJECT. Counted beside
	-- the shipped one so the entry can be corrected against a number.
	if not rec.voice_last or (now - rec.voice_last) >= VOICE_COOLDOWN_MS then
		rec.voice_last = now
		rec.voice_perobj = rec.voice_perobj + 1
		C47.voice_perobj = C47.voice_perobj + 1
	end
end

local function note_remove(obj)
	C47.removes_total = C47.removes_total + 1
	local rec = C47.by_handle and C47.by_handle[rawget(obj, "handle")]
	if not rec then return end
	rec.removes = rec.removes + 1
	if rec.add_gt then
		local held = GameTime() - rec.add_gt
		if held >= 0 then
			rec.held_total = rec.held_total + held
			rec.held_n = rec.held_n + 1
			if not rec.held_max or held > rec.held_max then rec.held_max = held end
		end
		rec.add_gt = nil
	end
end

function OnMsg.AddNotificationObject(notification, obj, object_params)
	if not C47.msgs_armed then return end
	if type(notification) ~= "table" or notification.id ~= NOTIF_ID then return end
	if type(obj) ~= "table" then return end
	pcall(note_add, obj)
end

function OnMsg.RemoveNotificationObject(notification, obj, object_params)
	if not C47.msgs_armed then return end
	if type(notification) ~= "table" or notification.id ~= NOTIF_ID then return end
	if type(obj) ~= "table" then return end
	pcall(note_remove, obj)
end

---- census ------------------------------------------------------------------

-- Two independent enumerations, both printed. City labels carry the
-- object_class ("OpenFarmBase" / "ForestationPlantBase") because
-- Building:AddToCityLabels does obj:AddToLabel(self.object_class, self)
-- (Building.lua:417-418); CurrentMap:MapGet("map", class) walks the map itself.
-- If the two disagree the log says so instead of one of them quietly winning.
local function enumerate(class_name)
	local by_label = C47.LabelAll(class_name)
	local map = rawget(_G, "CurrentMap")
	local ok, by_map = false, nil
	if map and map.MapGet then
		ok, by_map = pcall(function() return map:MapGet("map", class_name) end)
	end
	say(fmt("CENSUS enumerate %s :: city-labels=%d | CurrentMap:MapGet ok=%s count=%s — two independent walks; a disagreement is a finding, not a rounding",
		class_name, #by_label, tostring(ok),
		tostring(ok and type(by_map) == "table" and #by_map or "n/a")))
	local out, seen = {}, {}
	for _, o in ipairs(by_label) do
		local h = rawget(o, "handle")
		if h and not seen[h] then seen[h] = true; out[#out + 1] = o end
	end
	if ok and type(by_map) == "table" then
		for _, o in ipairs(by_map) do
			local h = rawget(o, "handle")
			if h and not seen[h] then seen[h] = true; out[#out + 1] = o end
		end
	end
	return out
end

-- ⛔ crops read with GetHarvestTypeInfo (pure), NEVER GetCurrentHarvestType.
local function crop_line(b)
	local names, total, n = {}, 0, 0
	for i = 1, 3 do
		local ok, info = pcall(function() return b:GetHarvestTypeInfo(i) end)
		if ok and type(info) == "table" then
			local amount = rawget(info, "placement_consumption_amount")
			names[#names + 1] = fmt("%s(%s)", tostring(rawget(info, "id")), tostring(amount))
			total = total + (amount or 0)
			n = n + 1
		else
			names[#names + 1] = "-"
		end
	end
	return table.concat(names, "/"), total, n
end

function C47.Census()
	if not C47.Guard("Census") then return false end
	C47.Label("ORGANIC", "census of every Seeds consumer on the map",
		"reads only; nothing is placed, enacted, cheated or set")
	C47.Now("census")

	C47.tracked, C47.by_handle = {}, {}

	-- the live notification preset, read rather than quoted
	local presets = rawget(_G, "NotificationPresets")
	local preset = type(presets) == "table" and presets[NOTIF_ID]
	if preset then
		VOICE_COOLDOWN_MS = rawget(preset, "VoiceCooldown") or preset.VoiceCooldown or VOICE_COOLDOWN_MS
		say(fmt("CENSUS preset %s :: VoicePerObject=%s | VoiceCooldown=%s ms | VoiceUnique=%s (class default is 'text', NotificationPreset.lua:34) | SuppressTime=%s | Suppressable=%s | UpdatePeriod=%s | VoicedText present=%s",
			NOTIF_ID, tostring(preset.VoicePerObject), tostring(preset.VoiceCooldown),
			tostring(preset.VoiceUnique), tostring(preset.SuppressTime),
			tostring(preset.Suppressable), tostring(preset.UpdatePeriod),
			tostring((preset.VoicedText or "") ~= "")))
	else
		say(fmt("CENSUS ⛔ NotificationPresets.%s did not resolve — the voice arithmetic below uses the SOURCE value %d ms and is a source claim, not a live read",
			NOTIF_ID, VOICE_COOLDOWN_MS))
	end
	say(fmt("CENSUS voice gate :: const.Voice.NoVoiceBefore=%s ms of map game time (QueueVoice returns early below it, Voice.lua:345-346)",
		tostring(const and const.Voice and const.Voice.NoVoiceBefore)))

	local groups = {
		{ class = "OpenFarmBase",         role = "SUBJECT" },
		{ class = "ForestationPlantBase", role = "CONTROL" },
	}
	local counts = {}
	for _, g in ipairs(groups) do
		local list = enumerate(g.class)
		counts[g.class] = #list
		for i, b in ipairs(list) do
			local crops, crop_sum, crop_n = crop_line(b)
			local rec = {
				obj = b, role = g.role, class = g.class,
				handle = rawget(b, "handle"),
				adds = 0, removes = 0,
				voice_shipped = 0, voice_perobj = 0, voice_last = nil,
				held_total = 0, held_n = 0, held_max = nil, add_gt = nil,
				min = nil, max = nil, sum = 0, n = 0, zero = 0,
				refills = 0, refill_amount = 0, last_buf = nil,
				not_working_samples = 0, consumption_reason_samples = 0,
				reasons = {},
			}
			C47.tracked[#C47.tracked + 1] = rec
			if rec.handle then C47.by_handle[rec.handle] = rec end

			local ok_r, reason = pcall(function() return b:GetNotWorkingReason() end)
			say(fmt("CENSUS %s [%d] %s :: crops=%s (sum over the 3 slots=%d, %d selected ⇒ mean per hex=%s) | consumption_amount=%s | consumption_max_storage=%s | consumption_stored_resources=%s | building_update_time=%s ms (%s game hours) | working=%s | GetNotWorkingReason=%s (ok=%s) | parent_dome=%s",
				g.role, i, idof(b), crops, crop_sum, crop_n,
				crop_n > 0 and tostring(crop_sum / crop_n) or "n/a",
				tostring(rawget(b, "consumption_amount")),
				tostring(rawget(b, "consumption_max_storage")),
				tostring(rawget(b, "consumption_stored_resources")),
				tostring(rawget(b, "building_update_time")),
				tostring((rawget(b, "building_update_time") or 0) / C47.MS_PER_HOUR),
				tostring(rawget(b, "working")), tostring(reason), tostring(ok_r),
				tostring(rawget(b, "parent_dome"))))
		end
		-- ⛔ the condition-sampled rule: a zero here is UNSAMPLED, not refuted
		C47.Applicable("C47-population(" .. g.class .. ", the " .. g.role .. ")",
			#list, #list,
			"city labels (Building.lua:417-418) plus CurrentMap:MapGet, deduped by handle, walked this load")
	end

	if counts.ForestationPlantBase == 0 then
		say("CENSUS ⛔⛔ NO ForestationPlantBase IN THIS COLONY. The A/B control this whole leg rests on is UNSAMPLED — not refuted, not passed. Every sibling comparison below stays a SOURCE ratio and the entry must say UNSAMPLED. This is the same failure mode the C39 bracket hit (1 of 8 families present).")
	end
	if counts.OpenFarmBase == 0 then
		say("CENSUS ⛔⛔ NO OpenFarmBase IN THIS COLONY — the subject is absent and this leg measured nothing. Check that the staged copy is the owner's C47FARM save.")
		return false
	end
	C47.census_done = true
	return true
end

---- the sampling window -----------------------------------------------------

local function sample_once()
	C47.samples = C47.samples + 1
	for _, rec in ipairs(C47.tracked or empty_table) do
		local b = rec.obj
		local cur = rawget(b, "consumption_stored_resources")
		if type(cur) == "number" then
			if not rec.min or cur < rec.min then rec.min = cur end
			if not rec.max or cur > rec.max then rec.max = cur end
			rec.sum = rec.sum + cur
			rec.n = rec.n + 1
			if cur == 0 then rec.zero = rec.zero + 1 end
			if rec.last_buf and cur > rec.last_buf then
				rec.refills = rec.refills + 1
				rec.refill_amount = rec.refill_amount + (cur - rec.last_buf)
			end
			rec.last_buf = cur
		end
		if not rawget(b, "working") then
			rec.not_working_samples = rec.not_working_samples + 1
			local ok, reason = pcall(function() return b:GetNotWorkingReason() end)
			if ok and reason then
				rec.reasons[tostring(reason)] = (rec.reasons[tostring(reason)] or 0) + 1
				if reason == "Consumption" then
					rec.consumption_reason_samples = rec.consumption_reason_samples + 1
				end
			end
		end
	end
end

local function window_line(tag)
	for _, rec in ipairs(C47.tracked or empty_table) do
		say(fmt("%sWINDOW %s %s %s :: adds=%d removes=%d | buffer now=%s min=%s max=%s mean=%s | zero-samples=%d/%d | refills=%d totalling %d | not-working samples=%d (of which reason='Consumption' %d)",
			C47.BannerVoid(), tostring(tag), rec.role, idof(rec.obj),
			rec.adds, rec.removes,
			tostring(rec.last_buf), tostring(rec.min), tostring(rec.max),
			rec.n > 0 and tostring(rec.sum / rec.n) or "n/a",
			rec.zero, rec.n, rec.refills, rec.refill_amount,
			rec.not_working_samples, rec.consumption_reason_samples))
	end
	say(fmt("%sWINDOW %s TOTALS :: adds(all objects on '%s')=%d removes=%d | adds on objects this leg does NOT track=%d | voice-eligible SHIPPED model (text-keyed, RealTime, colony-wide)=%d | voice-eligible under the entry's ASSUMED per-object model=%d",
		C47.BannerVoid(), tostring(tag), NOTIF_ID, C47.adds_total, C47.removes_total,
		C47.adds_untracked, C47.voice_eligible, C47.voice_perobj))
	C47.Now(tostring(tag))
end

-- The live witness that the cooldown really is keyed on the TEXT: print the
-- key set of VoicesCooldown (a GameVar, Voice.lua:316). One key for the whole
-- colony is the measurement; a key per farm would refute the source read.
function C47.VoiceTableWitness(when)
	local cd = rawget(_G, "VoicesCooldown")
	local q  = rawget(_G, "VoiceQueue")
	if type(cd) ~= "table" then
		say(fmt("VOICETABLE %s :: VoicesCooldown did not resolve — not sampled", tostring(when)))
		return
	end
	local keys, n = {}, 0
	for k in pairs(cd) do
		n = n + 1
		if n <= 12 then keys[#keys + 1] = tostring(k) end
	end
	say(fmt("VOICETABLE %s :: VoicesCooldown has %d key(s) [%s] | VoiceQueue length=%s — ⭐ the keys are voiced-TEXT ids, not object handles (Voice.lua:355-358), so one key covers every not-working building in the colony at once",
		tostring(when), n, table.concat(keys, ", "),
		tostring(type(q) == "table" and #q or "n/a")))
end

function C47.Sample()
	if not C47.Guard("Sample") then return false end
	if not C47.census_done then
		say("SAMPLE ⛔ census did not complete — nothing sampled")
		return false
	end
	C47.Label("ORGANIC", "buffer sampling and notification counting over game time",
		fmt("no mechanism is forced: the colony simply runs at %sx and the instrument watches. The only thing this leg does to the game is set the game speed.", tostring(C47.speed)))

	C47.msgs_armed = true
	C47.start_gt, C47.start_rt = GameTime(), RealTime()
	local target_gt = C47.start_gt + C47.target_sols * C47.MS_PER_SOL
	say(fmt("SAMPLE BEGIN :: start GameTime=%d (%s) | target=+%d sols = GameTime %d | speed=%sx | sample every %d real ms | real-time cap %d ms | ⛔ counters were armed at THIS line, so every add/remove below happened inside the window",
		C47.start_gt, C47.Sols(C47.start_gt), C47.target_sols, target_gt,
		tostring(GetTimeFactor() / const.DefaultTimeFactor), C47.sample_ms, C47.max_real_ms))
	C47.VoiceTableWitness("at window open")

	local next_window = C47.start_rt + C47.window_ms
	local window_idx = 0
	while true do
		sample_once()
		local gt, rt = GameTime(), RealTime()
		if gt - target_gt >= 0 then
			say(fmt("SAMPLE END :: reached the game-time target (GameTime=%d, %s)", gt, C47.Sols(gt)))
			break
		end
		if rt - (C47.start_rt + C47.max_real_ms) >= 0 then
			say(fmt("SAMPLE END ⚠️ REAL-TIME CAP HIT at GameTime=%d (%s) — the window is SHORTER than the %d sols asked for. Report the game-time span actually covered, never the target.",
				gt, C47.Sols(gt), C47.target_sols))
			break
		end
		if rt - next_window >= 0 then
			window_idx = window_idx + 1
			next_window = rt + C47.window_ms
			window_line("w" .. window_idx)
			C47.VoiceTableWitness("w" .. window_idx)
		end
		Sleep(C47.sample_ms)
	end
	C47.end_gt, C47.end_rt = GameTime(), RealTime()
	C47.msgs_armed = false
	C47.VoiceTableWitness("at window close")
	return true
end

---- the report --------------------------------------------------------------

function C47.Report()
	local span_gt = (C47.end_gt or GameTime()) - (C47.start_gt or 0)
	local span_rt = (C47.end_rt or RealTime()) - (C47.start_rt or 0)
	local sols = span_gt / C47.MS_PER_SOL
	say(fmt("REPORT SPAN :: game time %d ms = %s | real time %d ms | samples taken=%d | speed asked=%sx, speed at close=%sx",
		span_gt, C47.Sols(span_gt), span_rt, C47.samples,
		tostring(C47.speed), tostring(GetTimeFactor() / const.DefaultTimeFactor)))
	if span_gt <= 0 then
		say("REPORT ⛔ THE GAME CLOCK DID NOT ADVANCE. Every rate below would be a division by zero and nothing was measured. This is the corun-pt15 C4 failure (a loaded save arrives paused).")
		return false
	end

	for _, rec in ipairs(C47.tracked or empty_table) do
		-- ⛔ every derived figure is printed BESIDE its raw terms, so an auditor
		--    re-derives rather than inherits (the recorded-facts-are-claims rule).
		say(fmt("%sREPORT %s %s :: adds=%d removes=%d over %s | adds/sol=%s | flap pairs=%d | buffer min=%s max=%s (raw sum=%d over n=%d ⇒ mean %s) of max_storage=%s | samples at ZERO=%d of %d (%s%%) | refills observed=%d totalling %d seeds (mean %s per refill) | not-working samples=%d, of which reason='Consumption'=%d | time held in the notification: total %s game ms over %d closed pairs (mean %s, max %s) | voice-eligible SHIPPED=%d, under the entry's assumed per-object model=%d",
			C47.BannerVoid(), rec.role, idof(rec.obj),
			rec.adds, rec.removes, C47.Sols(span_gt),
			tostring(rec.adds * C47.MS_PER_SOL / span_gt),
			(rec.adds < rec.removes) and rec.adds or rec.removes,
			tostring(rec.min), tostring(rec.max),
			rec.sum, rec.n,
			rec.n > 0 and tostring(rec.sum / rec.n) or "n/a",
			tostring(rawget(rec.obj, "consumption_max_storage")),
			rec.zero, rec.n, rec.n > 0 and tostring(rec.zero * 100 / rec.n) or "n/a",
			rec.refills, rec.refill_amount,
			rec.refills > 0 and tostring(rec.refill_amount / rec.refills) or "n/a",
			rec.not_working_samples, rec.consumption_reason_samples,
			tostring(rec.held_total), rec.held_n,
			rec.held_n > 0 and tostring(rec.held_total / rec.held_n) or "n/a",
			tostring(rec.held_max),
			rec.voice_shipped, rec.voice_perobj))
		local reasons = {}
		for reason, count in pairs(rec.reasons) do
			reasons[#reasons + 1] = fmt("%s=%d", reason, count)
		end
		table.sort(reasons)
		say(fmt("%sREPORT %s %s :: not-working reason histogram over the window: %s",
			C47.BannerVoid(), rec.role, idof(rec.obj),
			#reasons > 0 and table.concat(reasons, ", ") or "(never sampled not-working)"))
	end

	say(fmt("%sREPORT COLONY :: adds on '%s' across ALL objects=%d | removes=%d | of those, adds on objects outside this leg's census=%d | voice-eligible SHIPPED model=%d over %d real seconds (ceiling is one per %d real s = %s) | voice-eligible under the entry's assumed per-object model=%d",
		C47.BannerVoid(), NOTIF_ID, C47.adds_total, C47.removes_total, C47.adds_untracked,
		C47.voice_eligible, span_rt / 1000, VOICE_COOLDOWN_MS,
		tostring(span_rt / VOICE_COOLDOWN_MS), C47.voice_perobj))

	C47.Applicable("C47-adds-sampled(the flap events this window contains)",
		C47.adds_total, C47.samples,
		fmt("%d samples taken over %s of game time with the two OnMsg subscriptions armed", C47.samples, C47.Sols(span_gt)))

	say("REPORT ⛔ NOT CLAIMED BY THIS RUN: that a banner appeared on a screen or that a voice line was audible. What is counted is that the game raised AddNotificationObject/RemoveNotificationObject for this id, and what the queueing rule in QueueVoice would have admitted. A screen and a speaker need the owner's eyes and ears.")
	say("REPORT ⚠️ CONFOUNDS, stated beside the numbers and not below them: CHEATS ARE ENABLED on this rig (standing fact) and this colony is an oversized terraforming speed-run. The template numbers are playstyle-independent; every RATE above is not. And the whole window ran at one speed — the drain is game-time driven while the voice cooldown is REAL-time driven, so the two ratios move against each other at any other speed.")
	return true
end

---- the leg body ------------------------------------------------------------

function C47.Boot()
	CreateRealTimeThread(function()
		Sleep(1)
		say("=== c47-farm C47S === payload loaded. Mode " .. tostring(C47.mode)
			.. ". Staged save: " .. STAGED_SAVE)

		CreateRealTimeThread(function()
			local deadline = RealTime() + C47.watchdog
			while C47.stage ~= "done" do
				if RealTime() - deadline >= 0 then
					say("TIMEOUT " .. tostring(C47.stage))
					pcall(C47.Report)
					local flush = rawget(_G, "FlushLogFile")
					if flush then flush() end
					Sleep(1000)
					quit()
					return
				end
				Sleep(1000)
			end
		end)

		-- an unattended session has no one to click the welcome popup
		if rawget(_G, "ShowStartGamePopup") then
			ShowStartGamePopup = function() end
			say("ShowStartGamePopup neutered for this unattended run")
		end

		C47.stage = "waiting-menu"
		if not C47.WaitFor(5 * 60 * 1000, function() return not not GetPreGameMainMenu() end) then
			say("TIMEOUT waiting-menu")
			C47.stage = "done" Sleep(2000) quit() return
		end
		Sleep(2000)

		C47.stage = "conditions"
		C47.ReadConditions("main menu, before the load")

		C47.stage = "loading"
		local load_ok, load_res = C47.TryYield("load " .. STAGED_SAVE, function()
			return C47.Load(STAGED_SAVE, C47.speed)
		end)
		if not (load_ok and load_res) then
			say("⛔ LOAD FAILED, or the speed did not take — no sampling window. Nothing below this line exists.")
			C47.ErrorWatchNote()
			C47.stage = "done" Sleep(2000) quit() return
		end
		C47.ReadConditions("after the load, in the colony")

		C47.stage = "census"
		local ok_c, res_c = C47.TryYield("leg Census", C47.Census)
		if not (ok_c and res_c) then
			say("⛔ CENSUS FAILED — no subject to sample.")
			C47.ErrorWatchNote()
			C47.stage = "done" Sleep(2000) quit() return
		end

		C47.stage = "sampling"
		C47.TryYield("leg Sample", C47.Sample)

		C47.stage = "reporting"
		C47.Try("C47.Report", C47.Report)

		C47.stage = "closing"
		C47.ReadConditions("at close")
		C47.Now("close")
		C47.ErrorWatchNote()
		say("=== c47-farm C47S === DONE. Quitting. ⛔ Nothing was saved: this run never calls SaveGame, so the staged copy on disk is byte-identical to the one staged before the launch.")
		C47.stage = "done"
		Sleep(3000)
		quit()
	end)
end

say("[C47S] 98_C47Sampler loaded — self-driving in 1 tick.")
C47.Boot()
