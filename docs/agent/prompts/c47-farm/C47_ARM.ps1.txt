# c47-farm ARM / DISARM script.  Parked source; copy to a scratch .ps1 and run.
#
# Rule C11: arming is a script FILE, never an inline one-liner - an inline edit's
# quoting was mangled once and the game launched unarmed.
# C11 corollary (unattended-1 I2): never pipe this script's output through
# Select-Object - a piped consumer terminates the upstream pipeline and can kill
# the script before its write executes. Nothing here is piped.
#
# G5 ARM GATE (binding): after writing, EVERY piece is read BACK OFF DISK and the
# script exits non-zero if any is missing. A launch never happens on the word of
# the writer.
#
# S3 (corun-batch-1 ledger): a parked .ps1 needs a BOM, because PS 5.1 reads a
#    no-BOM .ps1 as ANSI and any non-ASCII character breaks the parse. This body
#    is kept ASCII-only as well, belt and braces.
#
# ENCODING: PowerShell 5.1's Set-Content -Encoding utf8 writes a BOM and once
#    silently prefixed one onto metadata.lua. Use WriteAllLines with an explicit
#    no-BOM encoder for metadata.lua, as below.
#
#   Usage:  .\C47_ARM.ps1 arm-suite     # 96_AutoRunFlag only: the Job 1 full-suite launch
#           .\C47_ARM.ps1 arm-sampler   # 97_C47Common + 98_C47Sampler: the UNATTENDED leg
#           .\C47_ARM.ps1 arm-ride      # 97_C47Common + 98_C47Ride: the ATTENDED sitting
#           .\C47_ARM.ps1 disarm        # removes everything this script can add
#
# ------------------------------------------------------------------------------
# THE TWO GATES POINT IN OPPOSITE DIRECTIONS, AND THAT IS THE POINT.
#
#   arm-sampler is UNATTENDED. Nobody is at the keyboard, so the payload MUST
#   self-drive: it loads a staged save, sets a speed, samples and quits. A
#   payload that waited to be typed at would sit there until the watchdog killed
#   it, so the gate FAILS one that does not call C47.Boot() and quit().
#
#   arm-ride is ATTENDED. The owner is playing their own campaign, so the
#   payload MUST NOT drive the game at all: a LoadGame would throw away the
#   colony they are sitting in, a SetGameSpeed would fight their hands, and a
#   quit() would close the game under them. The gate therefore FAILS a payload
#   that calls ANY of those - the corun-pt15 rule, mechanised.
# ------------------------------------------------------------------------------
#
# ------------------------------------------------------------------------------
# WHAT THIS SCRIPT DOES *NOT* TOUCH: Code/61_Probes_Wave11.lua and its
# metadata.lua line. That probe is PERMANENT - it is the Job 1 half of this leg
# and ships with the kit. Only the two sampler files and the autorun flag are
# arming state. A disarm that removed the wave-11 line would silently drop the
# suite from 96 probes back to 95.
#
# THE SELF-DRIVE GATE, and why it points the way it does.
# corun-pt15 was an ATTENDED sitting: its gate FAILED a payload that started a
# flow at file scope, because a self-driving payload would race the owner's play.
# THIS chain is UNATTENDED - nobody is at the keyboard - so the gate is the
# unattended-2/3 one: it FAILS a sampler payload that does NOT self-drive.
# ------------------------------------------------------------------------------

param([Parameter(Mandatory = $true)][ValidateSet("arm-suite", "arm-sampler", "arm-ride", "arm-veg", "disarm")][string]$mode)

$ErrorActionPreference = "Stop"
$park = "c:\Dev\SMR-BugFixPack\docs\agent\prompts\c47-farm"
$kit  = "c:\Dev\SMR-BugFixPack-TestKit"
$meta = Join-Path $kit "metadata.lua"
$noBom = New-Object System.Text.UTF8Encoding($false)

$common  = @{ src = "97_C47Common.lua.txt";  dst = "Code\97_C47Common.lua" }
$payload = @{ src = "98_C47Sampler.lua.txt"; dst = "Code\98_C47Sampler.lua" }
$ride    = @{ src = "98_C47Ride.lua.txt";    dst = "Code\98_C47Ride.lua" }
$veg     = @{ src = "98_C48Veg.lua.txt";     dst = "Code\98_C48Veg.lua" }
$pairs   = @()
if ($mode -eq "arm-sampler") { $pairs = @($common, $payload) }
if ($mode -eq "arm-ride")    { $pairs = @($common, $ride) }
if ($mode -eq "arm-veg")     { $pairs = @($common, $veg) }

$autorunLine = 'Code/96_AutoRunFlag.lua'

# --- clean slate every time: strip anything this script can add ---------------
foreach ($p in @($common, $payload, $ride, $veg)) {
  $f = Join-Path $kit $p.dst
  if (Test-Path -LiteralPath $f) { Remove-Item -LiteralPath $f -Force; Write-Output ("DELETED " + $p.dst) }
}
$lines = Get-Content -LiteralPath $meta -Encoding UTF8
$kept = @()
foreach ($l in $lines) {
  # DEFECT FOUND 2026-08-15 (C48 leg, before its first launch) AND FIXED HERE.
  # (ASCII-only body, per the S3 note at the top of this file.)
  #    This pattern read 'Code/9[78]_C47', which matches 97_C47Common and
  #    98_C47Sampler/98_C47Ride but NOT 98_C48Veg - the payload named after the
  #    case it measures, C48, not C47. So every arm-veg added a metadata line
  #    that no later run could strip, the file itself WAS deleted by the loop
  #    above (it iterates the pair records, not this regex), and two commits that
  #    say "metadata returns to its shipped shape" (a9d5c43, f4d3bb0) each ADDED
  #    a stray line pointing at a file that is not on disk. The class family is
  #    what must match, never one case id.
  if ($l -match 'Code/9[78]_C4[78]') { continue }
  if ($l -match [regex]::Escape($autorunLine)) { continue }
  $kept += $l
}
[System.IO.File]::WriteAllLines($meta, $kept, $noBom)
Write-Output ("metadata.lua: stripped " + ($lines.Count - $kept.Count) + " arming line(s)")

# --- arm ----------------------------------------------------------------------
if ($mode -ne "disarm") {
  foreach ($p in $pairs) {
    Copy-Item -LiteralPath (Join-Path $park $p.src) -Destination (Join-Path $kit $p.dst) -Force
    Write-Output ("WROTE " + $p.dst)
  }
  $lines = Get-Content -LiteralPath $meta -Encoding UTF8
  $out = New-Object System.Collections.Generic.List[string]
  $inserted = $false
  foreach ($l in $lines) {
    $out.Add($l)
    if (-not $inserted -and $l -match '99_FixtureCarry\.lua') {
      if ($mode -eq "arm-suite") {
        $out.Add(("`t`t`"" + $autorunLine + "`",  -- TEMPORARY c47-farm suite launch"))
      } else {
        foreach ($p in $pairs) {
          $leaf = Split-Path $p.dst -Leaf
          $out.Add(("`t`t`"Code/" + $leaf + "`",  -- TEMPORARY c47-farm"))
        }
      }
      $inserted = $true
    }
  }
  if (-not $inserted) { Write-Output "ARM FAIL: anchor 99_FixtureCarry.lua not found"; exit 1 }
  [System.IO.File]::WriteAllLines($meta, $out, $noBom)
  Write-Output "WROTE metadata.lua (no BOM)"
}

# ---- gate --------------------------------------------------------------------
$fail = $false
$m = Get-Content -LiteralPath $meta -Raw -Encoding UTF8

# The permanent Job 1 probe must survive every mode of this script.
if ($m -match [regex]::Escape('"Code/61_Probes_Wave11.lua"')) {
  Write-Output "GATE OK       metadata still lists the permanent Code/61_Probes_Wave11.lua"
} else {
  Write-Output "GATE FAIL: metadata no longer lists Code/61_Probes_Wave11.lua - the Job 1 probe was dropped"; $fail = $true
}
if (Test-Path -LiteralPath (Join-Path $kit "Code\61_Probes_Wave11.lua")) {
  Write-Output "GATE OK       Code/61_Probes_Wave11.lua is on disk"
} else {
  Write-Output "GATE FAIL: Code/61_Probes_Wave11.lua is missing from disk"; $fail = $true
}

if ($mode -eq "arm-suite") {
  if ($m -match [regex]::Escape('"' + $autorunLine + '"')) { Write-Output "ARM GATE OK   metadata lists 96_AutoRunFlag.lua" }
  else { Write-Output "ARM GATE FAIL: metadata does not list 96_AutoRunFlag.lua"; $fail = $true }
  $flag = Get-Content -LiteralPath (Join-Path $kit "Code\96_AutoRunFlag.lua") -Raw -Encoding UTF8
  if ($flag -match '(?m)^\s*SMRTest_AutoRunEnabled\s*=\s*true') { Write-Output "ARM GATE OK   96_AutoRunFlag sets SMRTest_AutoRunEnabled = true" }
  else { Write-Output "ARM GATE FAIL: 96_AutoRunFlag does not set the flag"; $fail = $true }
  if ($flag -match '(?m)^\s*SMRTest_AutoRunSetupOnly\s*=\s*true') {
    Write-Output "ARM GATE FAIL: SetupOnly is UNCOMMENTED - that is the ATTENDED shape and this launch is unattended"; $fail = $true
  } else { Write-Output "ARM GATE OK   SetupOnly is commented out (unattended shape)" }
  foreach ($p in @($common, $payload, $ride, $veg)) {
    if (Test-Path -LiteralPath (Join-Path $kit $p.dst)) { Write-Output ("ARM GATE FAIL: sampler/ride payload still present " + $p.dst); $fail = $true }
  }
}

# ---- C48: same unattended gate as arm-sampler, on the veg payload ------------
if ($mode -eq "arm-veg") {
  if ($m -match [regex]::Escape('"' + $autorunLine + '"')) {
    Write-Output "ARM GATE FAIL: 96_AutoRunFlag is still listed - it would start a NEW COLONY and race this payload's LoadGame"; $fail = $true
  } else { Write-Output "ARM GATE OK   96_AutoRunFlag is NOT listed" }
  foreach ($p in $pairs) {
    $f = Join-Path $kit $p.dst
    if (-not (Test-Path -LiteralPath $f)) { Write-Output ("ARM GATE FAIL: missing " + $p.dst); $fail = $true; continue }
    $t = Get-Content -LiteralPath $f -Raw -Encoding UTF8
    if ($t -notmatch 'TEMPORARY') { Write-Output ("ARM GATE FAIL: no TEMPORARY marker in " + $p.dst); $fail = $true }
    else { Write-Output ("ARM GATE OK   " + $p.dst) }
    if ($m -match [regex]::Escape('"Code/' + (Split-Path $p.dst -Leaf) + '"')) { Write-Output ("ARM GATE OK   metadata lists " + $p.dst) }
    else { Write-Output ("ARM GATE FAIL: metadata does not list " + $p.dst); $fail = $true }
  }
  $vegText = Get-Content -LiteralPath (Join-Path $kit $veg.dst) -Raw -Encoding UTF8
  $vegCode = ($vegText -split "`r?`n" | ForEach-Object { ($_ -replace '--.*$', '') } | Where-Object { $_.Trim() -ne '' }) -join "`n"
  $vegCode = $vegCode -replace 'OnMsg\.[A-Za-z_]\w*', 'OnMsgHandler'
  if ($vegCode -match '(?m)^\s*C47\.Boot\s*\(') { Write-Output "ARM GATE OK   payload self-drives (C47.Boot() at file scope)" }
  else { Write-Output "ARM GATE FAIL: payload does not call C47.Boot() at file scope"; $fail = $true }
  if ($vegCode -match 'quit\s*\(') { Write-Output "ARM GATE OK   payload quits the process itself" }
  else { Write-Output "ARM GATE FAIL: payload never calls quit()"; $fail = $true }
  # read-only apart from the game speed
  $w = @()
  foreach ($banned in @('GetCurrentHarvestType', 'SaveGame', 'CheatFill', 'SetHarvestType',
                        'PlaceVegetation', 'Consume_Internal', 'DoneObject', 'SetAmount')) {
    if ($vegCode -match ([regex]::Escape($banned) + '\s*\(')) { $w += $banned }
  }
  if ($w.Count -gt 0) { Write-Output ("ARM GATE FAIL: the payload CALLS " + ($w -join ", ") + " - C48 is read-only apart from the game speed"); $fail = $true }
  else { Write-Output "ARM GATE OK   payload calls none of the mutating reads/writes" }
  # the C48 headline reading must actually be attempted
  if ($vegCode -match 'VegetationTaskRequester') { Write-Output "ARM GATE OK   payload scans VegetationTaskRequester (the C48 headline)" }
  else { Write-Output "ARM GATE FAIL: payload never mentions VegetationTaskRequester - it is not measuring C48"; $fail = $true }
  $comText = Get-Content -LiteralPath (Join-Path $kit $common.dst) -Raw -Encoding UTF8
  $both = $comText + "`n" + $vegText
  $defined = @{}
  foreach ($mm in [regex]::Matches($both, '(?m)^\s*function\s+C47\.([A-Za-z_]\w*)')) { $defined[$mm.Groups[1].Value] = $true }
  foreach ($mm in [regex]::Matches($both, '(?m)^\s*C47\.([A-Za-z_]\w*)\s*=')) { $defined[$mm.Groups[1].Value] = $true }
  $used = @{}
  foreach ($mm in [regex]::Matches($vegCode, 'C47\.([A-Za-z_]\w*)')) { $used[$mm.Groups[1].Value] = $true }
  $un = @(); foreach ($k in $used.Keys) { if (-not $defined.ContainsKey($k)) { $un += $k } }
  if ($un.Count -gt 0) { Write-Output ("ARM GATE FAIL: unresolved C47.* names: " + ($un -join ", ")); $fail = $true }
  else { Write-Output ("ARM GATE OK   all " + $used.Count + " C47.* names resolve") }
}

# ---- ATTENDED: the gate that points the other way ----------------------------
if ($mode -eq "arm-ride") {
  if ($m -match [regex]::Escape('"' + $autorunLine + '"')) {
    Write-Output "ARM GATE FAIL: 96_AutoRunFlag is still listed - it would start a NEW COLONY under the owner"; $fail = $true
  } else { Write-Output "ARM GATE OK   96_AutoRunFlag is NOT listed (no new-colony flow)" }
  if (Test-Path -LiteralPath (Join-Path $kit $payload.dst)) {
    Write-Output "ARM GATE FAIL: the UNATTENDED sampler is still in the kit - it would load a staged save out from under the owner"; $fail = $true
  } else { Write-Output "ARM GATE OK   the unattended sampler is NOT present" }
  foreach ($p in $pairs) {
    $f = Join-Path $kit $p.dst
    if (-not (Test-Path -LiteralPath $f)) { Write-Output ("ARM GATE FAIL: missing " + $p.dst); $fail = $true; continue }
    $t = Get-Content -LiteralPath $f -Raw -Encoding UTF8
    if ($t -notmatch 'TEMPORARY') { Write-Output ("ARM GATE FAIL: no TEMPORARY marker in " + $p.dst); $fail = $true }
    else { Write-Output ("ARM GATE OK   " + $p.dst) }
    if ($m -match [regex]::Escape('"Code/' + (Split-Path $p.dst -Leaf) + '"')) { Write-Output ("ARM GATE OK   metadata lists " + $p.dst) }
    else { Write-Output ("ARM GATE FAIL: metadata does not list " + $p.dst); $fail = $true }
  }

  $rideText = Get-Content -LiteralPath (Join-Path $kit $ride.dst) -Raw -Encoding UTF8
  $rideCode = ($rideText -split "`r?`n" | ForEach-Object { ($_ -replace '--.*$', '') } | Where-Object { $_.Trim() -ne '' }) -join "`n"
  # ⚠️ SUBSCRIBING to a message is not CALLING the thing it is named after, and
  #    the first arming attempt (2026-08-15 16:5x) failed on exactly that:
  #    `function OnMsg.LoadGame()` declares a handler the engine calls when the
  #    OWNER loads, and the banned-call scan below read it as a call to
  #    LoadGame(). Blank the OnMsg handler names first, so the scan sees calls
  #    only. ⛔ Deliberately NOT done by dropping LoadGame from the banned list:
  #    a real LoadGame() in an attended payload is exactly the disaster this
  #    gate exists to stop.
  $rideCode = $rideCode -replace 'OnMsg\.[A-Za-z_]\w*', 'OnMsgHandler'

  # ⛔ THE ATTENDED GATE. The owner is at the keyboard. The payload may observe
  #    and it may write to the log; it may not touch the game in any way.
  $drives = @()
  foreach ($banned in @('LoadGame', 'SetGameSpeed', 'SetTimeFactor', 'quit', 'SaveGame',
                        'GetCurrentHarvestType', 'CheatFill', 'SetHarvestType',
                        'Activate', 'Deactivate', 'PlaceVegetation', 'Consume_Internal',
                        'NewGame', 'ChangeMap')) {
    if ($rideCode -match ([regex]::Escape($banned) + '\s*\(')) { $drives += $banned }
  }
  if ($drives.Count -gt 0) {
    Write-Output ("ARM GATE FAIL: an ATTENDED payload CALLS " + ($drives -join ", ") + " - it would drive the game out from under the owner. Observe only.")
    $fail = $true
  } else {
    Write-Output "ARM GATE OK   payload drives NOTHING (no LoadGame / SetGameSpeed / SetTimeFactor / quit / SaveGame / NewGame / ChangeMap and no mutating building call) - it observes and writes to the log"
  }

  # It must arm itself on the OWNER's load rather than waiting to be typed at:
  # the symptom is unpredictable and anything before the first console line
  # would be missed.
  if ($rideText -match 'function\s+OnMsg\.LoadGame') { Write-Output "ARM GATE OK   payload arms on the owner's own load (OnMsg.LoadGame)" }
  else { Write-Output "ARM GATE FAIL: payload does not arm on OnMsg.LoadGame - it would miss everything before the owner reached the console"; $fail = $true }

  # And the owner needs the witness primitives, or the sitting measures nothing
  # a machine could not have measured alone.
  foreach ($needed in @('Saw', 'Heard', 'Mark', 'Report', 'Farms')) {
    if ($rideCode -match ('function\s+C47\.' + $needed + '\s*\(')) { Write-Output ("ARM GATE OK   owner primitive C47." + $needed + " is defined") }
    else { Write-Output ("ARM GATE FAIL: owner primitive C47." + $needed + " is missing - the human witness has no way into the log"); $fail = $true }
  }

  $comText = Get-Content -LiteralPath (Join-Path $kit $common.dst) -Raw -Encoding UTF8
  $both = $comText + "`n" + $rideText
  $defined = @{}
  foreach ($mm in [regex]::Matches($both, '(?m)^\s*function\s+C47\.([A-Za-z_]\w*)')) { $defined[$mm.Groups[1].Value] = $true }
  foreach ($mm in [regex]::Matches($both, '(?m)^\s*C47\.([A-Za-z_]\w*)\s*=')) { $defined[$mm.Groups[1].Value] = $true }
  $used = @{}
  foreach ($mm in [regex]::Matches($rideCode, 'C47\.([A-Za-z_]\w*)')) { $used[$mm.Groups[1].Value] = $true }
  $unresolved = @()
  foreach ($k in $used.Keys) { if (-not $defined.ContainsKey($k)) { $unresolved += $k } }
  if ($unresolved.Count -gt 0) {
    Write-Output ("ARM GATE FAIL: unresolved C47.* names used by the ride payload: " + ($unresolved -join ", ")); $fail = $true
  } else {
    Write-Output ("ARM GATE OK   all " + $used.Count + " C47.* names used by the ride payload resolve to a definition")
  }
}

if ($mode -eq "arm-sampler") {
  if ($m -match [regex]::Escape('"' + $autorunLine + '"')) {
    Write-Output "ARM GATE FAIL: 96_AutoRunFlag is still listed - it would start a NEW COLONY and race this payload's LoadGame"; $fail = $true
  } else { Write-Output "ARM GATE OK   96_AutoRunFlag is NOT listed (no new-colony flow)" }
  foreach ($p in $pairs) {
    $f = Join-Path $kit $p.dst
    if (-not (Test-Path -LiteralPath $f)) { Write-Output ("ARM GATE FAIL: missing " + $p.dst); $fail = $true; continue }
    $t = Get-Content -LiteralPath $f -Raw -Encoding UTF8
    if ($t -notmatch 'TEMPORARY') { Write-Output ("ARM GATE FAIL: no TEMPORARY marker in " + $p.dst); $fail = $true }
    else { Write-Output ("ARM GATE OK   " + $p.dst) }
    if ($m -match [regex]::Escape('"Code/' + (Split-Path $p.dst -Leaf) + '"')) { Write-Output ("ARM GATE OK   metadata lists " + $p.dst) }
    else { Write-Output ("ARM GATE FAIL: metadata does not list " + $p.dst); $fail = $true }
  }

  # --- SELF-DRIVE GATE. An unattended payload must start itself. ---
  $payText = Get-Content -LiteralPath (Join-Path $kit $payload.dst) -Raw -Encoding UTF8
  if ($payText -match '(?m)^\s*C47\.Boot\s*\(') { Write-Output "ARM GATE OK   payload self-drives (C47.Boot() at file scope)" }
  else { Write-Output "ARM GATE FAIL: payload does not call C47.Boot() at file scope - nobody would start it"; $fail = $true }
  if ($payText -match 'quit\s*\(') { Write-Output "ARM GATE OK   payload quits the process itself" }
  else { Write-Output "ARM GATE FAIL: payload never calls quit() - the process would hang"; $fail = $true }

  # --- THE READ-ONLY GATE. This leg measures a colony; it must not change one.
  #     GetCurrentHarvestType MUTATES current_harvest_idx on entry
  #     (ThreeHarvestTypesBuilding.lua:41-42), so calling it to "read the crop"
  #     would advance every farm's rotation on every sample.
  #     The scan runs over CODE ONLY: every Lua comment is removed first - whole
  #     comment lines AND trailing ones - because the payload names these calls
  #     in its header precisely to say it does not make them, and a gate that
  #     cannot tell a warning from a call is a gate nobody will trust. Both
  #     forms bit this leg on 2026-08-15: the header comments first, then a
  #     trailing "-- the model C47.md ASSUMED" on a real assignment, which the
  #     resolution check below read as a call to an undefined C47.md.
  #     ⚠️ Known limit, stated rather than hidden: this strips a literal "--"
  #     inside a Lua string too. The payload contains none (it uses em dashes in
  #     prose); if one is ever added, this gate reports a false PASS on that
  #     line, not a false FAIL, so re-check by eye when adding string content.
  $codeOnly = ($payText -split "`r?`n" | ForEach-Object { ($_ -replace '--.*$', '') } | Where-Object { $_.Trim() -ne '' }) -join "`n"
  $writes = @()
  foreach ($banned in @('GetCurrentHarvestType', 'SaveGame', 'CheatFill', 'SetHarvestType',
                        'Activate', 'Deactivate', 'PlaceVegetation', 'Consume_Internal')) {
    if ($codeOnly -match ([regex]::Escape($banned) + '\s*\(')) { $writes += $banned }
  }
  if ($writes.Count -gt 0) {
    Write-Output ("ARM GATE FAIL: the payload CALLS " + ($writes -join ", ") + " - this leg is read-only apart from the game speed, and GetCurrentHarvestType in particular MUTATES current_harvest_idx (ThreeHarvestTypesBuilding.lua:41-42), which would advance every farm's rotation on every sample")
    $fail = $true
  } else {
    Write-Output "ARM GATE OK   payload calls none of the mutating reads/writes (GetCurrentHarvestType, SaveGame, CheatFill, SetHarvestType, Activate, Deactivate, PlaceVegetation, Consume_Internal) - the only thing it changes is the game speed"
  }

  # --- G1 RESOLUTION CROSS-CHECK. A parse sweep is a SYNTAX verdict, not a
  #     resolution one (unattended-1 I1). Every C47.<Name> the payload USES must
  #     be DEFINED in one of the two files.
  $comText = Get-Content -LiteralPath (Join-Path $kit $common.dst) -Raw -Encoding UTF8
  $both = $comText + "`n" + $payText
  $defined = @{}
  foreach ($mm in [regex]::Matches($both, '(?m)^\s*function\s+C47\.([A-Za-z_]\w*)')) { $defined[$mm.Groups[1].Value] = $true }
  foreach ($mm in [regex]::Matches($both, '(?m)^\s*C47\.([A-Za-z_]\w*)\s*=')) { $defined[$mm.Groups[1].Value] = $true }
  # ⚠️ USES are collected from CODE ONLY, for the same reason the write-scan is:
  #    the payload's header cites "C47.md §(c)" and other prose, and the first
  #    arming attempt (2026-08-15 15:23) failed this gate on the unresolved name
  #    "md". A gate that cannot tell a citation from a call gets switched off by
  #    the next person in a hurry, which is worse than no gate.
  $used = @{}
  foreach ($mm in [regex]::Matches($codeOnly, 'C47\.([A-Za-z_]\w*)')) { $used[$mm.Groups[1].Value] = $true }
  $unresolved = @()
  foreach ($k in $used.Keys) { if (-not $defined.ContainsKey($k)) { $unresolved += $k } }
  if ($unresolved.Count -gt 0) {
    Write-Output ("ARM GATE FAIL: unresolved C47.* names used by the payload: " + ($unresolved -join ", "))
    $fail = $true
  } else {
    Write-Output ("ARM GATE OK   all " + $used.Count + " C47.* names used by the payload resolve to a definition")
  }
}

if ($mode -eq "disarm") {
  foreach ($p in @($common, $payload, $ride, $veg)) {
    if (Test-Path -LiteralPath (Join-Path $kit $p.dst)) { Write-Output ("DISARM FAIL: still present " + $p.dst); $fail = $true }
  }
  # Same defect, same fix: this gate could not see a C48 line either, so the
  #    disarm that left two of them behind still reported DISARM OK.
  if ($m -match 'Code/9[678]_(C4[78]|AutoRunFlag)') { Write-Output "DISARM FAIL: metadata still lists an arming line"; $fail = $true }
  if (-not $fail) { Write-Output "DISARM OK - kit is clean, and the permanent wave-11 probe is intact" }
}

if ($fail) { Write-Output "GATE RESULT: FAIL - DO NOT LAUNCH"; exit 1 }
Write-Output "GATE RESULT: PASS"
exit 0
