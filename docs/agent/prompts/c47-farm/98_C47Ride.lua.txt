---- the two subscriptions ---------------------------------------------------
--
-- Armed at FILE SCOPE, before any colony exists, so nothing can happen behind
-- the instrument's back. Identical accounting to the unattended sampler, so the
-- two runs' numbers are directly comparable.

local function note_add(obj)
	C47.adds_total = C47.adds_total + 1
	local rec = C47.by_handle and C47.by_handle[rawget(obj, "handle")]
	if not rec then C47.adds_untracked = C47.adds_untracked + 1 end

	-- The SHIPPED voice model, evaluated for EVERY add on this id: the cooldown
	-- table is keyed on the voiced TEXT (Voice.lua:355-358), so any stalled
	-- building anywhere consumes the same 60 real seconds. MEASURED 2026-08-15:
	-- one live key, 769547039933, the id of this notification's own VoicedText.
	local now = RealTime()
	if not C47.voice_last or (now - C47.voice_last) >= VOICE_COOLDOWN_MS then
		C47.voice_last = now
		C47.voice_eligible = C47.voice_eligible + 1
		if rec then rec.voice_shipped = rec.voice_shipped + 1 end
	end

	if not rec then return end
	rec.adds = rec.adds + 1
	rec.add_gt = GameTime()
	if not rec.voice_last or (now - rec.voice_last) >= VOICE_COOLDOWN_MS then
		rec.voice_last = now
		rec.voice_perobj = rec.voice_perobj + 1
		C47.voice_perobj = C47.voice_perobj + 1
	end
	-- ⭐ ATTENDED-ONLY: announce the add AS IT HAPPENS, so the owner can compare
	--    it against their own screen in the same second. This is the line whose
	--    presence-or-absence beside a C47.Saw() is the whole sitting.
	say(fmt("%s>>> LIVE ADD :: %s raised '%s' (add #%d for this object) | buffer=%s of %s | reason=%s | ⭐ IF YOU SEE A BANNER OR HEAR A LINE NOW, TYPE C47.Saw(\"banner\") OR C47.Heard(\"voice\")",
		C47.BannerVoid(), idof(obj), NOTIF_ID, rec.adds,
		tostring(rawget(obj, "consumption_stored_resources")),
		tostring(select(2, pcall(function() return obj.consumption_max_storage end))),
		tostring(select(2, pcall(function() return obj:GetNotWorkingReason() end)))))
end

local function note_remove(obj)
	C47.removes_total = C47.removes_total + 1
	local rec = C47.by_handle and C47.by_handle[rawget(obj, "handle")]
	if not rec then return end
	rec.removes = rec.removes + 1
	if rec.add_gt then
		local held = GameTime() - rec.add_gt
		if held >= 0 then
			rec.held_total = rec.held_total + held
			rec.held_n = rec.held_n + 1
			if not rec.held_max or held > rec.held_max then rec.held_max = held end
		end
		rec.add_gt = nil
	end
end

function OnMsg.AddNotificationObject(notification, obj, object_params)
	if not C47.msgs_armed then return end
	if type(notification) ~= "table" or notification.id ~= NOTIF_ID then return end
	if type(obj) ~= "table" then return end
	pcall(note_add, obj)
end

function OnMsg.RemoveNotificationObject(notification, obj, object_params)
	if not C47.msgs_armed then return end
	if type(notification) ~= "table" or notification.id ~= NOTIF_ID then return end
	if type(obj) ~= "table" then return end
	pcall(note_remove, obj)
end

---- census ------------------------------------------------------------------

local function enumerate(class_name)
	local by_label = C47.LabelAll(class_name)
	local map = rawget(_G, "CurrentMap")
	local ok, by_map = false, nil
	if map and map.MapGet then
		ok, by_map = pcall(function() return map:MapGet("map", class_name) end)
	end
	say(fmt("CENSUS enumerate %s :: city-labels=%d | CurrentMap:MapGet ok=%s count=%s",
		class_name, #by_label, tostring(ok),
		tostring(ok and type(by_map) == "table" and #by_map or "n/a")))
	local out, seen = {}, {}
	for _, o in ipairs(by_label) do
		local h = rawget(o, "handle")
		if h and not seen[h] then seen[h] = true; out[#out + 1] = o end
	end
	if ok and type(by_map) == "table" then
		for _, o in ipairs(by_map) do
			local h = rawget(o, "handle")
			if h and not seen[h] then seen[h] = true; out[#out + 1] = o end
		end
	end
	return out
end

-- ⛔ crops read with GetHarvestTypeInfo (pure). NEVER GetCurrentHarvestType,
--    which MUTATES current_harvest_idx on entry (ThreeHarvestTypesBuilding.lua
--    :41-42) and would advance every farm's rotation on every read.
local function crop_line(b)
	local names, total, n = {}, 0, 0
	for i = 1, 3 do
		local ok, info = pcall(function() return b:GetHarvestTypeInfo(i) end)
		if ok and type(info) == "table" then
			local amount = rawget(info, "placement_consumption_amount")
			names[#names + 1] = fmt("%s(%s)", tostring(rawget(info, "id")), tostring(amount))
			total = total + (amount or 0)
			n = n + 1
		else
			names[#names + 1] = "-"
		end
	end
	return table.concat(names, "/"), total, n
end

function C47.Census()
	C47.Label("ORGANIC", "census of every Seeds consumer on the map",
		"reads only; this sitting places nothing, enacts nothing, cheats nothing and sets no speed")
	C47.Now("census")
	C47.tracked, C47.by_handle = {}, {}

	local presets = rawget(_G, "NotificationPresets")
	local preset = type(presets) == "table" and presets[NOTIF_ID]
	if preset then
		VOICE_COOLDOWN_MS = preset.VoiceCooldown or VOICE_COOLDOWN_MS
		say(fmt("CENSUS preset %s :: VoicePerObject=%s | VoiceCooldown=%s ms | VoiceUnique=%s | SuppressTime=%s | UpdatePeriod=%s",
			NOTIF_ID, tostring(preset.VoicePerObject), tostring(preset.VoiceCooldown),
			tostring(preset.VoiceUnique), tostring(preset.SuppressTime),
			tostring(preset.UpdatePeriod)))
	end

	local groups = {
		{ class = "OpenFarmBase",         role = "SUBJECT" },
		{ class = "ForestationPlantBase", role = "CONTROL" },
	}
	local counts = {}
	for _, g in ipairs(groups) do
		local list = enumerate(g.class)
		counts[g.class] = #list
		for i, b in ipairs(list) do
			local crops, crop_sum, crop_n = crop_line(b)
			local ok_ms, max_store = pcall(function() return b.consumption_max_storage end)
			local ok_bu, upd = pcall(function() return b.building_update_time end)
			local rec = {
				obj = b, role = g.role, class = g.class,
				handle = rawget(b, "handle"),
				adds = 0, removes = 0,
				voice_shipped = 0, voice_perobj = 0, voice_last = nil,
				held_total = 0, held_n = 0, held_max = nil, add_gt = nil,
				min = nil, max = nil, sum = 0, n = 0, zero = 0,
				refills = 0, refill_amount = 0, last_buf = nil,
				drains = 0, drain_total = 0, drain_max = nil, hexes = {},
				mean_hex_cost = crop_n > 0 and (crop_sum / crop_n) or nil,
				not_working_samples = 0, consumption_reason_samples = 0, reasons = {},
				crops_at_census = crops,
			}
			C47.tracked[#C47.tracked + 1] = rec
			if rec.handle then C47.by_handle[rec.handle] = rec end
			-- the controls are numerous and quiet; only the farms get a line each
			if g.role == "SUBJECT" or i <= 3 then
				local ok_r, reason = pcall(function() return b:GetNotWorkingReason() end)
				say(fmt("CENSUS %s [%d] %s :: crops=%s (mean per hex=%s) | consumption_amount=%s | max_storage=%s | stored=%s | tick=%s ms | working=%s | reason=%s | ack-flag=%s",
					g.role, i, idof(b), crops,
					crop_n > 0 and tostring(crop_sum / crop_n) or "n/a",
					tostring(rawget(b, "consumption_amount")), tostring(max_store),
					tostring(rawget(b, "consumption_stored_resources")), tostring(upd),
					tostring(rawget(b, "working")), tostring(reason),
					tostring(rawget(b, "SMRFixPack_ack_notworking"))))
			end
		end
		C47.Applicable("C47-population(" .. g.class .. ", the " .. g.role .. ")",
			#list, #list, "city labels plus CurrentMap:MapGet, deduped by handle, walked this load")
	end
	if counts.OpenFarmBase == 0 then
		say("CENSUS ⛔⛔ NO OpenFarmBase ON THIS MAP — this sitting is watching a colony with no Open Farm in it. Check you loaded C47FARM.")
		return false
	end
	C47.census_done = true
	return true
end

---- sampling ----------------------------------------------------------------

local function sample_once()
	C47.samples = C47.samples + 1
	for _, rec in ipairs(C47.tracked or empty_table) do
		local b = rec.obj
		local cur = rawget(b, "consumption_stored_resources")
		if type(cur) == "number" then
			if not rec.min or cur < rec.min then rec.min = cur end
			if not rec.max or cur > rec.max then rec.max = cur end
			rec.sum = rec.sum + cur
			rec.n = rec.n + 1
			if cur == 0 then rec.zero = rec.zero + 1 end
			if rec.last_buf and cur > rec.last_buf then
				rec.refills = rec.refills + 1
				rec.refill_amount = rec.refill_amount + (cur - rec.last_buf)
			elseif rec.last_buf and cur < rec.last_buf then
				local spend = rec.last_buf - cur
				rec.drains = rec.drains + 1
				rec.drain_total = rec.drain_total + spend
				if not rec.drain_max or spend > rec.drain_max then rec.drain_max = spend end
				if rec.mean_hex_cost and rec.mean_hex_cost > 0 then
					local hexes = (spend + rec.mean_hex_cost / 2) / rec.mean_hex_cost
					rec.hexes[hexes] = (rec.hexes[hexes] or 0) + 1
				end
			end
			rec.last_buf = cur
		end
		if not rawget(b, "working") then
			rec.not_working_samples = rec.not_working_samples + 1
			local ok, reason = pcall(function() return b:GetNotWorkingReason() end)
			if ok and reason then
				rec.reasons[tostring(reason)] = (rec.reasons[tostring(reason)] or 0) + 1
				if reason == "Consumption" then
					rec.consumption_reason_samples = rec.consumption_reason_samples + 1
				end
			end
		end
	end
end

---- the owner's own primitives ----------------------------------------------
--
-- ⭐ These are the sitting. Everything above is arithmetic the machine could
--    have done alone; these four put a human witness into the same log.

local function stamp(kind, what)
	C47.witness_count = C47.witness_count + 1
	local colony = rawget(_G, "UIColony")
	say(fmt("⭐⭐ OWNER %s #%d :: \"%s\" :: GameTime()=%s | day=%s hour=%s | speed=%sx | RealTime()=%s | adds so far: farms=%s colony=%d | voice-eligible=%d",
		tostring(kind), C47.witness_count, tostring(what),
		tostring(GameTime()), tostring(colony and colony.day), tostring(colony and colony.hour),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), tostring(RealTime()),
		tostring(C47.FarmAdds()), C47.adds_total, C47.voice_eligible))
	C47.Farms()
end

-- ⛔ A SCREEN EVENT. No log line this project has ever written could stand in
--    for it, and it is the reason a human is sitting here.
function C47.Saw(what)      stamp("SAW-ON-SCREEN", what or "(unspecified)") end
function C47.Heard(what)    stamp("HEARD", what or "(unspecified)") end
-- Something the owner DID: built, enlarged, re-cropped, demolished.
function C47.Mark(what)     stamp("DID", what or "(unspecified)") end
-- Owner verbatim, relayed into the log the moment it is spoken (WORKFLOW).
function C47.Say2(what)     stamp("SAID", what or "(unspecified)") end

function C47.FarmAdds()
	local n = 0
	for _, rec in ipairs(C47.tracked or empty_table) do
		if rec.role == "SUBJECT" then n = n + rec.adds end
	end
	return n
end

-- One line per farm, right now. Cheap enough to call constantly.
function C47.Farms()
	for _, rec in ipairs(C47.tracked or empty_table) do
		if rec.role == "SUBJECT" then
			say(fmt("%sFARM %s :: buffer=%s (min %s, max %s) of %s | crops now=%s | adds=%d removes=%d | drains=%d spending %d | working=%s",
				C47.BannerVoid(), idof(rec.obj),
				tostring(rec.last_buf), tostring(rec.min), tostring(rec.max),
				tostring(select(2, pcall(function() return rec.obj.consumption_max_storage end))),
				(select(1, crop_line(rec.obj))),
				rec.adds, rec.removes, rec.drains, rec.drain_total,
				tostring(rawget(rec.obj, "working"))))
		end
	end
	return true
end

---- the report --------------------------------------------------------------

function C47.Report()
	local span_gt = GameTime() - (C47.start_gt or GameTime())
	local span_rt = RealTime() - (C47.start_rt or RealTime())
	say(fmt("REPORT SPAN :: game time %d ms (%s game hours, %s) | real time %d ms | samples=%d | owner witness marks=%d | speed now=%sx",
		span_gt, tostring(span_gt / C47.MS_PER_HOUR), C47.Sols(span_gt),
		span_rt, C47.samples, C47.witness_count,
		tostring(GetTimeFactor() / const.DefaultTimeFactor)))
	for _, rec in ipairs(C47.tracked or empty_table) do
		local quiet = rec.role == "CONTROL" and rec.adds == 0 and rec.drains == 0
		if not quiet then
			say(fmt("%sREPORT %s %s :: adds=%d removes=%d | buffer min=%s max=%s (sum=%d over n=%d) of %s | samples at ZERO=%d of %d | refills=%d totalling %d | drains=%d totalling %d (mean spend %s vs mean hex cost %s) | not-working samples=%d, reason='Consumption'=%d | voice-eligible SHIPPED=%d perobject-model=%d",
				C47.BannerVoid(), rec.role, idof(rec.obj), rec.adds, rec.removes,
				tostring(rec.min), tostring(rec.max), rec.sum, rec.n,
				tostring(select(2, pcall(function() return rec.obj.consumption_max_storage end))),
				rec.zero, rec.n, rec.refills, rec.refill_amount,
				rec.drains, rec.drain_total,
				rec.drains > 0 and tostring(rec.drain_total / rec.drains) or "n/a",
				tostring(rec.mean_hex_cost),
				rec.not_working_samples, rec.consumption_reason_samples,
				rec.voice_shipped, rec.voice_perobj))
			local parts = {}
			local hx = {}
			for h, c in pairs(rec.hexes) do hx[#hx + 1] = { h = h, c = c } end
			table.sort(hx, function(a, b) return a.h < b.h end)
			for _, e in ipairs(hx) do parts[#parts + 1] = fmt("%sx%d", tostring(e.h), e.c) end
			if #parts > 0 then
				say(fmt("%sREPORT %s %s :: implied-hex histogram (hexes x drain events): %s — the unattended runs measured ~2.2 on a fully planted farm against C47's assumed 4",
					C47.BannerVoid(), rec.role, idof(rec.obj), table.concat(parts, ", ")))
			end
		end
	end
	local quiet_controls = 0
	for _, rec in ipairs(C47.tracked or empty_table) do
		if rec.role == "CONTROL" and rec.adds == 0 and rec.drains == 0 then quiet_controls = quiet_controls + 1 end
	end
	say(fmt("%sREPORT CONTROL SUMMARY :: %d ForestationPlant(s) raised no add and spent nothing at all this sitting (printed as a count, not suppressed — every one that DID move has its own line above)",
		C47.BannerVoid(), quiet_controls))
	say(fmt("%sREPORT COLONY :: adds on '%s' across ALL objects=%d | removes=%d | on objects outside the census=%d | voice-eligible SHIPPED model=%d | owner witness marks=%d",
		C47.BannerVoid(), NOTIF_ID, C47.adds_total, C47.removes_total,
		C47.adds_untracked, C47.voice_eligible, C47.witness_count))
	C47.VoiceTableWitness("on demand")
	C47.ReadBannerGate()
	say("REPORT ⭐ WHAT THIS SITTING CAN CLAIM THAT THE UNATTENDED RUNS COULD NOT: whatever the owner typed into C47.Saw / C47.Heard. Counting the engine's message is a measurement; a banner on a screen and a line out of a speaker are witnessed by a human or not at all.")
	say("REPORT ⚠️ CONFOUNDS: cheats are enabled on this rig, this colony is an oversized terraforming speed-run, and the drone carry dial is at +2 so a Seeds trip is 3000, not the vanilla 1000.")
	return true
end

-- Live witness that the voice cooldown is keyed on the TEXT, not the object.
function C47.VoiceTableWitness(when)
	local cd = rawget(_G, "VoicesCooldown")
	local q  = rawget(_G, "VoiceQueue")
	if type(cd) ~= "table" then
		say(fmt("VOICETABLE %s :: VoicesCooldown did not resolve", tostring(when)))
		return
	end
	local keys, n = {}, 0
	for k in pairs(cd) do
		n = n + 1
		if n <= 12 then keys[#keys + 1] = tostring(k) end
	end
	say(fmt("VOICETABLE %s :: VoicesCooldown has %d key(s) [%s] | VoiceQueue length=%s — 769547039933 is this notification's own VoicedText id, and ONE key covers the whole colony",
		tostring(when), n, table.concat(keys, ", "),
		tostring(type(q) == "table" and #q or "n/a")))
end

---- arming, on the owner's own load -----------------------------------------
--
-- ⛔ NOTHING BELOW LOADS, SPEEDS, SAVES OR QUITS ANYTHING. It waits for the
--    owner to bring a colony up by hand and then starts watching it.

local function begin_watching(why)
	if not C47.GameIsLive() then return end
	C47.loads_seen = C47.loads_seen + 1
	say(fmt("=== c47-farm C47R === ATTENDED ride-along arming on %s (colony #%d this process). ⛔ This payload never loads, never sets a speed and never quits — the owner drives.", tostring(why), C47.loads_seen))
	C47.ReadConditions("attended sitting, colony up")
	if not C47.Census() then return end
	C47.msgs_armed = true
	C47.start_gt, C47.start_rt = GameTime(), RealTime()
	say("=== c47-farm C47R === WATCHING. Console (press Enter): C47.Saw(\"banner\") · C47.Heard(\"voice\") · C47.Mark(\"what you did\") · C47.Say2(\"your words\") · C47.Report · C47.Farms")
	C47.VoiceTableWitness("at arming")

	CreateRealTimeThread(function()
		local next_window = RealTime() + C47.window_ms
		while C47.msgs_armed do
			sample_once()
			if RealTime() - next_window >= 0 then
				next_window = RealTime() + C47.window_ms
				C47.Now("interim")
				C47.Farms()
				say(fmt("%sINTERIM :: colony adds=%d removes=%d | farm adds=%s | voice-eligible=%d | samples=%d | owner marks=%d",
					C47.BannerVoid(), C47.adds_total, C47.removes_total,
					tostring(C47.FarmAdds()), C47.voice_eligible, C47.samples, C47.witness_count))
			end
			Sleep(C47.sample_ms)
		end
	end)
end

function OnMsg.LoadGame()
	CreateRealTimeThread(function()
		local wait = rawget(_G, "WaitLoadingScreenClose")
		if wait then pcall(wait) end
		Sleep(3000)
		C47.msgs_armed = false          -- a new colony invalidates the old handles
		begin_watching("OnMsg.LoadGame")
	end)
end

		if wait then pcall(wait) end
		Sleep(3000)
		C47.msgs_armed = false
		begin_watching("OnMsg.CityStart")
	end)
end
