-- ⛔ READ `docs/agent/bugs/C48.md` FIRST. This file measures ONE question:
--    does the drone network feed a farm from scattered one-plant vegetation
--    seed offers in preference to a full Seeds depot, and does the density of
--    those offers explain why one of the owner's two farms starves and the
--    other never does?
--
-- WHY IT EXISTS. `Lua/Vegetation.lua` places a `VegetationTaskRequester` on each
-- grown, off-cooldown vegetation hex to offer that plant's seeds to drones
-- (:1081, :1774-1783). That class inherits `TaskRequester.supply_dist_modifier
-- = 100` — the default, NO distance penalty — while the only comparable
-- scattered source in the game, `SurfaceDeposit`, carries a deliberate 150
-- ("surface deposits are considered 50% further than storages",
-- SurfaceDeposit.lua:84). So one plant's yield competes with a 4,000-unit depot
-- on equal terms.
--
-- ⛔ THE SAMPLING RATE IS THE POINT. The C47 attended sitting polled every 2
--    real seconds and smeared individual deliveries together, which is why
--    "how much does one delivery actually bring" is still unanswered after
--    seven unattended launches and a sitting. This file polls at 200 ms.
--
-- ⛔ READ-ONLY apart from the game speed. It never calls GetCurrentHarvestType
--    (which MUTATES current_harvest_idx, ThreeHarvestTypesBuilding.lua:41-42),
--    never saves, never places or removes anything.

-- ⛔⛔ DEFECT IN THIS FILE, FOUND AT SOURCE BEFORE THE FIRST LAUNCH (2026-08-15)
--    AND REPAIRED HERE — it would have lost the headline reading entirely.
--
--    The first draft asked each farm for `work_radius or service_area_radius`
--    and counted offers in a world-distance circle of that size. Neither field
--    exists on a farm:
--      * `work_radius` is declared on `DroneNode` — drone hubs, RC Rovers and
--        drone extenders (`Lua/Buildings/DroneControl.lua:40`), NOT on
--        buildings. `OpenFarm.lua` never sets it and no building class does.
--      * `service_area_radius` does not appear anywhere in the shipped tree.
--    ⇒ both farms would have logged "no serving radius resolved, comparison
--      NOT SAMPLED", and the one number this whole leg exists to produce would
--      have been absent. Same defect family as the three the C47 leg disclosed:
--      trusting HOW a value is read instead of checking WHAT it is.
--
-- ⭐ THE REPAIR USES THE GAME'S OWN SEARCH SHAPE, which is also exactly the
--    owner's hypothesis. A drone does not search around the farm; it searches
--    around the COMMAND CENTRES that serve the farm, and it recurses through
--    their working extenders:
--        FindTaskRequesters(node)
--          local taskrequesters = node:MapGet(node, "hex", node.work_radius, "TaskRequester")
--          for _, extender in ipairs(node.linked_extenders) do
--              if extender.working then ... FindTaskRequesters(extender) ... end
--                                          -- Lua/Buildings/DroneControl.lua:315-324
--    Every building carries the list of centres serving it (`command_centers`,
--    `Lua/_TaskRequest.lua:298-323`), so the count this leg wants is: the union,
--    deduplicated by handle, of the VegetationTaskRequesters inside every centre
--    that serves this farm and every working extender hanging off those centres.
--    ⭐ That IS the owner's sentence — *"the drone extenders in that area give
--    the drones a much larger area to look for vegetation seeds"* — measured in
--    the units the engine uses (HEXES, not world distance).
-- ⚠️ A second, independent proxy is taken as well: offers within a FIXED hex
--    radius of the farm itself (`const.CommandCenterDefaultRadius`). It answers
--    the simpler half — "is this farm standing in denser vegetation" — without
--    depending on the colony's extender topology, and it still reads if
--    `command_centers` comes back empty.

-- The centres serving a farm, plus their working extenders, deduplicated.
local function serving_centers(farm)
	local out, seen = {}, {}
	local ok, list = pcall(function() return farm.command_centers end)
	if not ok or type(list) ~= "table" then return out, false end
	local function add(node, depth)
		if not node or depth > 4 then return end
		local h = rawget(node, "handle")
		if h and seen[h] then return end
		if h then seen[h] = true end
		out[#out + 1] = node
		local ok_e, ext = pcall(function() return node.linked_extenders end)
		if ok_e and type(ext) == "table" then
			for _, e in ipairs(ext) do
				local ok_w, working = pcall(function() return e.working end)
				if ok_w and working then add(e, depth + 1) end
			end
		end
	end
	for _, c in ipairs(list) do add(c, 1) end
	return out, true
end

-- ⭐ THE HEADLINE READING. Every live VegetationTaskRequester on the map, and
--    how many sit inside the drone coverage that actually serves each farm.
function C47.VegScan(when)
	local map = rawget(_G, "CurrentMap")
	if not (map and map.MapGet) then
		say(fmt("VEG %s :: CurrentMap:MapGet unavailable — NOT SAMPLED", tostring(when)))
		return
	end
	local ok, list = pcall(function() return map:MapGet("map", "VegetationTaskRequester") end)
	if not ok or type(list) ~= "table" then
		say(fmt("VEG %s :: MapGet('VegetationTaskRequester') failed ok=%s — NOT SAMPLED. ⛔ A zero here would be UNSAMPLED, not 'none exist'.",
			tostring(when), tostring(ok)))
		return
	end
	say(fmt("VEG %s :: %d live VegetationTaskRequester(s) on the map", tostring(when), #list))

	-- their declared weighting, read LIVE off an instance rather than trusted
	local sample = list[1]
	if sample then
		local ok_d, dist = pcall(function() return sample.supply_dist_modifier end)
		local ok_p, prio = pcall(function() return sample.priority end)
		local ok_r, req  = pcall(function() return sample.request end)
		local amount
		if ok_r and req then
			local ok_a, a = pcall(function() return req:GetActualAmount() end)
			amount = ok_a and a or nil
		end
		say(fmt("VEG %s :: sample %s | supply_dist_modifier=%s (ok=%s; TaskRequester default is 100, SurfaceDeposit overrides to 150) | priority=%s (ok=%s) | request actual amount=%s | preset=%s",
			tostring(when), idof(sample), tostring(dist), tostring(ok_d),
			tostring(prio), tostring(ok_p), tostring(amount),
			tostring(select(2, pcall(function() return sample.preset and sample.preset.id end)))))
	end

	-- ⭐ WHAT ONE OFFER IS WORTH, which is the other half of the delivery-size
	--    question. `VegetationTaskRequester:CreateResourceRequests` posts
	--    `AddSupplyRequest(preset.output_resource, GetVegOutputAmount(preset, self), 0, 4)`
	--    (`Lua/Vegetation.lua:1848-1851`), and `output_resource` defaults to
	--    "Seeds" with `output_amount` defaulting to 1000
	--    (`ClassDef-PresetDefs.generated.lua:1917-1920`). In the shipped data
	--    only three presets can raise one: Bush **200**, Tree **1000** (default,
	--    unset) and Broadleaf **2000**; Grass and Lichen sit at 0 and are gated
	--    out. `GetVegOutputAmount` then adds a soil-quality bonus of
	--    `(sq - min_soil_quality)/2` percent, so the ceiling at perfect soil is
	--    280 / 1300 / 2600. ⛔ This histogram is the SOURCE claim being measured,
	--    not an assumption to report as a reading.
	local hist, n_sampled, cap = {}, 0, 400
	for i = 1, #list do
		if n_sampled >= cap then break end
		local o = list[i]
		local ok_a, amt = pcall(function() return o.request and o.request:GetActualAmount() end)
		local ok_i, pid = pcall(function() return o.preset and o.preset.id end)
		local key = tostring(ok_i and pid or "?") .. ":" .. tostring(ok_a and amt or "?")
		hist[key] = (hist[key] or 0) + 1
		n_sampled = n_sampled + 1
	end
	local parts = {}
	for k, c in pairs(hist) do parts[#parts + 1] = fmt("%s x%d", k, c) end
	table.sort(parts)
	say(fmt("VEG %s :: OFFER SIZES (preset:amount x count) over %d of %d requester(s)%s :: %s",
		tostring(when), n_sampled, #list,
		#list > cap and " ⚠️ CAPPED at 400 — the rest were NOT sampled, so this is a sample of the distribution, not the whole of it" or "",
		#parts > 0 and table.concat(parts, ", ") or "(none)"))

	-- per farm: how many offers sit inside the drone coverage that serves it
	for _, rec in ipairs(C47.farms or empty_table) do
		local centers, ok_list = serving_centers(rec.obj)
		if #centers == 0 then
			say(fmt("VEG %s :: %s (%s) — command_centers resolved=%s but yielded ZERO centres, so the serving-coverage count is ⛔ NOT SAMPLED for this farm (⛔ not zero offers: no search area was resolved)",
				tostring(when), idof(rec.obj), rec.role, tostring(ok_list)))
		else
			local seen, offers, all_reqs, detail = {}, 0, 0, {}
			for _, c in ipairs(centers) do
				local ok_r, radius = pcall(function() return c.work_radius end)
				local n_here = "n/a"
				if ok_r and type(radius) == "number" and radius > 0 then
					local ok_g, list_v = pcall(function()
						return c:MapGet(c, "hex", radius, "VegetationTaskRequester")
					end)
					if ok_g and type(list_v) == "table" then
						n_here = #list_v
						for _, o in ipairs(list_v) do
							local h = rawget(o, "handle")
							if h and not seen[h] then seen[h] = true; offers = offers + 1 end
						end
					else
						n_here = "READ FAILED"
					end
					local ok_t, list_t = pcall(function()
						return c:MapGet(c, "hex", radius, "TaskRequester")
					end)
					if ok_t and type(list_t) == "table" then all_reqs = all_reqs + #list_t end
				end
				detail[#detail + 1] = fmt("%s r=%s working=%s veg=%s",
					idof(c), tostring(ok_r and radius), tostring(rawget(c, "working")), tostring(n_here))
			end
			rec.veg_offers = offers
			say(fmt("VEG %s :: %s (%s) ⭐ SERVING COVERAGE: %d centre(s)/extender(s) — %s | ⭐⭐ %d DISTINCT VegetationTaskRequester(s) reachable (deduplicated by handle) | %d TaskRequester(s) of every kind in the same areas (sum, NOT deduplicated) — ⭐ THIS IS THE COMPARISON: the complaining farm should hold many and the silent one few",
				tostring(when), idof(rec.obj), rec.role, #centers,
				table.concat(detail, " · "), offers, all_reqs))
		end
		-- the topology-independent proxy: density around the farm itself
		local ok_p, near = pcall(function()
			return rec.obj:MapGet(rec.obj, "hex", const.CommandCenterDefaultRadius, "VegetationTaskRequester")
		end)
		rec.veg_local = (ok_p and type(near) == "table") and #near or nil
		say(fmt("VEG %s :: %s (%s) LOCAL DENSITY PROXY: %s offer(s) within a fixed %s-hex radius of the farm itself (ok=%s) — ⚠️ a proxy for 'is this farm standing in thicker vegetation', NOT the drone search area",
			tostring(when), idof(rec.obj), rec.role,
			tostring(rec.veg_local), tostring(const and const.CommandCenterDefaultRadius), tostring(ok_p)))
	end
end

---- the depots ---------------------------------------------------------------

-- If these sit FLAT while a farm's buffer sawtooths, the farm is being fed from
-- the landscape and not from storage. That is the whole C48 claim.
function C47.DepotScan(when)
	for _, d in ipairs(C47.depots or empty_table) do
		local ok, stored = pcall(function()
			if d.GetStoredAmount then return d:GetStoredAmount() end
			return (d.stockpiled_amount or empty_table).Seeds
		end)
		say(fmt("DEPOT %s :: %s stored Seeds=%s (ok=%s) | desired_amount=%s | ⛔ a supply request's desired_amount is what it KEEPS BACK (StorageDepot.lua:466), so 0 means 'give everything' — it is NOT a reason to withhold",
			tostring(when), idof(d), tostring(stored), tostring(ok),
			tostring(rawget(d, "desired_amount"))))
	end
end

---- census -------------------------------------------------------------------

function C47.Census()
	if not C47.Guard("Census") then return false end
	C47.Label("ORGANIC", "census of farms, Seeds depots and vegetation seed offers",
		"reads only; the only thing this leg changes is the game speed")
	C47.Now("census")
	C47.farms, C47.depots, C47.veg_hist = {}, {}, {}

	local map = rawget(_G, "CurrentMap")
	local farms = C47.LabelAll("OpenFarmBase")
	for i, b in ipairs(farms) do
		-- ⛔ NOT b.work_radius: a farm has no such field (see the defect note above
		--    the veg scan). The serving area is the union of the CENTRES that
		--    serve it, resolved live at every scan.
		local centers_at_census = select(1, serving_centers(b))
		local rec = {
			obj = b, role = "FARM", handle = rawget(b, "handle"),
			veg_offers = nil, veg_local = nil,
			min = nil, max = nil, last_buf = nil,
			deliveries = 0, delivered = 0, sizes = {},
			drains = 0, drain_total = 0,
			stalled = 0, n = 0,
		}
		C47.farms[#C47.farms + 1] = rec
		local ok_ms, maxs = pcall(function() return b.consumption_max_storage end)
		say(fmt("CENSUS FARM [%d] %s :: stored=%s of %s | consumption_amount=%s | tick=%s ms | working=%s | serving centres+extenders=%d (⛔ a farm has NO work_radius of its own — DroneControl.lua:40)",
			i, idof(b), tostring(rawget(b, "consumption_stored_resources")), tostring(maxs),
			tostring(rawget(b, "consumption_amount")),
			tostring(select(2, pcall(function() return b.building_update_time end))),
			tostring(rawget(b, "working")), #centers_at_census))
	end
	C47.Applicable("C48-population(OpenFarmBase)", #C47.farms, #C47.farms,
		"city labels walked this load")
	if #C47.farms == 0 then
		say("CENSUS ⛔ no OpenFarmBase — wrong save staged. Nothing measured.")
		return false
	end

	-- every building that stores Seeds
	local seen = {}
	for _, cls in ipairs{ "StorageDepot", "UniversalStorageDepot", "MechanizedDepot", "StorageSeeds" } do
		local ok, list = pcall(function() return map and map:MapGet("map", cls) end)
		if ok and type(list) == "table" then
			for _, d in ipairs(list) do
				local h = rawget(d, "handle")
				local ok_a, acc = pcall(function() return d:DoesAcceptResource("Seeds") end)
				if h and not seen[h] and (not ok_a or acc) then
					seen[h] = true
					C47.depots[#C47.depots + 1] = d
				end
			end
		end
	end
	C47.Applicable("C48-population(Seeds-accepting depots)", #C47.depots, #C47.depots,
		"CurrentMap:MapGet over four depot classes, filtered on DoesAcceptResource('Seeds')")
	C47.DepotScan("census")
	C47.VegScan("census")
	return true
end

---- sampling -----------------------------------------------------------------

local function sample_once()
	C47.samples = C47.samples + 1
	for _, rec in ipairs(C47.farms or empty_table) do
		local cur = rawget(rec.obj, "consumption_stored_resources")
		if type(cur) == "number" then
			rec.n = rec.n + 1
			if not rec.min or cur < rec.min then rec.min = cur end
			if not rec.max or cur > rec.max then rec.max = cur end
			if rec.last_buf and cur > rec.last_buf then
				-- ⭐ at 200 ms this is ONE delivery, not a smear of several
				local size = cur - rec.last_buf
				rec.deliveries = rec.deliveries + 1
				rec.delivered = rec.delivered + size
				rec.sizes[size] = (rec.sizes[size] or 0) + 1
			elseif rec.last_buf and cur < rec.last_buf then
				rec.drains = rec.drains + 1
				rec.drain_total = rec.drain_total + (rec.last_buf - cur)
			end
			rec.last_buf = cur
		end
		if not rawget(rec.obj, "working") then rec.stalled = rec.stalled + 1 end
	end
end

-- Snapshot the per-farm counters, so a segment's numbers are that segment's and
-- not a running total. ⛔ Re-emit the DIFFERENCES in the report; never let a
-- reader subtract two cumulative figures by eye.
local function snapshot()
	local s = {}
	for i, rec in ipairs(C47.farms or empty_table) do
		s[i] = { stalled = rec.stalled, n = rec.n, deliveries = rec.deliveries,
			delivered = rec.delivered, drains = rec.drains, drain_total = rec.drain_total }
	end
	return s
end

function C47.Sample()
	if not C47.Guard("Sample") then return false end
	C47.Label("FORCED", "the game speed, and ONLY the game speed",
		"the ladder sets the time factor between segments; nothing else about the colony is touched, and the farms/drones/vegetation are the same objects throughout")
	C47.start_gt, C47.start_rt = GameTime(), RealTime()
	C47.segments = {}
	say(fmt("SAMPLE BEGIN :: GameTime=%d | ladder=%s | %d game ms per rung | sampling every %d real ms | real cap %d ms",
		C47.start_gt, table.concat(C47.ladder, "x, ") .. "x",
		C47.segment_gt, C47.sample_ms, C47.max_real_ms))

	local last_gt, still = C47.start_gt, 0
	local aborted = false

	for rung, speed in ipairs(C47.ladder) do
		if aborted then break end
		-- ⛔ read the speed BACK; a rung whose speed did not take is void, and
		--    saying so is cheaper than discovering it in the numbers later.
		local ok_set = C47.Try("UIColony:SetGameSpeed(" .. tostring(speed) .. ") [rung " .. rung .. "]",
			function() UIColony:SetGameSpeed(speed) return true end)
		Sleep(500)
		local actual = GetTimeFactor() / const.DefaultTimeFactor
		local seg = { speed = speed, wanted = speed, actual = actual,
			took = (actual == speed), start_gt = GameTime(), start_rt = RealTime(),
			before = snapshot() }
		say(fmt("LADDER rung %d :: wanted %sx, GetTimeFactor reads %sx, TOOK=%s (set ok=%s) | segment target %d game ms from GameTime=%d",
			rung, tostring(speed), tostring(actual), tostring(seg.took), tostring(ok_set),
			C47.segment_gt, seg.start_gt))
		C47.DepotScan("rung " .. rung .. " open")
		C47.VegScan("rung " .. rung .. " open")

		local target = seg.start_gt + C47.segment_gt
		while true do
			sample_once()
			local gt, rt = GameTime(), RealTime()
			if gt == last_gt then
				still = still + 1
				if still == C47.freeze_samples then
					say(fmt("SAMPLE ⛔ CLOCK STOPPED for %d consecutive samples during rung %d (%sx) — this save pauses at day 385 under a PopupNotification's XPauseLayer. Every rung after this one is UNSAMPLED, not zero.",
						still, rung, tostring(speed)))
					C47.FreezeDiagnostic("clock stopped at " .. tostring(gt))
					C47.frozen_at = gt
					aborted = true
					break
				end
			else still = 0; last_gt = gt end
			if gt - target >= 0 then break end
			if rt - (C47.start_rt + C47.max_real_ms) >= 0 then
				say(fmt("SAMPLE ⛔ REAL-TIME CAP during rung %d (%sx) — this rung is PARTIAL and every later rung is UNSAMPLED.", rung, tostring(speed)))
				aborted = true
				break
			end
			Sleep(C47.sample_ms)
		end

		seg.end_gt, seg.end_rt = GameTime(), RealTime()
		seg.after = snapshot()
		seg.complete = (not aborted)
		C47.segments[#C47.segments + 1] = seg
		say(fmt("LADDER rung %d CLOSED :: %sx | game %d ms | real %d ms | complete=%s",
			rung, tostring(speed), seg.end_gt - seg.start_gt, seg.end_rt - seg.start_rt,
			tostring(seg.complete)))
	end
	return true
end

---- report -------------------------------------------------------------------

function C47.Report()
	local span_gt = (C47.frozen_at or GameTime()) - (C47.start_gt or GameTime())
	local span_rt = RealTime() - (C47.start_rt or RealTime())
	say(fmt("REPORT SPAN :: game time %d ms (%s game hours) | real time %d ms | samples=%d at %d ms | frozen at=%s",
		span_gt, tostring(span_gt / C47.MS_PER_HOUR), span_rt, C47.samples,
		C47.sample_ms, tostring(C47.frozen_at or "not frozen")))
	for _, rec in ipairs(C47.farms or empty_table) do
		local sizes, parts = {}, {}
		for s, c in pairs(rec.sizes) do sizes[#sizes + 1] = { s = s, c = c } end
		table.sort(sizes, function(a, b) return a.s < b.s end)
		for _, e in ipairs(sizes) do parts[#parts + 1] = fmt("%dx%d", e.s, e.c) end
		say(fmt("REPORT FARM %s :: buffer min=%s max=%s | stalled %d of %d samples | deliveries=%d totalling %d (mean %s) | drains=%d totalling %d | veg offers reachable at the last scan=%s | offers within %s hexes of the farm=%s",
			idof(rec.obj), tostring(rec.min), tostring(rec.max),
			rec.stalled, rec.n, rec.deliveries, rec.delivered,
			rec.deliveries > 0 and tostring(rec.delivered / rec.deliveries) or "n/a",
			rec.drains, rec.drain_total, tostring(rec.veg_offers),
			tostring(const and const.CommandCenterDefaultRadius), tostring(rec.veg_local)))
		say(fmt("REPORT FARM %s :: ⭐ DELIVERY SIZE DISTRIBUTION (size x count): %s — a wall of small values is the C48 signature; values near %s x DroneResourceCarryAmount would mean bulk depot pickups",
			idof(rec.obj), #parts > 0 and table.concat(parts, ", ") or "(none sampled)",
			tostring(const and const.ResourceScale)))
	end
	-- ⭐⭐ THE LADDER TABLE — the owner's test, one row per speed per farm, every
	--    figure a DIFFERENCE across that segment rather than a running total.
	--    Their prediction, recorded 2026-08-15: stall frequency FALLS as speed
	--    RISES. The existing readings agree in direction (52% at 1x attended,
	--    3.6% at 5x unattended) but were taken in different processes; this is
	--    the same colony throughout, so it is the real test.
	for _, seg in ipairs(C47.segments or empty_table) do
		local gt = seg.end_gt - seg.start_gt
		local hours = gt / C47.MS_PER_HOUR
		for i, rec in ipairs(C47.farms or empty_table) do
			local b, a = seg.before[i], seg.after[i]
			if b and a then
				local n = a.n - b.n
				local stalled = a.stalled - b.stalled
				local dl = a.deliveries - b.deliveries
				local dv = a.delivered - b.delivered
				local dr = a.drains - b.drains
				local dt = a.drain_total - b.drain_total
				say(fmt("LADDER ROW :: speed %sx (took=%s, complete=%s) | %s | game %d ms = %s h | STALLED %d of %d samples = %s%% | deliveries=%d totalling %d (mean %s) | drains=%d totalling %d | per GAME HOUR: drained %s, received %s",
					tostring(seg.speed), tostring(seg.took), tostring(seg.complete),
					idof(rec.obj), gt, tostring(hours),
					stalled, n, n > 0 and tostring(stalled * 100 / n) or "n/a",
					dl, dv, dl > 0 and tostring(dv / dl) or "n/a",
					dr, dt,
					hours > 0 and tostring(dt / hours) or "n/a",
					hours > 0 and tostring(dv / hours) or "n/a"))
			end
		end
	end
	say("LADDER ⛔ READ THE PER-GAME-HOUR COLUMNS, NOT THE RAW COUNTS: the rungs are equal in GAME time and wildly unequal in real time, so a raw count comparison across speeds is meaningless. ⛔ A rung whose took=false is VOID. A rung whose complete=false is PARTIAL. A rung that never ran is UNSAMPLED, never zero.")
	say("LADDER ⭐ THE OWNER'S PREDICTION, committed before this ran: stall frequency FALLS as speed RISES. If the stalled-% column does not fall monotonically, say so plainly — a refutation here is worth more than a confirmation, because it would mean the 52%-vs-3.6% gap between the attended and unattended legs has some other cause still unfound.")

	C47.DepotScan("at close")
	C47.VegScan("at close")
	say("REPORT ⛔ NOT CLAIMED: that a drone CHOSE a vegetation offer over a depot. This leg measures offer counts, depot stock and delivery sizes. If the depots sit flat while a farm's buffer sawtooths on small deliveries, that is strong circumstantial evidence — it is not a read of the drone's decision.")
	say("REPORT ⚠️ CONFOUNDS: cheats enabled, oversized terraforming speed-run colony, drone carry dial at +2 (a trip is 3000, not the vanilla 1000).")
	return true
end

function C47.Boot()
	CreateRealTimeThread(function()
		Sleep(1)
		say("=== c47-farm C48V === payload loaded. Staged save: " .. STAGED_SAVE)
		CreateRealTimeThread(function()
			local deadline = RealTime() + C47.watchdog
			while C47.stage ~= "done" do
				if RealTime() - deadline >= 0 then
					say("TIMEOUT " .. tostring(C47.stage))
					pcall(C47.Report)
					local f = rawget(_G, "FlushLogFile"); if f then f() end
					Sleep(1000) quit() return
				end
				Sleep(1000)
			end
		end)
		if rawget(_G, "ShowStartGamePopup") then ShowStartGamePopup = function() end end
		C47.stage = "waiting-menu"
		if not C47.WaitFor(5 * 60 * 1000, function() return not not GetPreGameMainMenu() end) then
			say("TIMEOUT waiting-menu") C47.stage = "done" Sleep(2000) quit() return
		end
		Sleep(2000)
		C47.ReadConditions("main menu, before the load")
		C47.stage = "loading"
		local ok, res = C47.TryYield("load " .. STAGED_SAVE, function()
			return C47.Load(STAGED_SAVE, C47.speed)
		end)
		if not (ok and res) then
			say("⛔ LOAD FAILED or the speed did not take — nothing below this line exists.")
			C47.ErrorWatchNote() C47.stage = "done" Sleep(2000) quit() return
		end
		C47.ReadConditions("after the load")
		C47.stage = "census"
		local ok_c, res_c = C47.TryYield("leg Census", C47.Census)
		if not (ok_c and res_c) then
			C47.ErrorWatchNote() C47.stage = "done" Sleep(2000) quit() return
		end
		C47.stage = "sampling"
		C47.TryYield("leg Sample", C47.Sample)
		C47.stage = "reporting"
		C47.Try("C47.Report", C47.Report)
		C47.ReadConditions("at close")
		C47.Now("close")
		C47.ErrorWatchNote()
		say("=== c47-farm C48V === DONE. ⛔ Nothing was saved.")
		C47.stage = "done"
		Sleep(3000)
		quit()
	end)
end

say("[C48V] 98_C48Veg loaded — self-driving in 1 tick.")
C47.Boot()
