---- output ------------------------------------------------------------------
--
-- ModLog is the only print path proven to reach the log file on disk
-- (Mod.lua:109-132). Its emitter formats its single argument a second time
-- (lib.lua:164-174), so a literal '%' raises there — escape it. Flush per line
-- so a killed run still shows how far it got.
local function say(msg)
	msg = "[" .. tostring(SP.tag) .. "] " .. tostring(msg)
	local modlog = rawget(_G, "ModLog")
	if modlog then
		modlog((msg:gsub("%%", "%%%%")))
	else
		local dbg = rawget(_G, "DebugPrint")
		if dbg then dbg(msg .. "\n") end
	end
	print(msg)
	local flush = rawget(_G, "FlushLogFile")
	if flush then pcall(flush) end
end
SP.Say = say

local function fmt(...)
	local ok, s = pcall(string.format, ...)
	return ok and s or "<fmt error>"
end
SP.Fmt = fmt

local function id_of(obj)
	if not obj then return "nil" end
	if type(obj) ~= "table" then return tostring(obj) end
	return tostring(obj.class or "?") .. "#" .. tostring(rawget(obj, "handle"))
end
SP.IdOf = id_of

-- G3. The ONLY sanctioned call wrapper. Prints ok and the error text, always.
-- ⛔ pcall cannot survive a yield — use SP.TryYield for anything that Sleeps.
function SP.Try(what, fn, ...)
	local ok, a, b = pcall(fn, ...)
	say(fmt("TRY %s ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

function SP.TryYield(what, fn, ...)
	local ok, a, b = sprocall(fn, ...)
	say(fmt("TRY %s (yielding) ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

-- G2. The condition-sampled rule, mechanised.
function SP.Applicable(label, count, population, witness)
	local applicable = (population or 0) > 0
	say(fmt("READ %s = %s | population=%s | APPLICABLE=%s | witness=%s",
		tostring(label), tostring(count), tostring(population),
		tostring(applicable), tostring(witness or "-- NONE NAMED --")))
	return applicable
end

---- flow helpers ------------------------------------------------------------

local function shutdown(reason)
	SP.stage = "done"
	say("SHUTDOWN " .. tostring(reason))
	local flush = rawget(_G, "FlushLogFile")
	if flush then flush() end
	Sleep(SP.quit_delay)
	quit()
end
SP.Shutdown = shutdown

-- untimed = true suppresses the END timing line (EF-045 correction C8): a step
-- spanning a load or a save blocks render mode and under-reports itself by up
-- to 11.5x. Loads take their number from the engine's own line.
local function step(name, fn, untimed)
	SP.stage = name
	local t = RealTime()
	say("STEP " .. name .. " BEGIN")
	local ok, res = sprocall(fn)
	if not ok then
		say("STEP " .. name .. " ERROR")
		shutdown("ERROR " .. name)
		return false
	end
	if untimed then
		say("STEP " .. name .. " END (no timing — this step blocks render mode)")
	else
		say(fmt("STEP %s END %dms", name, RealTime() - t))
	end
	if res == false then
		shutdown("STOP " .. name .. " — the step asked to stop; its reason is the line above")
		return false
	end
	return true
end
SP.Step = step

local function wait_for(timeout, ready)
	local deadline = RealTime() + timeout
	while not ready() do
		if RealTime() - deadline >= 0 then return false end
		Sleep(500)
	end
	return true
end
SP.WaitFor = wait_for

---- load / save -------------------------------------------------------------

function SP.Load(savefile, speed)
	local err = LoadGame(savefile, {})            -- ⛔ never pass save_as_last
	if err then
		say("LOADERR " .. tostring(savefile) .. " " .. tostring(err))
		return false
	end
	SP.loads = SP.loads + 1
	say(fmt("LOAD OK #%d file=%s — LoadGame returned no error", SP.loads, tostring(savefile)))
	if not wait_for(5 * 60 * 1000, game_is_live) then
		say("LOADERR " .. tostring(savefile) .. " — city never came live")
		return false
	end
	Sleep(SP.settle_time)                          -- 15 s is the MEASURED datum
	speed = speed or 3
	local before = GetTimeFactor() / const.DefaultTimeFactor
	local ok = pcall(function() UIColony:SetGameSpeed(speed) end)
	Sleep(500)
	local after = GetTimeFactor() / const.DefaultTimeFactor
	say(fmt("READ speed: SetGameSpeed(%d) call=%s | before=%s after=%s",
		speed, tostring(ok), tostring(before), tostring(after)))
	return true
end

-- ⛔ save_as_last is NEVER passed — it would repoint the owner's Continue button.
-- ⛔⛔ `savename` IS WRITTEN VERBATIM — the engine appends NOTHING (MEASURED
--    2026-08-11, unattended-2 run 1: SaveGame("U2RT1", {savename = "U2RT1"})
--    produced a file literally called `U2RT1`, with no extension, which
--    LoadGame("U2RT1.savegame.sav") could not then find). That extensionless
--    file is still in the save folder today. The guard below is why this run
--    cannot repeat it.
local SAVE_EXT = ".savegame.sav"

function SP.Save(name)
	name = tostring(name)
	if name:sub(-#SAVE_EXT) ~= SAVE_EXT then
		say(fmt("SAVE ⚠️ name '%s' carries no %s suffix — the engine writes savename VERBATIM. Appending it.", name, SAVE_EXT))
		name = name .. SAVE_EXT
	end
	local landed = false
	SP.TryYield("SaveGame " .. tostring(name), function()
		local err, nm = SaveGame(name, { savename = name, silent = true, no_screenshot = true })
		say(fmt("SAVE SaveGame returned err=%s name=%s", tostring(err), tostring(nm)))
		landed = not err
		return err
	end)
	Sleep(1000)
	say("SAVE WITNESS for " .. tostring(name)
		.. " = the on-disk file read agent-side after this process exits, plus the LoadGame of it below. NOT ListForTag (EF-049).")
	return landed
end

---- ⛔⛔ THE TWO-REGISTRY GATE READ ------------------------------------------
--
-- ⛔ TWO LINES, NOT ONE (TestKit README, 2026-08-12 split): a leg that quotes
--    one number describes half the rig. Both are printed even when the registry
--    is absent, because "absent" is itself a cell's expected reading.
local function read_registry(global_name, label)
	local pack = rawget(_G, global_name)
	if type(pack) ~= "table" or type(pack.order) ~= "table" then
		say(fmt("GATEREAD %s :: REGISTRY ABSENT (global %s is %s) — the mod's code did not run in this process",
			tostring(label), tostring(global_name), type(pack)))
		return 0, 0, nil
	end
	local active, total, rows = 0, 0, {}
	for _, id in ipairs(pack.order) do
		total = total + 1
		local f = pack.fixes and pack.fixes[id]
		local st = f and f.status or "MISSING"
		if st == "active" then active = active + 1 end
		rows[#rows + 1] = id .. "=" .. st
	end
	say(fmt("GATEREAD %s :: %d/%d active", tostring(label), active, total))
	-- per-module rows, in chunks, so a status flip is greppable by NAME
	local chunk = {}
	for i = 1, #rows do
		chunk[#chunk + 1] = rows[i]
		if #chunk == 10 or i == #rows then
			say(fmt("GATEROW %s :: %s", tostring(label), table.concat(chunk, " ")))
			chunk = {}
		end
	end
	return active, total, pack
end
SP.ReadRegistry = read_registry

function SP.ReadGates(when)
	say(fmt("GATES %s ------------------------------------------------", tostring(when)))
	local fa, ft = read_registry("SMRFixPack",   "fix pack")
	local oa, ot = read_registry("SMROptInPack", "opt-in pack")
	say(fmt("GATES %s :: fix pack %d/%d | opt-in pack %d/%d | map=%s | speed=%s | loads_this_process=%d",
		tostring(when), fa, ft, oa, ot, tostring(rawget(_G, "CurrentMapName")),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), SP.loads))
	return fa, ft, oa, ot
end

-- Which mods actually loaded CODE this process, read off ModsLoaded rather than
-- inferred from our own globals — the independent second reader for the config
-- gate below (corun-batch-2 ledger 3: one live read per engine name a reader
-- consumes; ModsLoaded is Mod.lua:2110-2114's own output).
function SP.ReadModOrder(when)
	local loaded = rawget(_G, "ModsLoaded")
	if type(loaded) ~= "table" then
		say(fmt("MODORDER %s :: NOT INSPECTABLE — ModsLoaded is %s", tostring(when), type(loaded)))
		return {}
	end
	local ids = {}
	for i, m in ipairs(loaded) do
		ids[#ids + 1] = fmt("%d:%s", i, tostring(type(m) == "table" and m.id or m))
	end
	-- ⭐ EF-054's open question: the route says INTER-mod order is the account's
	--    ENABLE order. This is the rig's OBSERVED order, which the fact asks
	--    prompt 4 to record. Position 1 loads FIRST and its wrappers sit INNERMOST.
	say(fmt("MODORDER %s :: %d mod(s) loaded, in load order :: %s",
		tostring(when), #loaded, table.concat(ids, " ")))
	return ids
end

---- ⛔⛔ THE CONFIG GATE — IT STOPS THE RUN ----------------------------------
--
-- WORKFLOW batch-2 harness rule 7, AMENDED 2026-08-11: "A gate whose only
-- output is a line in a log is not a gate." unattended-2 run 1 started in a
-- state a PREVIOUS sitting had left it in, printed pack=0/0, and carried on for
-- six steps taking readings about code that never ran.
--
-- This matrix makes that failure mode the NORMAL case: three of its cells are
-- deliberately-different mod configurations, so every launch MUST prove it is
-- the cell it says it is before it measures anything. `want_fix` / `want_opt`
-- are the cell's DECLARED configuration; a mismatch in EITHER direction stops.
function SP.RequireConfig(cell, want_fix, want_opt)
	local fa, ft, oa, ot = SP.ReadGates("config-gate/" .. tostring(cell))
	SP.ReadModOrder("config-gate/" .. tostring(cell))
	-- ⛔ Every launch prints the account's dial/toggle positions at boot. Two
	--    reasons: cell (d) run 1 could not adjudicate its own dial miss without
	--    them, and cell (e) moves the dials in-session — so the NEXT launch's
	--    boot read is the independent witness that its restore actually took.
	SP.ReadDialOptions("config-gate/" .. tostring(cell))
	local have_fix = type(rawget(_G, "SMRFixPack"))   == "table"
	local have_opt = type(rawget(_G, "SMROptInPack")) == "table"
	local ok = (have_fix == not not want_fix) and (have_opt == not not want_opt)
	if ok then
		say(fmt("CONFIGGATE OK :: cell %s :: fix pack present=%s (wanted %s) | opt-in present=%s (wanted %s) :: %d/%d and %d/%d",
			tostring(cell), tostring(have_fix), tostring(not not want_fix),
			tostring(have_opt), tostring(not not want_opt), fa, ft, oa, ot))
		return true
	end
	say(fmt("CONFIGGATE ⛔ WRONG CONFIGURATION :: cell %s wanted fix=%s opt=%s but this process has fix=%s opt=%s. EVERY reading this run could take is VOID — the cell is not the cell. STOPPING.",
		tostring(cell), tostring(not not want_fix), tostring(not not want_opt),
		tostring(have_fix), tostring(have_opt)))
	say("CONFIGGATE ⛔ THIS IS NOT SCRIPTABLE FROM HERE. AccountStorage, SaveAccountStorage and ModsReloadItems are all ModEnvBlacklist keys (Mod.lua), so no mod-side path can change which mods loaded. The configuration is set agent-side with the game CLOSED, and it takes a FULL PROCESS RESTART (D13's four-states rule).")
	return false
end

---- ⛔ THE §4.4 OPTION INSTRUMENT, BOTH DIRECTIONS ---------------------------
--
-- ⛔ MUST fire AFTER ClassesBuilt, or Opt_NoHomeless's F100 second-pass Require
--    fails and the cell misses its prediction for a reason that is not a port
--    regression (design MUST-FIX 4). RunAll-time is the simplest correct hook.
-- ⚠️ `Mods[id].options`, NEVER `CurrentModOptions` — the latter is per-mod-env
--    and the TestKit's copy is the TestKit's (the failed 2026-07-29 leg; the
--    build had to repair exactly this in the D09 dial probe).
-- ⛔ It writes NO account storage. It returns a restore closure that puts every
--    entry back to the value read on ENTRY, and the payload calls it in EVERY
--    branch — the DroneStatDials probe's proven shape.
SP.OPT_IDS = { "ClassicRockets", "AcknowledgedWarnings", "ResidencyControl",
               "MultipleSuns", "DroneOverhaul", "CohortHousing", "NoHomeless" }

function SP.SetOptIns(value, when)
	local mods = rawget(_G, "Mods")
	local def  = type(mods) == "table" and mods["SMR_CommunityOptInPack"]
	local opts = def and def.options
	if type(opts) ~= "table" then
		say(fmt("OPTINSET %s ⛔ options object unreachable (Mods[...] = %s, .options = %s) — NOT armed, and nothing was written",
			tostring(when), type(def), type(opts)))
		return nil
	end
	local orig = {}
	for _, id in ipairs(SP.OPT_IDS) do orig[id] = rawget(opts, id) end
	local before = {}
	for _, id in ipairs(SP.OPT_IDS) do before[#before + 1] = id .. "=" .. tostring(orig[id]) end
	say(fmt("OPTINSET %s :: entry values (restored in every branch) :: %s", tostring(when), table.concat(before, " ")))
	for _, id in ipairs(SP.OPT_IDS) do rawset(opts, id, value and true or false) end
	SP.Try("Msg ApplyModOptions (set " .. tostring(not not value) .. ")", function()
		Msg("ApplyModOptions", "SMR_CommunityOptInPack")
		return "sent"
	end)
	-- read the gate back BEFORE restoring, so the log records what was armed
	local a, t = read_registry("SMROptInPack", "opt-in pack AFTER in-session set=" .. tostring(not not value))
	say(fmt("OPTINSET %s :: %d/%d modules active after in-session set to %s",
		tostring(when), a, t, tostring(not not value)))
	return function()
		for _, id in ipairs(SP.OPT_IDS) do rawset(opts, id, orig[id]) end
		SP.Try("Msg ApplyModOptions (RESTORE)", function()
			Msg("ApplyModOptions", "SMR_CommunityOptInPack")
			return "sent"
		end)
		read_registry("SMROptInPack", "opt-in pack AFTER restore")
		say(fmt("OPTINSET %s :: entry values restored", tostring(when)))
	end
end

-- ⛔ THE DIAL POSITIONS, READ RATHER THAN ASSUMED. Design §6.2 predicted the two
--    D09 modifiers ABSENT after load, and its whole reason was "the mod id
--    changed, so the dials read BASE". That premise is an ACCOUNT-STATE claim,
--    not a code claim, and the account moved on 2026-08-12 18:30. A modifier
--    present at a NON-base dial is the module working exactly as designed; a
--    modifier present at BASE is the stale-residue failure the design feared.
--    The two cases are indistinguishable without this read, which is why the
--    first cell-(d) run could not adjudicate its own miss.
function SP.ReadDialOptions(when)
	local mods = rawget(_G, "Mods")
	local def  = type(mods) == "table" and mods["SMR_CommunityOptInPack"]
	local opts = def and def.options
	if type(opts) ~= "table" then
		say(fmt("DIALOPTS %s :: NOT INSPECTABLE — Mods[SMR_CommunityOptInPack] = %s, .options = %s",
			tostring(when), type(def), type(opts)))
		return nil
	end
	local parts = {}
	for _, id in ipairs(SP.OPT_IDS) do
		parts[#parts + 1] = id .. "=" .. tostring(rawget(opts, id))
	end
	local speed, carry = rawget(opts, "DroneSpeedDial"), rawget(opts, "DroneCarryDial")
	say(fmt("DIALOPTS %s :: DroneSpeedDial=%s (type=%s) | DroneCarryDial=%s (type=%s) — ⛔ the choice VALUE is the choice STRING itself (Mod.lua:2764-2771); '1x (base)' and '+0 (base)' are the base positions",
		tostring(when), tostring(speed), type(speed), tostring(carry), type(carry)))
	say(fmt("DIALOPTS %s :: toggles :: %s", tostring(when), table.concat(parts, " ")))
	return { speed = speed, carry = carry }
end

---- ⛔⛔ THE PERSISTED-NAME INVENTORY (cell (d)) -----------------------------
--
-- Design §6.2's per-row table, read back LIVE by EXACT BYTES. These names are
-- the save contract: they kept their `SMRFixPack_` prefix through the split on
-- purpose, and a rename would show here as a silently-reset population.
--
-- ⛔ EVERY ROW REPORTS ITS POPULATION SIZE, not just its count — a zero found
--    among 400 domes and a zero found among 0 domes are different facts, and
--    only the first is evidence ("close cases completely").
local DIAL_ROWS = {
	{ label = "Drone",  id = "SMRFixPack_DroneSpeedDial" },
	{ label = "Consts", id = "SMRFixPack_DroneCarryDial" },
}

-- Rows 1-3: instance fields. Walked over EVERY labelled object of every city —
-- a guessed class name that resolves to nothing prints a plausible zero forever
-- (corun-batch-2 ledger, three reader defects in one function).
local FIELD_ROWS = {
	{ field = "SMRFixPack_ack_notworking",           owner = "Building (Opt_AcknowledgedWarnings)" },
	{ field = "SMRFixPack_closed_to_new_residents",  owner = "Dome/MicroGHabitatBase (Opt_ResidencyControl)" },
	{ field = "SMRFixPack_no_homeless",              owner = "Dome/MicroGHabitatBase (Opt_NoHomeless)" },
}

function SP.ReadInventory(when)
	say(fmt("INVENTORY %s ==========================================", tostring(when)))
	local colony = rawget(_G, "UIColony")

	-- rows 4-5: the D09 dial modifiers, by EXACT id, in their own containers
	local lm = type(colony) == "table" and rawget(colony, "label_modifiers")
	if type(lm) ~= "table" then
		say(fmt("INVROW %s :: dials NOT INSPECTABLE — UIColony.label_modifiers is %s. Absence here is NOT evidence of absence.",
			tostring(when), type(lm)))
	else
		for _, row in ipairs(DIAL_ROWS) do
			local bucket = lm[row.label]
			local n = 0
			if type(bucket) == "table" then for _ in pairs(bucket) do n = n + 1 end end
			local mod = type(bucket) == "table" and bucket[row.id]
			-- ⛔ the modifier's OWN fields, so presence can be adjudicated against
			--    the dial position rather than read as a bare true/false. A
			--    modifier at a non-base dial is the module working; a modifier
			--    present while the dial reads base is the stale residue.
			say(fmt("INVROW %s :: label_modifiers[%s][%s] = %s (type=%s truthy=%s) | prop=%s percent=%s amount=%s id=%s | modifiers on that label = %d — read this beside the DIALOPTS line: ABSENT is correct at a BASE dial, PRESENT is correct at a non-base dial (Opt_DroneStatDials.lua:42-44)",
				tostring(when), tostring(row.label), tostring(row.id), tostring(mod),
				type(mod), tostring(not not mod),
				tostring(type(mod) == "table" and mod.prop),
				tostring(type(mod) == "table" and mod.percent),
				tostring(type(mod) == "table" and mod.amount),
				tostring(type(mod) == "table" and mod.id), n))
		end
	end

	-- the positive control: fix-pack side, unmoved. It proves the read method
	-- works when the answer should be yes.
	local ctrl = type(colony) == "table" and rawget(colony, "SMRFixPack_F48_StationConnectors")
	say(fmt("INVROW %s :: CONTROL UIColony.SMRFixPack_F48_StationConnectors = %s (type=%s truthy=%s) — fix-pack side, never moved",
		tostring(when), tostring(ctrl), type(ctrl), tostring(not not ctrl)))

	-- rows 1-3: instance fields, with populations
	local city = rawget(_G, "MainCity") or rawget(_G, "UICity")
	if not city or type(city.labels) ~= "table" then
		say(fmt("INVROW %s :: instance fields NOT INSPECTABLE — no city/labels", tostring(when)))
		return
	end
	local counts, values, seen, scanned = {}, {}, {}, 0
	for _, row in ipairs(FIELD_ROWS) do counts[row.field] = 0; values[row.field] = {} end
	local domes, buildings = 0, 0
	for _, list in pairs(city.labels) do
		if type(list) == "table" then
			for _, obj in ipairs(list) do
				if type(obj) == "table" and not seen[obj] then
					seen[obj] = true
					scanned = scanned + 1
					if IsKindOf(obj, "Dome") or IsKindOf(obj, "MicroGHabitatBase") then domes = domes + 1 end
					if IsKindOf(obj, "Building") then buildings = buildings + 1 end
					for _, row in ipairs(FIELD_ROWS) do
						local v = rawget(obj, row.field)
						if v ~= nil then
							counts[row.field] = counts[row.field] + 1
							if #values[row.field] < 6 then
								values[row.field][#values[row.field] + 1] =
									fmt("%s=%s", id_of(obj), tostring(v))
							end
						end
					end
				end
			end
		end
	end
	say(fmt("INVPOP %s :: scanned %d distinct labelled object(s) | Dome/MicroGHabitatBase = %d | Building = %d",
		tostring(when), scanned, domes, buildings))
	for _, row in ipairs(FIELD_ROWS) do
		local pop = row.field == "SMRFixPack_ack_notworking" and buildings or domes
		say(fmt("INVROW %s :: %s on %d object(s) of %d %s | examples: %s",
			tostring(when), row.field, counts[row.field], pop, tostring(row.owner),
			#values[row.field] > 0 and table.concat(values[row.field], " ") or "<none>"))
		SP.Applicable("cell-d-row(" .. row.field .. ")", counts[row.field], pop,
			"every labelled object of every city walked this load; own keys only, `next` never walks the metatable chain")
	end
	say(fmt("INVENTORY %s END =======================================", tostring(when)))
	return { counts = counts, scanned = scanned, domes = domes, buildings = buildings }
end

---- ⛔⛔ THE ACTIVE SAVE-CONTRACT ROUND TRIP (cell (e)) ----------------------
--
-- WHY THIS CELL EXISTS, and it was not in the design. Cell (d) is a PASSIVE
-- witness: it can only read stamps a previous world happened to leave. Run on
-- `CP15PT15` it read 0 / 0 / 0 against a population of ONE dome and 116
-- buildings — the "thin witness" the design's own residual risk 3 and the QA's
-- weak claim 2 both named. A passive read cannot be made to say more by staging
-- another save; if the owner never used these policies, no save carries them.
--
-- So this cell WRITES the contract and reads it back across a real save. The
-- three fields are plain persisted instance fields whose only writer is a UI
-- handler doing exactly `obj[FLAG] = true` (Opt_AcknowledgedWarnings.lua:105,
-- and the same shape behind the two dome policy toggles), so writing them from
-- here is the same act the player's click performs.
--
-- ⛔ NAME WHAT IS FORCED (the forced-vs-organic rule): FORCED = the field write
--    and the dial position. MEASURED = that the EXACT BYTES survive
--    save -> reload, on the same objects, and that the ported modules read them
--    back. That is invariant 6(c) tested directly instead of hoped for.
-- ⛔ It runs ONLY on a staged COPY, which dies in the result commit.
local CONTRACT_DOME_FIELDS = { "SMRFixPack_closed_to_new_residents", "SMRFixPack_no_homeless" }
local CONTRACT_BLD_FIELD   = "SMRFixPack_ack_notworking"

-- Collects, for one field, the SET OF HANDLES carrying it. Handles survive a
-- save/reload; Lua table identity does not, so a handle set is the only exact
-- before/after comparison available.
function SP.HandlesWith(field)
	local out, seen = {}, {}
	local city = rawget(_G, "MainCity") or rawget(_G, "UICity")
	if not city or type(city.labels) ~= "table" then return out end
	for _, list in pairs(city.labels) do
		if type(list) == "table" then
			for _, o in ipairs(list) do
				if type(o) == "table" and not seen[o] then
					seen[o] = true
					if rawget(o, field) ~= nil then out[#out + 1] = rawget(o, "handle") end
				end
			end
		end
	end
	table.sort(out)
	return out
end

local function handles_line(list)
	local parts = {}
	for i = 1, #list do parts[#parts + 1] = tostring(list[i]) end
	return table.concat(parts, ",")
end
SP.HandlesLine = handles_line

function SP.WriteContract(when, how_many_buildings)
	how_many_buildings = how_many_buildings or 3
	local city = rawget(_G, "MainCity") or rawget(_G, "UICity")
	if not city or type(city.labels) ~= "table" then
		say(fmt("CONTRACT %s ⛔ no city/labels — nothing written", tostring(when)))
		return nil
	end
	local domes, buildings, seen = {}, {}, {}
	for _, list in pairs(city.labels) do
		if type(list) == "table" then
			for _, o in ipairs(list) do
				if type(o) == "table" and not seen[o] then
					seen[o] = true
					if IsKindOf(o, "Dome") or IsKindOf(o, "MicroGHabitatBase") then domes[#domes + 1] = o
					elseif IsKindOf(o, "Building") then buildings[#buildings + 1] = o end
				end
			end
		end
	end
	say(fmt("CONTRACT %s :: subjects available :: %d Dome/MicroGHabitatBase, %d other Building",
		tostring(when), #domes, #buildings))
	if #domes == 0 then
		say("CONTRACT ⛔ no dome to stamp — this world cannot carry the two dome-policy rows")
	end
	for _, dome in ipairs(domes) do
		for _, f in ipairs(CONTRACT_DOME_FIELDS) do rawset(dome, f, true) end
	end
	local stamped = 0
	for i = 1, math.min(how_many_buildings, #buildings) do
		rawset(buildings[i], CONTRACT_BLD_FIELD, true)
		stamped = stamped + 1
	end
	say(fmt("CONTRACT %s :: FORCED WRITE :: %s on every one of %d dome(s); %s on %d building(s)",
		tostring(when), table.concat(CONTRACT_DOME_FIELDS, " + "), #domes,
		CONTRACT_BLD_FIELD, stamped))
	local rec = { fields = {} }
	for _, f in ipairs(CONTRACT_DOME_FIELDS) do rec.fields[f] = SP.HandlesWith(f) end
	rec.fields[CONTRACT_BLD_FIELD] = SP.HandlesWith(CONTRACT_BLD_FIELD)
	for f, hs in sorted_pairs(rec.fields) do
		say(fmt("CONTRACT %s :: %s now on %d object(s), handles=[%s]",
			tostring(when), f, #hs, handles_line(hs)))
	end
	return rec
end

-- Sets both D09 dials OFF BASE through the module's REAL Apply path, so the
-- persisted modifier ids are written by the ported module itself rather than by
-- this instrument. Returns a restore closure; the caller runs it in EVERY branch.
function SP.SetDials(speed_value, carry_value, when)
	local mods = rawget(_G, "Mods")
	local def  = type(mods) == "table" and mods["SMR_CommunityOptInPack"]
	local opts = def and def.options
	if type(opts) ~= "table" then
		say(fmt("DIALSET %s ⛔ options object unreachable — nothing written", tostring(when)))
		return nil
	end
	local o_speed, o_carry = rawget(opts, "DroneSpeedDial"), rawget(opts, "DroneCarryDial")
	say(fmt("DIALSET %s :: entry values DroneSpeedDial=%s DroneCarryDial=%s (restored in every branch)",
		tostring(when), tostring(o_speed), tostring(o_carry)))
	rawset(opts, "DroneSpeedDial", speed_value)
	rawset(opts, "DroneCarryDial", carry_value)
	SP.Try("Msg ApplyModOptions (dials -> " .. tostring(speed_value) .. " / " .. tostring(carry_value) .. ")", function()
		Msg("ApplyModOptions", "SMR_CommunityOptInPack")
		return "sent"
	end)
	return function()
		rawset(opts, "DroneSpeedDial", o_speed)
		rawset(opts, "DroneCarryDial", o_carry)
		SP.Try("Msg ApplyModOptions (dials RESTORE)", function()
			Msg("ApplyModOptions", "SMR_CommunityOptInPack")
			return "sent"
		end)
		say(fmt("DIALSET %s :: entry values restored", tostring(when)))
	end
end

-- ⛔ The verdict is the EXACT BYTES surviving, on the SAME objects. A count that
--    matches while the handles differ is not a pass.
function SP.VerifyContract(rec, when)
	if type(rec) ~= "table" then
		say("CONTRACTVERIFY NOT-APPLICABLE (nothing was written) :: " .. tostring(when))
		return nil
	end
	local broken = 0
	for f, before in sorted_pairs(rec.fields) do
		local after = SP.HandlesWith(f)
		local same = (#before == #after)
		if same then
			for i = 1, #before do if before[i] ~= after[i] then same = false end end
		end
		if not same then broken = broken + 1 end
		say(fmt("CONTRACTVERIFY %s :: %s :: before %d object(s) [%s] -> after %d object(s) [%s] :: %s",
			tostring(when), f, #before, handles_line(before), #after, handles_line(after),
			same and "IDENTICAL — the exact bytes survived the round trip on the same objects"
			      or "⛔ CHANGED — a persisted name did not survive"))
	end
	say(fmt("CONTRACTVERIFY %s :: %d of %d field(s) broke the save contract",
		tostring(when), broken, 3))
	return broken
end

-- Names every row that moved between two takings, rather than printing a total.
function SP.DiffInventory(a, b, what)
	if type(a) ~= "table" or type(b) ~= "table" then
		say("INVDIFF NOT-APPLICABLE (a reading is missing) :: " .. tostring(what))
		return nil
	end
	local changed = {}
	for _, row in ipairs(FIELD_ROWS) do
		if a.counts[row.field] ~= b.counts[row.field] then
			changed[#changed + 1] = fmt("%s: %s -> %s", row.field,
				tostring(a.counts[row.field]), tostring(b.counts[row.field]))
		end
	end
	if a.scanned ~= b.scanned then
		changed[#changed + 1] = fmt("scanned: %s -> %s", tostring(a.scanned), tostring(b.scanned))
	end
	say(fmt("INVDIFF %s :: %d reading(s) changed%s", tostring(what), #changed,
		#changed > 0 and (" — " .. table.concat(changed, " | ")) or ""))
	return #changed
end

---- the suite ---------------------------------------------------------------
--
-- ⛔ SKIPs BY NAME, never as a total (STATE's standing rule). SMRTest.last
--    carries every probe's verdict after RunAll, so the per-probe rows can be
--    re-emitted as one greppable line per status class and diffed against
--    u2run3's archived list (design §7 residual risk 4).
function SP.RunSuite(when)
	local kit = rawget(_G, "SMRTest")
	if not (kit and kit.RunAll) then
		say("SUITE ⛔ SMRTest.RunAll is missing (00_TestCore.lua did not load) :: " .. tostring(when))
		return nil
	end
	local counts
	SP.Try("SMRTest.RunAll() " .. tostring(when), function()
		counts = kit.RunAll()
		return fmt("PASS=%s FAIL=%s SKIP=%s ERROR=%s",
			tostring(counts and counts.PASS), tostring(counts and counts.FAIL),
			tostring(counts and counts.SKIP), tostring(counts and counts.ERROR))
	end)
	local last = kit.last
	if type(last) ~= "table" then
		say("SUITE ⚠️ SMRTest.last is not a table — per-probe rows NOT re-emitted; read them off the RunAll block above")
		return counts
	end
	local by = { PASS = {}, FAIL = {}, SKIP = {}, ERROR = {} }
	for _, id in ipairs(kit.order or empty_table) do
		local r = last[id]
		local st = r and r.status or "MISSING"
		local bucket = by[st] or by.ERROR
		bucket[#bucket + 1] = id
	end
	for _, st in ipairs({ "FAIL", "ERROR", "SKIP", "PASS" }) do
		local names = by[st]
		say(fmt("SUITEBYNAME %s :: %s (%d) :: %s", tostring(when), st, #names,
			#names > 0 and table.concat(names, " ") or "<none>"))
	end
	return counts
end

---- error-watch marker ------------------------------------------------------
--
-- ⛔ It may NOT count engine error lines: the engine writes those straight to
--    the log file and the mod environment has no file read (io/os are both
--    ModEnvBlacklist keys). A count printed from here would be invented. It
--    prints the boundary a reader greps between.
-- ⚠️ The token an error sweep greps for is spelled AROUND, not written.
function SP.ErrorWatchNote()
	say(fmt("ERRORWATCH %s — every engine error line (the bracketed LUA-ERROR marker) in this log between the '=== split-optins %s ===' banner and this line happened inside this run. Report every unexplained line VERBATIM with its position and its age; never discount one. stage_at_close=%s loads_this_process=%d",
		tostring(SP.tag), tostring(SP.tag), tostring(SP.stage), SP.loads))
end

---- watchdog + flow ---------------------------------------------------------

function SP.Flow()
	CreateRealTimeThread(function()
		local deadline = RealTime() + SP.watchdog
		while SP.stage ~= "done" do
			if RealTime() - deadline >= 0 then
				say("WATCHDOG TIMEOUT at stage " .. tostring(SP.stage))
				local flush = rawget(_G, "FlushLogFile")
				if flush then flush() end
				Sleep(1000)
				quit()
				return
			end
			Sleep(1000)
		end
	end)

	say(fmt("=== split-optins %s === watchdog %dms", tostring(SP.tag), SP.watchdog))

	if not step("1-waiting-menu", function()
		if not wait_for(SP.menu_timeout, function() return not not GetPreGameMainMenu() end) then
			return false
		end
		Sleep(2000)
	end) then return end

	local ok = sprocall(SP.Payload)
	if not ok then
		say("PAYLOAD ERROR at stage " .. tostring(SP.stage))
		shutdown("ERROR payload")
		return
	end
	shutdown("done")
end

function SP.Boot()
	CreateRealTimeThread(function()
		-- CreateRealTimeThread runs its body synchronously up to the first yield,
		-- so this Sleep is what puts everything below after ModsLoadCode().
		Sleep(1)
		SP.Flow()
	end)
end
