---- PT-15 — where is the mystery? -------------------------------------------
--
-- ⚠️⚠️ THIS READ IS PARTLY DEGRADED BY DESIGN, and the degraded half is named
--    on its own line rather than papered over with a guess (prompt 1 stop
--    condition). What the engine exposes and what it does not:
--
--  ✅ THE FOUNDER STAGE IS FULLY MEASURABLE. g_ColonyNotViableUntil is a GameVar
--     (Lua\ColonyViability.lua:80) whose encoding is documented at :74-78:
--       -3 = before the first passenger rocket launched
--       -2 = first passenger rocket in transit
--       -1 = colony viable — ⭐ APPROVAL HAS PASSED
--       >0 = GameTime() at which the founder stage ends; OnMsg.NewHour fires
--            Msg("ColonyApprovalPassed") on the first hour after it
--            (:87-98). const.ColonyViableByDelay = 10 sols (:102).
--     ⇒ if it is > 0, the remaining founder time is EXACTLY (value - GameTime()).
--
--  ⛔ THE 10–20 SOL SLEEP IS NOT MEASURABLE. Mystery 11's Trigger sequence does
--     WaitMsg("ColonyApprovalPassed") then Sleep(7200000 + InteractionRand(
--     7200000, seq_id)) (Lua\Scenario\Mystery 11.generated.lua:75-76) inside a
--     game-time coroutine. Lua is given NO wake-time accessor: the sleeping
--     thread's deadline lives engine-side, ThreadsRegister is a dbg()-gated
--     debug registry, and script sequences carry no instruction pointer to read
--     (SequenceListPlayer:StartSequenceScript, CommonLua\Libs\Sequences\
--     SequenceListPlayer.lua:431-459 — one coroutine, no ip). ⇒ this leg reports
--     ARMED / NOT ARMED and the 0–20 sol BOUND, never a remaining figure. A
--     reader who wants a number must march and watch for the Sinkhole message.
--
--  ✅ WHAT IS ARMED IS DECIDABLE. The Trigger sequence's thread being alive plus
--     zero sinkholes plus spawn_fireflies=false places the sequence in the sleep
--     (or still in the WaitMsg, which g_ColonyNotViableUntil then separates).
function CP15.MysteryWhere()
	if not CP15.Guard("MysteryWhere") then return false end
	CP15.Label("ORGANIC", "mystery state read", "reads only; nothing in the sequence is touched")

	local gt = CP15.Now("mystery state read")

	-- 1. the founder stage, exactly
	local viable = rawget(_G, "g_ColonyNotViableUntil")
	local approval
	if viable == -1 then       approval = "PASSED (colony viable)"
	elseif viable == -3 then   approval = "NOT PASSED — first passenger rocket not launched yet"
	elseif viable == -2 then   approval = "NOT PASSED — first passenger rocket in transit"
	elseif type(viable) == "number" and viable > 0 then
		approval = fmt("NOT PASSED — founder stage ends at GameTime %d, i.e. %s from now",
			viable, CP15.Sols(viable - gt))
	else                       approval = "UNREADABLE" end
	say(fmt("MYSTERY approval :: g_ColonyNotViableUntil=%s (raw, type=%s) | GameTime()=%s | ⇒ %s | const.ColonyViableByDelay=%s (%s)",
		tostring(viable), type(viable), tostring(gt), approval,
		tostring(const.ColonyViableByDelay), CP15.Sols(const.ColonyViableByDelay)))

	-- 2. the sequence player, by the public accessor
	local ok_p, player = pcall(function() return GetScenarioPlayer("Mystery 11") end)
	say(fmt("MYSTERY player :: GetScenarioPlayer('Mystery 11') ok=%s -> %s | s_SeqListPlayers=%d entr(y/ies)",
		tostring(ok_p), tostring(player),
		#(rawget(_G, "s_SeqListPlayers") or empty_table)))
	-- ⚠️ MysteryBase.seq_player is declared (Lua\Mysteries\Mystery.lua:6) and
	--    NEVER ASSIGNED anywhere in the tree — do not reach for it.
	local colony  = rawget(_G, "UIColony")
	local mystery = colony and colony.mystery
	say(fmt("MYSTERY player :: UIColony.mystery.seq_player=%s ⚠️ declared-but-never-assigned in Src; the accessor above is the real handle",
		tostring(mystery and rawget(mystery, "seq_player"))))

	if not player then
		say("MYSTERY ⛔ NO SEQUENCE PLAYER for 'Mystery 11'. Either the mystery is not this scenario or the scenario never started. Every line below is NOT-APPLICABLE.")
		CP15.Applicable("PT15-sequence-player(a Mystery 11 player exists)", 0, 0,
			"GetScenarioPlayer('Mystery 11') returned nothing this load")
		return false
	end

	-- 3. every sequence's status, with the thread liveness read raw
	local names, running = {}, 0
	for name, st in sorted_pairs(player.seq_states or empty_table) do
		local alive = IsValidThread(st and st.thread)
		if alive then running = running + 1 end
		names[#names + 1] = name
		say(fmt("MYSTERY seq :: '%s' status=%s | thread=%s alive=%s (type=%s)",
			tostring(name), tostring(st and st.status), tostring(st and st.thread),
			tostring(not not alive), type(alive)))
	end
	say(fmt("MYSTERY seq :: %d sequence state(s), %d with a live thread", #names, running))

	-- 4. the Trigger sequence's own registers — this is where the choices live
	local trig = (player.seq_states or empty_table)["Trigger"]
	if trig and type(trig.GetRegisters) == "function" then
		CP15.Try("Trigger GetRegisters [FIRST EXECUTION]", function()
			local r = trig.GetRegisters()
			local parts = {}
			for k, v in sorted_pairs(r or empty_table) do
				parts[#parts + 1] = tostring(k) .. "=" .. tostring(v)
			end
			say("MYSTERY registers :: " .. table.concat(parts, " "))
			return #parts
		end)
	else
		say("MYSTERY registers :: NOT-APPLICABLE — no 'Trigger' seq_state with a GetRegisters closure")
	end

	-- 5. the arming verdict, stated as a bound and never as a remaining figure
	local trig_alive = IsValidThread(trig and trig.thread)
	local sink = #CP15.LabelAll("Sinkhole")
	local armed = (viable == -1) and trig_alive and sink == 0
		and not (mystery and mystery.spawn_fireflies)
	say(fmt("MYSTERY countdown :: Trigger thread alive=%s | sinkholes=%d | spawn_fireflies=%s ⇒ %s",
		tostring(not not trig_alive), sink, tostring(mystery and mystery.spawn_fireflies),
		armed and "ARMED — the 10-20 sol sleep is running" or
		((viable ~= -1) and "NOT ARMED — still waiting on ColonyApprovalPassed" or
		 "PAST IT — the sinkhole stage has begun or the sequence is elsewhere")))
	say("MYSTERY countdown :: ⛔ DEGRADED READ. The remaining sleep is NOT exposed to Lua (no wake-time accessor; script sequences carry no ip). The only true statement is the BOUND: 0 to 20 sols from the moment approval passed, uniform over 10-20 sols in total. Sleep(7200000 + InteractionRand(7200000, seq_id)) — Mystery 11.generated.lua:76.")

	-- ⭐ THE BOUND IS TIGHTER THAN "0 TO 20" WHEN THE APPROVAL MOMENT IS KNOWN,
	--    and on THIS fixture it is — from the owner, not from the game.
	-- ⛔ The stamp is DESTROYED when approval fires: g_ColonyNotViableUntil is
	--    overwritten to -1 (ColonyViability.lua:95-97) and OnMsg.
	--    ColonyApprovalPassed calls RemoveNotification("FounderStageDuration")
	--    (Legislature.lua:1522-1523). Nothing persists the time. So an approval
	--    moment can only ever enter this run as a CLAIM, and it is printed as
	--    one — never folded into a derived figure without its source attached.
	if armed then
		CP15.armed_seen = CP15.armed_seen or gt
		say(fmt("MYSTERY countdown :: CEILING (sound, no claim needed) — approval passed at or before this reading, so the sinkhole beat fires no later than GameTime %d, i.e. within %s of now. The march has a guaranteed end.",
			gt + 20 * CP15.MS_PER_SOL, CP15.Sols(20 * CP15.MS_PER_SOL)))
		if CP15.approval_claim then
			say(fmt("MYSTERY countdown :: FLOOR from a CLAIM, not a measurement :: %s ⇒ the beat cannot fire before roughly GameTime %d (about %s from now). Treat as a planning figure only.",
				tostring(CP15.approval_claim), gt + 10 * CP15.MS_PER_SOL,
				CP15.Sols(10 * CP15.MS_PER_SOL)))
			say("MYSTERY countdown :: ⇒ SCHEDULING CONSEQUENCE — the first stretch of the march is dead time by construction. Spend it on the C39 leg and the Ctrl-F9 check rather than watching an empty map.")
		else
			say("MYSTERY countdown :: NO FLOOR AVAILABLE — nothing in this run knows when approval passed, and the game does not keep it. Set CP15.approval_claim to the owner's own words if they have them; leave it unset rather than guessing.")
		end
	elseif CP15.approval_claim and viable ~= -1 then
		say(fmt("MYSTERY countdown :: ⛔⛔ THE CARRIED OWNER CLAIM IS REFUTED BY THIS READING. The claim was: %s — but g_ColonyNotViableUntil reads %s, which is not -1, so approval has NOT passed. Record the refutation, drop the floor, and re-plan the march from the founder-stage figure above (which IS exact).",
			tostring(CP15.approval_claim), tostring(viable)))
	end
	CP15.Applicable("PT15-countdown-read(a Trigger sequence state exists to read)",
		trig and 1 or 0, trig and 1 or 0,
		"player.seq_states walked this load; 'Trigger' is the mystery's main sequence (Mystery 11.generated.lua:37)")
	return armed
end

---- PT-15 — the trap and grid read (F07) ------------------------------------
--
-- ⭐ THE ONE-LINE VERDICT: broken = electricity_production equals the WISP COUNT;
--    fixed = 1000x the wisp count. A Solar Panel is 2000 internal units, so 3
--    wisps broken is 3 units (nothing) and 3 wisps fixed is 3000 (a real
--    producer). Route, re-derived:
--      SetLightTrapMode("free") -> trap.el_prod_modifier:Change(#trap.fireflies)
--      (Fireflies.lua:692, the defect — the two sibling sites :346 and :479 both
--      multiply by 1000). ObjectModifier:Change SETS ABSOLUTELY (Lua\Modifiers
--      .lua:321-331; the game UndefineClass'es CommonLua's version at :288, so
--      the CommonLua ModSet variant is NOT the one that runs).
--      -> UpdateModifier -> ElectricityProducer:OnModifiableValueChanged
--      (Lua\ElectricityProducer.lua:68-73) -> electricity:SetProduction(working
--      and GetPerformanceModifiedElectricityProduction() or 0).
--
-- ⛔⛔ TIMING — READ IMMEDIATELY AFTER THE CHOICE, BEFORE ANY WISP MOVES. The
--    defect SELF-HEALS: every later catch (Fireflies.lua:479) and every later
--    release (:346) re-Changes the same modifier WITH the 1000, so a broken
--    reading is destroyed by the next wisp event — typically the 4 AM
--    FirefliesGoHome (Fireflies.lua:257-263, hour == 4). This is why the entry
--    says the broken value "persists until wisp count changes".
--
-- ⚠️ `working` gates the GRID number but not the property. A trap that is not
--    working shows electricity_production correctly and grid production 0. Both
--    are printed so the two cannot be confused.
function CP15.TrapRead(when)
	if not CP15.Guard("TrapRead") then return false end
	CP15.Label("ORGANIC", "PT-15 trap/grid read", "reads only; the mode is whatever the played mystery set")

	local colony  = rawget(_G, "UIColony")
	local mystery = colony and colony.mystery
	local traps   = CP15.LabelAll("LightTrap")
	CP15.Now(when)
	say(fmt("TRAP %s :: lighttrap_mode=%s | %d LightTrap(s) | fireflies_caught=%s fireflies_destroyed=%s",
		tostring(when), tostring(mystery and mystery.lighttrap_mode), #traps,
		tostring(mystery and mystery.fireflies_caught),
		tostring(mystery and mystery.fireflies_destroyed)))

	local total_wisps, total_prod, verdicts = 0, 0, {}
	for i, t in ipairs(traps) do
		local wisps = #(rawget(t, "fireflies") or empty_table)
		total_wisps = total_wisps + wisps
		local mod  = rawget(t, "el_prod_modifier")
		local amt  = mod and rawget(mod, "amount")
		local pct  = mod and rawget(mod, "percent")
		local appl = mod and rawget(mod, "is_applied")
		-- the property itself, own field and resolved, both raw (a nil must show)
		local own = rawget(t, "electricity_production")
		local ok_r, resolved = pcall(function() return t.electricity_production end)
		local el   = rawget(t, "electricity")
		local prod = el and rawget(el, "production")
		local grid = el and rawget(el, "grid")
		local ok_u, ui = pcall(function() return t:GetUIPowerProduction() end)
		total_prod = total_prod + (type(prod) == "number" and prod or 0)
		say(fmt("TRAP %s :: #%d %s wisps=%d | el_prod_modifier{amount=%s percent=%s is_applied=%s} | electricity_production own=%s resolved=%s (ok=%s) | working=%s (type=%s) work_state=%s | grid element production=%s | GetUIPowerProduction()=%s (ok=%s)",
			tostring(when), i, idof(t), wisps,
			tostring(amt), tostring(pct), tostring(appl),
			tostring(own), tostring(resolved), tostring(ok_r),
			tostring(rawget(t, "working")), type(rawget(t, "working")),
			tostring(rawget(t, "work_state")),
			tostring(prod), tostring(ui), tostring(ok_u)))
		if grid then
			say(fmt("TRAP %s :: #%d grid :: production=%s current_production=%s consumption=%s current_consumption=%s current_reserve=%s | %d producer(s) %d consumer(s) %d storage(s)",
				tostring(when), i,
				tostring(rawget(grid, "production")), tostring(rawget(grid, "current_production")),
				tostring(rawget(grid, "consumption")), tostring(rawget(grid, "current_consumption")),
				tostring(rawget(grid, "current_reserve")),
				#(rawget(grid, "producers") or empty_table),
				#(rawget(grid, "consumers") or empty_table),
				#(rawget(grid, "storages") or empty_table)))
		else
			say(fmt("TRAP %s :: #%d grid :: NOT-APPLICABLE — trap has no electricity grid element", tostring(when), i))
		end
		-- the arithmetic verdict, per trap, with both candidate values spelled out
		if wisps > 0 and type(resolved) == "number" then
			local v
			if resolved == wisps * 1000 then      v = "FIXED (equals wisps x 1000)"
			elseif resolved == wisps then         v = "⛔ BROKEN (equals the wisp count — the missing x1000, Fireflies.lua:692)"
			else                                  v = "NEITHER — see the raw numbers" end
			verdicts[#verdicts + 1] = fmt("#%d %s: wisps=%d prod=%d broken-would-be=%d fixed-would-be=%d ⇒ %s",
				i, idof(t), wisps, resolved, wisps, wisps * 1000, v)
		end
	end
	for _, v in ipairs(verdicts) do say(fmt("%sTRAPVERDICT %s :: %s", CP15.Void(), tostring(when), v)) end
	say(fmt("%sTRAPSUM %s :: %d trap(s), %d wisp(s) held, %d total grid production | reference: a Solar Panel produces 2000 internal units",
		CP15.Void(), tostring(when), #traps, total_wisps, total_prod))

	CP15.Applicable("PT15-trap-population(a Light Trap exists)", #traps, #traps,
		"LightTrap label walked across every city this load")
	CP15.Applicable("PT15-wisp-population(a trap is HOLDING wisps — without this the F07 reading is meaningless, since 0 x 1 and 0 x 1000 are both 0)",
		total_wisps, total_wisps, "trap.fireflies counted per trap this load")
	return total_wisps, total_prod
end

---- F15 rider — the destroy-mode research reward ----------------------------
--
-- ⚠️⚠️ RE-DERIVED, AND THE ENTRY'S IMPLIED INSTRUMENT IS WRONG. There is no
--    research-point BANK to read. Research:AddResearchPoints (Lua\Research.lua:
--    810-846) adds to the CURRENT QUEUE'S TECH:
--      status.points = status.points + ModifyResearchPoints(points * scale)
--    with scale = const.ResearchPointsScale = 1000 (Lua\_GameConst.lua:44) and
--    ModifyResearchPoints (:722-737) applying FOUR multipliers. It also RETURNS
--    EARLY, silently, when there is no researchable tech — points vanish.
--    ⇒ the granted figure is Δstatus.points / scale, un-modified, and every
--      multiplier must be PRINTED beside it rather than assumed to be 100.
--
-- ⚠️ AND THE GRANT IS DELAYED. Each wisp's RP comes from its Die DESTRUCTOR
--    (Fireflies.lua:533-544) which Sleep(100000)s FIRST — 100,000 game-ms is
--    3.33 game hours. Read AFTER at least 4 game hours, not immediately.
function CP15.RPRead(when)
	if not CP15.Guard("RPRead") then return false end
	local colony = rawget(_G, "UIColony")
	local q      = colony and rawget(colony, "research_queue") or empty_table
	local tech   = q[1]
	local st     = tech and (colony.tech_status or empty_table)[tech]
	local ok_c, cost = pcall(function() return colony:TechCost(tech) end)
	local g = rawget(_G, "g_Consts") or empty_table
	local ok_o, omega = pcall(function() return OmegaTelescopeResearchBoostPercent(colony) end)
	say(fmt("RP %s :: queue[1]=%s | status.points=%s (raw, scaled by const.ResearchPointsScale=%s) | TechCost=%s (ok=%s) | queue length=%d",
		tostring(when), tostring(tech), tostring(st and st.points),
		tostring(const.ResearchPointsScale), tostring(cost), tostring(ok_c), #q))
	say(fmt("RP %s :: multipliers — BreakthroughResearchSpeedMod=%s ExperimentalResearchSpeedMod=%s IndependenceLawResearchSpeedMod=%s OmegaTelescopeResearchBoostPercent=%s (ok=%s). Research.lua:722-737 applies these to EVERY grant; a raw 100/wisp only appears if all are identity.",
		tostring(when), tostring(g.BreakthroughResearchSpeedMod),
		tostring(g.ExperimentalResearchSpeedMod), tostring(g.IndependenceLawResearchSpeedMod),
		tostring(omega), tostring(ok_o)))
	if not tech then
		say("RP ⛔ NO TECH IN THE QUEUE. AddResearchPoints returns early when it can find no researchable tech (Research.lua:816-819) — a grant taken in this state is LOST and unmeasurable. Queue a tech before running the F15 rider.")
	end
	CP15.Applicable("F15-RP-instrument(a queued tech exists to receive the points)",
		tech and 1 or 0, tech and 1 or 0, "UIColony.research_queue read this load")
	return tech, st and st.points
end

-- ⛔⛔ THE ONE FORCED-MODE PRIMITIVE, and it is deliberately awkward to call.
--    Chain README rule 12: the mystery is PLAYED, not forced. Calling this for
--    "free" FORFEITS the tested grant for PT-15 — the reading becomes a
--    measurement of the patched function, not of the shipped player route.
--    F15's destroy rider has no organic route on a save that chose "free", so
--    it uses this knowingly and says so in its own label.
function CP15.ForceMode(mode, confirm)
	if not CP15.Guard("ForceMode") then return false end
	if confirm ~= "I-KNOW-THIS-IS-FORCED" then
		say("FORCEMODE ⛔ REFUSED. Call CP15.ForceMode(mode, \"I-KNOW-THIS-IS-FORCED\"). This exists so a forced mystery payload can never be typed by reflex.")
		return false
	end
	CP15.Label("FORCED", "SetLightTrapMode(" .. tostring(mode) .. ")",
		"the mystery's own payload function called directly, bypassing the SA_WaitChoice the player would answer (Mystery 11.generated.lua:429-441/470-471)")
	local ok = CP15.Try("SetLightTrapMode(" .. tostring(mode) .. ") [FIRST EXECUTION]", function()
		SetLightTrapMode(mode)
		return (rawget(_G, "UIColony") or empty_table).mystery
			and UIColony.mystery.lighttrap_mode
	end)
	return ok
end

-- The full F15 rider: read, force destroy, wait out the destructor sleep, read.
-- ⚠️ ONE SHOT. After the mode is "destroy", SinkholeBase:BuildingUpdate stops
--    spawning wisps entirely (Fireflies.lua:248-250, `lighttrap_mode ~=
--    "destroy"` gates the spawn) — there is no second trapful to retry on.
function CP15.F15Destroy()
	if not CP15.Guard("F15Destroy") then return false end
	local wisps = select(1, CP15.TrapRead("F15 before destroy"))
	if not wisps or wisps == 0 then
		say("F15 ⛔ NOT RUN — no trapped wisps. The batch-destroy branch this rider measures (Fireflies.lua:676-688) grants reward = wisps x 100; with zero wisps there is nothing to measure and nothing to compare the notification against.")
		return false
	end
	local tech, before = CP15.RPRead("F15 before destroy")
	say(fmt("F15 :: expectation — notification says %d RP (wisps x 100, AddUniqueNotification 'Mystery11WispsKilled'); the SHIPPED FIX drops the batch AddResearchPoints so the Die destructors are the single payer. Granted should EQUAL displayed. The vanilla defect paid 200/wisp.", wisps * 100))
	CP15.ForceMode("destroy", "I-KNOW-THIS-IS-FORCED")
	say("F15 :: ⏳ each wisp's Die destructor Sleep(100000)s = 3.33 GAME hours before its +100 lands (Fireflies.lua:536-542). Let at least 4 game hours pass at whatever speed is set, THEN call CP15.F15After().")
	CP15.f15_before = before
	CP15.f15_tech   = tech
	CP15.f15_wisps  = wisps
	return true
end

function CP15.F15After()
	if not CP15.Guard("F15After") then return false end
	local tech, after = CP15.RPRead("F15 after destroy")
	CP15.TrapRead("F15 after destroy")
	local before = CP15.f15_before
	if type(before) ~= "number" or type(after) ~= "number" then
		say("F15 ⛔ NOT COMPARABLE — a points reading is missing or non-numeric. Report the two raw readings; do not compute a delta from them.")
		return false
	end
	if tech ~= CP15.f15_tech then
		say(fmt("F15 ⚠️ THE QUEUED TECH CHANGED (%s -> %s). A tech completing mid-measurement re-enters AddResearchPoints with the remainder (Research.lua:845-846), so the delta below spans two counters and is NOT a clean grant figure.",
			tostring(CP15.f15_tech), tostring(tech)))
	end
	local scale = const.ResearchPointsScale or 1000
	local delta = after - before
	say(fmt("%sF15 :: status.points %d -> %d, delta=%d raw = %d RP at scale %d | wisps destroyed this batch=%s | notification claimed %s RP | ⇒ granted/displayed ratio = %s",
		CP15.Void(), before, after, delta, delta / scale, scale,
		tostring(CP15.f15_wisps), tostring((CP15.f15_wisps or 0) * 100),
		((CP15.f15_wisps or 0) > 0) and fmt("%d/%d", delta / scale, CP15.f15_wisps * 100) or "n/a"))
	say("F15 :: ⚠️ the ratio is only 1:1 when every multiplier read by CP15.RPRead was identity. Quote both RP lines and the multipliers together, never the ratio alone.")
	return delta
end

---- C39 — Service Automation, observed ---------------------------------------
--
-- ⭐⭐ THE SHIFT-BOUNDARY QUESTION IS ANSWERED FROM Src, AND IT SPLITS IN TWO.
--
--  ✅ `performance` RECOMPUTES IMMEDIATELY — no shift change needed. The law's
--     only effect is LawEffectModifyLabel{ServiceBuildings, max_workers, -50%}
--     (Data\LawDef\LawDef-Technology.lua:227-234). OnStart calls
--     container:SetLabelModifier (ClassDef-Factions.generated.lua:2039-2059),
--     which drives each building's Workplace:OnModifiableValueChanged("max_
--     workers", ...) — and that handler ends with SetWorkshift(current_shift),
--     CheckWorkForUnemployed(), UpdatePerformance() and RebuildInfopanel(self)
--     (Lua\Buildings\Workplace.lua:376-409). UpdatePerformance (:241-250) is
--     SetBase("performance", GetWorkshiftPerformance(active or current shift)).
--     ⇒ read the infopanel the moment the law lands.
--
--  ⛔ THE COMFORT THE SHIFT PAYS IS ONLY OBSERVABLE AT A SHIFT BOUNDARY. It is
--     paid inside Workshop:OnChangeWorkshift, to the workers of the OUTGOING
--     shift, and only `if working`: comfort_boost = Min(MulDivRound(g_Consts.
--     WorkInWorkshopComfortBoost, self.performance, 100), 100*const.Scale.Stat)
--     (Lua\Buildings\ArtWorkshop.lua:24-27 — the entry's citation is EXACT).
--     ⇒ the ARITHMETIC is available instantly (this leg prints it); the PAYMENT
--       needs a shift change with the law active.
--
--  ⚠️ AND ENACTING FIRES WORKERS. old_value > new_value walks the shift and
--     FireWorker()s the overflow (Workplace.lua:378-390). A previously FULL
--     shift therefore stays full at the reduced max — which is exactly the
--     "full reduced shift" the entry's ~200 arithmetic assumes. ⇒ FILL THE
--     WORKSHOP'S ACTIVE SHIFT BEFORE ENACTING, or the reading measures a
--     half-staffed building and settles nothing. On revert the fired workers do
--     NOT snap back; CheckWorkForUnemployed re-hires over time, so the AFTER-
--     REVERT reading is not automatically the BEFORE reading.
local C39_LAW = "Policy_Automation_ServiceAutomation"

local function c39_read_one(tag, when, b)
	if not b then
		say(fmt("C39 %s %s :: NOT-APPLICABLE — no subject selected. Run CP15.C39Scan() then CP15.C39Set(i, j).", tostring(tag), tostring(when)))
		return nil
	end
	local ok_s, is_service = pcall(function() return b:IsKindOf("Service") end)
	local ok_w, is_wshop   = pcall(function() return b:IsKindOf("Workshop") end)
	local shifts = {}
	for i = 1, (rawget(b, "max_shifts") or 3) do
		local n = #((rawget(b, "workers") or empty_table)[i] or empty_table)
		local ok_p, p = pcall(function() return b:GetWorkshiftPerformance(i) end)
		shifts[#shifts + 1] = fmt("s%d{workers=%d perf=%s(ok=%s) closed=%s}",
			i, n, tostring(p), tostring(ok_p),
			tostring((rawget(b, "closed_shifts") or empty_table)[i]))
	end
	local mods = rawget(b, "modifications")
	local mw   = mods and mods.max_workers
	say(fmt("C39 %s %s :: %s | IsKindOf(Service)=%s (raw=%s ok=%s) IsKindOf(Workshop)=%s (raw=%s ok=%s) | base_max_workers=%s max_workers=%s | current_shift=%s active_shift=%s | performance=%s | working=%s | %s",
		tostring(tag), tostring(when), idof(b),
		tostring(not not is_service), tostring(is_service), tostring(ok_s),
		tostring(not not is_wshop), tostring(is_wshop), tostring(ok_w),
		tostring(rawget(b, "base_max_workers")), tostring(rawget(b, "max_workers")),
		tostring(rawget(b, "current_shift")), tostring(rawget(b, "active_shift")),
		tostring(rawget(b, "performance")), tostring(rawget(b, "working")),
		table.concat(shifts, " ")))
	if mw then
		say(fmt("C39 %s %s :: modifications.max_workers{amount=%s percent=%s min=%s max=%s} — this is where the law's -50%% lands",
			tostring(tag), tostring(when), tostring(mw.amount), tostring(mw.percent),
			tostring(mw.min), tostring(mw.max)))
	else
		say(fmt("C39 %s %s :: modifications.max_workers = none", tostring(tag), tostring(when)))
	end
	-- the Comfort arithmetic, printed for BOTH subjects so the KIND difference
	-- is visible: for the Workshop this figure is the player-visible payload;
	-- for the Service control `performance` scales service throughput instead.
	local g     = rawget(_G, "g_Consts") or empty_table
	local boost = g.WorkInWorkshopComfortBoost
	local stat  = (rawget(_G, "const") or empty_table).Scale
	stat = stat and stat.Stat
	local perf  = rawget(b, "performance")
	local comfort
	if type(boost) == "number" and type(perf) == "number" and type(stat) == "number" then
		comfort = Min(MulDivRound(boost, perf, 100), 100 * stat)
	end
	say(fmt("C39 %s %s :: comfort-if-this-were-a-Workshop = Min(MulDivRound(%s, performance=%s, 100), 100*%s) = %s  [ArtWorkshop.lua:24-27; PAID at the shift boundary to the OUTGOING shift, only if working]",
		tostring(tag), tostring(when), tostring(boost), tostring(perf),
		tostring(stat), tostring(comfort)))
	return { perf = perf, max_workers = rawget(b, "max_workers"), comfort = comfort }
end

			tostring(rawget(b, "max_workers")), tostring(rawget(b, "performance"))))

	say(fmt("%sC39SCAN :: %d building(s) on label 'ServiceBuildings' — %d ARE Service (reach the compensating branch, Workplace.lua:205-217), %d are NOT (take the max_workers cut and cannot be compensated). function CP15.C39Set(i, j)
	if not CP15.Guard("C39Set") then return false end
	local l = CP15.c39_list or empty_table
	CP15.c39_subject, CP15.c39_control = l[i], l[j]
	say(fmt("C39SET :: subject=[%s] %s | control=[%s] %s",
		tostring(i), idof(CP15.c39_subject), tostring(j), idof(CP15.c39_control)))
	if not CP15.c39_subject or not CP15.c39_control then
		say("C39SET ⛔ one or both indices did not resolve. Re-run CP15.C39Scan() and use the indices it printed.")
		return false
	end
	local ok_s = select(2, pcall(function() return CP15.c39_subject:IsKindOf("Service") end))
	local ok_c = select(2, pcall(function() return CP15.c39_control:IsKindOf("Service") end))
	if ok_s then say("C39SET ⚠️ the SUBJECT reports IsKindOf('Service')=true — it is INSIDE the compensated class and is not a C39 subject.") end
	if not ok_c then say("C39SET ⚠️ the CONTROL reports IsKindOf('Service')=false — it is on the same side of the line as the subject and controls nothing.") end
	local ds = rawget(CP15.c39_subject, "parent_dome")
	local dc = rawget(CP15.c39_control, "parent_dome")
	if ds ~= dc then
		say(fmt("C39SET ⚠️ subject dome=%s control dome=%s — DIFFERENT domes. The entry asks for the same dome so dome-level modifiers cannot explain a difference. Record this if it cannot be fixed.", idof(ds), idof(dc)))
	end
	return true
end

-- ⛔ FIRST EXECUTION. LawDefs[id]:Activate() is the EXACT call a passed vote
--    executes (Legislature:EnactLaw -> LawDefs[law_id]:Activate, Lua\Factions\
--    Legislature.lua:512-553); Activate swaps out the policy's previous law,
--    applies effects and fires Msg("LawActivated") (ClassDef-Factions.generated
--    .lua:1778-1794). Neither it nor :Deactivate() (:1796-1813) is cheat-gated.
function CP15.C39Enact()
	if not CP15.Guard("C39Enact") then return false end
	CP15.Label("FORCED", "LawDefs." .. C39_LAW .. ":Activate()",
		"the enactment a passed Assembly vote would perform, called directly — the vote flow is skipped, the effect path is identical")
	local law = c39_law()
	if not law then
		say("C39 ⛔ LawDefs." .. C39_LAW .. " is nil. Nothing enacted.")
		return false
	end
	CP15.Try("LawDefs." .. C39_LAW .. ":Activate() [FIRST EXECUTION]", function()
		law:Activate()
		return (rawget(_G, "ActiveLaws") or empty_table)[C39_LAW] ~= nil
	end)
	-- liveness witness: the law must now be in ActiveLaws, read back
	local active = (rawget(_G, "ActiveLaws") or empty_table)[C39_LAW]
	say(fmt("%sC39 ENACT :: ActiveLaws.%s after the call = %s (truthy=%s) — this is the liveness witness, not the absence of an error",
		CP15.Void(), C39_LAW, tostring(active), tostring(not not active)))
	return not not active
end

	say("C39 AFTER :: ⚖️ THE SIGN IS THE WHOLE QUESTION. If the control's performance RISES toward ~200 while the subject's stays near 100, the four Workshops are MISSING AN UPLIFT. If the control holds ~100 while the subject FALLS, they are taking a PENALTY. Report the four raw numbers; do not name the sign from one of them.")
	say("C39 AFTER :: ⚠️ if the subject's worker count fell below its new max_workers, the shift is NOT full and the ~200 arithmetic does not apply — say so instead of comparing.")
	say("C39 AFTER :: the COMFORT PAYMENT itself is only observable across a shift change (Workshop:OnChangeWorkshift). If the sitting can spare one shift boundary with the law active, take a third reading after it.")
	CP15.c39_after_s, CP15.c39_after_c = s, c

end

function CP15.C39Revert()
	if not CP15.Guard("C39Revert") then return false end
	CP15.Label("FORCED", "LawDefs." .. C39_LAW .. ":Deactivate()",
		"reverts the enactment and restores the policy default (ClassDef-Factions.generated.lua:1796-1813)")
	local law = c39_law()

	CP15.Try("LawDefs." .. C39_LAW .. ":Deactivate() [FIRST EXECUTION]", function()
		law:Deactivate()
		return (rawget(_G, "ActiveLaws") or empty_table)[C39_LAW] == nil
	end)
	local active = (rawget(_G, "ActiveLaws") or empty_table)[C39_LAW]
	say(fmt("%sC39 REVERT :: ActiveLaws.%s after the call = %s (truthy=%s)",
		CP15.Void(), C39_LAW, tostring(active), tostring(not not active)))

	c39_read_one("control", "after-revert", CP15.c39_control)
	say("C39 REVERT :: ⚠️ workers fired by the enactment do NOT snap back. CheckWorkForUnemployed re-hires over game time, so an after-revert reading below the baseline is expected and is NOT evidence of a second defect.")
	return true
end

---- ⚖️ OPTION A — the speed ceiling, MEASURED ------------------------------
--
-- Owner-authorised 2026-08-11 as an OPTION, decided at the keyboard. Parked and
-- callable; nothing runs it automatically.
--
-- ⭐ WHY THIS EXISTS. "Ultra" is 20x by UI convention only. const.MaxTimeFactor
--    is 100,000,000 against const.DefaultTimeFactor 1000 — a 100,000x ceiling —
--    and UIColony:SetGameSpeed applies its argument through
--    SetTimeFactor(const.DefaultTimeFactor * factor) with NO Lua-side clamp
--    (Lua\Colony.lua:564-592). So the plan's 25-50 real minutes of scripted
--    sleep is a choice, not a limit.
--
-- ⛔⛔ AND WHY IT CANNOT JUST BE CRANKED — the failure mode is SILENT.
--    Buildings poll on a game-time thread that Sleep()s building_update_time
--    and then reads UIColony.hour (Building:StartUpdateThread,
--    Lua\Buildings\Building.lua:776-795). The mystery's own logic tests that
--    hour for EQUALITY: wisps spawn at `hour == 22`, go home at `hour == 4`
--    (Fireflies.lua:231/257), and a Light Trap's water draw switches at
--    `hour == 21` / `hour == 5` (:566-572). SinkholeBase polls every 5000
--    game-ms. Once ONE rendered frame advances more than 5000 game-ms, a poll
--    is skipped, an hour value is never observed, the equality never matches —
--    and no wisps spawn that night, with no error and no log line.
--
-- ⇒ THE INSTRUMENT IS THE SAME CADENCE AS THE CODE AT RISK. A game-time thread
--   Sleep(5000)ing and recording every hour it observes reproduces exactly the
--   poll that would fail. A gap in the observed hour sequence IS the failure.
--   The effective multiplier is measured alongside, because a factor the sim
--   cannot sustain buys nothing: game-ms advanced divided by real-ms elapsed.
--
-- ⚠️ WHEN TO RUN IT: during the countdown's dead window. Before the sinkhole
--    exists there is nothing hour-gated to damage, the 10-20 sols have to pass
--    anyway, and the ramp IS the march. Running it later risks eating a wisp
--    night for a measurement.
local function speed_probe_body(factor, real_seconds)
	local requested = tonumber(factor) or 20
	local budget    = (tonumber(real_seconds) or 20) * 1000
	local hours_per_day = (rawget(_G, "const") or empty_table).HoursPerDay or 24

	local ok_set = pcall(function() UIColony:SetGameSpeed(requested) end)
	Sleep(500)
	local applied = GetTimeFactor() / const.DefaultTimeFactor

	local seen, gaps, last = 0, {}, nil
	local g0, r0 = GameTime(), RealTime()
	CP15.speed_probe_running = true
	CreateGameTimeThread(function()
		while CP15.speed_probe_running do
			local h = rawget(_G, "UIColony") and UIColony.hour
			if type(h) == "number" and h ~= last then
				if last ~= nil then
					local step = (h - last) % hours_per_day
					if step > 1 then
						gaps[#gaps + 1] = fmt("%d->%d (skipped %d)", last, h, step - 1)
					end
				end
				seen = seen + 1
				last = h
			end
			Sleep(5000)   -- the EXACT cadence SinkholeBase polls at
		end
	end)
	Sleep(budget)
	CP15.speed_probe_running = false
	local dg, dr = GameTime() - g0, Max(1, RealTime() - r0)
	local eff100 = MulDivRound(dg, 100, dr)

	local verdict
	if seen == 0 then
		verdict = "NOT-APPLICABLE — no hour transition was observed in the budget; give it more real seconds or more speed"
	elseif #gaps > 0 then
		verdict = fmt("UNSAFE — %d skipped hour(s): %s. At this factor the mystery's hour==22 spawn can silently never fire.",
			#gaps, table.concat(gaps, ", "))
	else
		verdict = "SAFE at this factor — every hour in the window was observed by a 5000ms poller"
	end
	say(fmt("%sSPEED requested=%sx applied=%sx (SetGameSpeed ok=%s) | effective=%d.%02dx MEASURED (%d game-ms over %d real-ms) | hour transitions seen=%d | %s",
		CP15.Void(), tostring(requested), tostring(applied), tostring(ok_set),
		eff100 / 100, eff100 % 100, dg, dr, seen, verdict))
	return { requested = requested, applied = applied, eff100 = eff100,
		seen = seen, gaps = #gaps, safe = (seen > 0 and #gaps == 0) }
end

function CP15.SpeedProbe(factor, real_seconds)
	if not CP15.Guard("SpeedProbe") then return false end
	CP15.Label("FORCED", "game speed probe at " .. tostring(factor) .. "x",
		"speed is a harness accelerator and is disclosed per reading; it changes only how fast game time advances, never what the sim computes")
	CreateRealTimeThread(function() speed_probe_body(factor, real_seconds) end)
	say("SPEED probe started in a real-time thread; its line lands when the budget expires. Do not stack probes.")
	return true
end

-- The ramp. Default rungs bracket the ~300x boundary the poll arithmetic
-- predicts (a 60fps frame at 300x advances ~5000 game-ms) — but the number that
-- matters is MEASURED, not predicted, which is the whole point of the leg.
function CP15.SpeedRamp(list, real_seconds)
	if not CP15.Guard("SpeedRamp") then return false end
	CP15.Label("FORCED", "game speed ramp",
		"a sequence of speed probes; the sitting's own march runs inside it")
	list = list or { 20, 50, 100, 200, 400 }
	real_seconds = real_seconds or 20
	CreateRealTimeThread(function()
		local rows, best = {}, nil
		for _, f in ipairs(list) do
			local r = speed_probe_body(f, real_seconds)
			rows[#rows + 1] = fmt("%sx->eff %d.%02dx %s", tostring(f),
				r.eff100 / 100, r.eff100 % 100, r.safe and "SAFE" or "UNSAFE")
			if r.safe then best = f end
		end
		say(fmt("%sSPEEDRAMP :: %s", CP15.Void(), table.concat(rows, " | ")))
		if best then
			say(fmt("%sSPEEDRAMP ⭐ HIGHEST FACTOR WITH ZERO SKIPPED HOURS = %sx. Use it for the march. ⚠️ This is one machine on one save at one moment — record it as a measurement with those conditions, not as a property of the game.",
				CP15.Void(), tostring(best)))
		else
			say("SPEEDRAMP ⛔ NO SAFE RUNG. Every factor tried skipped hours or saw no transition. Fall back to 20x and say so.")
		end
		local ok = pcall(function() UIColony:SetGameSpeed(best or 20) end)
		Sleep(500)
		say(fmt("SPEEDRAMP restored speed: SetGameSpeed(%s) ok=%s, read back %sx",
			tostring(best or 20), tostring(ok),
			tostring(GetTimeFactor() / const.DefaultTimeFactor)))
	end)
	return true
end

--  immediately — that one DOES bypass the night cycle. Not used, not wrapped.)
function CP15.MassWisps(confirm)
	if not CP15.Guard("MassWisps") then return false end
	if confirm ~= "OWNER-SAID-YES" then
		say("MASSWISPS ⛔ REFUSED. Call CP15.MassWisps(\"OWNER-SAID-YES\"). This is an owner decision taken at the keyboard, and it amends chain rule 12 — it may not be typed by reflex.")
		return false
	end
	local sinks = CP15.LabelAll("Sinkhole")
	if #sinks == 0 then
		say("MASSWISPS ⛔ NOT RUN — no Sinkhole exists yet. The mystery has not spawned one, so there is nothing to fill.")
		CP15.Applicable("PT15-masswisps-population(a sinkhole exists to fill)", 0, 0,
			"Sinkhole label walked across every city this load")
		return false
	end
	CP15.Label("FORCED", "MassFireflySpawn on every sinkhole",
		"⚠️ AMENDS chain rule 12 by explicit owner decision; wisp SUPPLY is manufactured, the catch and the trap arithmetic stay untouched")
	local colony = rawget(_G, "UIColony")
	local mystery = colony and colony.mystery
	say(fmt("MASSWISPS BEFORE :: fireflies_caught=%s | %d sinkhole(s)",
		tostring(mystery and mystery.fireflies_caught), #sinks))
	for i, s in ipairs(sinks) do
		say(fmt("MASSWISPS BEFORE :: #%d %s fireflies=%d max_firefly_number=%s",
			i, idof(s), #(rawget(s, "fireflies") or empty_table),
			tostring(rawget(s, "max_firefly_number"))))
	end
	for i, s in ipairs(sinks) do
		CP15.Try(fmt("MassFireflySpawn(sinkhole #%d) [FIRST EXECUTION]", i), function()
			MassFireflySpawn(s)
			return #(rawget(s, "fireflies") or empty_table)
		end)
	end
	for i, s in ipairs(sinks) do
		say(fmt("%sMASSWISPS AFTER :: #%d %s fireflies=%d max_firefly_number=%s",
			CP15.Void(), i, idof(s), #(rawget(s, "fireflies") or empty_table),
			tostring(rawget(s, "max_firefly_number"))))
	end
	say("MASSWISPS :: ⏳ the new wisps are DORMANT (SetCommand(false)). They fly on the NEXT night's hour==22 FirefliesStart — so this saves nights of accumulation, not the night cycle itself. Watch fireflies_caught climb from the following morning.")
	say("MASSWISPS :: ⛔ THE RECORD MUST CARRY THIS. Any PT-15 verdict taken on a trapful that this leg supplied says so in the same sentence, and a `tested` grant names it as forced supply.")
	return true
end
