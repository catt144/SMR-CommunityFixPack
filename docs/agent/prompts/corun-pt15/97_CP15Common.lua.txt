CP15 = rawget(_G, "CP15") or {}

CP15.tag         = CP15.tag or "CP15"

CP15.loads       = 0
CP15.mode        = CP15.mode or "REAL"   -- the payload's MODE banner overwrites

---- output ------------------------------------------------------------------
--
-- ModLog is the only print path proven to reach the log file on disk
-- (Mod.lua:109-132). Its emitter formats its single argument a second time
-- (lib.lua:164-174), so a literal '%' raises there — escape it. Flush per line
-- so an abandoned sitting still shows how far it got.
local function say(msg)
	msg = "[" .. tostring(CP15.tag) .. "] " .. tostring(msg)
	local modlog = rawget(_G, "ModLog")
	if modlog then
		modlog((msg:gsub("%%", "%%%%")))
	else
		local dbg = rawget(_G, "DebugPrint")
		if dbg then dbg(msg .. "\n") end
	end
	print(msg)
	local cp = rawget(_G, "ConsolePrint")
	if cp then pcall(cp, msg .. "\n") end   -- attended: the owner reads the console too
	local flush = rawget(_G, "FlushLogFile")
	if flush then pcall(flush) end
end
CP15.Say = say

local function fmt(...)
	local ok, s = pcall(string.format, ...)
	return ok and s or "<fmt error>"
end
CP15.Fmt = fmt

local function id_of(obj)
	if not obj then return "nil" end
	if type(obj) ~= "table" then return tostring(obj) end
	return tostring(obj.class or "?") .. "#" .. tostring(rawget(obj, "handle"))
end
CP15.IdOf = id_of

-- G3. The ONLY sanctioned call wrapper. Prints ok and the error text, always.
-- ⛔ pcall cannot survive a yield — use CP15.TryYield for anything that Sleeps.
function CP15.Try(what, fn, ...)
	local ok, a, b = pcall(fn, ...)
	say(fmt("TRY %s ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

function CP15.TryYield(what, fn, ...)
	local ok, a, b = sprocall(fn, ...)
	say(fmt("TRY %s (yielding) ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

-- G2. The condition-sampled rule, mechanised.
function CP15.Applicable(label, count, population, witness)
	local applicable = (population or 0) > 0
	say(fmt("READ %s = %s | population=%s | APPLICABLE=%s | witness=%s",
		tostring(label), tostring(count), tostring(population),
		tostring(applicable), tostring(witness or "-- NONE NAMED --")))
	return applicable
end

-- ⭐ chain README rule 11. EVERY reading declares which side of the line it is
--    on. A sitting mixes both freely, so the label travels with the reading
--    rather than sitting in a preamble the audit has to remember.
function CP15.Label(kind, what, detail)
	kind = tostring(kind):upper()
	if kind ~= "FORCED" and kind ~= "ORGANIC" then
		kind = "UNLABELLED(⚠️ neither FORCED nor ORGANIC was declared)"
	end
	say(fmt("%s%s :: %s :: %s", CP15.Void(), kind, tostring(what), tostring(detail or "-")))
	return kind
end

---- small helpers -----------------------------------------------------------

local function wait_for(timeout, ready)
	local deadline = RealTime() + timeout
	while not ready() do
		if RealTime() - deadline >= 0 then return false end
		Sleep(500)
	end
	return true
end
CP15.WaitFor = wait_for

local function game_is_live()
	local city   = rawget(_G, "MainCity")
	local colony = rawget(_G, "UIColony")
	return not not (city and city.labels and colony and colony.labels)
end
CP15.GameIsLive = game_is_live

local function label_all(name)
	local out = {}
	for _, city in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, o in ipairs((city.labels or empty_table)[name] or empty_table) do
			out[#out + 1] = o
		end
	end
	return out
end
CP15.LabelAll = label_all

-- Game time, printed raw beside every derived figure (prompt 1, Job 1 rule).
-- 30,000 game-ms = 1 game hour is MEASURED (PLAYTEST_HELP); a sol is 24 of them.
CP15.MS_PER_HOUR = 30000
CP15.MS_PER_SOL  = 24 * 30000

function CP15.Sols(ms)
	if type(ms) ~= "number" then return "n/a" end
	return fmt("%d.%02d sols", ms / CP15.MS_PER_SOL,
		((ms % CP15.MS_PER_SOL) * 100) / CP15.MS_PER_SOL)
end

function CP15.Now(when)
	local colony = rawget(_G, "UIColony")
	local gt = GameTime()
	say(fmt("CLOCK %s :: GameTime()=%s (%s since game start) | UIColony.day=%s hour=%s | speed=%sx | RealTime()=%s",
		tostring(when), tostring(gt), CP15.Sols(gt),
		tostring(colony and colony.day), tostring(colony and colony.hour),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), tostring(RealTime())))
	return gt
end

---- load / save -------------------------------------------------------------

function CP15.Load(savefile, speed)
	local err = LoadGame(savefile, {})            -- ⛔ never pass save_as_last
	if err then
		say("LOADERR " .. tostring(savefile) .. " " .. tostring(err))
		return false
	end
	CP15.loads = CP15.loads + 1
	say(fmt("LOAD OK #%d file=%s — LoadGame returned no error", CP15.loads, tostring(savefile)))
	if not wait_for(5 * 60 * 1000, game_is_live) then
		say("LOADERR " .. tostring(savefile) .. " — city never came live")
		return false
	end
	Sleep(CP15.settle_time)                        -- 15 s is the MEASURED datum
	speed = speed or 3
	local before = GetTimeFactor() / const.DefaultTimeFactor
	local ok = pcall(function() UIColony:SetGameSpeed(speed) end)
	Sleep(500)
	local after = GetTimeFactor() / const.DefaultTimeFactor
	say(fmt("READ speed: SetGameSpeed(%d) call=%s | before=%s after=%s",
		speed, tostring(ok), tostring(before), tostring(after)))
	return true
end

-- ⛔ save_as_last is NEVER passed — it would repoint the owner's Continue button.
-- ⛔ G4: the witness is the on-disk file (read agent-side, game closed) plus the
--    LoadGame that follows in this very run. ListForTag is printed and is NOT
--    the witness (EF-049).
-- ⛔⛔ `savename` IS WRITTEN VERBATIM — the engine appends NOTHING (EF-050,
--    MEASURED 2026-08-11: SaveGame("U2RT1", {savename = "U2RT1"}) returned
--    err=false and produced a file literally called `U2RT1`, with no extension,
--    which LoadGame("U2RT1.savegame.sav") then could not find). Resurrecting a
--    primitive means resurrecting its ARGUMENT CONVENTION, not just its body.
local SAVE_EXT = ".savegame.sav"

function CP15.Save(name)
	name = tostring(name)
	if name:sub(-#SAVE_EXT) ~= SAVE_EXT then
		say(fmt("SAVE ⚠️ name '%s' carries no %s suffix — the engine writes savename VERBATIM, so this would produce an extensionless file no LoadGame could find. Appending it.", name, SAVE_EXT))
		name = name .. SAVE_EXT
	end
	local landed = false
	CP15.TryYield("SaveGame " .. tostring(name), function()
		local err, nm = SaveGame(name, { savename = name, silent = true, no_screenshot = true })
		say(fmt("SAVE SaveGame returned err=%s name=%s", tostring(err), tostring(nm)))
		landed = not err
		return err
	end)
	Sleep(1000)
	local sg = rawget(_G, "Savegame")
	if sg then
		CP15.Try("Savegame.ListForTag (EF-049 open question, NOT the witness)", function()
			local list = sg.ListForTag("savegame")
			return type(list) == "table" and #list or ("non-table:" .. type(list))
		end)
	end
	say("SAVE WITNESS for " .. tostring(name)
		.. " = the on-disk file read agent-side after this process exits, plus any LoadGame of it. NOT ListForTag (EF-049).")
	return landed
end

---- run conditions ----------------------------------------------------------
--
-- WORKFLOW EXTERNAL VALIDITY rule: the pack count is READ, never assumed.
function CP15.ReadConditions(note, named)
	local pack = rawget(_G, "SMRFixPack")
	local active, total, inactive = 0, 0, {}
	if pack and pack.order then
		for _, id in ipairs(pack.order) do
			total = total + 1
			local f = pack.fixes and pack.fixes[id]
			local st = f and f.status or "missing"
			if st == "active" then active = active + 1 else inactive[#inactive + 1] = id .. "=" .. st end
		end
	end
	say(fmt("CONDITIONS pack=%d/%d active | map=%s | speed=%s | loads_this_process=%d | mode=%s | note=%s",
		active, total, tostring(rawget(_G, "CurrentMapName")),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), CP15.loads,
		tostring(CP15.mode), tostring(note or "-")))
	if #inactive > 0 then
		say("CONDITIONS not-active: " .. table.concat(inactive, ", "))
	end
	for _, id in ipairs(named or empty_table) do
		local f = pack and pack.fixes and pack.fixes[id]
		say(fmt("CONDITIONS by-name %s = %s | detail=%s", tostring(id),
			tostring(f and f.status or "NOT REGISTERED"), tostring(f and f.detail or "-")))
	end
	return active, total
end

---- ⛔ THE RUN-CONDITION GATE (WORKFLOW rule 7 as amended 2026-08-11) --------
--
-- unattended-2 run 1 read `pack=0/0 active` and carried on for six more steps,
-- producing readings that could not mean anything: the pack was DISABLED in the
-- Mod Manager, left that way by corun-batch-2's leg T uninstall test. The
-- amended rule puts the gate at every run top and requires it to STOP.
--
-- ⛔ A SITTING HAS NO SINGLE RUN TOP. So the gate lives at the top of every leg
--    (CP15.Guard) and the standalone CP15.Gate() is the sitting's opening line.
--    The DEFAULT IS TO STOP. `CP15.rehearsal` exists only for a declared
--    HARNESS REHEARSAL, and when it is set every verdict is void by
--    construction and CP15.Void() stamps that on each line.
function CP15.RequirePackLoaded(note)
	local pack = rawget(_G, "SMRFixPack")
	local active, total = 0, 0
	if pack and pack.order then
		for _, id in ipairs(pack.order) do
			total = total + 1
			local f = pack.fixes and pack.fixes[id]
			if f and f.status == "active" then active = active + 1 end
		end
	end
	CP15.pack_loaded = active > 0
	if CP15.pack_loaded then
		say(fmt("GATE pack-loaded OK :: %d/%d modules active :: %s", active, total, tostring(note)))
		return true
	end
	say(fmt("GATE ⛔ PACK NOT LOADED :: %d/%d active. The Community Fix Pack mod is present on disk but not enabled in the Mod Manager. Every reading about F07, F15, C39 or F85's save behaviour is VOID in this state — the pack's code never ran. :: %s",
		active, total, tostring(note)))
	say("GATE ⛔ THIS IS NOT SCRIPTABLE. AccountStorage, SaveAccountStorage and ModsReloadItems are all ModEnvBlacklist keys (Mod.lua), and there is no console at the main menu — so no mod-side or console-side path can tick the box. It is one human click in the Mod Manager, followed by a FULL RESTART (D13). ⚠️ THE RE-ENABLE IS HANDED BACK TO THE OWNER, EXPLICITLY.")
	if CP15.rehearsal then
		say("GATE ⚠️ CONTINUING ANYWAY: this sitting is a declared HARNESS REHEARSAL. It exists to prove the legs run, and NOTHING it prints is a result. Every verdict below is VOID.")
		return true
	end
	return false
end

-- Prefix for any line that would otherwise read like a verdict.
function CP15.Void(label)
	if CP15.pack_loaded then return "" end
	return "VOID(pack not loaded) "
end

-- ⛔ THE PER-LEG GATE. Every leg's first line. Returns false ⇒ the leg must
--    return immediately without reading or mutating anything.
function CP15.Guard(leg)
	CP15.stage = tostring(leg)
	if CP15.RequirePackLoaded("leg " .. tostring(leg)) then
		return true
	end
	say(fmt("GATE ⛔ LEG '%s' NOT RUN. Nothing was read and nothing was changed. Fix the pack state and start the sitting over.", tostring(leg)))
	return false
end

---- error-watch marker ------------------------------------------------------

-- ⛔ It may NOT count engine error lines: the engine writes those straight to
--    the log file and the mod environment has no file read (io/os are both
--    ModEnvBlacklist keys). A count printed from here would be invented. It
--    prints the boundary a reader greps between.
-- ⚠️ The token an error sweep greps for is spelled AROUND, not written.
function CP15.ErrorWatchNote()
	say(fmt("ERRORWATCH %s — every engine error line (the bracketed LUA-ERROR marker) in this log between the '=== corun-pt15 %s ===' banner and this line happened inside this sitting. Report every unexplained line verbatim with its position and its age; never discount one. stage_at_close=%s loads_this_process=%d mode=%s",
		tostring(CP15.tag), tostring(CP15.tag), tostring(CP15.stage), CP15.loads, tostring(CP15.mode)))
end
