-- One line. Prints the fixture, picks the case-A track, snapshots it.
function CB1.Leg1Read()
	local n, st, tr, routed, healthy, damaged = CB1.ReadTrackFixture()
	local pick = nil
	-- Prefer a LONG healthy track: more elements = more of the rewrite exercised.
	for _, t in ipairs(healthy or empty_table) do
		if not pick or #(t.elements or empty_table) > #(pick.elements or empty_table) then pick = t end
	end
	CB1.caseA = pick
	if pick then
		say(fmt("LEG1 case-A track picked: %s with %d elements", id_of(pick), #(pick.elements or empty_table)))
		CB1.ReadWalkability(pick)
		CB1.SnapTrack(pick, "leg1-caseA-BEFORE")
	else
		say("LEG1 NO healthy multi-element track — case A is SKIP <no fixture>")
	end
	return pick
end

-- Case A. The CORRECTED call the repair would ship, on a healthy track.
-- ⚠️ F48's procedure also assigns start_el/end_el by hand afterwards. That is
--    REDUNDANT on the success path (ProcessTrackElements sets both itself,
--    Tracks.lua:820-822) and, worse, on the FAILURE path it would write
--    endpoints the shipped repair would never write — making case B look more
--    destructive than the repair actually is. We do NOT hand-assign; we run
--    exactly what would ship and read what it did.
function CB1.Leg1CaseA()
	local t = CB1.caseA
	if not t then say("LEG1A SKIP <no case-A track; run CB1.Leg1Read() first>") return end
	local before = CB1.SnapTrack(t, "leg1-caseA-BEFORE-call")
	say("LEG1A FORCED: the corrected savegame-fixup call, run by hand. MEASURED: whether the track survives it.")
	CB1.Try("ProcessTrackElements(ResolveMap(t), t.elements) [FIRST EXECUTION]", function()
		return ProcessTrackElements(ResolveMap(t), t.elements)
	end)
	local after = CB1.SnapTrack(t, "leg1-caseA-AFTER-call")
	say(fmt("LEG1A VERDICT signature %s", before == after and "UNCHANGED (stable no-op)" or "CHANGED — compare the two SNAP lines above"))
	return before, after
end

-- Stage case B's damage. FORCED, and it says so.
-- HELP's meteor row is binding: pos is the THIRD argument, always pass one.
-- A mid-track element is chosen deliberately — BreakTracks' own filter skips
-- both endpoints and any station element (Meteors.lua:601).
function CB1.Leg1Damage(track)
	track = track or CB1.caseA
	if not track then say("LEG1DMG SKIP <no track>") return end
	local list = track.elements or empty_table
	if #list < 5 then say("LEG1DMG SKIP <track too short to break a middle element safely>") return end
	local el = list[1 + DivRound(#list, 2)]
	if el == track.start_el or el == track.end_el or el.station then
		say("LEG1DMG SKIP <mid element is an endpoint or a station element; BreakTracks would skip it>")
		return
	end
	say(fmt("LEG1DMG FORCED: meteor at %s (mid element %s of %s). MEASURED: nothing yet — this only stages the fixture.",
		tostring(el:GetPos()), id_of(el), id_of(track)))
	CB1.Try("CheatMeteors single at the element position", function()
		return CheatMeteors("single", nil, el:GetPos())
	end)
	CB1.damaged = track
	say("LEG1DMG watch for the impact, then run CB1.Leg1CaseBCheck()")
	return el
end

-- The gate. Reads whether the damage actually made the track unwalkable.
function CB1.Leg1CaseBCheck()
	local t = CB1.damaged or CB1.caseA
	if not t then say("LEG1B SKIP <no damaged track>") return end
	local broken, sites = 0, #(t.elements_under_construction or empty_table)
	for _, el in ipairs(t.elements or empty_table) do if el.broken then broken = broken + 1 end end
	say(fmt("LEG1B damage state: %d broken element(s), %d repair site(s), %d repair_cgs",
		broken, sites, #(t.repair_cgs or empty_table)))
	local shadowed, missing = CB1.ReadWalkability(t)
	CB1.SnapTrack(t, "leg1-caseB-BEFORE")
	CB1.caseB_samples = ((shadowed or 0) > 0 or (missing or 0) > 0)
	if not CB1.caseB_samples then
		say("⛔ LEG1B THE DECIDER IS UNSAMPLED: the hex grid still hands the walk the track's own elements, so OrderTrackElements will SUCCEED and this is case A again. Running the call anyway would produce a 'clean pass' that proves nothing about the risk F48 is blocked on. Break another element, or record leg 1 as SKIP <case B could not be made unwalkable> and leave F48 blocked.")
	else
		say("LEG1B the walk is predicted to FAIL — this track DOES sample the decider. CB1.Leg1CaseB() is now allowed.")
	end
	return CB1.caseB_samples
end

-- Case B. ⛔ Refuses to run unless the gate above says the decider is sampled.
function CB1.Leg1CaseB()
	if not CB1.caseB_samples then
		say("⛔ LEG1B REFUSED — run CB1.Leg1CaseBCheck() first, and it must report a FAILING predicted walk. (G2: the liveness witness for 'the assert fired' is that reading, taken before the call.)")
		return
	end
	local t = CB1.damaged
	local before = CB1.SnapTrack(t, "leg1-caseB-BEFORE-call")
	say("LEG1B FORCED: meteor damage + the corrected fixup call. MEASURED: what the failure path leaves behind.")
	say("LEG1B ⛔ EXPECT an assert line 'unable to find the expected number of track elements' in the log. That is the DESIGNED path, not a crash — EF-008, it does not unwind.")
	CB1.Try("ProcessTrackElements on the DAMAGED track [FIRST EXECUTION]", function()
		return ProcessTrackElements(ResolveMap(t), t.elements)
	end)
	local after = CB1.SnapTrack(t, "leg1-caseB-AFTER-call")
	say(fmt("LEG1B VERDICT signature %s", before == after and "UNCHANGED" or "CHANGED — read duplicate_node_idx and connections_total in the two SNAP lines"))
	say("LEG1B now the owner's eyes: does the route still form, does the train still run, is the repair site still salvageable (F45 cursor)? Then save + reload and run CB1.Leg1Read() again.")
	return before, after
end

function CB1.PickDome(index)
	local domes = CB1.LabelAll("Dome")
	local main = rawget(_G, "MainMap")
	local on_main = {}
	for i, x in ipairs(domes) do
		local m = IsValid(x) and ResolveMap(x) or nil
		local is_main = (m ~= nil and m == main)
		if is_main then on_main[#on_main + 1] = x end
		say(fmt("DOME[%d] %s pos=%s on_MainMap=%s", i, id_of(x),
			tostring(IsValid(x) and x:GetPos() or "?"), tostring(is_main)))
	end
	local d = index and domes[index] or on_main[1] or domes[1]
	say(fmt("DOME %d of %d dome(s) are on MainMap — WaitBombard hard-codes MainMap, so only these are usable",
		#on_main, #domes))
	say(fmt("PickDome -> %s (pass an index to choose another; put the camera on it and keep it LOW)", id_of(d)))
	CB1.dome = d
	return d
end

-- One line at the moment the owner is watching. count/radius are the entry's.
function CB1.Leg2Bombard(dome, count, radius)
	dome = dome or CB1.dome or CB1.LabelAll("Dome")[1]
	if not IsValid(dome) then say("LEG2 SKIP <no valid dome to centre the volley on>") return end
	count = count or 8
	radius = radius or 40 * guim
	-- ⛔ CORRECTED 2026-08-04 (see the PickDome header): the map is MainMap by
	--    construction, because that is what WaitBombard uses. Reading any other
	--    map's g_IncomingMissiles would measure a table nothing writes to.
	local map = rawget(_G, "MainMap")
	local dome_map = ResolveMap(dome)
	if dome_map ~= map then
		say(fmt("⛔ LEG2 REFUSED — dome %s resolves to map %s but WaitBombard hard-codes MainMap (Bombardment.lua:55 / Fix_BombardmentSpread.lua:91). The volley would land somewhere the camera is not, and the watcher would poll a table nothing writes to. Pick a dome whose PickDome line reads on_MainMap=true.",
			id_of(dome), tostring(dome_map)))
		return
	end
	local speed = GetTimeFactor() / const.DefaultTimeFactor
	if speed == 0 then
		say("⛔ LEG2 REFUSED — the game is PAUSED and StartBombard only spawns a game-time thread; nothing would happen. Unpause and call again.")
		return
	end
	local before = 0
	for _ in pairs(map.g_IncomingMissiles or empty_table) do before = before + 1 end
	say(fmt("LEG2 FORCED: StartBombard(<dome %s>, %d, %d, 500, 1500) — a volley that would not otherwise happen. MEASURED: the SPREAD (owner's eyes) and the five integrity readings. speed=%s g_IncomingMissiles_before=%d",
		id_of(dome), radius, count, tostring(speed), before))
	say("LEG2 ⛔ FIRST EXECUTION of StartBombard by this project. The entry's own recipe was measured BROKEN in prep; this is the corrected form.")
	CB1.bombard_seen = false
	CB1.Try("StartBombard [FIRST EXECUTION]", function()
		return StartBombard(dome, radius, count, 500, 1500)
	end)
	-- G2 LIVENESS WITNESS for "the volley ENDED": an empty missile table also
	-- describes a volley that never started, so we watch it become NON-EMPTY
	-- first and record that as the witness.
	CreateRealTimeThread(function()
		local deadline = RealTime() + 3 * 60 * 1000
		local peak = 0
		while RealTime() < deadline do
			local n = 0
			for _ in pairs(map.g_IncomingMissiles or empty_table) do n = n + 1 end
			if n > peak then peak = n end
			if peak > 0 then CB1.bombard_seen = true end
			if peak > 0 and n == 0 then
				say(fmt("LEG2 VOLLEY ENDED — peak in-flight missiles = %d (WITNESS: the table was non-empty, so this zero is a real end, not a volley that never started)", peak))
				CB1.ReadBombardResidue(map)
				return
			end
			Sleep(1000)
		end
		say(fmt("⛔ LEG2 3-MINUTE BOUND HIT — peak in-flight missiles = %d, still %s. If peak is 0 the volley NEVER STARTED (read the TRY line above for the raise); if peak > 0 the volley did not end and F26's fix should be reverted per the entry.",
			peak, peak > 0 and "in flight" or "nothing ever spawned"))
	end)
	say("LEG2 watching for up to 3 minutes; the END line prints itself. Owner: eyes on the sky NOW — scatter (different incoming angles) or a rank of parallel trails?")
end

-- The agent-side half of the five integrity checks.
function CB1.ReadBombardResidue(map)
	map = map or MainMap
	local decals = 0
	local each = rawget(_G, "MapGet")
	local ok, res = pcall(function() return map:MapGet("map", "DecRocketSplatter") end)
	if ok and type(res) == "table" then decals = #res end
	local cracked, domes = 0, CB1.LabelAll("Dome")
	for _, d in ipairs(domes) do
		local n = d.crack_visuals or d.cracks or d.dome_cracks
		if type(n) == "table" and #n > 0 then cracked = cracked + 1
		elseif type(n) == "number" and n > 0 then cracked = cracked + 1 end
	end
	say(fmt("LEG2 residue: scorch decals on map=%s (read ok=%s) | %d of %d dome(s) reading as cracked",
		tostring(decals), tostring(ok), cracked, #domes))
	CB1.Applicable("PT47-decals", decals, decals, "DecRocketSplatter queried on the map after the volley")
	CB1.Applicable("PT47-dome-crack", cracked, #domes, "Dome label walked after the volley")
	-- ⛔ interception: with no defence tower on this save it is UNSAMPLED, and
	-- prep measured exactly that (0 defence towers on CB1STAGE).
	local towers = CB1.LabelAll("DefenceTowerBase")
	CB1.Applicable("PT47-interception", 0, #towers,
		"DefenceTowerBase label — 0 towers means UNSAMPLED, not 'no interception happened'")
end

function CB1.Leg4Pick(index)
	local free_senior_by_dome, free_senior_total = {}, 0
	for _, r in ipairs(CB1.LabelAll("Residence")) do
		local trait = r.exclusive_trait
		if trait == "Senior" then
			local cap  = tonumber(r.capacity) or 0
			local used = type(r.colonists) == "table" and #r.colonists or 0
			local free = cap - used
			if free > 0 then
				local d = r.parent_dome
				free_senior_total = free_senior_total + free
				if d then free_senior_by_dome[d] = (free_senior_by_dome[d] or 0) + free end
			end
		end
	end
	local subjects, in_dome_with_slot = {}, 0
	local each = rawget(_G, "AllMapsForEach")
	if type(each) == "function" then
		each(true, "Colonist", function(c)
			if not IsValid(c) then return end
			local traits = c.traits
			if type(traits) ~= "table" or not traits.Senior then return end
			local res = c.residence
			if not res then return end
			local trait = res.exclusive_trait
			if trait and trait ~= "" and trait ~= false then return end
			local d = c.dome or res.parent_dome
			local free_here = free_senior_by_dome[d] or 0
			if free_here > 0 then in_dome_with_slot = in_dome_with_slot + 1 end
			subjects[#subjects + 1] = { c = c, dome = d, free_here = free_here, res = res }
		end)
	end
	-- Strongest sample first: the module has an in-dome slot it wants them in.
	table.sort(subjects, function(a, b) return a.free_here > b.free_here end)
	for i, s in ipairs(subjects) do
		if i > 5 then break end
		say(fmt("LEG4PICK candidate[%d] %s residence=%s dome=%s free_SENIOR_slots_in_THAT_dome=%d",
			i, id_of(s.c), id_of(s.res), id_of(s.dome), s.free_here))
	end
	say(fmt("LEG4PICK %d Senior(s) live in a NORMAL residence | %d of them are in a dome that HAS a free Senior slot | %d free Senior slot(s) colony-wide",
		#subjects, in_dome_with_slot, free_senior_total))
	-- Prediction 10's falsifier, stated as its own reading.
	CB1.Applicable("D07-trigger-E-in-dome-pass(prediction 10)", in_dome_with_slot, #subjects,
		"every Senior in a normal residence walked against free Senior slots IN THEIR OWN DOME")

	else
		say("LEG4PICK prediction 10 HOLDS so far — no Senior in a normal residence has a free Senior slot in their own dome, so reachability is the gate and the module is behaving on the in-dome pass.")
	end
	local s = subjects[index or 1]

	end

	end
	CB1.Try("ViewAndSelectObject(subject)", function() return ViewAndSelectObject(s.c) end)
	say(fmt("LEG4PICK ⇒ OWNER: %s is now SELECTED and on screen, living in %s (dome %s). Manually assign them a residence, then say 'assigned'.",

end

-- Popup trio. ⛔ The route documented in POPUP_CONSEQUENCE_AUDIT §8 item 3 —
-- "a launch-issue prompt is the cheapest" — was MEASURED NOT AVAILABLE on
-- CB1STAGE by prep: of 6 rockets, 5 report a launch issue but 4 of those are
-- `no_target`, which is NOT a key of RocketBase:GetLaunchIssuePopups
-- (RocketBase.lua:2210-2219: missions_suspended / not_landed / fuel /
-- maintenance / disabled / unloading), and the one `fuel` rocket is on Earth.
-- Worse, GetLaunchIssuePopups RAISED on every rocket queried (the pcall result
-- was printed, per G3 — a nil return and a raise print identically otherwise).
-- ⇒ The route below is the Src-verified alternative, and it has the virtue of
--    being one of F85's OWN four sites rather than a stand-in:
--    ShowBreakthroughChoicePopup (Anomaly.lua:696-714), preset
--    "SubsurfaceAnomalyBreakthrough" (Anomaly.lua:393).
--    ⚠️ Precondition: UIColony:UpdateAvailableBreackthroughTechs() must be
--    non-empty. This prints the count first and refuses if it is zero, rather
--    than opening nothing and calling it a refusal.
--    ⚠️ Answering it DISCOVERS a breakthrough. Staged copy only.
function CB1.PopupOpen()
	local ok, techs = pcall(function() return UIColony:UpdateAvailableBreackthroughTechs() end)
	local n = (ok and type(techs) == "table") and #techs or -1
	say(fmt("POPUP available breakthrough techs = %s (read ok=%s)", tostring(n), tostring(ok)))
	if n <= 0 then
		say("⛔ POPUP REFUSED — no undiscovered breakthroughs left on this save, so ShowBreakthroughChoicePopup returns immediately (Anomaly.lua:698). The popup trio is SKIP <no choice popup reachable> unless another route is found live.")
		return false
	end
	say("POPUP FORCED: opening a breakthrough choice popup. MEASURED: whether a rebound Quick Save can land inside its window (F85's U tier).")
	CB1.Try("ShowBreakthroughChoicePopup [FIRST EXECUTION]", function()
		return ShowBreakthroughChoicePopup("SubsurfaceAnomalyBreakthrough", techs)
	end)
	say("POPUP owner's HANDS: with the popup open, press the rebound Quick Save key (F9). Did a save land? Then answer the popup, load that save, and check whether the popup — and its breakthrough — are gone.")
	return true
end
