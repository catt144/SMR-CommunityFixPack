---- output ------------------------------------------------------------------

-- ModLog is the only print path proven to reach the log file on disk
-- (Mod.lua:109-132). Its emitter formats its single argument a second time
-- (lib.lua:164-174), so a literal '%' raises there — escape it. Flush per line
-- so a killed run still shows how far it got.
local function say(msg)
	msg = "[" .. tostring(CB1.tag) .. "] " .. tostring(msg)
	local modlog = rawget(_G, "ModLog")
	if modlog then
		modlog((msg:gsub("%%", "%%%%")))
	else
		local dbg = rawget(_G, "DebugPrint")
		if dbg then dbg(msg .. "\n") end
	end
	print(msg)
	local flush = rawget(_G, "FlushLogFile")
	if flush then pcall(flush) end
end
CB1.Say = say

local function fmt(...)
	local ok, s = pcall(string.format, ...)
	return ok and s or "<fmt error>"
end
CB1.Fmt = fmt

local function id_of(obj)
	if not obj then return "nil" end
	if type(obj) ~= "table" then return tostring(obj) end
	return tostring(obj.class or "?") .. "#" .. tostring(rawget(obj, "handle"))
end
CB1.IdOf = id_of

-- G3. The ONLY sanctioned call wrapper. Prints ok and the error text, always.
-- ⛔ pcall cannot survive a yield — use CB1.TryYield for anything that Sleeps.
function CB1.Try(what, fn, ...)
	local ok, a, b = pcall(fn, ...)
	say(fmt("TRY %s ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

-- Same contract for steps that yield. sprocall is the engine's yield-safe pcall.
function CB1.TryYield(what, fn, ...)
	local ok, a, b = sprocall(fn, ...)
	say(fmt("TRY %s (yielding) ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

-- G2. The condition-sampled rule, mechanised. A count is only a result if the
-- population it was counted over is non-empty; otherwise it is UNSAMPLED.
function CB1.Applicable(label, count, population, witness)
	local applicable = (population or 0) > 0
	say(fmt("READ %s = %s | population=%s | APPLICABLE=%s | witness=%s",
		tostring(label), tostring(count), tostring(population),
		tostring(applicable), tostring(witness or "-- NONE NAMED --")))
	return applicable
end

---- flow helpers ------------------------------------------------------------

local function shutdown(reason)
	CB1.stage = "done"
	say("SHUTDOWN " .. tostring(reason))
	local flush = rawget(_G, "FlushLogFile")
	if flush then flush() end
	Sleep(CB1.quit_delay)
	quit()
end
CB1.Shutdown = shutdown

-- Steps Sleep, so sprocall is mandatory (pcall cannot survive a yield).
-- untimed = true suppresses the END timing line entirely (EF-045 correction C8:
-- do not even PRINT a RealTime() delta for such a step). Set for SAVE steps as
-- well as LOAD steps — SaveGame's first act is WaitRenderMode("ui")
-- (CommonLua/Savegame.lua:1079), the same no-frames block that made LoadGame
-- under-report itself 11.5x.
local function step(name, fn, untimed)
	CB1.stage = name
	local t = RealTime()
	say("STEP " .. name .. " BEGIN")
	local ok, res = sprocall(fn)
	if not ok then
		say("STEP " .. name .. " ERROR")
		shutdown("ERROR " .. name)
		return false
	end
	if untimed then
		say("STEP " .. name .. " END (no timing — this step blocks render mode; a load takes its number from the engine's own 'Game loaded on map ... in Nms' line, a save has none)")
	else
		say(fmt("STEP %s END %dms", name, RealTime() - t))
	end
	if res == false then
		shutdown("STOP " .. name .. " — the step asked to stop; its reason is the last line before this one")
		return false
	end
	return true
end
CB1.Step = step

local function wait_for(timeout, ready)
	local deadline = RealTime() + timeout
	while not ready() do
		if RealTime() - deadline >= 0 then return false end
		Sleep(500)
	end
	return true
end
CB1.WaitFor = wait_for

local function game_is_live()
	local city   = rawget(_G, "MainCity")
	local colony = rawget(_G, "UIColony")
	return not not (city and city.labels and colony and colony.labels)
end
CB1.GameIsLive = game_is_live

local function label_all(name)
	local out = {}
	for _, city in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, o in ipairs((city.labels or empty_table)[name] or empty_table) do
			out[#out + 1] = o
		end
	end
	return out
end
CB1.LabelAll = label_all

---- load / speed ------------------------------------------------------------

function CB1.Load(savefile, speed)
	local err = LoadGame(savefile, {})            -- ⛔ never pass save_as_last
	if err then
		say("LOADERR " .. tostring(savefile) .. " " .. tostring(err))
		return false
	end
	CB1.loads = CB1.loads + 1
	say(fmt("LOAD OK #%d file=%s — LoadGame returned no error", CB1.loads, tostring(savefile)))
	if not wait_for(5 * 60 * 1000, game_is_live) then
		say("LOADERR " .. tostring(savefile) .. " — city never came live")
		return false
	end
	Sleep(CB1.settle_time)                         -- 15 s is the MEASURED-sufficient datum
	speed = speed or 3
	local before = GetTimeFactor() / const.DefaultTimeFactor
	local ok = pcall(function() UIColony:SetGameSpeed(speed) end)
	Sleep(500)
	local after = GetTimeFactor() / const.DefaultTimeFactor
	say(fmt("READ speed: SetGameSpeed(%d) call=%s | before=%s after=%s",
		speed, tostring(ok), tostring(before), tostring(after)))
	return true
end

--
-- WORKFLOW EXTERNAL VALIDITY rule. Pack count is READ, never assumed — the
-- owner's campaign runs every opt-in ON, which is not default config.
-- `named` is a list of module ids whose status must be reported BY NAME
-- (leg 4 stands on Opt_CohortHousing being active, and "81/81" does not say so).
function CB1.ReadConditions(note, named)
	local pack = rawget(_G, "SMRFixPack")
	local active, total, inactive = 0, 0, {}
	if pack and pack.order then
		for _, id in ipairs(pack.order) do
			total = total + 1
			local f = pack.fixes and pack.fixes[id]
			local st = f and f.status or "missing"
			if st == "active" then active = active + 1 else inactive[#inactive + 1] = id .. "=" .. st end
		end
	end
	say(fmt("CONDITIONS pack=%d/%d active | map=%s | speed=%s | loads_this_process=%d | note=%s",
		active, total, tostring(rawget(_G, "CurrentMapName")),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), CB1.loads, tostring(note or "-")))
	if #inactive > 0 then
		say("CONDITIONS not-active: " .. table.concat(inactive, ", "))
	end
	for _, id in ipairs(named or empty_table) do
		local f = pack and pack.fixes and pack.fixes[id]
		say(fmt("CONDITIONS by-name %s = %s", tostring(id), tostring(f and f.status or "NOT REGISTERED")))
	end
	return active, total
end

-- Leg 3 gate — PT-42 / F22 / F75. "Is Last Transmission an ACTIVE faction on
-- this save?" has a literal answer in the engine: FactionsHolder.active_factions
-- (Lua/Factions/Factions.lua:181,196,201). Seats come from g_Legislature
-- .legislature_members[faction_id] (:644) and the approval record from
-- .factions_approval[faction_id] (:182). All three are read, because "present
-- as a preset" and "active in this colony" are different claims.
function CB1.ReadFactions()
	local holder = rawget(_G, "g_FactionsHolder")
	local leg    = rawget(_G, "g_Legislature")
	local assembly = CB1.LabelAll("AssemblyBase")
	if #assembly == 0 then assembly = CB1.LabelAll("MartianAssembly") end
	if type(holder) ~= "table" then
		say("FACTIONS NOT-APPLICABLE (g_FactionsHolder is " .. type(holder) .. ") — politics never initialised on this save")
		CB1.Applicable("LastTransmission-active", "n/a", 0, "g_FactionsHolder absent")
		return nil
	end
	local active, n = {}, 0
	for k, v in pairs(holder.active_factions or empty_table) do
		local id = type(v) == "table" and (v.id or v.class) or tostring(v)
		if type(k) == "string" and type(v) ~= "table" then id = k end
		n = n + 1
		active[#active + 1] = tostring(id)
	end
	table.sort(active)
	say(fmt("FACTIONS assembly_buildings=%d | active_factions(%d): %s",
		#assembly, n, table.concat(active, ", ")))
	local seats = {}
	for id, s in pairs((leg and leg.legislature_members) or empty_table) do
		seats[#seats + 1] = fmt("%s=%s", tostring(id), tostring(s))
	end
	table.sort(seats)
	say("FACTIONS legislature seats: " .. (next(seats) and table.concat(seats, " ") or "<none>"))

	local lt = (holder.factions_approval or empty_table)["LastTransmission"]
	local lt_active = false
	for _, id in ipairs(active) do if id == "LastTransmission" then lt_active = true end end
	if lt then
		say(fmt("FACTIONS LastTransmission approval=%s polarization=%s likes_data=%s",
			tostring(lt.approval), tostring(lt.polarization),
			tostring(type(lt.likes_data) == "table" and #lt.likes_data or lt.likes_data)))
		-- The six F75 storage likes, by id, with the value Eval returned.
		local want = { TLEPowerStorage2Sols = true, TLENoPowerStorage = true,
			TLEWaterStorage2Sols = true, TLENoWaterStorage = true,
			TLEOxygenStorage2Sols = true, TLENoOxygenStorage = true }
		for _, ld in ipairs(lt.likes_data or empty_table) do
			if want[ld.id] then
				say(fmt("FACTIONS   like %s value=%s", tostring(ld.id), tostring(ld.value)))
			end
		end
	else
		say("FACTIONS LastTransmission has NO factions_approval record")
	end
	CB1.Applicable("LastTransmission-active", tostring(lt_active), #active,
		"active_factions list read off g_FactionsHolder, non-empty")
	return lt_active, #active
end

-- Leg 4 fixture — PT-53 E / D07. Seniors, the normal residences they live in,
-- and free cohort slots. Cohort detection mirrors the built module:
-- residence.exclusive_trait ("Child" via children_only, Residence.lua:26-28;
-- "Senior" from the SeniorsResidence template).
function CB1.ReadCohort()
	local residences = CB1.LabelAll("Residence")
	local cohort, normal, free_cohort, free_normal = 0, 0, 0, 0
	local by_trait = {}
	for _, r in ipairs(residences) do
		local trait = rawget(r, "exclusive_trait") or r.exclusive_trait
		local cap   = tonumber(r.capacity) or 0
		local used  = type(r.colonists) == "table" and #r.colonists or 0
		local free  = cap - used
		if trait and trait ~= "" and trait ~= false then
			cohort = cohort + 1
			by_trait[tostring(trait)] = (by_trait[tostring(trait)] or 0) + (free > 0 and free or 0)
			if free > 0 then free_cohort = free_cohort + free end
		else
			normal = normal + 1
			if free > 0 then free_normal = free_normal + free end
		end
	end
	local detail = {}
	for t, f in pairs(by_trait) do detail[#detail + 1] = fmt("%s=%d", t, f) end
	table.sort(detail)
	local seniors, seniors_housed, seniors_in_normal, employed = 0, 0, 0, 0
	local each = rawget(_G, "AllMapsForEach")
	if type(each) == "function" then
		each(true, "Colonist", function(c)
			if not IsValid(c) then return end
			local traits = c.traits
			if type(traits) == "table" and traits.Senior then
				seniors = seniors + 1
				local res = c.residence
				if res then
					seniors_housed = seniors_housed + 1
					local trait = res.exclusive_trait
					if not trait or trait == "" or trait == false then
						seniors_in_normal = seniors_in_normal + 1
					end
				end
				if IsValid(c.workplace) or c.user_forced_workplace then employed = employed + 1 end
			end
		end)
	end
	say(fmt("COHORT residences=%d (%d cohort / %d normal) | free slots: cohort=%d [%s] normal=%d",
		#residences, cohort, normal, free_cohort, table.concat(detail, " "), free_normal))
	say(fmt("COHORT seniors=%d (%d housed, %d of them in a NORMAL residence, %d employed)",
		seniors, seniors_housed, seniors_in_normal, employed))
	-- Leg 4's manual-assign half needs a Senior living in a normal residence AND
	-- a free cohort slot to move them to; either being 0 makes the leg UNSAMPLED.
	CB1.Applicable("PT53E-fixture(senior-in-normal x free-cohort-slot)",
		fmt("%d x %d", seniors_in_normal, free_cohort),
		(seniors_in_normal > 0 and free_cohort > 0) and 1 or 0,
		"both halves counted off live labels this load")
	return seniors, seniors_in_normal, free_cohort
end

-- Leg 1 fixture — PT-37 / F48. Tracks, stations, trains with routes.
function CB1.ReadTrackFixture()
	local tracks   = CB1.LabelAll("TrackBase")
	local stations = CB1.LabelAll("Station")
	local trains   = CB1.LabelAll("Train")
	local els, broken, sites, connected = 0, 0, 0, 0
	local damaged_tracks, healthy_tracks = {}, {}
	for _, tr in ipairs(tracks) do
		local list = tr.elements or empty_table
		els = els + #list
		local nsites = #(tr.elements_under_construction or empty_table)
		local nrepair = #(tr.repair_cgs or empty_table)
		sites = sites + nsites
		local nbroken = 0
		for _, el in ipairs(list) do if el.broken then nbroken = nbroken + 1 end end
		broken = broken + nbroken
		-- ⚠️ NOT "is this track connecting two stations". stations_connected is
		-- set only on the <=2-element branch of OnMsg.StationsConnected
		-- (Track.lua:668-677); longer tracks take ConnectToGrids instead and
		-- leave it false. The confirm cycle read 0 of 17 and that is CORRECT and
		-- MEANINGLESS — kept, renamed, and never read as a fixture verdict.
		if tr.stations_connected then connected = connected + 1 end
		if nbroken > 0 or nrepair > 0 then
			damaged_tracks[#damaged_tracks + 1] = tr
		elseif #list > 1 then
			healthy_tracks[#healthy_tracks + 1] = tr
		end
	end
	-- ⚠️ CORRECTED IN PREP, 2026-08-04, visibly (chain rule 5). The first version
	-- tested `t.route or t.train_route or t.command == "Move"` and reported
	-- 0 of 8 trains active — while the F21 reader on the very same load showed
	-- all four sampled trains on command=GotoStation carrying 9-12 passengers.
	-- The fixture was fine; the READER was wrong, and a leg that trusted it
	-- would have declared leg 1's train activity absent. The command name is the
	-- evidence, so the command name is what is counted, and the raw commands are
	-- printed so the next reader can check rather than trust.
	local routed, cmds = 0, {}
	for _, t in ipairs(trains) do
		local c = tostring(t.command)
		cmds[c] = (cmds[c] or 0) + 1
		if c ~= "nil" and c ~= "Idle" and c ~= "false" then routed = routed + 1 end
	end
	local cmd_detail = {}
	for c, n in pairs(cmds) do cmd_detail[#cmd_detail + 1] = fmt("%s=%d", c, n) end
	table.sort(cmd_detail)
	say(fmt("TRACKS %d track(s) (%d with stations_connected — see the note, NOT a connectivity verdict), %d element(s), %d broken element(s), %d repair site(s)",
		#tracks, connected, els, broken, sites))
	say("TRACKS train commands: " .. table.concat(cmd_detail, " "))
	say(fmt("TRACKS %d station(s), %d train(s) (%d with a route or moving)", #stations, #trains, routed))
	say(fmt("TRACKS healthy multi-element tracks=%d | already-damaged tracks=%d",
		#healthy_tracks, #damaged_tracks))
	for i, tr in ipairs(healthy_tracks) do
		if i > 3 then break end
		say(fmt("TRACKS   healthy[%d] %s start_el=%s end_el=%s #elements=%d",
			i, id_of(tr), id_of(tr.start_el), id_of(tr.end_el), #(tr.elements or empty_table)))
	end
	CB1.Applicable("PT37-caseA-fixture(healthy multi-element track)", #healthy_tracks,
		#tracks, "TrackBase label walked this load")
	CB1.Applicable("PT37-train-activity(F21 ride-along)", routed, #trains,
		"Train label walked this load")
	return #tracks, #stations, #trains, routed, healthy_tracks, damaged_tracks
end

-- ⭐ Leg 1's DECIDER PRE-CHECK, and it mutates nothing.
--
-- Whether case B actually samples the risky path is a C-side question the
-- source cannot settle, and prep found it (see 01_OPUS_PREP's Job 2 report):
-- BreakTrackElement (Track.lua:618-659) does NOT remove the element from
-- track.elements — it sets element.broken = <site>, hides it, and places a
-- TrackConstructionSite, whose __parents are { "ConstructionSite",
-- "TrackGridElement" } (TrackElement.lua:580-583). So TWO TrackGridElements now
-- sit on the broken hex, while OrderTrackElements walks with
-- HexGetTrackGridElement = hex_grid:GetObject(q, r, "TrackGridElement")
-- (TrackElement.lua:1-3), which returns exactly ONE of them.
--   * returns the hidden original  → the walk succeeds → case B is just another
--     case A and the DECIDER IS UNSAMPLED. Say so; do not report a pass.
--   * returns the construction site → is_track_element[e] is false → the walk
--     stalls → assert fires → case B is the real failure case.
-- This reader answers that BEFORE anything is mutated, by asking the hex grid
-- for each element's own hex and comparing identity.
function CB1.ReadWalkability(track)
	if not track then say("WALK NOT-APPLICABLE (no track passed)") return nil end
	local map = ResolveMap(track)
	local grid = map and map.object_hex_grid
	if not grid then say("WALK NOT-APPLICABLE (no object_hex_grid on the resolved map)") return nil end
	local list = track.elements or empty_table
	local shadowed, checked, missing = 0, 0, 0
	for _, el in ipairs(list) do
		if el.q and el.r then
			checked = checked + 1
			local at = HexGetTrackGridElement(grid, el.q, el.r)
			if at == nil then
				missing = missing + 1
			elseif at ~= el then
				shadowed = shadowed + 1
				say(fmt("WALK   hex(%s,%s) holds %s, not the track's own element %s (is_construction_site=%s)",
					tostring(el.q), tostring(el.r), id_of(at), id_of(el),
					tostring(rawget(at, "is_construction_site"))))
			end
		end
	end
	say(fmt("WALK track=%s elements=%d checked=%d shadowed=%d missing=%d | PREDICTED OrderTrackElements walk: %s",
		id_of(track), #list, checked, shadowed, missing,
		(shadowed > 0 or missing > 0) and "FAILS (assert path — case B samples the decider)"
			or "SUCCEEDS (no gap — a damaged track here does NOT sample the decider)"))
	CB1.Applicable("PT37-caseB-samples-the-failure-path", tostring(shadowed > 0 or missing > 0),
		checked, "every element's own hex queried against the live object_hex_grid")
	return shadowed, missing, checked
end

-- Leg 1 / leg 4 shared: the exact state the F48 procedure compares before and
-- after. node_idx is included because the failure path forces start_el.node_idx
-- = 1 (Tracks.lua:579) while the success-path renumbering (:632-635) never runs
-- — a DUPLICATE node_idx is the residue signature.
function CB1.SnapTrack(track, when)
	if not track then say("SNAP NOT-APPLICABLE (no track)") return nil end
	local list = track.elements or empty_table
	local idx, conn, dup = {}, 0, {}
	local seen = {}
	for i, el in ipairs(list) do
		local n = el.node_idx
		idx[#idx + 1] = tostring(n)
		if seen[n] then dup[#dup + 1] = tostring(n) end
		seen[n] = true
		conn = conn + #(el.connections or empty_table)
	end
	local sig = fmt("start_el=%s end_el=%s n=%d node_idx=[%s] connections_total=%d duplicate_node_idx=[%s]",
		id_of(track.start_el), id_of(track.end_el), #list,
		table.concat(idx, ","), conn, table.concat(dup, ","))
	say(fmt("SNAP %s %s :: %s", tostring(when), id_of(track), sig))
	return sig
end

-- C42 ride-along. A stale entry is a unit in a passage element's `units` list
-- whose own `holder` is something else. ⛔ The denominator is UNIT ENTRIES, not
-- passages — leg F's zero was over a population of 0 entries and is UNSAMPLED.
function CB1.ReadC42(when)
	local stale, elements, units, passages = 0, 0, 0, 0
	for _, c in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, p in ipairs((c.labels or empty_table).Passage or empty_table) do
			passages = passages + 1
			for _, el in ipairs(p.elements or empty_table) do
				elements = elements + 1
				for _, u in ipairs(el.units or empty_table) do
					units = units + 1
					if u.holder ~= el then stale = stale + 1 end
				end
			end
		end
	end
	say(fmt("C42STALE %d  | %d passage(s), %d element(s), %d unit entr(ies) | when=%s",
		stale, passages, elements, units, tostring(when)))
	CB1.Applicable("C42-stale-entries", stale, units,
		"unit ENTRIES is the denominator; passages>0 is NOT the condition")
	return stale, passages, elements, units
end

-- Leg 2 fixture — PT-47 / F26. Interception needs a defence tower; without one
-- the interception reading is UNSAMPLED and must say so.
function CB1.ReadDefences()
	local towers = CB1.LabelAll("DefenceTowerBase")
	if #towers == 0 then towers = CB1.LabelAll("DefenceTower") end
	local working = 0
	for _, t in ipairs(towers) do
		if t.working ~= false and not t.destroyed then working = working + 1 end
	end
	local domes = CB1.LabelAll("Dome")
	say(fmt("DEFENCES %d defence tower(s) (%d working) | %d dome(s)", #towers, working, #domes))
	for i, d in ipairs(domes) do
		if i > 5 then break end
		say(fmt("DEFENCES   dome[%d] %s pos=%s", i, id_of(d), tostring(IsValid(d) and d:GetPos() or "?")))
	end
	CB1.Applicable("PT47-interception-check", working, #towers,
		"DefenceTowerBase label walked this load")
	CB1.Applicable("PT47-dome-crack-check", #domes, #domes,
		"Dome label walked this load")
	return #towers, working, #domes, domes
end

-- Popup trio fixture. The documented cheapest choice popup is a rocket
-- launch-issue prompt (POPUP_CONSEQUENCE_AUDIT §8 item 3). The route is named
-- concretely: which rocket, and which issue key it currently reports.
-- LanderRocket.lua:89-104 maps issue keys to popup functions via
-- RocketBase:GetLaunchIssuePopups(); an issue with no popup entry is a refusal,
-- not a prompt, so both halves are printed.
function CB1.ReadPopupRoute()
	local names = { "RocketBase", "SupplyRocket", "LanderRocket", "UniversalRocket", "CargoRocket" }
	local seen, rockets = {}, {}
	for _, n in ipairs(names) do
		for _, r in ipairs(CB1.LabelAll(n)) do
			if not seen[r] then seen[r] = true rockets[#rockets + 1] = r end
		end
	end
	local with_issue = 0
	for _, r in ipairs(rockets) do
		local ok, issue = pcall(function() return r:GetLaunchIssue() end)
		local ok2, popups = pcall(function() return r:GetLaunchIssuePopups() end)
		local has_popup = ok2 and type(popups) == "table" and issue and popups[issue] ~= nil
		if issue then with_issue = with_issue + 1 end
		say(fmt("POPUP rocket %s state=%s GetLaunchIssue ok=%s issue=%s | popup-for-that-issue=%s (GetLaunchIssuePopups ok=%s)",
			id_of(r), tostring(r.command or r.status), tostring(ok), tostring(issue),
			tostring(has_popup), tostring(ok2)))
	end
	CB1.Applicable("popup-route(rocket launch issue)", with_issue, #rockets,
		"every rocket on the map queried for a live launch issue")
	return #rockets, with_issue
end

-- F21 ride-along reads. The wrapper's claim is that neither the Comfort penalty
-- nor the rolling-average stat counts platform waiting.
function CB1.ReadTrainStats(when)
	local stations = CB1.LabelAll("Station")
	local trains   = CB1.LabelAll("Train")
	local waiting = 0
	for _, s in ipairs(stations) do
		waiting = waiting + #(s.waiting_for_train or empty_table)
	end
	for i, t in ipairs(trains) do
		if i > 4 then break end
		local ts = t.ts_spent_time or t.spent_time
		say(fmt("F21 train %s command=%s spent_time=%s passengers=%s",
			id_of(t), tostring(t.command), tostring(ts),
			tostring(type(t.units) == "table" and #t.units or "?")))
	end
	say(fmt("F21 %d station(s) holding %d colonist(s) waiting for a train | %d train(s) | when=%s",
		#stations, waiting, #trains, tostring(when)))
	CB1.Applicable("F21-platform-wait-population", waiting, #stations,
		"station.waiting_for_train walked this load")
	return waiting, #trains
end

---- error-watch marker ------------------------------------------------------
--
-- ⛔ It may NOT count [LUA ERROR] lines. The engine writes those straight to the
-- log file and the mod environment has no file read (io and os are both keys of
-- ModEnvBlacklist, Mod.lua:1424/:1425), so a count printed from here would be
-- invented. It prints the boundary a reader greps between, and names the
-- attribution duty (WORKFLOW "never silently discount a line").
-- ⚠️ CORRECTED IN PREP, 2026-08-04, visibly (chain rule 5). The first version of
-- this line contained the literal token an error sweep greps for, so the marker
-- MATCHED ITS OWN SEARCH: the confirm cycle's error grep returned 2 hits, both
-- of them this sentence, and a hurried reader would have called a clean run
-- dirty. The token is spelled around below. Inherited from unattended-1, where
-- it had the same defect and nobody noticed because that grep was run by hand.
function CB1.ErrorWatchNote()
	say(fmt("ERRORWATCH %s — every engine error line (the bracketed LUA-ERROR marker) in this log file between the '=== corun-batch-1 %s ===' banner and this line happened inside this run. Attribute each one against the FORCED / ORGANIC window markers printed above; report every unexplained line verbatim with its position, never discount one. stage_at_close=%s loads_this_process=%d",
		tostring(CB1.tag), tostring(CB1.tag), tostring(CB1.stage), CB1.loads))
end

---- watchdog + flow ---------------------------------------------------------
--
-- Used only by the UNATTENDED payload (98_CB1Confirm). The attended sitting
-- library (99_CB1Legs) does not call this at all — it is invoked from the
-- console, one line per moment.

function CB1.Flow()
	CreateRealTimeThread(function()
		local deadline = RealTime() + CB1.watchdog
		while CB1.stage ~= "done" do
			if RealTime() - deadline >= 0 then
				say("WATCHDOG TIMEOUT at stage " .. tostring(CB1.stage))
				local flush = rawget(_G, "FlushLogFile")
				if flush then flush() end
				Sleep(1000)
				quit()
				return
			end
			Sleep(1000)
		end
	end)

	say(fmt("=== corun-batch-1 %s === watchdog %dms", tostring(CB1.tag), CB1.watchdog))

	if not step("1-waiting-menu", function()
		if not wait_for(CB1.menu_timeout, function() return not not GetPreGameMainMenu() end) then
			return false
		end
		Sleep(2000)
	end) then return end

	local ok = sprocall(CB1.Payload)
	if not ok then
		say("PAYLOAD ERROR at stage " .. tostring(CB1.stage))
		shutdown("ERROR payload")
		return
	end
	shutdown("done")
end

function CB1.Boot()
	CreateRealTimeThread(function()
		-- CreateRealTimeThread runs its body synchronously up to the first
		-- yield, so this Sleep is what puts everything below after
		-- ModsLoadCode().
		Sleep(1)
		if rawget(_G, "SMRTest_EnablePathLeg") then
			say("standing down: 98_EnablePathLeg.lua drives this session")
			return
		end
		CB1.Flow()
	end)
end
