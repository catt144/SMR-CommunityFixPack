--   tracks / stations / running trains (leg 1) + a walkability control ·

CB1.tag      = "CB1CONFIRM"
CB1.watchdog = 12 * 60 * 1000

local SAVE = "CB1STAGE.savegame.sav"

function CB1.Payload()
	if not CB1.Step("2-load-staged-copy", function()
		return CB1.Load(SAVE, 3)
	end, "untimed") then return end

	CB1.Step("3-conditions", function()
		CB1.ReadConditions("corun-batch-1 fixture confirm, one cycle, READ ONLY",
			{ "CohortHousing", "Opt_CohortHousing", "BombardmentSpread",
			  "TrainWaitTime", "GridGlobalStorage", "LastTransmissionStorage" })
	end)

	CB1.Step("4-leg3-gate-factions", function()
		CB1.ReadFactions()
	end)

	CB1.Step("5-leg4-cohort-fixture", function()
		CB1.ReadCohort()
	end)

	CB1.Step("6-leg1-track-fixture", function()
		local _, _, _, _, healthy, damaged = CB1.ReadTrackFixture()
		-- Walkability CONTROL. Run on a HEALTHY track, where the predicted
		-- answer is "SUCCEEDS": that is what proves the reader can distinguish,
		-- rather than always printing the same verdict. If this save already
		-- carries a damaged track, read that one too — the sitting would then
		-- not have to stage the damage at all.
		if healthy and healthy[1] then
			CB1.Say("WALK control on a HEALTHY track — prediction: SUCCEEDS")
			CB1.ReadWalkability(healthy[1])
			CB1.SnapTrack(healthy[1], "confirm-cycle-healthy")
		else
			CB1.Say("WALK NOT-APPLICABLE — no healthy multi-element track on this save")
		end
		if damaged and damaged[1] then
			CB1.Say("WALK on an ALREADY-DAMAGED track found on this save — prediction: see the reader's own line")
			CB1.ReadWalkability(damaged[1])
			CB1.SnapTrack(damaged[1], "confirm-cycle-damaged")
		else
			CB1.Say("WALK no already-damaged track on this save — the sitting stages case B's damage itself (FORCED)")
		end
	end)

	CB1.Step("7-c42-ride-along", function()
		-- ⛔ This is a POST-LOAD read and therefore carries leg F's own limit:
		-- Holder.units is populated only by runtime holder transitions
		-- (Unit:SetHolderOnMap is the sole caller of OnEnterHolder), so a load
		-- may simply rebuild it empty. It is recorded as the sitting's BEFORE
		-- baseline, not as a C42 result.
		CB1.ReadC42("post-load baseline, NOT a C42 result")
	end)

	CB1.Step("8-leg2-defences", function()
		CB1.ReadDefences()
	end)

	CB1.Step("9-popup-route", function()
		CB1.ReadPopupRoute()
	end)

	CB1.Step("10-leg5-turbine-fixture", function()
		CB1.ReadTurbineFixture()
	end)

	CB1.Step("11-f21-ride-along", function()
		CB1.ReadTrainStats("confirm-cycle baseline")
	end)

	CB1.ErrorWatchNote()
end

CB1.Boot()
