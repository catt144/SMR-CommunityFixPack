-- SMR Community Fix Pack — Test Kit :: d13-rescue MATRIX PAYLOAD. TEMPORARY
--
-- Declared by: d13-rescue chain prompt 4 (04_OPUS_VERIFY.md).
-- Predictions: agent/reports/D13_VERIFICATION.md §2, committed BEFORE launch 1.
--
-- ONE file, one CELL per launch. RS_ARM.ps1 rewrites the CELL line below off
-- disk and reads it back, so a cell can never be mistaken for another after the
-- fact (the REHEARSAL-flag shape from unattended-2, generalised).
--
-- ⛔ EVERY CELL IS A DIFFERENT MOD CONFIGURATION, set by JUNCTION presence
--    (EF-055: a pull is a REAL uninstall, D13 state (4); the account and the
--    owner's ticks are untouched), and a junction change needs a FULL PROCESS
--    RESTART (D13's four-states rule). RS.RequireConfig STOPS the run when the
--    process is not the cell it claims to be — the amended batch-2 rule 7: a
--    gate whose only output is a line in a log is not a gate.
--
-- THE CELLS (predictions are D13_VERIFICATION.md §2; the letters match):
--   r0        fix + opt + kit            RESCUESTAGE  the standing-config control
--   r1        fix + opt + kit + rescue   RESCUESTAGE  the artifact as a measurable no-op
--   r2pre     kit only                   RESCUEWIT    before-inventory + the manufacture
--   r2        kit + rescue               RESCUEDMG    ⭐ the rescue case, AUTOMATIC path
--   r2b       kit + rescue               RESCUECLEAN  idempotence, fresh process
--   r3        kit only                   RESCUECLEAN  the artifact leaves nothing
--   rrestore  fix + opt + kit            RESCUESTAGE  the close-out control
--   (r3z, the zero-mods read, has no instrument at all — no kit, no payload.)
--
-- ⛔ THE SUITE WORLD IS FIXED ACROSS THE TALLY CELLS. r0 and r1 load the SAME
--    staged copy, because the audit-recounted baseline (78/0/10/0 of 88) was
--    measured on exactly these bytes and a tally taken on a different world is
--    not comparable — the [state] probes read the colony.

local CELL = "r0"   -- ⛔ REWRITTEN BY RS_ARM.ps1 AND READ BACK OFF DISK

-- ⛔ Copy-only, every one of them (EF-051 HOLD). Names are VERBATIM (EF-050).
local SUITE_SAVE = "RESCUESTAGE.savegame.sav"   -- byte copy of PT35FIXTURE (== SPSTAGE, the baseline world)
local WITNESS    = "RESCUEWIT.savegame.sav"     -- byte copy of SPCONRT, the NATIVE residue witness
local DAMAGED    = "RESCUEDMG.savegame.sav"     -- written by r2pre: native residue + the declared manufacture
local CLEANED    = "RESCUECLEAN.savegame.sav"   -- written by r2, after the artifact's automatic pass

RS.tag = "RS-" .. CELL

local CONFIG = {
	r0       = { fix = true,  opt = true,  rescue = false },
	r1       = { fix = true,  opt = true,  rescue = true  },
	r2pre    = { fix = false, opt = false, rescue = false },
	r2       = { fix = false, opt = false, rescue = true  },
	r2b      = { fix = false, opt = false, rescue = true  },
	r3       = { fix = false, opt = false, rescue = false },
	rrestore = { fix = true,  opt = true,  rescue = false },
}

local WATCHDOG = {
	r0 = 30 * 60 * 1000, r1 = 30 * 60 * 1000, r2pre = 30 * 60 * 1000,
	r2 = 30 * 60 * 1000, r2b = 20 * 60 * 1000, r3 = 20 * 60 * 1000,
	rrestore = 20 * 60 * 1000,
}
RS.watchdog = WATCHDOG[CELL] or 25 * 60 * 1000

function RS.Payload()
	RS.Say("MODE ON DISK = CELL " .. tostring(CELL)
		.. " — this banner is the cell's identity; the ARM gate printed the same value off disk before launch")

	local cfg = CONFIG[CELL]
	if not cfg then
		RS.Say("PAYLOAD ⛔ unknown CELL '" .. tostring(CELL) .. "' — nothing will be measured")
		return
	end

	-- ⛔ THE CONFIG GATE, BEFORE ANY READING. It stops.
	if not RS.Step("2-config-gate", function()
		return RS.RequireConfig(CELL, cfg.fix, cfg.opt, cfg.rescue)
	end) then return end

	----------------------------------------------------------------------------
	if CELL == "r0" then
		-- The standing-config control. The artifact is NOT installed; this is
		-- what the rig looks like before it ever arrives, and r-restore re-reads
		-- the same rows to prove the round trips gave it back unchanged.
		if not RS.Step("3-load-suite-world", function()
			return RS.Load(SUITE_SAVE, 3) and true or false
		end, true) then return end

		RS.Step("4-baseline-inventory", function()
			RS.ReadGates("suite-world-loaded")
			RS.ReadResidue("r0/baseline-world")
			RS.ReadRescueTable("r0")   -- expected NOT AVAILABLE: the mod is absent
		end)

		RS.Step("5-suite", function()
			RS.ReadGates("immediately-before-RunAll")
			RS.RunSuite(CELL)
		end)
		RS.ErrorWatchNote()
		return
	end

	----------------------------------------------------------------------------
	if CELL == "r1" then
		-- The artifact beside both packs. Per spec 10.1 the automatic pass is a
		-- measurable NO-OP here and prints its zero, and a stand-down dialog is
		-- raised once per SESSION rather than per load.
		RS.Step("3-rescue-surfaces-at-menu", function()
			RS.ReadRescueTable("r1/at-menu")
		end)

		local after_load, after_scan, after_suite
		if not RS.Step("4-load-suite-world", function()
			-- ⭐ the artifact's OnMsg.PostLoadGame fires inside this call; its
			--    own log lines are the evidence, not anything printed from here
			return RS.Load(SUITE_SAVE, 3) and true or false
		end, true) then return end

		RS.Step("5-inventory-after-automatic-pass", function()
			RS.ReadGates("suite-world-loaded")
			after_load = RS.ReadResidue("r1/after-automatic-pass")
		end)

		RS.Step("6-read-only-scan", function()
			-- Scan() is scan_only + force: it counts what the artifact would act
			-- on WITHOUT the stand-down hiding it, and mutates nothing. On a live
			-- colony that "mutates nothing" is a claim worth measuring, so the
			-- inventory is re-read against it immediately.
			local mod = rawget(_G, "SMRSaveRescue")
			if type(mod) == "table" and type(mod.Scan) == "function" then
				RS.Try("SMRSaveRescue.Scan()", function()
					local r = mod.Scan()
					return RS.Fmt("removed(counted)=%s scanned=%s", tostring(r and r.removed), tostring(r and r.scanned))
				end)
			end
			after_scan = RS.ReadResidue("r1/after-read-only-scan")
			RS.DiffResidue(after_load, after_scan, "r1 automatic-pass -> Scan()")
		end)

		RS.Step("7-suite", function()
			RS.ReadGates("immediately-before-RunAll")
			RS.RunSuite(CELL)
		end)

		RS.Step("8-interlock-check", function()
			-- ⛔ THE INTERLOCK TEST. The six rescue probes drive FORCED passes;
			--    they are supposed to touch only synthetic state (WithGlobals,
			--    g_RainDisaster stubbed false, MainMap false). A moved handle set
			--    here means a probe reached the owner's live colony.
			after_suite = RS.ReadResidue("r1/after-RunAll")
			RS.DiffResidue(after_load, after_suite, "r1 BEFORE RunAll -> AFTER RunAll (the probe interlock)")
			RS.CargoCheck("r1")
		end)
		RS.ErrorWatchNote()
		return
	end

	----------------------------------------------------------------------------
	if CELL == "r2pre" then
		-- Kit only: no pack, no artifact. Two jobs — read the NATIVE residue off
		-- a real witness with nothing installed that could clean it, and
		-- MANUFACTURE the pre-rewrite-lineage rows no save in the folder carries.
		local before, after_mfg, after_rt, keep_kind
		if not RS.Step("3-load-witness", function()
			return RS.Load(WITNESS, 1) and true or false
		end, true) then return end

		RS.Step("4-before-inventory", function()
			RS.Say("WITNESS " .. WITNESS .. " = a byte copy of SPCONRT.savegame.sav (MD5 8A7639341C536816BFE18E35150AEDDE), the split-optins cell-(e) contract round trip — savename VERBATIM (EF-050), copy-only (EF-051)")
			before = RS.ReadResidue("r2pre/before")
			RS.CargoCheck("r2pre/before")
			local kit = rawget(_G, "SMRTest")
			if kit and kit.FixtureCarry then RS.Try("SMRTest.FixtureCarry() before", kit.FixtureCarry) end
		end)

		RS.Step("5-manufacture", function()
			keep_kind = RS.EnsureKeepWitness("r2pre", before and before.keep and before.keep.f35)
			RS.Manufacture("r2pre", 25)
			after_mfg = RS.ReadResidue("r2pre/after-manufacture")
			RS.Say("MANUFACTURE r2pre :: KEEP witness kind = " .. tostring(keep_kind))
		end)

		if not RS.Step("6-round-trip", function()
			-- R4: the manufacture is only real if it survives disk. EF-049 —
			-- the witness is the on-disk bytes plus the load-back, never the API.
			if not RS.Save(DAMAGED) then return false end
			return RS.Load(DAMAGED, 1) and true or false
		end, true) then return end

		RS.Step("7-after-round-trip", function()
			after_rt = RS.ReadResidue("r2pre/after-round-trip")
			RS.DiffResidue(after_mfg, after_rt, "r2pre manufacture -> save -> reload (R4)")
			RS.Say("ROUNDTRIP r2pre :: ⚠️ the one row that could legitimately move is the Meteors thread — a thread killed in-session may or may not persist as dead. Its value in the line above IS the answer, and if it reads valid the meteor restart stays UNSAMPLED rather than refuted.")
			local kit = rawget(_G, "SMRTest")
			if kit and kit.FixtureCarry then RS.Try("SMRTest.FixtureCarry() after-round-trip", kit.FixtureCarry) end
		end)
		RS.ErrorWatchNote()
		return
	end

	----------------------------------------------------------------------------
	if CELL == "r2" then
		-- ⭐ THE RESCUE CASE. Both packs genuinely absent, so the artifact's own
		--    OnMsg.PostLoadGame fires for real. The pass is sampled through its
		--    REAL trigger, never through force = true: the probe surface exists
		--    because the rig's normal config stands the pass down, and here it
		--    does not.
		local after, survivors, kept
		RS.Step("3-rescue-surfaces-at-menu", function()
			RS.ReadRescueTable("r2/at-menu")
		end)

		if not RS.Step("4-load-damaged", function()
			return RS.Load(DAMAGED, 1) and true or false
		end, true) then return end

		RS.Step("5-read-back", function()
			RS.ReadGates("damaged-save-loaded")
			after = RS.ReadResidue("r2/after-automatic-pass")
			survivors, kept = RS.AssertCleared(after, "r2/after-automatic-pass")
			-- ⚠️ spec 10.2's disclosed incompleteness: an object in no label is
			--    never visited, and the real case is a colonist riding a rocket.
			--    A surviving reserved_at/shelter_try is checked against THAT
			--    population before anything is filed as a defect.
			RS.CargoCheck("r2/after-automatic-pass")
			local kit = rawget(_G, "SMRTest")
			if kit and kit.FixtureCarry then RS.Try("SMRTest.FixtureCarry() after-pass", kit.FixtureCarry) end
		end)

		RS.Step("6-second-pass-in-process", function()
			-- Idempotence, cheap half: a forced pass immediately after must find
			-- nothing. The expensive half is cell r2b, a whole fresh process.
			local mod = rawget(_G, "SMRSaveRescue")
			if type(mod) == "table" and type(mod.RunPass) == "function" then
				RS.Try("SMRSaveRescue.RunPass{force=true} (second pass, same process)", function()
					local r = mod.RunPass({ force = true, quiet = true })
					return RS.Fmt("removed=%s meteors=%s rains_restarted=%s rains_finished=%s skipped=%s",
						tostring(r and r.removed), tostring(r and r.heals and r.heals.meteors),
						tostring(r and r.heals and r.heals.rains_restarted),
						tostring(r and r.heals and r.heals.rains_finished),
						tostring(r and r.skipped and #r.skipped))
				end)
			end
			RS.ReadResidue("r2/after-second-pass")
		end)

		RS.Step("7-save-cleaned", function()
			if not RS.Save(CLEANED) then
				RS.Say("SAVE ⛔ RESCUECLEAN did not land — cells r2b and r3 have no input and must be re-planned")
				return false
			end
		end, true)
		RS.ErrorWatchNote()
		return
	end

	----------------------------------------------------------------------------
	if CELL == "r2b" then
		-- Idempotence in a FRESH PROCESS: `stand_down_told` is reset, nothing is
		-- carried over, and the second consecutive load of a cleaned save must
		-- perform no removal and no heal. ⭐ This is what the artifact has
		-- INSTEAD of a persisted latch — the rain heal's only detector (D4) was
		-- removed by step 4 of the previous pass, so the heal cannot re-fire.
		if not RS.Step("3-load-cleaned", function()
			return RS.Load(CLEANED, 1) and true or false
		end, true) then return end

		RS.Step("4-read-back", function()
			RS.ReadGates("cleaned-save-loaded")
			local rec = RS.ReadResidue("r2b/after-automatic-pass")
			RS.AssertCleared(rec, "r2b/second-consecutive-load")
			RS.Say("IDEMPOTENCE r2b :: the artifact's own 'pass: removed N' line above IS the verdict — it is printed even when every number is zero, precisely so that an absence of lines cannot be told from an absence of the pass (spec 10.9(3)).")
		end)
		RS.ErrorWatchNote()
		return
	end

	----------------------------------------------------------------------------
	if CELL == "r3" then
		-- ⛔ THE ARTIFACT LEAVES NOTHING, half one. The rescue's junction is
		--    pulled, so anything of its namespace still readable here is residue
		--    it left in the save. Half two is this log itself, read after
		--    archiving (EF-047: absence verdicts come from the archived file).
		if not RS.Step("3-load-cleaned", function()
			return RS.Load(CLEANED, 1) and true or false
		end, true) then return end

		RS.Step("4-artifact-absent-inventory", function()
			RS.ReadGates("cleaned-save-loaded-with-no-mods-but-the-kit")
			local rec = RS.ReadResidue("r3/artifact-absent")
			RS.AssertCleared(rec, "r3/artifact-absent")
			RS.ScanArtifactNames("r3")
			RS.ReadRescueTable("r3")   -- expected NOT AVAILABLE
			local kit = rawget(_G, "SMRTest")
			if kit and kit.FixtureCarry then RS.Try("SMRTest.FixtureCarry() artifact-absent", kit.FixtureCarry) end
		end)
		RS.ErrorWatchNote()
		return
	end

	----------------------------------------------------------------------------
	if CELL == "rrestore" then
		-- The close-out control: the rig is back in its STANDING configuration
		-- after seven junction round trips, both registries read what r0 read,
		-- and the owner's dials and toggles are where they left them. The config
		-- gate above already printed all three; this loads the baseline world one
		-- more time so the residue inventory can be compared with r0's row for
		-- row — the account is one witness, the save is the other.
		if not RS.Step("3-load-suite-world", function()
			return RS.Load(SUITE_SAVE, 1) and true or false
		end, true) then return end

		RS.Step("4-standing-config-readback", function()
			RS.ReadGates("standing-config")
			RS.ReadModOrder("standing-config")
			RS.ReadDialOptions("standing-config")
			RS.ReadResidue("rrestore/baseline-world")
			RS.Say("RESTORE rrestore :: compare this inventory with r0/baseline-world — same bytes, same world, seven junction round trips apart. EF-055's claim is that a junction pull leaves the account untouched; the DIALOPTS line and this inventory are the two independent witnesses.")
		end)
		RS.ErrorWatchNote()
		return
	end

	RS.Say("PAYLOAD ⛔ CELL '" .. tostring(CELL) .. "' has no body — nothing was measured")
end

RS.Boot()
