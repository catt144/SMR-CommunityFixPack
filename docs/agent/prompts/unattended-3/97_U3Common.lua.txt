U3 = rawget(_G, "U3") or {}

U3.tag         = U3.tag or "U3"
U3.settle_time = 15 * 1000        -- the MEASURED-sufficient datum (co-run #1)
U3.stage       = "boot"
U3.loads       = 0
U3.mode        = U3.mode or "REAL"

---- output ------------------------------------------------------------------
--
-- ModLog is the only print path proven to reach the log file on disk
-- (Mod.lua:109-132). Its emitter formats its single argument a second time
-- (lib.lua:164-174), so a literal '%' raises there — escape it. Flush per line
-- so an abandoned run still shows how far it got.
local function say(msg)
	msg = "[" .. tostring(U3.tag) .. "] " .. tostring(msg)
	local modlog = rawget(_G, "ModLog")
	if modlog then
		modlog((msg:gsub("%%", "%%%%")))
	else
		local dbg = rawget(_G, "DebugPrint")
		if dbg then dbg(msg .. "\n") end
	end
	print(msg)
	local flush = rawget(_G, "FlushLogFile")
	if flush then pcall(flush) end
end
U3.Say = say

local function fmt(...)
	local ok, s = pcall(string.format, ...)
	return ok and s or "<fmt error>"
end
U3.Fmt = fmt

local function id_of(obj)
	if not obj then return "nil" end
	if type(obj) ~= "table" then return tostring(obj) end
	return tostring(obj.class or "?") .. "#" .. tostring(rawget(obj, "handle"))
end
U3.IdOf = id_of

-- G3. The ONLY sanctioned call wrapper. Prints ok and the error text, always.
-- ⛔ pcall cannot survive a yield — use U3.TryYield for anything that Sleeps.
function U3.Try(what, fn, ...)
	local ok, a, b = pcall(fn, ...)
	say(fmt("TRY %s ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

function U3.TryYield(what, fn, ...)
	local ok, a, b = sprocall(fn, ...)
	say(fmt("TRY %s (yielding) ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

-- G2. The condition-sampled rule, mechanised.
function U3.Applicable(label, count, population, witness)
	local applicable = (population or 0) > 0
	say(fmt("READ %s = %s | population=%s | APPLICABLE=%s | witness=%s",
		tostring(label), tostring(count), tostring(population),
		tostring(applicable), tostring(witness or "-- NONE NAMED --")))
	return applicable
end

-- EVERY reading declares which side of the forced/organic line it is on.
function U3.Label(kind, what, detail)
	kind = tostring(kind):upper()
	if kind ~= "FORCED" and kind ~= "ORGANIC" then
		kind = "UNLABELLED(⚠️ neither FORCED nor ORGANIC was declared)"
	end
	say(fmt("%s%s :: %s :: %s", U3.Void(), kind, tostring(what), tostring(detail or "-")))
	return kind
end

---- small helpers -----------------------------------------------------------

local function wait_for(timeout, ready)
	local deadline = RealTime() + timeout
	while not ready() do
		if RealTime() - deadline >= 0 then return false end
		Sleep(500)
	end
	return true
end
U3.WaitFor = wait_for

local function game_is_live()
	local city   = rawget(_G, "MainCity")
	local colony = rawget(_G, "UIColony")
	return not not (city and city.labels and colony and colony.labels)
end
U3.GameIsLive = game_is_live

local function label_all(name)
	local out = {}
	for _, city in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, o in ipairs((city.labels or empty_table)[name] or empty_table) do
			out[#out + 1] = o
		end
	end
	return out
end
U3.LabelAll = label_all

-- Game time, printed raw beside every derived figure.
-- 30,000 game-ms = 1 game hour is MEASURED (PLAYTEST_HELP); a sol is 24 of them.
U3.MS_PER_HOUR = 30000
U3.MS_PER_SOL  = 24 * 30000

function U3.Sols(ms)
	if type(ms) ~= "number" then return "n/a" end
	return fmt("%d.%02d sols", ms / U3.MS_PER_SOL,
		((ms % U3.MS_PER_SOL) * 100) / U3.MS_PER_SOL)
end

-- ⭐ THE LINE THE 08-11 BRACKET WAS MISSING AT REVERT. Called at EVERY read,
--    including the revert, so the pause state is evidenced at every point and
--    not merely at the two the reader happens to look at.
function U3.Now(when)
	local colony = rawget(_G, "UIColony")
	local gt = GameTime()
	say(fmt("CLOCK %s :: GameTime()=%s (%s since game start) | UIColony.day=%s hour=%s | speed=%sx | RealTime()=%s",
		tostring(when), tostring(gt), U3.Sols(gt),
		tostring(colony and colony.day), tostring(colony and colony.hour),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), tostring(RealTime())))
	return gt
end

---- load --------------------------------------------------------------------
--
-- ⛔ NO SPEED IS SET. See the header: this leg's bracket is a PAUSED one and a
--    speed would be the confound the 08-12 ruling's method exists to exclude.
--    The speed is READ BACK, and forced to 0 if the engine handed us anything
--    else, with both values printed.
function U3.Load(savefile)
	local err = LoadGame(savefile, {})            -- ⛔ never pass save_as_last
	if err then
		say("LOADERR " .. tostring(savefile) .. " " .. tostring(err))
		return false
	end
	U3.loads = U3.loads + 1
	say(fmt("LOAD OK #%d file=%s — LoadGame returned no error", U3.loads, tostring(savefile)))
	if not wait_for(5 * 60 * 1000, game_is_live) then
		say("LOADERR " .. tostring(savefile) .. " — city never came live")
		return false
	end
	Sleep(U3.settle_time)                          -- 15 s is the MEASURED datum
	local before = GetTimeFactor() / const.DefaultTimeFactor
	local forced = false
	if before ~= 0 then
		forced = true
		pcall(function() UIColony:SetGameSpeed(0) end)
		Sleep(500)
	end
	local after = GetTimeFactor() / const.DefaultTimeFactor
	say(fmt("READ speed after load: before=%sx after=%sx | forced_to_zero=%s | PAUSED=%s — the bracket below is void if this is not 0x",
		tostring(before), tostring(after), tostring(forced), tostring(after == 0)))
	return true
end

---- run conditions ----------------------------------------------------------
--
-- WORKFLOW EXTERNAL VALIDITY rule: the pack count is READ, never assumed.
function U3.ReadConditions(note, named)
	local pack = rawget(_G, "SMRFixPack")
	local active, total, inactive = 0, 0, {}
	if pack and pack.order then
		for _, id in ipairs(pack.order) do
			total = total + 1
			local f = pack.fixes and pack.fixes[id]
			local st = f and f.status or "missing"
			if st == "active" then active = active + 1 else inactive[#inactive + 1] = id .. "=" .. st end
		end
	end
	say(fmt("CONDITIONS pack=%d/%d active | map=%s | speed=%s | loads_this_process=%d | mode=%s | note=%s",
		active, total, tostring(rawget(_G, "CurrentMapName")),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), U3.loads,
		tostring(U3.mode), tostring(note or "-")))
	if #inactive > 0 then
		say("CONDITIONS not-active: " .. table.concat(inactive, ", "))
	end
	-- the other two mods, read the same way — a leg that quotes one line
	-- describes a third of the rig (TestKit README)
	local opt = rawget(_G, "SMROptInPack")
	if opt and opt.order then
		local oa, ot = 0, 0
		for _, id in ipairs(opt.order) do
			ot = ot + 1
			local m = opt.fixes and opt.fixes[id]
			if m and m.status == "active" then oa = oa + 1 end
		end
		say(fmt("CONDITIONS opt-in pack=%d/%d active", oa, ot))
	else
		say("CONDITIONS opt-in pack = NOT PRESENT in this process")
	end
	for _, id in ipairs(named or empty_table) do
		local f = pack and pack.fixes and pack.fixes[id]
		say(fmt("CONDITIONS by-name %s = %s | detail=%s", tostring(id),
			tostring(f and f.status or "NOT REGISTERED"), tostring(f and f.detail or "-")))
	end
	return active, total
end

---- ⛔ THE RUN-CONDITION GATE (WORKFLOW rule 7 as amended 2026-08-11) --------
--
-- unattended-2 run 1 read `pack=0/0 active` and carried on for six more steps,
-- producing readings that could not mean anything: the pack was DISABLED in the
-- Mod Manager. The amended rule puts the gate at the run top and requires it to
-- STOP. This run has a single run top (U3.Boot), so the gate is there — and
-- U3.Guard is kept for any leg called out of band.
function U3.RequirePackLoaded(note)
	local pack = rawget(_G, "SMRFixPack")
	local active, total = 0, 0
	if pack and pack.order then
		for _, id in ipairs(pack.order) do
			total = total + 1
			local f = pack.fixes and pack.fixes[id]
			if f and f.status == "active" then active = active + 1 end
		end
	end
	U3.pack_loaded = active > 0
	if U3.pack_loaded then
		say(fmt("GATE pack-loaded OK :: %d/%d modules active :: %s", active, total, tostring(note)))
		return true
	end
	say(fmt("GATE ⛔ PACK NOT LOADED :: %d/%d active. The Community Fix Pack mod is present on disk but not enabled in the Mod Manager. Every reading about C39 or F85 is VOID in this state — the pack's code never ran. :: %s",
		active, total, tostring(note)))
	say("GATE ⛔ THIS IS NOT SCRIPTABLE. AccountStorage, SaveAccountStorage and ModsReloadItems are all ModEnvBlacklist keys (Mod.lua), and there is no console at the main menu — so no mod-side or console-side path can tick the box. It is one human click in the Mod Manager, followed by a FULL RESTART (D13).")
	return false
end

-- Prefix for any line that would otherwise read like a verdict.
function U3.Void()
	if U3.pack_loaded then return "" end
	return "VOID(pack not loaded) "
end

function U3.Guard(leg)
	U3.stage = tostring(leg)
	if U3.RequirePackLoaded("leg " .. tostring(leg)) then return true end
	say(fmt("GATE ⛔ LEG '%s' NOT RUN. Nothing was read and nothing was changed.", tostring(leg)))
	return false
end

---- error-watch marker ------------------------------------------------------
--
-- ⛔ It may NOT count engine error lines: the engine writes those straight to
--    the log file and the mod environment has no file read (io/os are both
--    ModEnvBlacklist keys). A count printed from here would be invented. It
--    prints the boundary a reader greps between.
-- ⚠️ The token an error sweep greps for is spelled AROUND, not written.
function U3.ErrorWatchNote()
	say(fmt("ERRORWATCH %s — every engine error line (the bracketed LUA-ERROR marker) in this log between the '=== unattended-3 %s ===' banner and this line happened inside this run. Report every unexplained line verbatim with its position and its age; never discount one. stage_at_close=%s loads_this_process=%d mode=%s",
		tostring(U3.tag), tostring(U3.tag), tostring(U3.stage), U3.loads, tostring(U3.mode)))
end
