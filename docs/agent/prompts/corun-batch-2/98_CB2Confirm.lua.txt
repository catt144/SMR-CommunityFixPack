CB2.tag      = "CB2CONFIRM"
CB2.watchdog = 15 * 60 * 1000

local SAVE = "CB2STAGE.savegame.sav"

function CB2.Payload()
	if not CB2.Step("2-load-staged-copy", function()
		return CB2.Load(SAVE, 3)
	end, "untimed") then return end

	CB2.Step("3-conditions", function()
		-- ⚠️ REGISTERED ids, not file names (batch-1 asked for "Opt_CohortHousing"
		-- and got NOT REGISTERED beside "CohortHousing = active").
		CB2.ReadConditions("corun-batch-2 fixture confirm, one cycle, READ ONLY",
			{ "BombardmentSpread", "TrainWaitTime", "CohortHousing", "SaveSanitizer" })
	end)

	CB2.Step("4-legP-popup-subjects", function()
		CB2.ReadPopupSubjects()
	end)

	CB2.Step("5-legQ-track-candidates", function()
		CB2.ReadTrackQ(2)
		CB2.ReadF99Residue("confirm-cycle baseline, before anything is broken")
	end)

	CB2.Step("6-legR-defences", function()
		CB2.ReadDefences()
	end)

	CB2.Step("7-legS-turbine-fixture", function()
		CB2.ReadTurbineFixture("confirm cycle")
	end)

	CB2.Step("8-f21-reads-and-subject", function()
		CB2.ReadF21("confirm-cycle baseline, speed 3")
		CB2.ReadF21Subjects()
	end)

	-- ⭐ The long one, and the most valuable thing this cycle can do: find out
	-- whether the C42 leg has a subject AT ALL before the owner is asked to sit
	-- for it. Two sessions read C42STALE 0 over ZERO unit entries; if this
	-- witness comes back empty again, the C42 ride-along is briefed as SKIP with
	-- its try count, not as a moment.
	CB2.Step("9-c42-baseline-then-witness", function()
		CB2.ReadC42("post-load baseline at speed 3, NOT a C42 result")
		CB2.Say("C42 raising speed to 20 for the witness window — a traversal witness needs GAME TIME to pass. This is the second and last thing this cycle changes, and it changes nothing on disk.")
		CB2.Try("SetGameSpeed(20) for the C42 witness window", function()
			return UIColony:SetGameSpeed(20)
		end)
		Sleep(2000)
		CB2.Say(CB2.Fmt("C42 speed read back = %s", tostring(GetTimeFactor() / const.DefaultTimeFactor)))
		CB2.WitnessC42(90, 1000, 20000)
	end)

	CB2.ErrorWatchNote()
end

CB2.Boot()
