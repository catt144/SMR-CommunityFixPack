---- shared: the PROVEN save primitive ---------------------------------------
--
-- ⭐ RESURRECTED, not re-derived. Envelope-proven 2026-08-04 (unattended-1
--    cycle 0, log u1c0): SaveGame from a real-time thread returns err=false, the
--    file appears to Savegame.ListForTag("savegame"), LoadGame brings it back.
--    Measured cost 0.58–0.63 s on the 56 MB save.
-- ⛔ save_as_last is NEVER passed — it would repoint the owner's Continue button.
-- ⛔ G4: the proof is a LIVE list-before/list-after, never a remembered flag.
--    (Batch-1's version computed a "before" count and then never used it.)
function CB2.SaveNamed(name)
	if type(name) ~= "string" or name == "" then
		say("SAVE SKIP <no name given>")
		return
	end
	CreateRealTimeThread(function()
		local sg = rawget(_G, "Savegame")
		local before = -1
		CB2.Try("Savegame.ListForTag before", function()
			local list = sg.ListForTag("savegame")
			before = type(list) == "table" and #list or -1
			return before
		end)
		CB2.TryYield("SaveGame " .. name, function()
			local err, nm = SaveGame(name, { savename = name, silent = true, no_screenshot = true })
			say(fmt("SAVE SaveGame returned err=%s name=%s", tostring(err), tostring(nm)))
			return err
		end)
		Sleep(1000)
		local after = -1
		CB2.Try("Savegame.ListForTag after", function()
			local list = sg.ListForTag("savegame")
			after = type(list) == "table" and #list or -1
			return after
		end)
		say(fmt("SAVE WITNESS tagged savegame files %s -> %s (a save that did not land leaves this UNCHANGED — that is the liveness witness, not SaveGame's own return)",
			tostring(before), tostring(after)))
		say("SAVE ⛔ save_as_last was NOT passed — the owner's Continue button is untouched.")
	end)
end

-- ⭐ F21, the restamp witness — the reader gap closed 2026-08-10.
--
-- WHAT IT WATCHES, and why it is a witness rather than an inference: the whole
-- triple-spend is computed from ONE field, `colonist.transport_ticket
-- .start_wait` (ColonistTransport.lua:493 stamps it on reaching the platform;
-- :551 spends it at ExitVehicle three times over). Our wrapper restamps it at
-- the boarding moment. `ticket.stage` moves "Inbound" -> "Waiting" (:496) ->
-- "Traveling" (BoardVehicle's destructor, :515). So watching ONE subject across
-- that transition answers the question directly:
--    start_wait CHANGED at boarding  -> the wrapper fired, ride-only from here
--    start_wait UNCHANGED            -> it did not, and F21 is live for them
-- ⛔ Neither outcome can be produced by an absent fixture: the subject is only
--    chosen once they are already in station.waiting_for_train with a ticket.
-- ⚠️ NOT a `tested` re-earn on its own. That stays what the rider says it is.
function CB2.RideF21(tries)
	tries = tries or 120
	local n, c = CB2.ReadF21Subjects()
	if not c then
		say("⛔ F21 UNSAMPLED <no colonist is waiting on a platform with a ticket right now>. Let the sim run at speed and call CB2.RideF21() again. This is a missing subject, NOT a negative result.")
		return false
	end
	local tk = c.transport_ticket
	if type(tk) ~= "table" or not tk.start_wait then
		say("⛔ F21 UNSAMPLED <the chosen subject's transport_ticket vanished between the find and the watch>")
		return false
	end
	local before_stage, before_stamp = tostring(tk.stage), tk.start_wait
	say(fmt("F21WITNESS subject %s BEFORE: stage=%s start_wait=%s (GameTime=%s, waited so far=%s ms). FORCED: nothing — the boarding is the game's own.",
		id_of(c), before_stage, tostring(before_stamp), tostring(GameTime()), tostring(GameTime() - before_stamp)))
	CreateRealTimeThread(function()
		for attempt = 1, tries do
			local t = c.transport_ticket
			if type(t) ~= "table" then
				say(fmt("F21WITNESS subject %s lost their ticket at try %d/%d (discarded, or they left the queue) — UNSAMPLED, pick another subject.", id_of(c), attempt, tries))
				return
			end
			if tostring(t.stage) == "Traveling" then
				local moved = (t.start_wait ~= before_stamp)
				say(fmt("F21WITNESS subject %s BOARDED at try %d/%d — stage %s -> Traveling | start_wait %s -> %s | RESTAMPED=%s (GameTime at read=%s)",
					id_of(c), attempt, tries, before_stage,
					tostring(before_stamp), tostring(t.start_wait), tostring(moved), tostring(GameTime())))
				if moved then
					say(fmt("⭐ F21WITNESS THE WRAPPER FIRED — start_wait advanced by %s ms to the boarding moment, so the train, the track and the 'travel time' Comfort charge will all measure the RIDE only. This is the fix observed doing its work, not inferred from an average.",
						tostring(t.start_wait - before_stamp)))
				else
					say("⛔ F21WITNESS THE WRAPPER DID NOT FIRE on this boarding — start_wait is unchanged, so ExitVehicle will spend wait+ride three times exactly as F21 describes. That is a DEFECT reading and it goes on the entry verbatim.")
				end
				CB2.ReadF21("immediately after a witnessed boarding")
				return
			end
			Sleep(500)
		end
		say(fmt("⛔ F21WITNESS BOUND HIT — %d tries and subject %s never reached stage 'Traveling' (last stage=%s). UNSAMPLED with its try count stated; no train came for them in the window.",
			tries, id_of(c), tostring(c.transport_ticket and c.transport_ticket.stage)))
	end)
	say("F21WITNESS watching in the background; the verdict line prints itself. Carry on with the next leg.")
	return true
end

-- The thread-validity read the audit names. Its own liveness witness: the
-- ACTIVE-count is printed alongside, so "thread invalid" cannot be confused
-- with "no storybit was ever running".
function CB2.PKeystoneRead(when)
	local active = rawget(_G, "g_StoryBitActive") or empty_table
	local n = #active
	local first = active[1]
	local thr = type(first) == "table" and first.run_thread or nil
	local ok, valid = pcall(function() return IsValidThread(thr) end)
	say(fmt("PKEYSTONE g_StoryBitActive=%d entr(ies) | first.id=%s run_thread=%s IsValidThread=%s (read ok=%s) | when=%s",
		n, tostring(type(first) == "table" and first.id or "-"), tostring(thr),
		tostring(valid), tostring(ok), tostring(when)))
	CB2.Applicable("popup-keystone-thread-survival", tostring(valid), n,
		"g_StoryBitActive is the population; a validity verdict over 0 running bits samples nothing")
	return valid, n
end

---- LEG Q — F99's untested cell: repair path x multi-track merge -------------
--
-- ⛔ WHAT THIS LEG IS AND IS NOT. It FORCES BOTH HALVES — the breaks and the
--    completion. It samples ONE CELL of F99's 2x2 for a RATE datum with its
--    condition stated. It can NEVER speak to reachability, and no line of its
--    output may be read as doing so.
--
-- THE CELL (F99 entry, 2026-08-05): repair sites on TWO OR MORE DIFFERENT
-- tracks, completed together. Sample 1 had the repair path without a merge;
-- sample 2 had the merge without the repair path. Nobody has run both.
--
-- THE TRAP, Src-re-verified 2026-08-10: BreakTracks (Meteors.lua:599-613) skips
-- any element that is its track's start_el or end_el or carries a station, so an
-- endpoint aim SILENTLY DOES NOTHING. Every break is witnessed here, never
-- assumed. ⚠️ `el.broken` is a TABLE (the repair site), not a boolean — a
-- truthiness test is correct, `el.broken == true` would be false forever.
function CB2.QBreak(want_tracks)
	want_tracks = want_tracks or 2
	local usable = CB2.ReadTrackQ(want_tracks)
	if #usable < want_tracks then
		say(fmt("⛔ LEGQ SKIP <the cell needs %d or more DISTINCT tracks with a breakable mid-element; this save offers %d>", want_tracks, #usable))
		return false
	end
	CB2.q_broken = {}
	local witnessed, distinct = 0, {}
	for i = 1, want_tracks do
		local u = usable[i]
		local el, tr = u.el, u.track
		local sites_before = #((type(tr) == "table" and tr.elements_under_construction) or empty_table)
		local cgs_before   = #((type(tr) == "table" and tr.repair_cgs) or empty_table)
		say(fmt("LEGQ FORCED: %s:CheatBreakElement() on mid-element %s of track %s (sites_before=%d repair_cgs_before=%d). MEASURED: nothing yet — this only builds the cell.",
			id_of(el), id_of(el), id_of(tr), sites_before, cgs_before))
		CB2.Try("CheatBreakElement", function() return el:CheatBreakElement() end)
		Sleep(1500)   -- settling window: the site is placed asynchronously
		local sites_after = #((type(tr) == "table" and tr.elements_under_construction) or empty_table)
		local cgs_after   = #((type(tr) == "table" and tr.repair_cgs) or empty_table)
		-- THE BREAK WITNESS, three independent signs. `broken` is a table.
		local broken_flag = el.broken and true or false
		local hit = broken_flag or (sites_after > sites_before) or (cgs_after > cgs_before)
		say(fmt("LEGQ BREAK WITNESS element=%s broken=%s (type=%s) | repair sites %d -> %d | repair_cgs %d -> %d | WITNESSED=%s",
			id_of(el), tostring(broken_flag), type(el.broken), sites_before, sites_after,
			cgs_before, cgs_after, tostring(hit)))
		if hit then
			witnessed = witnessed + 1
			distinct[tr] = true
			CB2.q_broken[#CB2.q_broken + 1] = { track = tr, el = el }
		else
			say("⛔ LEGQ THIS BREAK DID NOT TAKE — BreakTracks skipped it (endpoint or station element), or nothing changed. It is NOT counted. Do not treat a later zero as covering this element.")
		end
	end
	local n_distinct = 0
	for _ in pairs(distinct) do n_distinct = n_distinct + 1 end
	CB2.q_distinct = n_distinct
	CB2.Applicable("F99-cell-built(DISTINCT tracks carrying a WITNESSED repair site)",
		n_distinct, want_tracks,
		"each break confirmed by el.broken truthy OR a new repair site OR repair_cgs growth, after a settling window")
	if n_distinct < 2 then
		say(fmt("⛔ LEGQ THE CELL IS NOT BUILT — %d distinct track(s) carry a witnessed repair site and the cell needs 2 or more. Running the completion now would re-sample F99's ALREADY-TESTED single-track repair cell (sample 1, 2026-08-05) and add nothing. Break more, or record leg Q as SKIP <cell could not be built>.", n_distinct))
		return false
	end
	say(fmt("⭐ LEGQ CELL BUILT — %d DISTINCT tracks each carrying a witnessed repair site. This is the configuration no sample has covered. CB2.QComplete() is now allowed.", n_distinct))
	CB2.ReadF99Residue("after the breaks, before the completion")
	return true
end

-- The completion. ⛔ Refuses unless the cell is actually built.
function CB2.QComplete()
	if (CB2.q_distinct or 0) < 2 then
		say("⛔ LEGQ REFUSED — run CB2.QBreak() first and it must report 2 or more DISTINCT tracks with WITNESSED breaks. Completing without that samples a cell F99 already has two zeros for.")
		return
	end
	say(fmt("LEGQ FORCED: CheatCompleteAllConstructions() on map %s, with %d DISTINCT tracks carrying repair sites. MEASURED: whether TrackElement.lua:805 throws. FORCED BOTH HALVES — this is a rate datum for one cell and says NOTHING about reachability.",
		tostring(rawget(_G, "CurrentMapName")), CB2.q_distinct))
	say("LEGQ ⛔ WATCH LINE: the throw to look for is `Lua/Buildings/TrackElement.lua:805: attempt to index a nil value (local 'start_el')`, stacked under ConstructionSite.lua:2483 -> Cheats.lua:115 -> MapForEach -> Cheats.lua:126. Zero of them is a RATE BOUND, not a refutation.")
	CB2.Try("CheatCompleteAllConstructions", function() return CheatCompleteAllConstructions() end)
	Sleep(3000)
	-- COMPLETION WITNESS: sites must have gone to 0, or the cell never completed
	-- and a zero-throw reading would be about nothing.
	local sites_left, tracks_seen = 0, 0
	for _, r in ipairs(CB2.q_broken or empty_table) do
		tracks_seen = tracks_seen + 1
		sites_left = sites_left + #((type(r.track) == "table" and r.track.elements_under_construction) or empty_table)
	end
	say(fmt("LEGQ COMPLETION WITNESS %d track(s) watched, %d repair site(s) still outstanding after the cheat", tracks_seen, sites_left))
	CB2.Applicable("F99-cell-completion-witness", sites_left, tracks_seen,
		"repair sites must reach 0 for the completion to have happened at all; a zero-throw reading over uncompleted sites samples nothing")
	CB2.ReadF99Residue("immediately after the completion, BEFORE any save or load")
	say("LEGQ ⛔ read F99RESIDUE above BEFORE saving or loading anything — load runs SavegameFixups.RebuildBrokenTracksAndConnect, which sweeps exactly what that counter looks for.")
end

function CB2.RPickDome(index)
	local domes = CB2.LabelAll("Dome")
	local main = rawget(_G, "MainMap")
	local on_main = {}
	for i, x in ipairs(domes) do
		local m = IsValid(x) and ResolveMap(x) or nil
		local is_main = (m ~= nil and m == main)
		if is_main then on_main[#on_main + 1] = x end
		if i <= 8 then
			say(fmt("DOME[%d] %s pos=%s on_MainMap=%s", i, id_of(x),
				tostring(IsValid(x) and x:GetPos() or "?"), tostring(is_main)))
		end
	end
	local d = index and domes[index] or on_main[1] or domes[1]
	say(fmt("DOME %d of %d dome(s) are on MainMap — WaitBombard hard-codes MainMap, so only these are usable",
		#on_main, #domes))
	CB2.Applicable("PT47-usable-dome(on MainMap)", #on_main, #domes,
		"every dome's ResolveMap compared against the MainMap global this load")
	say(fmt("RPickDome -> %s (pass an index to choose another; put the camera on it and keep it LOW so the trails are visible)", id_of(d)))
	CB2.dome = d
	return d
end

function CB2.RBombard(dome, count, radius)
	dome = dome or CB2.dome or CB2.LabelAll("Dome")[1]
	if not IsValid(dome) then say("⛔ LEGR SKIP <no valid dome to centre the volley on>") return end
	count = count or 8
	radius = radius or 40 * guim
	local map = rawget(_G, "MainMap")
	local dome_map = ResolveMap(dome)
	if dome_map ~= map then
		say(fmt("⛔ LEGR REFUSED — dome %s resolves to map %s but WaitBombard hard-codes MainMap. The volley would land off camera and the watcher would poll a table nothing writes to. Pick a dome whose RPickDome line reads on_MainMap=true.",
			id_of(dome), tostring(dome_map)))
		return
	end
	local speed = GetTimeFactor() / const.DefaultTimeFactor
	if speed == 0 then
		say("⛔ LEGR REFUSED — the game is PAUSED and StartBombard only spawns a game-time thread; nothing would happen. Unpause and call again.")
		return
	end
	local before = 0
	for _ in pairs(map.g_IncomingMissiles or empty_table) do before = before + 1 end
	say(fmt("LEGR FORCED: StartBombard(<dome %s>, %d, %d, 500, 1500) — a volley that would not otherwise happen. MEASURED: the SPREAD (owner's eyes, the one reading that is eyes by nature) and the five integrity readings. speed=%s g_IncomingMissiles_before=%d",
		id_of(dome), radius, count, tostring(speed), before))
	say("LEGR ⛔ FIRST EXECUTION of StartBombard by this project. Its documented recipe was measured BROKEN; this is the corrected form, and the pcall result below is printed whatever it says.")
	CB2.bombard_seen = false
	CB2.Try("StartBombard [FIRST EXECUTION]", function()
		return StartBombard(dome, radius, count, 500, 1500)
	end)
	-- G2 LIVENESS WITNESS for "the volley ENDED": an empty missile table also
	-- describes a volley that never started, so the table must be seen NON-EMPTY
	-- first and that is recorded as the witness.
	CreateRealTimeThread(function()
		local deadline = RealTime() + 3 * 60 * 1000
		local peak = 0
		while RealTime() < deadline do
			local n = 0
			for _ in pairs(map.g_IncomingMissiles or empty_table) do n = n + 1 end
			if n > peak then peak = n end
			if peak > 0 then CB2.bombard_seen = true end
			if peak > 0 and n == 0 then
				say(fmt("LEGR VOLLEY ENDED — peak in-flight missiles = %d (WITNESS: the table was non-empty, so this zero is a real end, not a volley that never started). A volley that never finishes is F26's stated FAIL.", peak))
				CB2.RResidue(map)
				return
			end
			Sleep(1000)
		end
		say(fmt("⛔ LEGR 3-MINUTE BOUND HIT — peak in-flight missiles = %d, still %s. peak 0 means the volley NEVER STARTED (read the TRY line for the raise); peak > 0 means it did not end, which is the FAIL condition F26 says should revert the fix.",
			peak, peak > 0 and "in flight" or "nothing ever spawned"))
	end)
	say("LEGR watching up to 3 minutes; the END line prints itself. ⇒ OWNER, EYES ON THE SKY NOW: a SCATTER (trails arriving from visibly different angles) or a RANK of parallel trails? Parallel is the old behaviour.")
end

-- The agent-side half of F26's five integrity checks. Decals, dome cracks and
-- interception are read here; the notification and the volley ENDING are read
-- by the watcher above and the owner's eyes.
function CB2.RResidue(map)
