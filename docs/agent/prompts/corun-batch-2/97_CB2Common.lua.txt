-- ModLog is the only print path proven to reach the log file on disk
-- (Mod.lua:109-132). Its emitter formats its single argument a second time
-- (lib.lua:164-174), so a literal '%' raises there — escape it. Flush per line
-- so a killed run still shows how far it got.
local function say(msg)
	msg = "[" .. tostring(CB2.tag) .. "] " .. tostring(msg)
	local modlog = rawget(_G, "ModLog")
	if modlog then
		modlog((msg:gsub("%%", "%%%%")))
	else
		local dbg = rawget(_G, "DebugPrint")
		if dbg then dbg(msg .. "\n") end
	end
	print(msg)
	local flush = rawget(_G, "FlushLogFile")
	if flush then pcall(flush) end
end
CB2.Say = say

local function fmt(...)
	local ok, s = pcall(string.format, ...)
	return ok and s or "<fmt error>"
end
CB2.Fmt = fmt

local function id_of(obj)
	if not obj then return "nil" end
	if type(obj) ~= "table" then return tostring(obj) end
	return tostring(obj.class or "?") .. "#" .. tostring(rawget(obj, "handle"))
end
CB2.IdOf = id_of

-- G3. The ONLY sanctioned call wrapper. Prints ok and the error text, always.
-- ⛔ pcall cannot survive a yield — use CB2.TryYield for anything that Sleeps.
function CB2.Try(what, fn, ...)
	local ok, a, b = pcall(fn, ...)
	say(fmt("TRY %s ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

-- Same contract for steps that yield. sprocall is the engine's yield-safe pcall.
function CB2.TryYield(what, fn, ...)
	local ok, a, b = sprocall(fn, ...)
	say(fmt("TRY %s (yielding) ok=%s result=%s%s", tostring(what), tostring(ok),
		tostring(a), b ~= nil and (" / " .. tostring(b)) or ""))
	return ok, a, b
end

-- G2. The condition-sampled rule, mechanised.
function CB2.Applicable(label, count, population, witness)
	local applicable = (population or 0) > 0
	say(fmt("READ %s = %s | population=%s | APPLICABLE=%s | witness=%s",
		tostring(label), tostring(count), tostring(population),
		tostring(applicable), tostring(witness or "-- NONE NAMED --")))
	return applicable
end

---- flow helpers ------------------------------------------------------------

local function shutdown(reason)
	CB2.stage = "done"
	say("SHUTDOWN " .. tostring(reason))
	local flush = rawget(_G, "FlushLogFile")
	if flush then flush() end
	Sleep(CB2.quit_delay)
	quit()
end
CB2.Shutdown = shutdown

local function step(name, fn, untimed)
	CB2.stage = name
	local t = RealTime()
	say("STEP " .. name .. " BEGIN")
	local ok, res = sprocall(fn)
	if not ok then
		say("STEP " .. name .. " ERROR")
		shutdown("ERROR " .. name)
		return false
	end
	if untimed then
		say("STEP " .. name .. " END (no timing — this step blocks render mode; a load takes its number from the engine's own 'Game loaded on map ... in Nms' line, a save has none)")
	else
		say(fmt("STEP %s END %dms", name, RealTime() - t))
	end
	if res == false then
		shutdown("STOP " .. name .. " — the step asked to stop; its reason is the last line before this one")
		return false
	end
	return true
end
CB2.Step = step

local function wait_for(timeout, ready)
	local deadline = RealTime() + timeout
	while not ready() do
		if RealTime() - deadline >= 0 then return false end
		Sleep(500)
	end
	return true
end
CB2.WaitFor = wait_for

function CB2.Load(savefile, speed)
	local err = LoadGame(savefile, {})            -- ⛔ never pass save_as_last
	if err then
		say("LOADERR " .. tostring(savefile) .. " " .. tostring(err))
		return false
	end
	CB2.loads = CB2.loads + 1
	say(fmt("LOAD OK #%d file=%s — LoadGame returned no error", CB2.loads, tostring(savefile)))
	if not wait_for(5 * 60 * 1000, game_is_live) then
		say("LOADERR " .. tostring(savefile) .. " — city never came live")
		return false
	end
	Sleep(CB2.settle_time)                         -- 15 s is the MEASURED-sufficient datum
	speed = speed or 3
	local before = GetTimeFactor() / const.DefaultTimeFactor
	local ok = pcall(function() UIColony:SetGameSpeed(speed) end)
	Sleep(500)
	local after = GetTimeFactor() / const.DefaultTimeFactor
	say(fmt("READ speed: SetGameSpeed(%d) call=%s | before=%s after=%s",
		speed, tostring(ok), tostring(before), tostring(after)))
	return true
end

-- ⚠️ `named` ids are the REGISTERED ids (SMRFixPack.Register's first argument),
--    NOT file names: batch-1 asked for "Opt_CohortHousing" and got

function CB2.ReadConditions(note, named)
	local pack = rawget(_G, "SMRFixPack")
	local active, total, inactive = 0, 0, {}
	if pack and pack.order then
		for _, id in ipairs(pack.order) do
			total = total + 1
			local f = pack.fixes and pack.fixes[id]
			local st = f and f.status or "missing"
			if st == "active" then active = active + 1 else inactive[#inactive + 1] = id .. "=" .. st end
		end
	end
	say(fmt("CONDITIONS pack=%d/%d active | map=%s | speed=%s | loads_this_process=%d | note=%s",
		active, total, tostring(rawget(_G, "CurrentMapName")),
		tostring(GetTimeFactor() / const.DefaultTimeFactor), CB2.loads, tostring(note or "-")))
	if #inactive > 0 then
		say("CONDITIONS not-active: " .. table.concat(inactive, ", "))
	end
	for _, id in ipairs(named or empty_table) do
		local f = pack and pack.fixes and pack.fixes[id]
		say(fmt("CONDITIONS by-name %s = %s", tostring(id), tostring(f and f.status or "NOT REGISTERED")))
	end
	return active, total
end

function CB2.ReadC42(when)
	local stale, elements, units, passages = 0, 0, 0, 0
	for _, c in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, p in ipairs((c.labels or empty_table).Passage or empty_table) do
			passages = passages + 1
			for _, el in ipairs(p.elements or empty_table) do
				elements = elements + 1
				for _, u in ipairs(el.units or empty_table) do
					units = units + 1
					if u.holder ~= el then stale = stale + 1 end
				end
			end
		end
	end
	say(fmt("C42STALE %d  | %d passage(s), %d element(s), %d unit entr(ies) | when=%s",
		stale, passages, elements, units, tostring(when)))
	CB2.Applicable("C42-stale-entries", stale, units,
		"unit ENTRIES is the denominator; passages>0 is NOT the condition")
	return stale, passages, elements, units
end

-- ⭐ NEW FOR THIS CHAIN, and it is the whole point of the C42 ride-along.
--
-- THE GAP, restated from the entry (2026-08-10 block): two sessions have read
-- C42STALE 0 over ZERO unit entries — once post-load, once within-session after
-- speed-20 traffic. Traffic is NOT the condition. Nobody has yet seen a
-- colonist inside a passage element at all, so "0 stale" and "no traversal was
-- ever in flight" are indistinguishable and the entry cannot move either way.
--
-- WHAT THIS DOES (S4/S5 discipline, all four parts named):
--   * WITNESS — polls every passage element until `units` is non-empty ONCE and
--     prints the exact (element, unit, holder) triple that witnessed it. No
--     witness => the leg is UNSAMPLED with its try count stated, never a zero.
--   * BOUNDED — tries and interval are arguments with sane defaults; it cannot
--     spin.
--   * TYPE-CHECKED — every field is read through type()/IsValid before use.
--   * SETTLING WINDOW — a witness is a TRANSIENT (a colonist mid-traversal is
--     legitimately in that list). The entry's claim is that the entry PERSISTS
--     after the traversal ends. So the witnessed pair is re-read after a
--     settling window and the verdict distinguishes:
--        persists AND u.holder ~= el  -> CONFIRMS the stale-entry mechanism
--        cleared                      -> REFUTES it (the exit deregistered them)
--        persists AND u.holder == el  -> still traversing; extend and re-read
function CB2.WitnessC42(tries, interval_ms, settle_ms)
	tries       = tries or 60
	interval_ms = interval_ms or 1000
	settle_ms   = settle_ms or 20000
	local seen_el, seen_unit, seen_holder, n = nil, nil, nil, 0
	local passages, elements = 0, 0
	for attempt = 1, tries do
		passages, elements, n = 0, 0, 0
		for _, c in ipairs(rawget(_G, "Cities") or empty_table) do
			for _, p in ipairs((c.labels or empty_table).Passage or empty_table) do
				passages = passages + 1
				for _, el in ipairs((type(p) == "table" and p.elements) or empty_table) do
					elements = elements + 1
					local list = (type(el) == "table" and el.units) or empty_table
					for _, u in ipairs(list) do
						n = n + 1
						if not seen_el then
							seen_el, seen_unit, seen_holder = el, u, u.holder
						end
					end
				end
			end
		end
		if seen_el then
			say(fmt("C42WITNESS HIT on try %d/%d — element %s holds unit %s (unit.holder=%s, same_object=%s) | %d passage(s), %d element(s), %d entr(ies) this sweep",
				attempt, tries, id_of(seen_el), id_of(seen_unit), id_of(seen_holder),
				tostring(seen_holder == seen_el), passages, elements, n))
			break
		end
		Sleep(interval_ms)
	end
	if not seen_el then
		say(fmt("⛔ C42WITNESS NONE — %d tries at %dms over %d passage(s) / %d element(s) and `units` was EMPTY every sweep. The C42 leg is UNSAMPLED, and this is the try count that must be stated beside it. It is NOT a zero and NOT a refutation: no colonist was ever observed inside a passage element, so the stale-entry population was never created.",
			tries, interval_ms, passages, elements))
		CB2.Applicable("C42-traversal-witness", 0, 0,
			fmt("%d bounded tries at %dms; `units` non-empty was the condition and it never held", tries, interval_ms))
		return false
	end
	CB2.Applicable("C42-traversal-witness", 1, 1,
		fmt("a unit was OBSERVED inside passage element %s — the population C42STALE counts over now demonstrably exists", id_of(seen_el)))
	-- SETTLING WINDOW. A mid-traversal entry is legitimate; a persisting one is
	-- the defect. Re-read the SAME pair.
	say(fmt("C42WITNESS settling %dms before re-reading the SAME pair — a transient entry is not the defect, a persisting one is.", settle_ms))
	Sleep(settle_ms)
	local still, holder_now = false, nil
	for _, u in ipairs((type(seen_el) == "table" and seen_el.units) or empty_table) do
		if u == seen_unit then still = true holder_now = u.holder end
	end
	if not still then
		say(fmt("⭐ C42WITNESS CLEARED after %dms — unit %s is no longer in element %s's `units`. That REFUTES the stale-entry mechanism for this traversal: something deregistered them. Record it as a measured negative WITH its witness, and say so on the entry.",
			settle_ms, id_of(seen_unit), id_of(seen_el)))
	elseif holder_now ~= seen_el then
		say(fmt("⭐⭐ C42WITNESS PERSISTS AND IS STALE after %dms — unit %s is still in element %s's `units` while unit.holder = %s. That is the defect this entry describes, WITNESSED for the first time.",
			settle_ms, id_of(seen_unit), id_of(seen_el), id_of(holder_now)))
	else
		say(fmt("C42WITNESS persists but unit.holder is STILL the element after %dms — the colonist is very likely mid-traversal, not stale. Re-run CB2.WitnessC42 with a longer settle before reading anything into it.", settle_ms))
	end
	return CB2.ReadC42("within-session, AFTER a named traversal witness, no save/load since")
end

---- F21 — the reader gap, CLOSED 2026-08-10 --------------------------------
--
-- ⛔ BATCH-1'S READER WAS THE DEFECT, and this replaces it rather than
--    inheriting it. CB1.ReadTrainStats read `t.ts_spent_time or t.spent_time`
--    and printed `spent_time=nil` on every train. NEITHER FIELD EXISTS on any
--    class in the tree. `TransportStatistics` (Lua\TransportStatistics.lua:1-11)
--    declares spent_time_sum / spent_time_values / visitors_per_day /
--    visitors_lifetime / full_today / full_yesterday / trip_time / last_reset,
--    and exactly three classes flatten it: Station (Station.lua:52),
--    StationsLink (StationsLink.lua:5) and Train (Train.lua:17). So that nil was
--    guaranteed on any save in any state — a reader that could not have printed
--    anything else. Full derivation on the F21 entry, 2026-08-10.
function CB2.ReadF21(when)
	local stations = CB2.LabelAll("Station")
	local trains   = CB2.LabelAll("Train")
	local waiting, riding = 0, 0
	for _, s in ipairs(stations) do
		waiting = waiting + #((type(s) == "table" and s.waiting_for_train) or empty_table)
	end
	for i, t in ipairs(trains) do
		local aboard = (type(t.units) == "table") and #t.units or 0
		riding = riding + aboard
		if i <= 4 then
			local ok, avg = pcall(function() return t:GetSpentTimeAverage() end)
			local ok_h, hrs = pcall(function() return t:GetSpentTimeAverageInHours() end)
			say(fmt("F21 train %s command=%s aboard=%d | spent_time_sum=%s samples=%s avg=%s (ok=%s) avg_hours=%s (ok=%s) trip_time=%s",
				id_of(t), tostring(t.command), aboard,
				tostring(rawget(t, "spent_time_sum")),
				tostring(type(t.spent_time_values) == "table" and #t.spent_time_values or t.spent_time_values),
				tostring(avg), tostring(ok), tostring(hrs), tostring(ok_h),
				tostring(rawget(t, "trip_time"))))
		end
	end
	for i, s in ipairs(stations) do
		if i > 3 then break end
		local ok, avg = pcall(function() return s:GetSpentTimeAverage() end)
		say(fmt("F21 station %s waiting=%d | spent_time_sum=%s samples=%s avg=%s (ok=%s)",
			id_of(s), #((type(s) == "table" and s.waiting_for_train) or empty_table),
			tostring(rawget(s, "spent_time_sum")),
			tostring(type(s.spent_time_values) == "table" and #s.spent_time_values or s.spent_time_values),
			tostring(avg), tostring(ok)))
	end
	say(fmt("F21 %d station(s) holding %d colonist(s) waiting | %d train(s) carrying %d rider(s) | when=%s",
		#stations, waiting, #trains, riding, tostring(when)))
	CB2.Applicable("F21-platform-wait-population", waiting, #stations,
		"station.waiting_for_train walked this load")
	return waiting, riding, #trains
end

-- ⭐ SUBJECT-FINDER (S2) for the F21 restamp witness. Names ONE colonist who is
-- on a platform with a live ticket, so the sitting has a subject instead of a
-- hunt. Prints the ticket fields the witness is taken on.
function CB2.ReadF21Subjects(limit)
	limit = limit or 3
	local found, shown = 0, 0
	local pick = nil
	for _, s in ipairs(CB2.LabelAll("Station")) do
		for _, c in ipairs((type(s) == "table" and s.waiting_for_train) or empty_table) do
			local tk = (type(c) == "table") and c.transport_ticket or nil
			if type(tk) == "table" and tk.start_wait then
				found = found + 1
				if not pick then pick = { c = c, s = s, tk = tk } end
				if shown < limit then
					shown = shown + 1
					say(fmt("F21SUBJ candidate[%d] colonist=%s station=%s ticket.stage=%s ticket.start_wait=%s (GameTime now=%s, waited=%s ms)",
						shown, id_of(c), id_of(s), tostring(tk.stage), tostring(tk.start_wait),
						tostring(GameTime()), tostring(GameTime() - tk.start_wait)))
				end
			end
		end
	end
	CB2.Applicable("F21-restamp-subject", found, found,
		"a colonist in station.waiting_for_train carrying a transport_ticket with a start_wait stamp")
	if pick then
		CB2.f21_subject = pick.c
		-- ⚠️ CORRECTED IN PREP by the G1 pass, 2026-08-10. This line named a
		-- "RideF21" + "Witness" function that does not exist; at an attended
		-- console a dead name is the owner typing a line that cannot work. Same
		-- failure class as unattended-1's missing ErrorWatchNote, found the same
		-- way, this time inside a message string rather than a call.
		-- ⛔ The bad name is spelled AROUND rather than written out, because a
		-- correction note that contains the token makes the G1 grep match itself
		-- and report RED forever — batch-1's ERRORWATCH marker, exactly.
		say(fmt("F21SUBJ ⇒ subject recorded: %s at %s. CB2.RideF21() polls them across boarding.", id_of(pick.c), id_of(pick.s)))
	else
		say("F21SUBJ NONE — no colonist is waiting on a platform with a ticket right now. The restamp witness is UNSAMPLED until one is; let the sim run and call again.")
	end
	return found, pick and pick.c or nil
end

---- Leg Q — F99's untested cell: repair path x multi-track merge -------------
--
-- ⛔ WHAT THE CELL IS (F99 entry, 2026-08-05 2x2): sample 1 was the repair path
--    on ONE track (0 throws); sample 2 was a confirmed multi-track merge with
--    NEW-BUILD sites (0 throws). The empty cell is repair sites on TWO OR MORE
--    DIFFERENT tracks completing together — the configuration all seven filed
--    throws happened under.
--
-- ⛔ THE INSTRUMENT AND ITS TRAP, both Src-re-verified on this tree 2026-08-10:
--    `TrackGridElement:CheatBreakElement()` = `BreakTracks({self})`
--    (TrackElement.lua:436-438) — an ungated global, works on retail. Its
--    sibling `CheatMeteorHit` three lines below calls the Platform.cheats-only
--    `TestMeteor` and THROWS on retail: that is F101.
--    `BreakTracks` (Meteors.lua:599-613) skips any element that is its track's
--    `start_el` or `end_el` or carries a `station` — so an ENDPOINT CLICK
--    SILENTLY DOES NOTHING. Every break must be witnessed, never assumed.
--
-- This reader finds MID-elements on DISTINCT tracks that BreakTracks will
-- actually accept, so the owner/console never aims at an element that will be
-- skipped.
function CB2.ReadTrackQ(want_tracks)
	want_tracks = want_tracks or 2
	local tracks = CB2.LabelAll("TrackBase")
	local usable, els, broken_now, sites_now = {}, 0, 0, 0
	for _, tr in ipairs(tracks) do
		local list = (type(tr) == "table" and tr.elements) or empty_table
		els = els + #list
		sites_now = sites_now + #((type(tr) == "table" and tr.elements_under_construction) or empty_table)
		for _, el in ipairs(list) do if el.broken then broken_now = broken_now + 1 end end
		-- the exact BreakTracks filter, applied here so a pick cannot be a no-op
		local cand = nil
		for i = 2, #list - 1 do
			local el = list[i]
			if el ~= tr.start_el and el ~= tr.end_el and not el.station and not el.broken then
				cand = el
				break
			end
		end
		if cand then usable[#usable + 1] = { track = tr, el = cand, n = #list } end
	end
	say(fmt("LEGQ %d track(s), %d element(s), %d already-broken, %d existing repair site(s)",
		#tracks, els, broken_now, sites_now))
	for i, u in ipairs(usable) do
		if i > 4 then break end
		say(fmt("LEGQ   breakable[%d] track=%s (%d elements) mid-element=%s — passes the BreakTracks filter (not start_el, not end_el, no station, not already broken)",
			i, id_of(u.track), u.n, id_of(u.el)))
	end
	CB2.q_candidates = usable
	CB2.Applicable("F99-cell-fixture(DISTINCT tracks with a breakable mid-element)",
		#usable, #tracks,
		fmt("BreakTracks' own filter applied per track; the cell needs %d or more DISTINCT tracks", want_tracks))
	if #usable < want_tracks then
		say(fmt("⛔ LEGQ SKIP <only %d track(s) offer a breakable mid-element; the untested cell needs %d or more DISTINCT tracks and cannot be built on this save>", #usable, want_tracks))
	end
	return usable
end

function CB2.ReadF99Residue(when)
	local a, b = 0, 0
	for _, c in ipairs(rawget(_G, "Cities") or empty_table) do
		for _, t in ipairs((c.labels or empty_table).TrackBase or empty_table) do
			if #((type(t) == "table" and t.elements) or empty_table) == 0 then a = a + 1 end
			if t.repair_cgs and #t.repair_cgs > 0

		end
	end

	return a, b
end

---- Leg P — popup subjects (S2 subject-finder) ------------------------------
--
-- ⛔ THE DOCUMENTED ROUTE IS DEAD ON THIS SAVE and batch-1 measured it:
--    POPUP_CONSEQUENCE_AUDIT §8 item 3 calls a rocket launch-issue prompt the
--    cheapest choice popup. Of 6 rockets, 5 reported an issue and 4 of those
--    were `no_target`, which is not a key of RocketBase:GetLaunchIssuePopups;
--    the one `fuel` rocket was on Earth; and GetLaunchIssuePopups RAISED on
--    every rocket queried. So this finder does not re-try it — it finds the two
--    routes that are actually reachable and reports whether each is live.
--
--    (a) THE KEYSTONE (audit §8 item 1): a storybit with replies. Popup path is
--        StoryBitState:...WaitStoryBitPopup (_StoryBits.lua:622), reached when
--        the bit carries StoryBitReply children. `StoryBits[id]` is the preset
--        table; `.Enabled` is the gate the engine itself checks
--        (_StoryBits.lua:876, :952).
--    (b) F85's OWN SITE (not a stand-in): ShowBreakthroughChoicePopup
--        (Anomaly.lua:696-714). It returns immediately when
--        UIColony:UpdateAvailableBreackthroughTechs() is empty (:698), so the
--        count is read FIRST and a zero is a refusal to open, not a popup that
--        failed. ⚠️ Answering it calls SetTechDiscovered — staged copy only.
function CB2.ReadPopupSubjects(limit)
	limit = limit or 4
	-- (a) storybits
	local bits = rawget(_G, "StoryBits")
	local enabled_with_replies, shown = 0, 0
	local first = nil
	if type(bits) == "table" then
		for id, def in sorted_pairs(bits) do
			if type(def) == "table" and def.Enabled == true then
				local replies = 0
				for _, child in ipairs(def) do
					if type(child) == "table" and child.IsKindOf and child:IsKindOf("StoryBitReply") then
						replies = replies + 1
					end
				end
				if replies > 0 then
					enabled_with_replies = enabled_with_replies + 1
					if not first then first = id end
					if shown < limit then
						shown = shown + 1
						say(fmt("POPUPSUBJ storybit[%d] id=%s replies=%d Enabled=true — ForceActivateStoryBit(\"%s\", MainMap) would open a popup",
							shown, tostring(id), replies, tostring(id)))
					end
				end
			end
		end
	else
		say(fmt("POPUPSUBJ ⛔ global StoryBits is %s, not a table — the keystone route cannot be found from here", type(bits)))
	end
	CB2.storybit = first
	CB2.Applicable("popup-keystone-subject(enabled storybit carrying replies)",
		enabled_with_replies, enabled_with_replies,
		"global StoryBits walked; .Enabled == true and at least one StoryBitReply child is the engine's own popup condition")
	-- (b) breakthrough choice popup — F85's own site
	local ok, techs = pcall(function() return UIColony:UpdateAvailableBreackthroughTechs() end)
	local n = (ok and type(techs) == "table") and #techs or -1
	say(fmt("POPUPSUBJ breakthrough choice popup: available techs = %s (read ok=%s) — ShowBreakthroughChoicePopup returns immediately at Anomaly.lua:698 when this is 0",
		tostring(n), tostring(ok)))
	CB2.Applicable("F85-choice-popup-route(ShowBreakthroughChoicePopup)", n, (n > 0) and n or 0,
		"UIColony:UpdateAvailableBreackthroughTechs() read live; 0 is a refusal to open, not a failed popup")
	-- (c) the sol-change corner's own precondition (audit §3.6) — is a sol near?
	local ok_d, day = pcall(function() return UIColony.day end)
	say(fmt("POPUPSUBJ sol-change corner (audit 3.6): UIColony.day=%s (ok=%s) — the rider is OPTIONAL and only rides if a sol boundary is near",
		tostring(day), tostring(ok_d)))
	return enabled_with_replies, first, n
end

---- watchdog + flow ---------------------------------------------------------
--
-- Used only by the UNATTENDED payload (98_CB2Confirm). The attended sitting
-- library (99_CB2Legs) does not call this at all — it is invoked from the
-- console, one line per moment.

function CB2.Flow()
	CreateRealTimeThread(function()
		local deadline = RealTime() + CB2.watchdog
		while CB2.stage ~= "done" do
			if RealTime() - deadline >= 0 then
				say("WATCHDOG TIMEOUT at stage " .. tostring(CB2.stage))
				local flush = rawget(_G, "FlushLogFile")
				if flush then flush() end
				Sleep(1000)
				quit()
				return
			end
			Sleep(1000)
		end
	end)

	say(fmt("=== corun-batch-2 %s === watchdog %dms", tostring(CB2.tag), CB2.watchdog))

	if not step("1-waiting-menu", function()
		if not wait_for(CB2.menu_timeout, function() return not not GetPreGameMainMenu() end) then
			return false
		end
		Sleep(2000)
	end) then return end

	local ok = sprocall(CB2.Payload)
	if not ok then
		say("PAYLOAD ERROR at stage " .. tostring(CB2.stage))
		shutdown("ERROR payload")
		return
	end
	shutdown("done")
end

function CB2.Boot()
	CreateRealTimeThread(function()
		-- CreateRealTimeThread runs its body synchronously up to the first
		-- yield, so this Sleep is what puts everything below after
		-- ModsLoadCode().
		Sleep(1)
		if rawget(_G, "SMRTest_EnablePathLeg") then
			say("standing down: 98_EnablePathLeg.lua drives this session")
			return
		end
		CB2.Flow()
	end)
end
